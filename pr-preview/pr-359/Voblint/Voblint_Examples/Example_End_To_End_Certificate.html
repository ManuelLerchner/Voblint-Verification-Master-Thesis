section ‹One configuration, one program, no premise left standing›

text ‹
  ‹run_voblint_source_sound› is stated for an arbitrary configuration
  and an arbitrary program, so on its own it says nothing about any particular
  run: a caller still owes the answer. This theory computes it, by evaluation,
  for one program at one configuration --- the
  product domain, a call-string context of length one, and always-join named
  explicitly rather than defaulted --- and reads the theorem's conclusion off
  the result.

  Three dimensions at once is the point. A composite domain exercises the
  product's transfer; a call-string policy keys the solved table by call history;
  and naming a discipline the pairing does not default to shows the certificate
  travels with the solver actually run, not with the one the registry prefers.
›

subsection ‹The program›

text ‹
  ‹bump› is called twice with different actuals, so a call string of length one
  files two activations of it under two contexts. The check reads the second
  result.

subsection ‹One source run, constructed›

text ‹
  The endpoint speaks about a source execution, so this one is built rather than
  assumed: start every variable at ‹0›, run both calls and the check, and end with
  ‹a = 2› and ‹b = 42›. Two facts about the source semantics are needed and are
  general rather than about this program --- a check steps to \<^const>‹SKIP›
  without touching the store, and a call whose body is a single \<^const>‹Return›
  completes in three steps: enter the callee, publish the return value, and pop
  the frame writing the destination.
›

lemma pcompletes_Check: "pcompletes 𝒢 Π (VIMP_Proc.gcom.Check () l cond) s s"
  by (rule star.step) (rule pstep.Check, rule star.refl)

lemma pcompletes_Call_return:
  assumes p: "Π p = Some decl"
      and bd: "body decl = VIMP_Proc.gcom.Return () (Some e)"
      and arity: "length actuals = length (formals decl)"
      and dist: "distinct (formals decl)"
      and glob: "¬ 𝒢 ret_var"
  shows "pcompletes 𝒢 Π (VIMP_Proc.gcom.Call () dst p actuals) s
           (combine_assign dst
              (⟦e⟧⇩e (bind_formals (formals decl) (map (λa. ⟦a⟧⇩e s) actuals)
                         (enter_state 𝒢 s)))
              (combine_env 𝒢 s
                 (bind_formals (formals decl) (map (λa. ⟦a⟧⇩e s) actuals)
                    (enter_state 𝒢 s))))"
proof -
  let ?callee = "bind_formals (formals decl) (map (λa. ⟦a⟧⇩e s) actuals) (enter_state 𝒢 s)"
  let ?entered =
    "VIMP_Proc.gcom.Seq (VIMP_Proc.gcom.Return () (Some e)) (VIMP_Proc.gcom.Restore ())"
  let ?unwinding = "VIMP_Proc.gcom.Seq (VIMP_Proc.gcom.Unwind ()) (VIMP_Proc.gcom.Restore ())"
  have c1: "𝒢, Π ⊢ (VIMP_Proc.gcom.Call () dst p actuals, s, [])
              →⇩p (?entered, ?callee, [Frame s dst])"
    using arity bd dist p pstep_Call by fastforce
  have c2: "𝒢, Π ⊢ (?entered, ?callee, [Frame s dst])
              →⇩p (?unwinding, ?callee(ret_var := ⟦e⟧⇩e ?callee), [Frame s dst])"
    by (rule pstep.Seq2) (rule pstep.ReturnSome)
  have c3: "𝒢, Π ⊢ (?unwinding, ?callee(ret_var := ⟦e⟧⇩e ?callee), [Frame s dst])
              →⇩p (VIMP_Proc.gcom.SKIP (),
               combine_assign dst (⟦e⟧⇩e ?callee) (combine_env 𝒢 s ?callee), [])"
    using pstep.UnwindAct [where 𝒢 = 𝒢 and Π = Π and dst = dst and fr = s and frs = "[]"
                             and s = "?callee(ret_var := ⟦e⟧⇩e ?callee)"]
    by (simp add: glob)
  show ?thesis
    using c1 c2 c3 by (meson star.refl star.step)
qed

subsection ‹The same run in the graph, at a node this theory can name›

text ‹
  \<^const>‹csim› is structural and not functional, so the endpoint below can only
  say ∗‹some› node represents where the run stopped. Naming the check's own node
  takes a second witness, built in the graph rather than in the source: the
  collecting semantics at ‹Statement 4› really does contain the store this
  program computes, which is what makes the verdict claim there bite.

  Walking it takes one hop the collecting vocabulary does not close over
  step by step: \<^const>‹node_collect› is closed under intra edges but not under a
  call or a return alone, since a return has to pop the trace's ∗‹own› caller.
  A call and its matching return together do compose, and that is
  ‹node_collect_call_return_step› in
  \<^theory>‹Voblint_CFG.Activation_Trace_Collect›; the two hops through ‹bump› below are its two
  instances.
›

theorem certificate_demo_check_semantically_true:
  assumes ans: "run_voblint certificate_config
                  certificate_demo_prog = Analysed res"
  shows "∀s ∈ 𝒞⇘declared_global certificate_demo_prog,prog_cfg certificate_demo_prog,cinit_stores (declared_global certificate_demo_prog)⇙ (Statement 4).
           truthy (⟦Less (N 0) (V (STR ''b''))⟧⇩e s)"
proof (intro ballI)
  fix s :: store
  assume mem:
    "s ∈ 𝒞⇘declared_global certificate_demo_prog,prog_cfg certificate_demo_prog,cinit_stores (declared_global certificate_demo_prog)⇙ (Statement 4)"
  obtain chk where r: "report_checks res = [chk]" and rp: "check_point chk = Statement 4"
    and re: "check_exp chk = Less (N 0) (V (STR ''b''))"
    and rv: "check_verdict chk = Decided Check_Proved"
    using certificate_demo_check [OF ans] by blast
  from certificate_demo_verdicts_at_check [OF ans mem] r rp re rv
  show "truthy (⟦Less (N 0) (V (STR ''b''))⟧⇩e s)"
    unfolding verdict_stores_eq by simp
qed

text ‹

  is existential in the node and the frame stack because \<^const>‹csim› is
  structural; here the node is ‹Statement 4›, the store is the one the program
  computes, and the answer is the one \<^const>‹run_voblint› returns --- so each
  conjunct of the headline theorem can be read off separately rather than inferred
  from an existential.

theorem certificate_demo_full_certificate:
  obtains res where
    "run_voblint certificate_config
       certificate_demo_prog = Analysed res"
    "map (λchk. (check_point chk, check_exp chk, check_verdict chk)) (report_checks res)
       = [(Statement 4, Less (N 0) (V (STR ''b'')), Decided Check_Proved)]"
    "pcompletes (declared_global certificate_demo_prog) (prog_table certificate_demo_prog)
       (main_body (prog_table certificate_demo_prog)) (λ_. 0)
       ((λ_. 0)(STR ''a'' := 2, STR ''b'' := 42))"
    "(λ_. 0)(STR ''a'' := 2, STR ''b'' := 42)
       ∈ 𝒞⇘declared_global certificate_demo_prog,prog_cfg certificate_demo_prog,cinit_stores (declared_global certificate_demo_prog)⇙ (Statement 4)"
    "(λ_. 0)(STR ''a'' := 2, STR ''b'' := 42) ∈ ℛ⇘res⇙ (Statement 4)"
    "(λ_. 0)(STR ''a'' := 2, STR ''b'' := 42) ∈ 𝒱⇘res⇙ (Statement 4)"
    "truthy (⟦Less (N 0) (V (STR ''b''))⟧⇩e ((λ_. 0)(STR ''a'' := 2, STR ''b'' := 42)))"

theorem certificate_demo_check_listed_sound:
  "∃res. run_voblint certificate_config
           certificate_demo_prog = Analysed res
     ∧ (∃chk ∈ set (report_checks res). check_exp chk = Less (N 0) (V (STR ''b''))

              ∈ 𝒞⇘declared_global certificate_demo_prog,prog_cfg certificate_demo_prog,cinit_stores (declared_global certificate_demo_prog)⇙ (check_point chk)
          ∧ check_verdict chk = Decided Check_Proved
          ∧ truthy (⟦Less (N 0) (V (STR ''b''))⟧⇩e
                     ((λ_. 0)(STR ''a'' := 2, STR ''b'' := 42))))"

         [OF certificate_demo_init certificate_demo_to_check next_check.simps(1) Analysed]

                  ∈ 𝒞⇘declared_global certificate_demo_prog,prog_cfg certificate_demo_prog,cinit_stores (declared_global certificate_demo_prog)⇙ (check_point chk)"

    by blast

    unfolding checks [symmetric] using listed by auto

  with Analysed listed re mem pr show ?thesis by blast
