Theory Example_Run_Voblint_Precision

theory Example_Run_Voblint_Precision
  imports "Voblint_CLI.Run_Voblint"
begin

section ‹Precision of the public run, per analysis and option›

text ‹
  \<^const>‹run_voblint› on small programs whose verdicts change with the activated
  analyses, the context policy or the signed-overflow option, pinned by evaluation.
›

definition prec_conf :: "analysis list ⇒ context_policy ⇒ sem_conf ⇒ voblint_conf" where
  "prec_conf as cx sem = ⦇conf_analyses = as, conf_context = cx,
     conf_globals = Globals_Join, conf_program_globals = Program_Globals_Flow_Sensitive,
     conf_sem = sem⦈"

definition verdicts :: "voblint_conf ⇒ imp_prog ⇒ contextual_verdict list option" where
  "verdicts conf p =
     (case run_voblint conf [] p of
        Analysed res ⇒ Some (map check_verdict (report_checks res))
      | _ ⇒ None)"

subsection ‹One check, six analyses›

text ‹After the loop, ‹x == 99› needs both the bound and the class: Interval sees
  ‹[99, 101]›, Congruence alone loses the class to a possible wraparound, and only
  their reduced product decides the check.›

definition step_prog :: imp_prog where
  "step_prog = program {
     fun main() { x = 0; while (x < 99) { x = x + 3; } __voblint_check(x == 99); } }"

lemma step_verdicts:
  "verdicts (prec_conf [Interval] (Call_String 0) default_sem_conf) step_prog
     = Some [Decided Check_Unknown]"
  "verdicts (prec_conf [Congruence] (Call_String 0) default_sem_conf) step_prog
     = Some [Decided Check_Unknown]"
  "verdicts (prec_conf [Ana_Int Refine_Fixpoint] (Call_String 0) default_sem_conf) step_prog
     = Some [Decided Check_Proved]"
  "verdicts (prec_conf [Order] (Call_String 0) default_sem_conf) step_prog
     = Some [Decided Check_Unknown]"
  "verdicts (prec_conf [Sign] (Call_String 0) default_sem_conf) step_prog
     = Some [Decided Check_Unknown]"
  "verdicts (prec_conf [Parity] (Call_String 0) default_sem_conf) step_prog
     = Some [Decided Check_Unknown]"
  by eval+

subsection ‹Contexts›

text ‹‹inc› is called with ‹1› and with ‹10›. Without contexts its two activations
  share one state, so each call's result lies in ‹[2, 11]› and neither check is decided.
  A call string of length ‹1› separates the two call sites, and so does the entry state,
  whose context is the value of ‹x›: both checks are proved either way.›

definition calls_prog :: imp_prog where
  "calls_prog = program {
     fun inc(x) { return x + 1; }
     fun main() { a = inc(1); __voblint_check(a == 2); b = inc(10); __voblint_check(b == 11); } }"

lemma call_string_verdicts:
  "verdicts (prec_conf [Interval] (Call_String 0) default_sem_conf) calls_prog
     = Some [Decided Check_Unknown, Decided Check_Unknown]"
  "verdicts (prec_conf [Interval] (Call_String 1) default_sem_conf) calls_prog
     = Some [Decided Check_Proved, Decided Check_Proved]"
  "verdicts (prec_conf [Interval] Entry_State default_sem_conf) calls_prog
     = Some [Decided Check_Proved, Decided Check_Proved]"
  by eval+

text ‹A longer call string pays off once calls nest. ‹g› is called only from ‹f›, and ‹f›
  from two sites in ‹main›. Length ‹1› keys both activations of ‹g› by ‹f›'s one call site,
  so ‹g› returns the join of ‹3› and ‹-10›; length ‹2› also records the site in ‹main›, and
  each activation returns its own argument.›

definition nested_prog :: imp_prog where
  "nested_prog = program {
     fun g(p) { return p; }
     fun f(p) { t = g(p); return t; }
     fun main() { x = f(3); y = f(0 - 10); __voblint_check(x > 0); __voblint_check(y < 0); } }"

lemma call_string_depth_verdicts:
  "verdicts (prec_conf [Interval] (Call_String 1) default_sem_conf) nested_prog
     = Some [Decided Check_Unknown, Decided Check_Unknown]"
  "verdicts (prec_conf [Interval] (Call_String 2) default_sem_conf) nested_prog
     = Some [Decided Check_Proved, Decided Check_Proved]"
  by eval+

subsection ‹The order analysis asks Interval›

text ‹‹f› is called with both outcomes of its guard and analyzed in one context. Both
  branches leave ‹x› below ‹y›, but at different values, so Interval's join
  (‹x ∈ [0, 20]›, ‹y ∈ [10, 30]›) cannot decide ‹x <= y›. The order analysis records the
  pair when it learns it, and it learns it by asking Interval at each assignment how the
  new value compares with the other variable. Alone it has no one to ask.›

definition order_asks_prog :: imp_prog where
  "order_asks_prog = program {
     fun f(c) {
       if (c > 0) { x = 0; y = 10; } else { x = 20; y = 30; }
       __voblint_check(x <= y); }
     fun main() { f(1); f(0); } }"

lemma order_asks_interval_verdicts:
  "verdicts (prec_conf [Interval] (Call_String 0) default_sem_conf) order_asks_prog
     = Some [Decided Check_Unknown]"
  "verdicts (prec_conf [Order] (Call_String 0) default_sem_conf) order_asks_prog
     = Some [Decided Check_Unknown]"
  "verdicts (prec_conf [Interval, Order] (Call_String 0) default_sem_conf) order_asks_prog
     = Some [Decided Check_Proved]"
  by eval+

subsection ‹Signed overflow›

text ‹
  The concrete semantics wraps ▩‹INT_MAX + 1› to ▩‹INT_MIN›. Under Goblint's default
  ▩‹assume_top› the signed overflow gives Interval the kind's top, and the check is
  unknown; under ▩‹assume_wraparound› Interval wraps as the concrete semantics does and
  proves it.
›

definition overflow_prog :: imp_prog where
  "overflow_prog = program {
     fun main() { x = 2147483647; x = x + 1; __voblint_check(x < 0); } }"

definition wrap_sem :: sem_conf where
  "wrap_sem = ⦇sem_int_signed_overflow = Assume_Wraparound⦈"

lemma overflow_verdicts:
  "verdicts (prec_conf [Interval] (Call_String 0) default_sem_conf) overflow_prog
     = Some [Decided Check_Unknown]"
  "verdicts (prec_conf [Interval] (Call_String 0) wrap_sem) overflow_prog
     = Some [Decided Check_Proved]"
  "verdicts (prec_conf [Ana_Int Refine_Never] (Call_String 0) default_sem_conf) overflow_prog
     = Some [Decided Check_Unknown]"
  "verdicts (prec_conf [Ana_Int Refine_Never] (Call_String 0) wrap_sem) overflow_prog
     = Some [Decided Check_Proved]"
  by eval+

end
