Theory Cooperation_Regression

theory Cooperation_Regression
  imports "Voblint_CLI.MCP"
begin

section ‹Regression: two analyses that ask each other›

text ‹
  Interval and Order run in one solve and ask each other through ‹EvalInt›, as Goblint's
  ▩‹base› asks its partners. ‹f› is entered only by ‹f(100, 0)›, so inside the branch
  ‹x = 100› and ‹y = 0›; without call strings both parameters range over ‹[0, 100]›.

  ▪ ‹(y <= x) + 1 == 2›: Interval evaluates the comparison by asking it, and Order
    answers ‹1›. Neither analysis decides it alone.
  ▪ The check under ‹x < y›: Interval's guard asks the condition, Order answers ‹0›, and
    the branch is unreachable. Interval alone reaches it and refutes the check.
  ▪ ‹z == 1› after ‹x = 0›, which removes Order's fact: Interval stored ‹z = 1› when it
    evaluated the assignment through the channel, so its own state proves the check.
  ▪ ‹y == 1› fails at the one store reaching it. No answer of either analysis decides it,
    and the product leaves it unknown.
›

definition coop_prog :: imp_prog where
  "coop_prog = program {
     fun f(x, y) {
       if (y <= x) {
         __voblint_check((y <= x) + 1 == 2);
         z = (y <= x);
         if (x < y) { __voblint_check(0 == 1); }
         x = 0;
         __voblint_check(z == 1);
         __voblint_check(y == 1);
       }
     }
     fun main() { f(0, 100); f(100, 0); } }"

definition verdicts_of :: "('v × 'c × 'r) list ⇒ 'r list" where
  "verdicts_of rs = map (λ(_, _, r). r) rs"

lemma coop_verdicts:
  "verdicts_of (mcp_cs.verdict_report default_sem_conf [] [Interval, Order] 0 Globals_Join is_global coop_prog)
     = [Decided Check_Proved, Dead, Decided Check_Proved, Decided Check_Unknown]"
  "verdicts_of (nonrel.verdict_report TYPE(ivl) default_sem_conf [] 0 Globals_Join is_global coop_prog)
     = [Decided Check_Unknown, Decided Check_Refuted, Decided Check_Unknown, Decided Check_Unknown]"
  "verdicts_of (order_analysis.report [] Globals_Join is_global coop_prog ())
     = [Check_Unknown, Check_Unknown, Check_Unknown, Check_Unknown]"
  by eval+

end
