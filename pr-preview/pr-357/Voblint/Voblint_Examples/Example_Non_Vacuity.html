Theory Example_Non_Vacuity

theory Example_Non_Vacuity
  imports "Voblint_CLI.Run_Voblint"
begin

section ‹The headline theorems are not vacuous›

text ‹
  A theorem whose premises no configuration meets is true and says nothing. This
  theory shows, for the endpoints of \<^theory>‹Voblint_CLI.Run_Voblint›, that their
  premises hold together on one concrete program, and that the conclusion is then
  informative: the verdict it constrains is \<^const>‹Check_Proved› or \<^const>‹Dead›, not
  \<^const>‹Check_Unknown›. It then shows five conditions to be load-bearing: the
  definite verdicts a report states, the transfer obligation of the congruence
  remainder, the context correlation of RETURN and the TOTAL obligation in
  \<^const>‹activation_coverage›, and the pairing in \<^const>‹entry_pairs_cover›. Dropping
  or weakening each admits an answer that misses a concrete execution.

  Non-vacuity is a property of the premises. It says nothing about whether
  \<^const>‹pstep› models the intended language.
›

subsection ‹The two-call program›

text ‹
  ‹bump› is called with ‹5› and with ‹4›. Entry-state contexts keep the two
  activations apart, so both checks are proved; without contexts the entry of
  ‹bump› joins ‹[4, 5]› and neither is.
›

definition nv_prog :: imp_prog where
  "nv_prog = program {
     fun bump(n) { return n + 1; }
     fun main() {
       a = bump(5);
       b = bump(4);
       __voblint_check(a == 6);
       __voblint_check(b == 5);
     }
   }"

definition nv_conf :: voblint_conf where
  "nv_conf = ⦇conf_analyses = [Interval], conf_context = Entry_State,
     conf_globals = Globals_Warrow, conf_program_globals = Program_Globals_Flow_Sensitive,
     conf_sem = default_sem_conf⦈"

lemma nv_report:
  "(case run_voblint nv_conf [] nv_prog of
      Analysed res ⇒
        map (λchk. (check_point chk, check_verdict chk)) (report_checks res)
          = [(Statement 4, Decided Check_Proved), (Statement 5, Decided Check_Proved)]
    | _ ⇒ False)"
  by eval

lemma nv_analysed:
  obtains res where
    "run_voblint nv_conf [] nv_prog = Analysed res"
    "map (λchk. (check_point chk, check_verdict chk)) (report_checks res)
       = [(Statement 4, Decided Check_Proved), (Statement 5, Decided Check_Proved)]"
  using nv_report by (cases "run_voblint nv_conf [] nv_prog") auto

subsection ‹The premises, discharged›

text ‹
  Each premise of the source-level endpoints, proved for ‹nv_prog›: the zero store is
  initial, and the typed run reaches the first check (‹nv_to_check›).
›

abbreviation nv_var :: "vname ⇒ var_id" where
  "nv_var x ≡ ScopedId (STR ''main'') x"

lemma nv_init: "(λ_. 0) ∈ init_stores (kenv_of [])"
  by (simp add: init_stores_def all_in_kinds_def kenv_of_def)

lemma nv_bump:
  "elab_ptable nv_prog [] (STR ''bump'')
     = Some ⦇formals = [ScopedId (STR ''bump'') (STR ''n'')],
             body = Return () (Some (TArith I32 AAdd (TVar I32 (ScopedId (STR ''bump'') (STR ''n'')))
                                       (TN I32 1)))⦈"
  by eval

lemma nv_main:
  "main_body (elab_ptable nv_prog [])
     = Seq (Seq (Seq (gcom.Call () (Some (nv_var (STR ''a''))) (STR ''bump'') [TN I32 5])
                     (gcom.Call () (Some (nv_var (STR ''b''))) (STR ''bump'') [TN I32 4]))
                (Check () (0, 0) (TCmp CEq (TVar I32 (nv_var (STR ''a''))) (TN I32 6))))
           (Check () (0, 0) (TCmp CEq (TVar I32 (nv_var (STR ''b''))) (TN I32 5)))"
  by eval

text ‹One call of ‹bump›: enter, return ‹n + 1›, and resume the caller.›

lemma nv_call:
  assumes k: "k ∈ ik_range I32" and k1: "k + 1 ∈ ik_range I32"
  shows
    "(kenv_of []),(elab_ptable nv_prog []) ⊢ (gcom.Call () (Some (nv_var x)) (STR ''bump'') [TN I32 k], t, []) →⇩p⇧* (SKIP (), t(nv_var x := k + 1), [])"
proof -
  let ?κ = "kenv_of []" and ?Π = "elab_ptable nv_prog []"
  let ?n = "ScopedId (STR ''bump'') (STR ''n'')"
  let ?E = "TArith I32 AAdd (TVar I32 ?n) (TN I32 1)"
  let ?c = "bind_formals [?n] (map2 (λy. kind_conv ?κ (Some y)) [?n] [eval (TN I32 k) t])
              (enter_state is_global t)"
  let ?c' = "?c(ReturnVar := eval ?E ?c)"
  have s1:
    "?κ,?Π ⊢ (gcom.Call () (Some (nv_var x)) (STR ''bump'') [TN I32 k], t, []) →⇩p (Seq (Return () (Some ?E)) (Restore ()), ?c, [Frame t (Some (nv_var x))])"
    using pstep_Call[where Π = ?Π and p = "STR ''bump''", OF nv_bump,
        of "[TN I32 k]" ?κ "()" "Some (nv_var x)" t "[]"]
    by simp
  have s2:
    "?κ,?Π ⊢ (Seq (Return () (Some ?E)) (Restore ()), ?c, [Frame t (Some (nv_var x))]) →⇩p (Seq (Unwind ()) (Restore ()), ?c', [Frame t (Some (nv_var x))])"
    by (rule Seq2) (rule ReturnSome)
  have s3:
    "?κ,?Π ⊢ (Seq (Unwind ()) (Restore ()), ?c', [Frame t (Some (nv_var x))]) →⇩p (SKIP (), combine_assign (Some (nv_var x)) (kind_conv ?κ (Some (nv_var x)) (?c' ReturnVar)) (combine_env is_global t ?c'), [])"
    by (rule UnwindAct)
  have eq: "combine_assign (Some (nv_var x)) (kind_conv ?κ (Some (nv_var x)) (?c' ReturnVar))
              (combine_env is_global t ?c') = t(nv_var x := k + 1)"
    using k k1
    by (auto simp: fun_eq_iff combine_env_def enter_state_def kenv_of_def simp del: ik_range_iff)
  show ?thesis
    using s1 s2 s3 by (simp only: eq[symmetric]) (meson star.step star.refl)
qed

abbreviation nv_final :: store where
  "nv_final ≡ (λ_. 0)(nv_var (STR ''a'') := 6, nv_var (STR ''b'') := 5)"

abbreviation nv_checks :: "(unit, var_id, exp) gcom" where
  "nv_checks ≡ Seq (Check () (0, 0) (TCmp CEq (TVar I32 (nv_var (STR ''a''))) (TN I32 6)))
                  (Check () (0, 0) (TCmp CEq (TVar I32 (nv_var (STR ''b''))) (TN I32 5)))"

text ‹The typed run, stopped with the first check about to execute.›

lemma nv_to_check:
  "(kenv_of []),(elab_ptable nv_prog []) ⊢ (main_body (elab_ptable nv_prog []), λ_. 0, []) →⇩p⇧* (nv_checks, nv_final, [])"
proof -
  let ?κ = "kenv_of []" and ?Π = "elab_ptable nv_prog []"
  let ?C1 = "gcom.Call () (Some (nv_var (STR ''a''))) (STR ''bump'') [TN I32 5]"
  let ?C2 = "gcom.Call () (Some (nv_var (STR ''b''))) (STR ''bump'') [TN I32 4]"
  let ?K1 = "Check () (0, 0) (TCmp CEq (TVar I32 (nv_var (STR ''a''))) (TN I32 6))"
  let ?K2 = "Check () (0, 0) (TCmp CEq (TVar I32 (nv_var (STR ''b''))) (TN I32 5))"
  let ?t1 = "(λ_. 0)(nv_var (STR ''a'') := 6) :: store"
  have r: "(4::int) ∈ ik_range I32" "(5::int) ∈ ik_range I32" "(6::int) ∈ ik_range I32"
    by eval+
  have c1: "?κ,?Π ⊢ (?C1, λ_. 0, []) →⇩p⇧* (SKIP (), ?t1, [])"
    using nv_call[where k = 5 and x = "STR ''a''" and t = "λ_. 0", OF r(2)] r(3) by simp
  have c2: "?κ,?Π ⊢ (?C2, ?t1, []) →⇩p⇧* (SKIP (), nv_final, [])"
    using nv_call[where k = 4 and x = "STR ''b''" and t = ?t1, OF r(1)] r(2) by simp
  have calls: "?κ,?Π ⊢ (Seq ?C1 ?C2, λ_. 0, []) →⇩p⇧* (SKIP (), nv_final, [])"
    using psteps_Seq2[OF c1, of ?C2] c2 by (meson Seq1 star.step star_trans)
  have "?κ,?Π ⊢ (Seq (Seq ?C1 ?C2) ?K1, λ_. 0, []) →⇩p⇧* (?K1, nv_final, [])"
    using psteps_Seq2[OF calls, of ?K1] by (meson Seq1 star.step star.refl star_trans)
  then show ?thesis unfolding nv_main by (rule psteps_Seq2)
qed

subsection ‹The endpoints, instantiated›

text ‹
  With the premises discharged, ‹run_voblint_source_sound› and its siblings apply to
  ‹nv_prog› and to ‹nv_dead_prog›, and the verdicts they constrain are ‹Check_Proved›
  and ‹Dead›.
›

theorem nv_source_certified:
  "∃res v stk.
     run_voblint nv_conf [] nv_prog = Analysed res
   ∧ map (λchk. (check_point chk, check_verdict chk)) (report_checks res)
       = [(Statement 4, Decided Check_Proved), (Statement 5, Decided Check_Proved)]
   ∧ elab_ptable nv_prog [], elab_cfg nv_prog [], prog_main_name
       ⊢ (nv_checks, nv_final, []) ≈ (v, nv_final, stk)
   ∧ nv_final ∈ reaching [] nv_prog v
   ∧ (∃d. Lifted d ∈ report_states_at res v
          ∧ nv_final ∈ mcp.act_gamma [] (activation_of (conf_analyses nv_conf)) d)
   ∧ ¬ DEAD res v"
proof -
  obtain res where ans: "run_voblint nv_conf [] nv_prog = Analysed res"
    and checks: "map (λchk. (check_point chk, check_verdict chk)) (report_checks res)
                   = [(Statement 4, Decided Check_Proved), (Statement 5, Decided Check_Proved)]"
    by (rule nv_analysed)
  from run_voblint_source_sound [OF ans nv_init nv_to_check]
  obtain v stk where
    sim: "elab_ptable nv_prog [], elab_cfg nv_prog [], prog_main_name
            ⊢ (nv_checks, nv_final, []) ≈ (v, nv_final, stk)"
    and reach: "nv_final ∈ reaching [] nv_prog v"
    and st: "∃d. Lifted d ∈ report_states_at res v
               ∧ nv_final ∈ mcp.act_gamma [] (activation_of (conf_analyses nv_conf)) d"
    and live: "¬ DEAD res v"
    by blast
  show ?thesis
    by (rule exI[of _ res], rule exI[of _ v], rule exI[of _ stk])
       (simp only: ans checks sim reach st live simp_thms)
qed

text ‹
  Read at the check: the store the run holds there reaches a listed check for
  ‹a == 6›, that check says \<^const>‹Check_Proved›, and the condition holds.
›

theorem nv_check_proved_sound:
  "∃res. run_voblint nv_conf [] nv_prog = Analysed res
     ∧ (∃chk ∈ set (report_checks res).
          check_exp chk = TCmp CEq (TVar I32 (nv_var (STR ''a''))) (TN I32 6)
          ∧ nv_final ∈ reaching [] nv_prog (check_point chk)
          ∧ check_verdict chk = Decided Check_Proved
          ∧ truthy (eval (TCmp CEq (TVar I32 (nv_var (STR ''a''))) (TN I32 6)) nv_final))"
proof -
  obtain res where ans: "run_voblint nv_conf [] nv_prog = Analysed res"
    and checks: "map (λchk. (check_point chk, check_verdict chk)) (report_checks res)
                   = [(Statement 4, Decided Check_Proved), (Statement 5, Decided Check_Proved)]"
    by (rule nv_analysed)
  have nc:
    "next_check nv_checks = Some ((0, 0), TCmp CEq (TVar I32 (nv_var (STR ''a''))) (TN I32 6))"
    by simp
  from run_voblint_check_sound [OF ans nv_init nv_to_check nc]
  obtain chk where listed: "chk ∈ set (report_checks res)"
    and re: "check_exp chk = TCmp CEq (TVar I32 (nv_var (STR ''a''))) (TN I32 6)"
    and mem: "nv_final ∈ reaching [] nv_prog (check_point chk)"
    and pr: "check_verdict chk = Decided Check_Proved
               ⟶ truthy (eval (TCmp CEq (TVar I32 (nv_var (STR ''a''))) (TN I32 6)) nv_final)"
    by blast
  have "(check_point chk, check_verdict chk)
          ∈ set [(Statement 4, Decided Check_Proved), (Statement 5, Decided Check_Proved)]"
    unfolding checks [symmetric] using listed by auto
  then have "check_verdict chk = Decided Check_Proved" by auto
  with ans listed re mem pr show ?thesis by blast
qed

section ‹Load-bearing conditions›

text ‹
  Each subsection below drops or weakens one condition and exhibits an answer that
  meets everything else yet misses a concrete execution.
›

subsection ‹Soundness alone is cheap›

text ‹
  The guarantees constrain only definite verdicts, so a report that answers
  \<^const>‹Check_Unknown› at every check meets them at every store. The endpoint is
  informative because the verdicts \<^const>‹run_voblint› actually returns above are
  \<^const>‹Check_Proved› and \<^const>‹Dead›.
›

definition answer_all :: "contextual_verdict ⇒ report ⇒ report" where
  "answer_all vd res =
     res⦇report_checks := map (λchk. chk⦇check_verdict := vd⦈) (report_checks res)⦈"

lemma unknown_everywhere_sound:
  "∀chk ∈ set (report_checks (answer_all (Decided Check_Unknown) res)).
     (check_verdict chk = Decided Check_Proved ⟶ P chk)
     ∧ (check_verdict chk = Decided Check_Refuted ⟶ Q chk)"
  by (auto simp: answer_all_def)

text ‹
  An answer of \<^const>‹Check_Proved› everywhere is not sound: the store after
  ‹x = 1› reaches the check ‹x == 0›, which it falsifies.
›

lemma nv_dead_reaches_first_check:
  "(λ_. 0)(nv_var (STR ''x'') := 1) ∈ reaching [] nv_dead_prog (Statement 1)"
proof -
  have intra: "(FunctionEntry (STR ''main''), EA_Body (STR ''main''), Statement 0)
                 ∈ intra (elab_cfg nv_dead_prog [])"
    "(Statement 0, EA_Assign (nv_var (STR ''x'')) (TN I32 1), Statement 1)
                 ∈ intra (elab_cfg nv_dead_prog [])"
    by eval+
  have entry: "cfg_entry (elab_cfg nv_dead_prog []) = FunctionEntry (STR ''main'')"
    by eval
  have e0: "(λ_. 0) ∈ reaching [] nv_dead_prog (FunctionEntry (STR ''main''))"
    using node_collect_init [OF nv_init, of "kenv_of []" "elab_cfg nv_dead_prog []"]
    by (simp add: entry)
  have s0: "(λ_. 0) ∈ reaching [] nv_dead_prog (Statement 0)"
    by (rule node_collect_intra_step [OF e0 intra(1)]) simp
  show ?thesis
    by (rule node_collect_intra_step [OF s0 intra(2)]) (simp add: kenv_of_def)
qed

theorem proved_everywhere_unsound:
  assumes "run_voblint nv_dead_conf [] nv_dead_prog = Analysed res"
  shows "∃chk ∈ set (report_checks (answer_all (Decided Check_Proved) res)).
           check_verdict chk = Decided Check_Proved
           ∧ (λ_. 0)(nv_var (STR ''x'') := 1) ∈ reaching [] nv_dead_prog (check_point chk)
           ∧ ¬ truthy (eval (check_exp chk) ((λ_. 0)(nv_var (STR ''x'') := 1)))"
proof -
  have exps: "map (λchk. (check_point chk, check_exp chk)) (report_checks res)
      = [(Statement 1, TCmp CEq (TVar I32 (nv_var (STR ''x''))) (TN I32 0)),
         (Statement 3, TCmp CEq (TVar I32 (nv_var (STR ''x''))) (TN I32 5))]"
  proof -
    have "(case run_voblint nv_dead_conf [] nv_dead_prog of
             Analysed res ⇒ map (λchk. (check_point chk, check_exp chk)) (report_checks res)
                = [(Statement 1, TCmp CEq (TVar I32 (nv_var (STR ''x''))) (TN I32 0)),
                   (Statement 3, TCmp CEq (TVar I32 (nv_var (STR ''x''))) (TN I32 5))]
           | _ ⇒ False)"
      by eval
    then show ?thesis using assms by simp
  qed
  then obtain chk where mem: "chk ∈ set (report_checks res)" and pt: "check_point chk = Statement 1"
      and ex: "check_exp chk = TCmp CEq (TVar I32 (nv_var (STR ''x''))) (TN I32 0)"
    by (cases "report_checks res" rule: remdups_adj.cases) auto
  have mem': "chk⦇check_verdict := Decided Check_Proved⦈
                ∈ set (report_checks (answer_all (Decided Check_Proved) res))"
    using mem by (simp add: answer_all_def)
  show ?thesis
    using nv_dead_reaches_first_check pt ex
    by (intro bexI[OF _ mem']) simp
qed

subsection ‹The pre-fix Goblint congruence remainder›

text ‹
  Before Goblint pull request 1161, the remainder of the odd class ‹1 + 2ℤ› by the
  constant ‹2› was the constant ‹1›. Any abstract remainder that answers so violates
  the obligation ‹congruence_mod_sound› discharges for the shipped one: ‹-5› is
  odd, and its truncating remainder by ‹2› is ‹-1›.
›

theorem prefix_congruence_mod_unsound:
  assumes const: "f (mk_congruence 1 2) (mk_congruence 2 0) = mk_congruence 1 0"
  shows "¬ (∀a b i j. i ∈ gamma_congruence a ⟶ j ∈ gamma_congruence b
                     ⟶ c_mod i j ∈ gamma_congruence (f a b))"
proof
  assume sound: "∀a b i j. i ∈ gamma_congruence a ⟶ j ∈ gamma_congruence b
                     ⟶ c_mod i j ∈ gamma_congruence (f a b)"
  have "c_mod (-5) 2 ∈ gamma_congruence (mk_congruence 1 0)"
    using sound [rule_format, of "-5" "mk_congruence 1 2" 2 "mk_congruence 2 0"]
    by (simp add: const gamma_congruence_def)
  then show False by (simp add: gamma_congruence_def c_mod_def c_div_def)
qed

section ‹The paired-coverage, RETURN and TOTAL counterexamples›

text ‹
  Three weakenings of ‹activation_coverage› fail: RETURN read at the caller's
  context, TOTAL dropped, and entry coverage without pairing. They are shown
  on core VIMP, the language the analyzer runs. ‹ret_prog›
  calls ‹f› with ‹1› and then ‹5›; every variable has the default kind
  \<^const>‹I32›.
›

definition ret_prog :: imp_prog where
  "ret_prog = program {
     fun f(n) { return n; }
     fun main() { a = f(1); b = f(5); }
   }"

abbreviation ret_n :: var_id where
  "ret_n ≡ ScopedId (STR ''f'') (STR ''n'')"

abbreviation ret_κ :: kenv where
  "ret_κ ≡ kenv_of []"

abbreviation ret_g :: "(var_id, exp) gcfg" where
  "ret_g ≡ elab_cfg ret_prog []"

lemma ret_norm [simp]: "ik_norm I32 1 = 1" "ik_norm I32 5 = 5" "(1::int) ∈ ik_range I32"
  by eval+

lemma ret_intra:
  "intra ret_g =
     {(FunctionEntry (STR ''f''), EA_Body (STR ''f''), Statement 0),
      (Statement 0, EA_Ret (Some (TVar I32 ret_n)) (STR ''f''), FunctionResult (STR ''f'')),
      (Statement 1, EA_Ret None (STR ''f''), FunctionResult (STR ''f'')),
      (FunctionEntry (STR ''main''), EA_Body (STR ''main''), Statement 2),
      (Statement 4, EA_Ret None (STR ''main''), FunctionResult (STR ''main''))}"
  by eval

lemma ret_calls:
  "calls ret_g =
     {(Statement 2, CallEdge (Some (nv_var (STR ''a''))) [ret_n] [TN I32 1],
       FunctionEntry (STR ''f''), Statement 3),
      (Statement 3, CallEdge (Some (nv_var (STR ''b''))) [ret_n] [TN I32 5],
       FunctionEntry (STR ''f''), Statement 4)}"
  by eval

lemma ret_entry: "cfg_entry ret_g = FunctionEntry (STR ''main'')"
  by eval

lemma ret_enter:
  "call_enter ret_κ (CallEdge dst [ret_n] [e]) s
     = (enter_state is_global s)(ret_n := ik_norm I32 (eval e s))"
  by (simp add: call_enter_CallEdge kenv_of_def)

lemma ret_enter_zero [simp]: "enter_state is_global (λ_. 0) = (λ_. 0)"
  by (simp add: enter_state_def enter_frame_def fun_eq_iff)

lemma ret_combine:
  assumes "⋀v. is_global v ⟹ t v = 0"
  shows
    "combine_collect κ (Some x) (λ_. 0) t = (λ_. 0)(x := kind_conv κ (Some x) (t ReturnVar))"
  using assms by (auto simp: combine_collect_def fun_eq_iff)

text ‹The policy of the illustration: the callee context is the entry value of ‹n›.›

definition ret_R :: "int call_context_policy" where
  "ret_R = context_policy_of_fun (λ_ _ es. es ret_n)"

text ‹
  The claim: ‹main› is covered in the initial context ‹0› up to its first call site
  and nowhere after it; ‹f› is covered everywhere, but only in the contexts its
  calls create.
›

definition ret_cover :: "cfg_node ⇒ int ⇒ store set" where
  "ret_cover v ctx =
     (if v ∈ {FunctionEntry (STR ''f''), Statement 0, FunctionResult (STR ''f'')}
      then (if ctx = 0 then {} else UNIV)
      else if ctx = 0 ∧ v ∈ {FunctionEntry (STR ''main''), Statement 2} then UNIV else {})"

lemma ret_weak_obligations:
  shows "∀s. s ∈ init_stores ret_κ ⟶ s ∈ ret_cover (cfg_entry ret_g) 0"
    and "∀u a v c s s'. (u, a, v) ∈ intra ret_g ⟶ s ∈ ret_cover u c
              ⟶ s' ∈ edge_step ret_κ a s ⟶ s' ∈ ret_cover v c"
    and "∀u dst pars args p cont c c' s.
              (u, CallEdge dst pars args, FunctionEntry p, cont) ∈ calls ret_g
              ⟶ s ∈ ret_cover u c
              ⟶ c' ∈ ret_R u c (call_info_of (CallEdge dst pars args) p) s
                    (call_enter ret_κ (CallEdge dst pars args) s)
              ⟶ call_enter ret_κ (CallEdge dst pars args) s
                    ∈ ret_cover (FunctionEntry p) c'"
    and "∀cl dst pars args p cont c1 s t.
              (cl, CallEdge dst pars args, FunctionEntry p, cont) ∈ calls ret_g
              ⟶ s ∈ ret_cover cl c1 ⟶ t ∈ ret_cover (FunctionResult p) c1
              ⟶ combine_collect ret_κ dst s t ∈ ret_cover cont c1"
    and "call_context_total_on ret_κ ret_cover ret_R ret_g"
  by (auto simp: ret_entry ret_intra ret_calls ret_cover_def ret_R_def ret_enter

lemma ret_not_coverage:
  "¬ activation_coverage ret_κ ret_g
       (init_stores ret_κ) ret_cover ret_R 0"
proof
  assume "activation_coverage ret_κ ret_g
            (init_stores ret_κ) ret_cover ret_R 0"
  then interpret C: activation_coverage ret_κ
    ret_g "init_stores ret_κ" ret_cover ret_R 0 .
  have e: "(Statement 2, CallEdge (Some (nv_var (STR ''a''))) [ret_n] [TN I32 1],
             FunctionEntry (STR ''f''), Statement 3) ∈ calls ret_g"
    by (simp add: ret_calls)
  have adm: "admits_call_context ret_κ ret_g ret_R
               (Statement 2) 0 (STR ''f'') (λ_. 0)
               (call_enter ret_κ
                  (CallEdge (Some (nv_var (STR ''a''))) [ret_n] [TN I32 1]) (λ_. 0)) 1"
    unfolding admits_call_context_def
    by (rule exI [of _ "Some (nv_var (STR ''a''))"], rule exI [of _ "[ret_n]"],
        rule exI [of _ "[TN I32 1]"], rule exI [of _ "Statement 3"])
       (simp add: e ret_R_def ret_enter)
  have "combine_collect ret_κ (Some (nv_var (STR ''a''))) (λ_. 0) (λ_. 0)
          ∈ ret_cover (Statement 3) 0"
    by (rule C.RETURN [OF e _ adm]) (simp_all add: ret_cover_def)
  then show False by (simp add: ret_cover_def)
qed

subsection ‹TOTAL is load-bearing›

text ‹
  A resumed caller carries its own context whatever its callee carries, so the
  continuation of a call is collected in the caller's bucket even when the relation
  admits no callee context. Such a relation makes CALL and RETURN vacuous.
›

lemma resume_keeps_context:
  assumes s: "s ∈ 𝒜⇘κ,adm,c0,g,S⇙ u ctx"
    and ce: "(u, CallEdge dst pars args, FunctionEntry q, cont) ∈ calls g"
    and body: "star (cfg_intra_step κ g)
                 (FunctionEntry q, call_enter κ (CallEdge dst pars args) s)
                 (FunctionResult q, t)"
  shows "combine_collect κ dst s t ∈ 𝒜⇘κ,adm,c0,g,S⇙ cont ctx"
proof -
  from s obtain caller where cv: "caller ∈ 𝒯⇘κ,g,S⇙"
    and cn: "sink_node caller = u" and ck: "activation_context_rel κ adm c0 g caller ctx"
    and cs: "sink_store caller = s"
    by (rule activation_collect_E)
  let ?entered = "Call caller [(FunctionEntry q, call_enter κ (CallEdge dst pars args) s)]"
  have ev: "?entered ∈ 𝒯⇘κ,g,S⇙"
    using valid_activation_trace.call [OF cv, where dst = dst and pars = pars and args = args
                            and p = q and cont = cont]
      ce cn cs by simp
  obtain callee where dv: "callee ∈ 𝒯⇘κ,g,S⇙"
    and dn: "sink_node callee = FunctionResult q" and ds: "sink_store callee = t"
    and dc: "caller_of callee = caller_of ?entered"
    by (rule valid_activation_trace_intra_path_extend [OF _ ev]) (use body in simp_all)
  let ?r = "Resume caller callee
              (path_of caller @ [(cont, combine_collect κ dst
                  (sink_store caller) (sink_store callee))])"
  have rv: "?r ∈ 𝒯⇘κ,g,S⇙"
    by (rule valid_activation_trace.ret [OF dv _ dn]) (use dc ce cn in simp_all)
  have "sink_store ?r ∈ 𝒜⇘κ,adm,c0,g,S⇙ (sink_node ?r) ctx"
    by (rule activation_collect_I [OF rv refl]) (use ck in simp)
  then show ?thesis using cs ds by (simp add: sink_node_def sink_store_def)
qed

text ‹The relation that admits no context, and the claim that covers ‹main› up to its first
  call site in the initial context and nothing else.›

definition tot_R :: "int call_context_policy" where
  "tot_R u ctx ci s es = {}"

definition tot_cover :: "cfg_node ⇒ int ⇒ store set" where
  "tot_cover v ctx =
     (if ctx = 0 ∧ v ∈ {FunctionEntry (STR ''main''), Statement 2} then UNIV else {})"

lemma tot_weak_obligations:
  shows "∀s. s ∈ init_stores ret_κ ⟶ s ∈ tot_cover (cfg_entry ret_g) 0"
    and "∀u a v c s s'. (u, a, v) ∈ intra ret_g ⟶ s ∈ tot_cover u c
              ⟶ s' ∈ edge_step ret_κ a s ⟶ s' ∈ tot_cover v c"
    and "∀u dst pars args p cont c c' s.
              (u, CallEdge dst pars args, FunctionEntry p, cont) ∈ calls ret_g
              ⟶ s ∈ tot_cover u c
              ⟶ c' ∈ tot_R u c (call_info_of (CallEdge dst pars args) p) s
                    (call_enter ret_κ (CallEdge dst pars args) s)
              ⟶ call_enter ret_κ (CallEdge dst pars args) s
                    ∈ tot_cover (FunctionEntry p) c'"
    and "∀cl dst pars args p cont c1 c' p' s t es.
              (cl, CallEdge dst pars args, FunctionEntry p, cont) ∈ calls ret_g
              ⟶ s ∈ tot_cover cl c1
              ⟶ admits_call_context ret_κ ret_g tot_R cl c1 p' s es c'
              ⟶ t ∈ tot_cover (FunctionResult p) c'
              ⟶ combine_collect ret_κ dst s t ∈ tot_cover cont c1"

  have ce: "(Statement 2, CallEdge (Some (nv_var (STR ''a''))) [ret_n] [TN I32 1],

  show "(λ_. 0)(nv_var (STR ''a'') := 1)
          ∈ 𝒜⇘ret_κ,tot_R,0,ret_g,?S⇙ (Statement 3) 0"
    using resume_keeps_context [OF s2 ce bd]
    by (simp add: ret_combine [OF ret_globals_zero] kenv_of_def)
  show "(λ_. 0)(nv_var (STR ''a'') := 1) ∉ tot_cover (Statement 3) 0"
    by (simp add: tot_cover_def)
qed

subsection ‹Entry coverage must be paired›

text ‹
  ‹y = p(x)› with ‹fun p(a) { return a; }› and ‹x = 1›, all of kind \<^const>‹I32›.
  Abstract values are represented by their concretizations, which hold values of kind
  \<^const>‹I32› at ‹a›, so
  \<^const>‹entry_pairs_cover› is taken at \<^const>‹id›. The answer
  ‹[(x > 0, a < 0), (⊥, a > 0)]› covers the caller store with its first pair and the
  entered store with its second. Each pair's callee result, ‹a› published as the return
  value, is combined with that pair's own continuation; the union excludes the run's
  ‹y = 1›.
›

abbreviation up_x :: var_id where "up_x ≡ (nv_var (STR ''x''))"
abbreviation up_y :: var_id where "up_y ≡ (nv_var (STR ''y''))"
abbreviation up_a :: var_id where "up_a ≡ ScopedId (STR ''p'') (STR ''a'')"

definition unpaired_answer :: "(store set × store set) list" where
  "unpaired_answer =
     [({t. 0 < t up_x}, {t. t up_a ∈ ik_range I32 ∧ t up_a < 0}),
      ({}, {t. t up_a ∈ ik_range I32 ∧ 0 < t up_a})]"

definition pairwise_contribution :: "(store set × store set) list ⇒ store set" where
  "pairwise_contribution P =
     (⋃(q, e) ∈ set P.
        {combine_collect (ret_κ) (Some up_y) s0 (t(ReturnVar := t up_a))
         | s0 t. s0 ∈ q ∧ t ∈ e})"

theorem unpaired_entry_cover_unsound:
  defines "s ≡ (λ_. 0)(up_x := 1) :: store"
      and "s' ≡ (λ_. 0)(up_a := 1) :: store"
  shows "s' = call_enter (ret_κ) (CallEdge (Some up_y) [up_a] [TVar I32 up_x]) s"
    and "(∃(q, e) ∈ set unpaired_answer. s ∈ q)"
    and "(∃(q, e) ∈ set unpaired_answer. s' ∈ e)"
    and "¬ entry_pairs_cover id s s' unpaired_answer"
    and "∀u ∈ pairwise_contribution unpaired_answer. u up_y < 0"
    and "combine_collect (ret_κ) (Some up_y) s (s'(ReturnVar := s' up_a)) up_y = 1"
  by (auto simp: s_def s'_def unpaired_answer_def pairwise_contribution_def
                 entry_pairs_cover_def call_enter_CallEdge enter_state_def enter_frame_def
                 combine_collect_def kenv_of_def fun_eq_iff)
