section ‹One concrete call, two overlapping entry alternatives, two contexts›

text ‹
  A call's entry transfer answers a ∗‹list› of (continuation, callee entry)
  alternatives, and an entry-state route selects one context per alternative. When two
  alternatives both cover the same concrete call, that call is represented under two
  contexts at once. This theory builds that situation on the core VIMP non-relational
  analysis at the Interval domain, all variables at \<^const>‹I32›, by overriding the
  component's entry alone. Alternative 1 is the stock entry: the formal ‹a› holds
  ‹[1, 1]› and the continuation is the caller's state. Alternative 2 forgets the formals
  on entry ∗‹and› forgets the caller's own ‹x› in its continuation, both to
  \<^const>‹KTop›. The concrete call with ‹x = 1› lies in both entries and both
  continuations, and the entry-state route sends the two entries to the contexts
  ‹[KD I32 [1, 1]]› and ‹[KTop]›.

  Forgetting ‹x› itself, rather than an artificial marker, keeps the pairing of a
  continuation with its entry observable per alternative, while each continuation stays a
  sound description of the caller: \<^const>‹KTop› describes every value.

  The executable half pins the solved equations: the two alternatives stay
  distinguishable through the call and return programs, and each is read back and joined.
  The relational half interprets the routed context locale at this instance, so the
  generic routed soundness argument covers a list-valued entry without change, and shows
  the one concrete caller activation admitted under both contexts. The negative half
  replaces the entry by one answering no alternative: then no callee context is solved,
  the continuation stays \<^const>‹Bot›, and the entry law fails.
›

subsection ‹The program and its graph›

  "ov_program = program {
     fun p(a) { return a; }
     fun main() { x = 1; y = p(x); }

abbreviation p_result :: cfg_node where "p_result ≡ FunctionResult (STR ''p'')"

  "ov_ca ≡ CallEdge (Some y_id) [a_id] [TVar I32 x_id]"

  "ov_ci ≡ call_info_of ov_ca (STR ''p'')"

subsection ‹The overriding entry›

text ‹‹forget_vars vs› sets every variable of ‹vs› to \<^const>‹KTop›, so it only
  enlarges what a state describes.›

definition forget_vars :: "var_id list ⇒ 'a::kind_int_domain tst lifted ⇒ 'a tst lifted" where
  "forget_vars vs = map_lift (λst. fold (λv st. st⟨nonrel_loc v := KTop⟩) vs st)"

lemma nonrel_rd_forget:
  "nonrel_rd (fold (λv st. st⟨nonrel_loc v := KTop⟩) vs st)
     = (λv. if v ∈ set vs then KTop else nonrel_rd (st :: 'a::kind_int_domain tst) v)"
  by (induction vs arbitrary: st) (auto simp: fun_eq_iff)

lemma gamma_exec_forget_vars: "gamma_exec κ d ⊆ gamma_exec κ (forget_vars vs d)"
  by (cases d) (auto simp: forget_vars_def nonrel_rd_forget gamma_tabs_def)

text ‹The stock alternatives, followed by each with ‹x› forgotten in the continuation and
  the formals forgotten in the entry.›

definition ov_enter ::
  "sem_conf ⇒ kenv ⇒ channel ⇒ call_info
   ⇒ 'a::kind_int_domain tst lifted × 'a tst lifted ⇒ ('a tst lifted × 'a tst lifted) list" where
  "ov_enter sem κ ch ci p =
     (let alts = ls_enter (nonrel_exec_spec sem κ) ch ci p
      in alts @ map (λ(c, e). (forget_vars [x_id] c, forget_vars (ci_formals ci) e)) alts)"

definition ov_comp :: "ivl tst lifted local_spec" where
  "ov_comp = (nonrel_exec_spec default_sem_conf ov_kenv)⦇ls_enter := ov_enter default_sem_conf ov_kenv⦈"

definition ov_spec ::
  "(pp × ivl kd list, (unit, ivl kd list) global_unknown, unit, ivl tst lifted, ivl tst lifted) dg_spec"
    where
  "ov_spec = dg_spec_of ov_comp"

text ‹The alternatives the specification's entry answers at a caller state.›

definition ov_alts :: "call_info ⇒ ivl tst lifted
    ⇒ (ivl tst lifted × ivl tst lifted) list" where
  "ov_alts ci d = ls_enter ov_comp (ls_channel ov_comp d) ci (d, d)"

lemma enter_ov_spec: "enter⇧♯ ov_spec ci = local_enter_transfer (ov_alts ci)"
  by (simp add: ov_spec_def ov_alts_def[abs_def])

text ‹The entry-state route of \<^const>‹nonrel_es_ctx›: a context is the formals'
  values in the entered state.›

definition ov_route ::
  "pp ⇒ ivl kd list ⇒ ivl tst lifted ⇒ (var_id, exp) gcall_action ⇒ ivl kd list" where
  "ov_route u ctx d ca = nonrel_es_ctx is_global d ca"

subsection ‹The override is sound›

text ‹The stock alternatives are among the override's, so the entry law carries over;
  every other field is the stock component's.›

lemma ov_enter_sound:
  "sound_enter κ (gamma_exec κ)
     (ov_enter sem κ :: _ ⇒ _ ⇒ 'a::kind_int_domain tst lifted × _ ⇒ _)"
proof -
  note stock = sound_local_spec_enterD[OF nonrel_exec_spec_sound[of κ sem, where 'a = 'a]]
  have sub: "⋀ch ci p. set (ls_enter (nonrel_exec_spec sem κ) ch ci p)
      ⊆ set (ov_enter sem κ ch ci (p :: 'a tst lifted × 'a tst lifted))"
    by (auto simp: ov_enter_def Let_def)
  show ?thesis
    using stock sub unfolding sound_enter_def by blast
qed

lemma ov_comp_sound:
  "sound_local_spec ov_kenv (gamma_exec ov_kenv) ov_comp"
  unfolding ov_comp_def
  by (rule sound_local_spec_update_enter[OF nonrel_exec_spec_sound ov_enter_sound])

lemma analysis_contract_ov_spec:
  "analysis_contract ov_kenv ov_spec
     (λd _. gamma_exec ov_kenv d)"
  unfolding ov_spec_def by (rule dg_spec_of_contract[OF ov_comp_sound])

subsection ‹The routed equation system, solved›

text ‹The equations the non-relational pipeline builds with entry-state contexts, at the
  overriding specification: same generator, route, buffered seed protocol and solver.›

definition ov_eqs :: "(pp × ivl kd list, (unit, ivl kd list) global_unknown,
    (ivl tst lifted, ivl tst lifted) dg_state) eqsT" where
  "ov_eqs = compiled_routed_eqs_for (Analysis_Global ()) Analysis_Global Activation_Seed ov_route
     ov_spec ov_cfg (Lifted (nonrel_init_st [])) Bot"

definition ov_sol :: "(pp × ivl kd list) set
    × (pp × ivl kd list + (unit, ivl kd list) global_unknown
         ⇒ (ivl tst lifted, ivl tst lifted) dg_state)" where
  "ov_sol = TD_side_rule_Interp_solve Globals_Join ov_eqs (cfg_exit ov_cfg, [])"

abbreviation ov_contexts :: "pp ⇒ ivl kd list set" where
  "ov_contexts v ≡ snd ` Set.filter (λ(v', _). v' = v) (fst ov_sol)"

abbreviation ov_read :: "pp ⇒ ivl kd list ⇒ var_id ⇒ ivl kd lifted" where
  "ov_read v ctx x ≡ map_lift (λst. nonrel_rd st x) (dg_local (snd ov_sol (Inl (v, ctx))))"

abbreviation ov_seed :: "ivl kd list ⇒ var_id ⇒ ivl kd lifted" where
  "ov_seed ctx x ≡ map_lift (λst. nonrel_rd st x)
     (dg_local (snd ov_sol (Inr (Activation_Seed p_entry ctx))))"

abbreviation ov_caller :: "ivl tst lifted" where
  "ov_caller ≡ dg_local (snd ov_sol (Inl (Statement 3, [])))"

text ‹What alternative ‹i› contributes at the continuation, traversed on its own
  against the solution.›

definition ov_alt_answer :: "nat ⇒ ivl tst lifted" where
  "ov_alt_answer i =
     dg_local (traverse_program
       (routed_call_alternative_program ov_spec Analysis_Global Activation_Seed ov_route
          (λd. d = Bot) [] ov_ca (Statement 3) (STR ''p'') (ov_alts ov_ci ov_caller ! i))
       (snd ov_sol))"

abbreviation ov_alt_read :: "nat ⇒ var_id ⇒ ivl kd lifted" where
  "ov_alt_read i x ≡ map_lift (λst. nonrel_rd st x) (ov_alt_answer i)"

abbreviation ov_ctx_exact :: "ivl kd list" where "ov_ctx_exact ≡ [KD I32 (Ivl (Fin 1) (Fin 1))]"
abbreviation ov_ctx_top :: "ivl kd list" where "ov_ctx_top ≡ [KTop]"

text ‹The full range of \<^const>‹I32›: what the callee returns from an unknown argument.›

abbreviation i32_range :: "ivl kd" where
  "i32_range ≡ KD I32 (Ivl (Fin (- 2147483648)) (Fin 2147483647))"

text ‹The solver runs once. Every executable observation below is a projection of this
  snapshot.›

lemma ov_solution_snapshot:
  "TD_side_rule_Interp_solve_c Globals_Join ov_eqs (cfg_exit ov_cfg, []) ≠ None
   ∧ ov_contexts p_entry = {ov_ctx_exact, ov_ctx_top}
   ∧ ov_contexts (Statement 3) = {[]}
   ∧ (cfg_entry ov_cfg, []) ∈ fst ov_sol
   ∧ ov_seed ov_ctx_exact a_id = Lifted (KD I32 (Ivl (Fin 1) (Fin 1)))
   ∧ ov_seed ov_ctx_top a_id = Lifted KTop
   ∧ ov_read p_entry ov_ctx_exact a_id = Lifted (KD I32 (Ivl (Fin 1) (Fin 1)))
   ∧ ov_read p_entry ov_ctx_top a_id = Lifted KTop
   ∧ ov_read p_result ov_ctx_exact ReturnVar = Lifted (KD I32 (Ivl (Fin 1) (Fin 1)))
   ∧ ov_read p_result ov_ctx_top ReturnVar = Lifted i32_range
   ∧ ov_alt_read 0 x_id = Lifted (KD I32 (Ivl (Fin 1) (Fin 1)))
   ∧ ov_alt_read 0 y_id = Lifted (KD I32 (Ivl (Fin 1) (Fin 1)))
   ∧ ov_alt_read 1 x_id = Lifted KTop
   ∧ ov_alt_read 1 y_id = Lifted i32_range
   ∧ ov_read (Statement 4) [] x_id = Lifted KTop
   ∧ ov_read (Statement 4) [] y_id = Lifted i32_range
   ∧ ov_caller = Lifted ((nonrel_init_st [])⟨nonrel_loc x_id := KD I32 (Ivl (Fin 1) (Fin 1))⟩)
   ∧ map (λ(c, e). ov_route (Statement 3) [] e ov_ca) (ov_alts ov_ci ov_caller)
       = [ov_ctx_exact, ov_ctx_top]
   ∧ (∀(u, ctx) ∈ fst ov_sol. ∀(u', a, v) ∈ intra ov_cfg. u = u' ⟶ (v, ctx) ∈ fst ov_sol)
   ∧ (∀(cl, ctx) ∈ fst ov_sol. ∀(cl', ca, ce, cont) ∈ calls ov_cfg.
        cl = cl' ⟶ (cont, ctx) ∈ fst ov_sol)
   ∧ (∀(u, ctx) ∈ fst ov_sol. u = Statement 3 ⟶
        (∀(c, e) ∈ set (ov_alts ov_ci (dg_local (snd ov_sol (Inl (u, ctx))))).
          (p_entry, ov_route u ctx e ov_ca) ∈ fst ov_sol))"
  by eval

lemma ov_terminates_c:
  "TD_side_rule_Interp_solve_c Globals_Join ov_eqs (cfg_exit ov_cfg, []) ≠ None"
  by (insert ov_solution_snapshot, elim conjE, assumption)

lemma ov_entry_solved: "(cfg_entry ov_cfg, []) ∈ fst ov_sol"
  by (insert ov_solution_snapshot, elim conjE, assumption)

lemma ov_caller_eq:
  "ov_caller = Lifted ((nonrel_init_st [])⟨nonrel_loc x_id := KD I32 (Ivl (Fin 1) (Fin 1))⟩)"
  by (insert ov_solution_snapshot, elim conjE, assumption)

lemma ov_route_alts:
  "map (λ(c, e). ov_route (Statement 3) [] e ov_ca) (ov_alts ov_ci ov_caller)
     = [ov_ctx_exact, ov_ctx_top]"
  by (insert ov_solution_snapshot, elim conjE, assumption)

lemma ov_fwd_closed:
  "∀(u, ctx) ∈ fst ov_sol. ∀(u', a, v) ∈ intra ov_cfg. u = u' ⟶ (v, ctx) ∈ fst ov_sol"
  by (insert ov_solution_snapshot, elim conjE, assumption)

lemma ov_comb_fwd_closed:
  "∀(cl, ctx) ∈ fst ov_sol. ∀(cl', ca, ce, cont) ∈ calls ov_cfg.
     cl = cl' ⟶ (cont, ctx) ∈ fst ov_sol"
  by (insert ov_solution_snapshot, elim conjE, assumption)

lemma ov_enter_fwd_at_call:
  "∀(u, ctx) ∈ fst ov_sol. u = Statement 3 ⟶
     (∀(c, e) ∈ set (ov_alts ov_ci (dg_local (snd ov_sol (Inl (u, ctx))))).
       (p_entry, ov_route u ctx e ov_ca) ∈ fst ov_sol)"
  by (insert ov_solution_snapshot, elim conjE, assumption)

subsubsection ‹3. Each activation returns what it was entered with, and stays paired›

text ‹The exact context returns the exact argument; the forgetful one returns the whole

lemma ov_result_exact:
  "ov_read p_result ov_ctx_exact ReturnVar = Lifted (KD I32 (Ivl (Fin 1) (Fin 1)))"

lemma ov_result_forgotten: "ov_read p_result ov_ctx_top ReturnVar = Lifted i32_range"
  by (insert ov_solution_snapshot, elim conjE, assumption)

text ‹
  Continuation and entry stay correlated per alternative. The joined continuation cannot
  show it, since the domain is non-relational. Traversing each alternative's own program
  against the solution does: alternative 1's return writes ‹y› from the exact callee into
  its untouched continuation, alternative 2's writes the callee's unknown result into its
  continuation with ‹x› forgotten.
›

lemma ov_alt1_pairing:
  "ov_alt_read 0 x_id = Lifted (KD I32 (Ivl (Fin 1) (Fin 1)))"

  by (insert ov_solution_snapshot, elim conjE, assumption)+

  "ov_alt_read 1 x_id = Lifted KTop"
  "ov_alt_read 1 y_id = Lifted i32_range"
  by (insert ov_solution_snapshot, elim conjE, assumption)+

subsubsection ‹4. The continuation holds the join of both alternatives›

text ‹The caller's continuation joins what both alternatives return, so it keeps neither

lemma ov_continuation_join:
  "ov_read (Statement 4) [] x_id = Lifted KTop"
  "ov_read (Statement 4) [] y_id = Lifted i32_range"
  by (insert ov_solution_snapshot, elim conjE, assumption)+

subsection ‹Soundness through the routed context layer›

text ‹
  \<^const>‹routed_entry_context_rel› at the override's alternatives is the context policy
  this instance keys its collecting semantics by. It admits a context for a concrete call
  whenever some alternative covers the caller store in its continuation and the entered
  store in its entry, and routes to that context.
›

abbreviation ov_R :: "(ivl kd list) call_context_policy" where
  "ov_R ≡ routed_entry_context_rel ov_alts (λd _. gamma_exec ov_kenv d) (snd ov_sol)
            Analysis_Global ov_route"

abbreviation ov_call_enter :: "store ⇒ store" where
  "ov_call_enter ≡ call_enter ov_kenv ov_ca"

text ‹The concrete caller store at the call site: ‹x = 1›, everything else ‹0›.›

definition ov_caller_store :: store where
  "ov_caller_store = (λ_. 0)(x_id := 1)"

lemma ov_caller_store_covered: "ov_caller_store ∈ gamma_exec ov_kenv ov_caller"
proof -
  have z: "(λ_. 0) ∈ gamma_tabs ov_kenv (nonrel_rd (nonrel_init_st [] :: ivl tst))"
    by (rule subsetD[OF nonrel_init_sound])
       (simp add: init_stores_def all_in_kinds_def)
  have "ov_caller_store ∈ gamma_tabs ov_kenv
      ((nonrel_rd (nonrel_init_st [] :: ivl tst))(x_id := KD I32 (Ivl (Fin 1) (Fin 1))))"
  proof (rule gamma_tabsI, goal_cases Kinds Values)
    case Kinds show ?case
      by (auto simp: in_kinds_def ov_caller_store_def kenv_of_def simp del: ik_range_iff)
  next
    case (Values v) show ?case
      using gamma_tabsD(2)[OF z, of v]
      by (auto simp: ov_caller_store_def kenv_of_def simp del: ik_range_iff)
  qed
  then show ?thesis by (simp add: ov_caller_eq)
qed

text ‹The stock alternative is first, so the stock entry law covers the entered store.›

definition ov_entry1 :: "ivl tst lifted" where
  "ov_entry1 = transfer_lift st_is_bot
     (enter_st default_sem_conf ov_kenv (ls_channel ov_comp ov_caller) [a_id] [TVar I32 x_id])
     ov_caller"

lemma ov_alts_caller:
  "ov_alts ov_ci ov_caller
     = [(ov_caller, ov_entry1), (forget_vars [x_id] ov_caller, forget_vars [a_id] ov_entry1)]"
  by (simp add: ov_alts_def ov_comp_def ov_enter_def nonrel_exec_spec_def ov_entry1_def)

lemma ov_routes:
  "ov_route (Statement 3) [] ov_entry1 ov_ca = ov_ctx_exact"
  "ov_route (Statement 3) [] (forget_vars [a_id] ov_entry1) ov_ca = ov_ctx_top"
  using ov_route_alts by (simp_all add: ov_alts_caller)

lemma ov_entry1_covered:
  "ov_call_enter ov_caller_store ∈ gamma_exec ov_kenv ov_entry1"
proof -
  have ch: "eval_query.channel_holds (ls_channel ov_comp ov_caller) ov_caller_store"
    by (rule ls_channel_sound[OF ov_comp_sound ov_caller_store_covered])
  note law = sound_local_spec_enterD[OF nonrel_exec_spec_sound[of ov_kenv default_sem_conf,
      where 'a = ivl], unfolded sound_enter_def, rule_format]
  from law[where s = ov_caller_store and p = "(ov_caller, ov_caller)"
      and ch = "ls_channel ov_comp ov_caller" and ci = ov_ci] ov_caller_store_covered ch
  show ?thesis by (simp add: nonrel_exec_spec_def ov_entry1_def)
qed

lemma ov_ci_edge: "CallEdge (ci_dst ov_ci) (ci_formals ov_ci) (ci_args ov_ci) = ov_ca"
  by simp

lemma ov_exact_context_admitted:
  "ov_ctx_exact ∈ ov_R (Statement 3) [] ov_ci ov_caller_store (ov_call_enter ov_caller_store)"
  unfolding routed_entry_context_rel_def ov_alts_caller
  using ov_caller_store_covered ov_entry1_covered ov_routes
  by (simp add: ov_ci_edge, intro exI[of _ ov_caller] exI[of _ ov_entry1], simp)

lemma ov_top_context_admitted:
  "ov_ctx_top ∈ ov_R (Statement 3) [] ov_ci ov_caller_store (ov_call_enter ov_caller_store)"
  unfolding routed_entry_context_rel_def ov_alts_caller
  using subsetD[OF gamma_exec_forget_vars ov_caller_store_covered, of "[x_id]"]
    subsetD[OF gamma_exec_forget_vars ov_entry1_covered, of "[a_id]"] ov_routes
  by (simp add: ov_ci_edge,
      intro exI[of _ "forget_vars [x_id] ov_caller"] exI[of _ "forget_vars [a_id] ov_entry1"], simp)

text ‹The one concrete call admits its callee activation under both contexts. The caller
  itself stays at the root context.›

lemma ov_two_contexts_admitted:
  "ov_ctx_exact ∈ ov_R (Statement 3) [] ov_ci ov_caller_store (ov_call_enter ov_caller_store)
   ∧ ov_ctx_top ∈ ov_R (Statement 3) [] ov_ci ov_caller_store (ov_call_enter ov_caller_store)
   ∧ ov_ctx_exact ≠ ov_ctx_top"
  using ov_exact_context_admitted ov_top_context_admitted by simp

subsubsection ‹The routed context locale at this instance›

text ‹The buffered post-solution, reconciled with the unbuffered generator the routed
  locale is stated over. The reconciliation inspects only that the entry is a pure
  function of the caller state, not how many alternatives it answers.›

lemma ov_pp_routed:
  "part_post_solution
     (routed_node_rhs intra_predecessor_addr_list call_site_list (λ_. Analysis_Global ()) ov_route
        (λctx' src a. dg_spec_edge_program ov_spec a src Analysis_Global)
        (routed_call_program ov_spec Analysis_Global Activation_Seed (static_resolve ov_cfg)
           (λd. d = Bot))
        (routed_entry_seed_programs Activation_Seed)
        ov_cfg Bot (Lifted (nonrel_init_st [])) Bot)
     (cfg_exit ov_cfg, []) (snd ov_sol) (fst ov_sol)"
proof -
  have pp: "part_post_solution ov_eqs (cfg_exit ov_cfg, []) (snd ov_sol) (fst ov_sol)"
    unfolding ov_sol_def
    by (rule TD_side_rule_Interp.part_post_solution_of_solve_c[OF ov_terminates_c])
  have wf: "dg_spec_wf ov_spec"
    unfolding ov_spec_def by (rule dg_spec_wf_dg_spec_of)
  show ?thesis
    by (rule pp_routed_of_buffered[OF wf _
          pp[unfolded ov_eqs_def compiled_routed_eqs_for_def bot_lifted_eq]]) simp
qed

lemma ov_dg_base:
  "dg_context_base ov_spec (λd _. gamma_exec ov_kenv d) ov_cfg (Analysis_Global ()) Analysis_Global
     ov_route (routed_call_program ov_spec Analysis_Global Activation_Seed (static_resolve ov_cfg)
       (λd. d = Bot))
     (routed_entry_seed_programs Activation_Seed) Bot (Lifted (nonrel_init_st [])) Bot
     (snd ov_sol) (fst ov_sol) (cfg_exit ov_cfg, []) (solved_local_reader (fst ov_sol) (snd ov_sol))
     (gamma_exec ov_kenv)"
proof (unfold_locales, goal_cases CmbWf ExtraWf FinE PP SgCov SgUncov Fwd)
  case CmbWf show ?case
    unfolding ov_spec_def by (rule sp_wf_routed_call_program[OF dg_spec_wf_dg_spec_of])
next

text ‹Adequacy, totality and return forwarding: every alternative the relation admits
  is one the entry ran, its context is solved, and the stock alternative covers every

interpretation ov_routed: routed_context ov_kenv ov_spec
  "λd _. gamma_exec ov_kenv d" ov_cfg "Analysis_Global ()" Analysis_Global ov_route
  Bot "Lifted (nonrel_init_st [])" Bot "snd ov_sol" "fst ov_sol" "(cfg_exit ov_cfg, [])"

  "λd. d = Bot" "gamma_exec ov_kenv" ov_R

  show ?case by (rule dg_context_activation.intro[OF analysis_contract_ov_spec ov_dg_base])

  show ?case
  proof (rule routed_context_base.intro[OF ov_dg_base], unfold_locales,
      goal_cases FinC CallsUnique SeedNeBuf SeedNeGlobal IsBotBot IsBotSound ResolveSound)
    case FinC show ?case by (rule ov_cfg_finite(2))

      unfolding ov_cfg_def by (rule elab_cfg_calls_source_unique)
  next

  next

  next

  next

        enter_deps_local_enter_transfer_mk_dg_man[of "ov_alts ?ci" ?d Analysis_Global "snd ov_sol"]
      by (fastforce simp: enter_ov_spec)
  next
    case (Total u ctx dst pars args p cont s)
    let ?ci = "call_info_of (CallEdge dst pars args) p"
    let ?d = "dg_local (snd ov_sol (Inl (u, ctx)))"

        Total(3) ch

  next
    case (CombFwd cl c1 dst pars args p cont)
    then show ?case using ov_comb_fwd_closed by auto
  qed
