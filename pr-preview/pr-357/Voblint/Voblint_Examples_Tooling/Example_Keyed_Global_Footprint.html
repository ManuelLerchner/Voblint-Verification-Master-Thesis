Theory Example_Keyed_Global_Footprint

theory Example_Keyed_Global_Footprint
  imports
    "Voblint_Framework.DG_Keyed_Split_Spec"
begin

section ‹Keyed globals: an edge depends on what it reads and publishes what it writes›

text ‹
  Two globals ‹g› and ‹h› of core VIMP, the \<^const>‹GlobalId› identities, and a
  procedure-scoped local ‹x›. Under the keyed lifter an edge's program depends on its
  source point and on the unknowns of the globals its expressions mention, and it
  publishes exactly at the unknowns of the globals it may assign. The footprints are the
  generic \<^const>‹edge_global_reads› and \<^const>‹edge_global_writes› at the parameters
  the analyzer uses: \<^const>‹exp_vars›, the return slot \<^const>‹ReturnVar› and
  \<^const>‹is_global›. Updating ‹h› therefore cannot destabilize a point that reads only
  ‹g›: no equation reading ‹g› names ‹h›'s unknown.
›

abbreviation g_id :: var_id where "g_id ≡ GlobalId (STR ''g'')"
abbreviation h_id :: var_id where "h_id ≡ GlobalId (STR ''h'')"

text ‹‹x := g + 1› and ‹h := 1›, at the 32-bit signed kind.›

abbreviation read_g :: "(var_id, exp) gedge_action" where
  "read_g ≡ EA_Assign (ScopedId (STR ''main'') (STR ''x''))
     (TArith I32 AAdd (TVar I32 g_id) (TN I32 1))"

abbreviation write_h :: "(var_id, exp) gedge_action" where
  "write_h ≡ EA_Assign h_id (TN I32 1)"

lemma read_g_footprint:
  "edge_global_reads is_global read_g = [g_id]"
  "edge_global_writes is_global read_g = []"
  by (simp_all add: edge_global_reads_def edge_global_writes_def edge_reads_def
      edge_writes.simps global_names_in_def)

lemma write_h_footprint:
  "edge_global_reads is_global write_h = []"
  "edge_global_writes is_global write_h = [h_id]"
  by (simp_all add: edge_global_reads_def edge_global_writes_def edge_reads_def
      edge_writes.simps global_names_in_def)

text ‹The keyed program for ‹x := g + 1› depends on ‹g›'s unknown and nothing else global.›

lemma read_g_depends_on_g_only:
  "dep_program τ (transfer_program (keyed_transfer cmb rl rg free
      (edge_global_reads is_global read_g)
      (edge_global_writes is_global read_g) f) src key)
     = {src, Inr (key g_id)}"
  by (simp add: dep_transfer_program keyed_transfer_def read_g_footprint
      sp_compile_with_def sp_bind_def sp_return_def)

text ‹The keyed program for ‹h := 1› publishes at ‹h›'s unknown and nowhere else.›

lemma write_h_publishes_h_only:
  "side_path τ (sp_compile (transfer_program (keyed_transfer cmb rl rg free
      (edge_global_reads is_global write_h)
      (edge_global_writes is_global write_h) f) src key))
     = [key h_id]"
  by (cases src)
     (simp_all add: write_h_footprint transfer_program_def transfer_program_at_def
        keyed_transfer_def sp_compile_def sp_compile_with_def sp_map_def sp_bind_def
        sp_return_def dg_read_at_def sp_read_local_def sp_read_global_def dg_sideg_def
        sp_publish_def)

end
