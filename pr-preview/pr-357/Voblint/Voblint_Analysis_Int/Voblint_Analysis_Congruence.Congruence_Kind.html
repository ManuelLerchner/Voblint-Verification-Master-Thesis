section ‹Congruences at a kind›

text ‹
  Normalizing to kind ‹k› changes a value by a multiple of ‹2 ^ w›, the kind's modulus.
  A class ‹c + mℤ› therefore survives the normalization modulo ‹gcd m (2 ^ w)›, and when
  that divisor is ‹2 ^ w› itself the class holds one value of the kind. This is
  Goblint's ▩‹handle_overflow› in ▩‹congruenceDomain.ml›, which keeps the largest
  power of two dividing ‹m›; it applies to every kind that wraps under the overflow
  policy.
›

lemma ik_mod_dvd_ik_norm: "ik_mod k dvd ik_norm k n - n"
proof -
  have "ik_norm k n - n = - ((n - ik_min k) - (n - ik_min k) mod ik_mod k)"
    by (simp add: ik_norm_mod)
  then show ?thesis by (simp only: dvd_minus_iff dvd_minus_mod)
qed

lemma ik_norm_eq_if_dvd:
  assumes "ik_mod k dvd n - c"
  shows "ik_norm k n = ik_norm k c"
proof -
  have "(n - ik_min k) mod ik_mod k = (c - ik_min k) mod ik_mod k"
    using assms by (simp add: mod_eq_dvd_iff)
  then show ?thesis by (simp add: ik_norm_mod)
qed

definition norm_cong_ik :: "ikind ⇒ congruence ⇒ congruence" where
  "norm_cong_ik k a =
     (case Rep_congruence a of
        None ⇒ bot
      | Some (c, m) ⇒
          (let g = gcd m (ik_mod k)
           in if g = ik_mod k then congruence_of_int (ik_norm k c) else mk_congruence c g))"

lemma gamma_congruence_Some:
  "Rep_congruence a = Some (c, m) ⟹ gamma_congruence a = {n. m dvd n - c}"
  by (simp add: gamma_congruence_def)

lemma norm_cong_ik_sound:
  assumes n: "n ∈ gamma_congruence a"
  shows "ik_norm k n ∈ gamma_congruence (norm_cong_ik k a)"
proof (cases a rule: congruence_rep_cases)
  case Bot
  then show ?thesis using n by (simp add: gamma_congruence_def)
next
  case (Class c m)
  let ?g = "gcd m (ik_mod k)"
  have "m dvd n - c" using n Class by (simp add: gamma_congruence_Some)
  then have gn: "?g dvd n - c" by (rule dvd_trans[rotated]) simp
  have "?g dvd ik_norm k n - n" by (rule dvd_trans[OF _ ik_mod_dvd_ik_norm]) simp
  with gn have g: "?g dvd ik_norm k n - c"
    using dvd_add[of ?g "ik_norm k n - n" "n - c"] by simp
  show ?thesis
  proof (cases "?g = ik_mod k")
    case True
    with gn have "ik_norm k n = ik_norm k c" by (intro ik_norm_eq_if_dvd) simp
    with True Class show ?thesis by (simp add: norm_cong_ik_def)
  next
    case False
    with g Class show ?thesis by (simp add: norm_cong_ik_def Let_def)
  qed
qed

text ‹The least value of a class at or above a bound decides whether the class meets a
  kind's range.›

lemma cong_least_above:
  fixes m c lo :: int
  assumes m: "0 < m"
  defines "r ≡ lo + (c - lo) mod m"
  shows "lo ≤ r" and "m dvd r - c"
    and "m dvd n - c ⟹ lo ≤ n ⟹ r ≤ n"
proof -
  show "lo ≤ r" using m by (simp add: r_def)
  have "r - c = - ((c - lo) - (c - lo) mod m)" by (simp add: r_def)
  then show "m dvd r - c" by (simp only: dvd_minus_iff dvd_minus_mod)
  assume d: "m dvd n - c" and lo: "lo ≤ n"
  have "(n - lo) mod m = (c - lo) mod m"
    using d by (simp add: mod_eq_dvd_iff)
  moreover have "(n - lo) mod m ≤ n - lo"
    using lo m by (simp add: zmod_le_nonneg_dividend)
  ultimately show "r ≤ n" by (simp add: r_def)
qed

instantiation congruence :: ik_domain
begin

definition widen_ik_congruence :: "ikind ⇒ congruence ⇒ congruence ⇒ congruence" where
  "widen_ik_congruence k a b = a ⊔ b"

definition narrow_ik_congruence :: "ikind ⇒ congruence ⇒ congruence ⇒ congruence" where
  "narrow_ik_congruence k a b = a ⊓ b"

definition empty_ik_congruence :: "ikind ⇒ congruence ⇒ bool" where
  "empty_ik_congruence k a =
     (case Rep_congruence a of
        None ⇒ True
      | Some (c, m) ⇒
          if m = 0 then ¬ (ik_min k ≤ c ∧ c ≤ ik_max k)
          else ik_max k < ik_min k + (c - ik_min k) mod m)"

instance proof
  fix k :: ikind and a b :: congruence
  show "a ≤ widen_ik k a b" "b ≤ widen_ik k a b"
    by (simp_all add: widen_ik_congruence_def)
  show "b ≤ a ⟹ b ≤ narrow_ik k a b" "b ≤ a ⟹ narrow_ik k a b ≤ a"
    by (simp_all add: narrow_ik_congruence_def)
  show "empty_ik k a ⟷ numeric_domain_class.gamma a ∩ ik_range k = {}"
  proof (cases a rule: congruence_rep_cases)
    case Bot
    then show ?thesis by (simp add: empty_ik_congruence_def gamma_congruence_def)
  next
    case (Class c m)
    show ?thesis
    proof (cases "m = 0")
      case True
      with Class show ?thesis
        by (auto simp: empty_ik_congruence_def gamma_congruence_Some)
    next
      case False
      then have m: "0 < m" using normalized_Rep_congruence[of a] Class by auto
      define r where "r = ik_min k + (c - ik_min k) mod m"
      have r: "ik_min k ≤ r" "m dvd r - c" "⋀n. m dvd n - c ⟹ ik_min k ≤ n ⟹ r ≤ n"
        using cong_least_above[OF m, where lo = "ik_min k" and c = c] by (simp_all add: r_def)
      have "empty_ik k a ⟷ ik_max k < r"
        using Class False by (simp add: empty_ik_congruence_def r_def)
      moreover have "ik_max k < r ⟷ gamma_congruence a ∩ ik_range k = {}"
        using r by (fastforce simp: Class gamma_congruence_Some)
      ultimately show ?thesis by simp
    qed
  qed
qed

end

section ‹Congruences as a kind-aware integer domain›

text ‹
  An operation at kind ‹k› computes the class of the mathematical result and then
  treats a possible overflow as Goblint's ▩‹congruenceDomain.ml› does under the
  overflow policy; a conversion follows its ▩‹cast_to›.
›

fun cong_aop :: "aop ⇒ congruence ⇒ congruence ⇒ congruence" where
  "cong_aop AAdd a b = a + b"
| "cong_aop ASub a b = a - b"
| "cong_aop AMul a b = a * b"
| "cong_aop ADiv a b = congruence_div a b"
| "cong_aop AMod a b = congruence_mod a b"

lemma cong_aop_sound:
  "m ∈ gamma_congruence a ⟹ n ∈ gamma_congruence b
     ⟹ aop_eval op m n ∈ gamma_congruence (cong_aop op a b)"
  by (cases op) (simp_all add: congruence_plus_sound congruence_minus_sound
      congruence_times_sound congruence_div_sound congruence_mod_sound)

text ‹Two classes with no common value are unequal, as in Goblint's ▩‹eq›.›

definition cong_eq :: "congruence ⇒ congruence ⇒ bool option" where
  "cong_eq a b =
     (case congruence_eqb a b of
        Some r ⇒ Some r
      | None ⇒ if is_bottom_congruence (a ⊓ b) then Some False else None)"

lemma cong_eq_sound:
  "cong_eq a b = Some r ⟹ m ∈ gamma_congruence a ⟹ n ∈ gamma_congruence b ⟹ (m = n) = r"
  by (auto simp: cong_eq_def is_bottom_congruence_correct split: option.splits if_splits
      dest: congruence_eqb_sound)


fun cong_refine :: "cop ⇒ congruence ⇒ congruence ⇒ congruence × congruence" where
  "cong_refine CEq a b = (a ⊓ b, a ⊓ b)"
| "cong_refine _ a b = (a, b)"

lemma cong_refine_sound:
  "cop_eval c m n ⟹ m ∈ gamma_congruence a ⟹ n ∈ gamma_congruence b
     ⟹ m ∈ gamma_congruence (fst (cong_refine c a b))
       ∧ n ∈ gamma_congruence (snd (cong_refine c a b))"
  by (cases c) simp_all

definition cong_fits :: "ikind ⇒ congruence ⇒ bool" where
  "cong_fits k a =
     (case Rep_congruence a of
        None ⇒ True
      | Some (c, m) ⇒ m = 0 ∧ ik_min k ≤ c ∧ c ≤ ik_max k)"

lemma cong_fits_sound: "cong_fits k a ⟹ n ∈ gamma_congruence a ⟹ n ∈ ik_range k"
  by (cases a rule: congruence_rep_cases)
     (auto simp: cong_fits_def gamma_congruence_Some gamma_congruence_def)

definition cong_excludes :: "int ⇒ congruence ⇒ bool" where
  "cong_excludes x a = (case Rep_congruence a of None ⇒ True | Some (c, m) ⇒ ¬ m dvd x - c)"

lemma cong_excludes_sound: "cong_excludes x a ⟹ n ∈ gamma_congruence a ⟹ n ≠ x"
  by (auto simp: cong_excludes_def gamma_congruence_def split: option.splits)

text ‹
  A conversion keeps the class when the source kind's range lies inside the target's,
  wraps a constant, and answers top otherwise, as Goblint's ▩‹cast_to› does under every
  overflow policy.
›

definition cast_cong_ik :: "ikind ⇒ congruence ⇒ congruence" where
  "cast_cong_ik k a =
     (case Rep_congruence a of
        None ⇒ bot
      | Some (c, m) ⇒ if m = 0 then congruence_of_int (ik_norm k c) else top)"

lemma cast_cong_ik_sound:
  "n ∈ gamma_congruence a ⟹ ik_norm k n ∈ gamma_congruence (cast_cong_ik k a)"
  by (cases a rule: congruence_rep_cases)
     (simp add: gamma_congruence_def, auto simp: cast_cong_ik_def gamma_congruence_Some)

text ‹
  An arithmetic result follows Goblint's ▩‹add›, ▩‹sub›, ▩‹mul›, ▩‹div› and ▩‹rem›.
  A remainder of in-range operands, and a quotient by a divisor other than ‹-1›, stays
  in range and keeps its class. Otherwise a kind that wraps under the overflow policy
  normalizes the class to the kind (▩‹handle_overflow›). A signed kind under
  ▩‹assume_top› keeps only a difference of two constants that lies in range and answers
  top for everything else. The Int product passes Goblint's ▩‹no_ov› case itself, when
  its interval shows that the result fits.
›

definition cong_arith_ik ::
    "signed_overflow_policy ⇒ ikind ⇒ aop ⇒ bool ⇒ congruence ⇒ congruence ⇒ congruence" where
  "cong_arith_ik ovp k op operands_fit a b =
     (let r = cong_aop op a b
      in if operands_fit ∧ (op = AMod ∨ op = ADiv ∧ cong_excludes (-1) b) then r
         else if should_wrap ovp k then norm_cong_ik k r
         else if op = ASub ∧ cong_fits k r then r
         else top)"

lemma cong_arith_ik_sound:
  assumes fit: "operands_fit ⟹ m ∈ ik_range k ∧ n ∈ ik_range k"
    and ma: "m ∈ gamma_congruence a" and nb: "n ∈ gamma_congruence b"
  shows "ik_norm k (aop_eval op m n) ∈ gamma_congruence (cong_arith_ik ovp k op operands_fit a b)"
proof -
  let ?r = "cong_aop op a b"
  have r: "aop_eval op m n ∈ gamma_congruence ?r" using ma nb by (rule cong_aop_sound)
  show ?thesis
  proof (cases "operands_fit ∧ (op = AMod ∨ op = ADiv ∧ cong_excludes (-1) b)")
    case True
    then have "aop_eval op m n ∈ ik_range k"
      using fit cong_excludes_sound[of "-1" b n] nb c_mod_in_range c_div_in_range by auto
    with True r show ?thesis by (simp add: cong_arith_ik_def)
  next
    case nr: False
    show ?thesis
    proof (cases "should_wrap ovp k")
      case True
      with nr r show ?thesis by (auto simp: cong_arith_ik_def Let_def norm_cong_ik_sound)
    next
      case False
      have "op = ASub ∧ cong_fits k ?r ⟹ aop_eval op m n ∈ ik_range k"
        using r cong_fits_sound by blast
      with nr False r show ?thesis by (auto simp: cong_arith_ik_def Let_def)
    qed
  qed
qed

instantiation congruence :: kind_int_domain
begin

definition top_ik_congruence :: "ikind ⇒ congruence" where "top_ik_congruence k = top"

definition of_int_ik_congruence :: "ikind ⇒ int ⇒ congruence" where
  "of_int_ik_congruence k n = congruence_of_int (ik_norm k n)"

definition cast_ik_congruence ::
    "signed_overflow_policy ⇒ ikind option ⇒ ikind ⇒ congruence ⇒ congruence" where
  "cast_ik_congruence ovp sk k a =
     (case sk of Some s ⇒ if ik_widens s k then a else cast_cong_ik k a | None ⇒ cast_cong_ik k a)"

definition arith_ik_congruence ::
    "signed_overflow_policy ⇒ ikind ⇒ aop ⇒ bool ⇒ congruence ⇒ congruence ⇒ congruence" where
  "arith_ik_congruence = cong_arith_ik"

definition cmp_ik_congruence :: "cop ⇒ congruence ⇒ congruence ⇒ bool option" where
  "cmp_ik_congruence = cop_of congruence_lt cong_eq"

definition meet_ik_congruence :: "congruence ⇒ congruence ⇒ congruence" where
  "meet_ik_congruence a b = a ⊓ b"

definition refine_ik_congruence :: "cop ⇒ congruence ⇒ congruence ⇒ congruence × congruence" where
  "refine_ik_congruence = cong_refine"

definition raw_aop_ik_congruence :: "aop ⇒ congruence ⇒ congruence ⇒ congruence" where
  "raw_aop_ik_congruence = cong_aop"

definition fits_ik_congruence :: "ikind ⇒ congruence ⇒ bool" where
  "fits_ik_congruence = cong_fits"

definition answer_ik_congruence :: "congruence ⇒ answer" where
  "answer_ik_congruence a = answer_with (int_congruence_update (λ_. a))"

definition of_answer_ik_congruence :: "ikind option ⇒ answer ⇒ congruence" where
  "of_answer_ik_congruence sk q = (case q of QBot ⇒ ⊥ | QTop ⇒ ⊤ | QLifted d ⇒ int_congruence d)"

instance proof
  fix n m :: int and k :: ikind and a b :: congruence and op :: aop and c :: cop and r :: bool
  show "n ∈ ik_range k ⟹ n ∈ numeric_domain_class.gamma (top_ik k :: congruence)"
    by (simp add: top_ik_congruence_def)
  show "ik_norm k n ∈ numeric_domain_class.gamma (of_int_ik k n :: congruence)"
    by (simp add: of_int_ik_congruence_def)
  show "(⋀k'. sk = Some k' ⟹ n ∈ ik_range k') ⟹ n ∈ numeric_domain_class.gamma a
      ⟹ ik_norm k n ∈ numeric_domain_class.gamma (cast_ik ovp sk k a)" for ovp sk
    by (cases sk)
       (auto simp: cast_ik_congruence_def ik_widens_def cast_cong_ik_sound simp del: ik_range_iff,
        auto)
  show "(fits ⟹ m ∈ ik_range k ∧ n ∈ ik_range k) ⟹ m ∈ numeric_domain_class.gamma a
      ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ ik_norm k (aop_eval op m n) ∈ numeric_domain_class.gamma (arith_ik ovp k op fits a b)" for ovp fits
    using cong_arith_ik_sound[where operands_fit = fits and ovp = ovp and op = op and k = k and m =
      m and n = n]
    by (simp add: arith_ik_congruence_def)
  show "cmp_ik c a b = Some r ⟹ m ∈ numeric_domain_class.gamma a
      ⟹ n ∈ numeric_domain_class.gamma b ⟹ cop_eval c m n = r"
    unfolding cmp_ik_congruence_def
    by simp (rule cop_of_sound[OF congruence_lt_sound cong_eq_sound])
  show "m ∈ numeric_domain_class.gamma a ⟹ m ∈ numeric_domain_class.gamma b
      ⟹ m ∈ numeric_domain_class.gamma (meet_ik a b)"
    by (simp add: meet_ik_congruence_def)
  show "cop_eval c m n ⟹ m ∈ numeric_domain_class.gamma a ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ m ∈ numeric_domain_class.gamma (fst (refine_ik c a b))
        ∧ n ∈ numeric_domain_class.gamma (snd (refine_ik c a b))"
    by (simp add: refine_ik_congruence_def cong_refine_sound)
  show "m ∈ numeric_domain_class.gamma a ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ aop_eval op m n ∈ numeric_domain_class.gamma (raw_aop_ik op a b)"
    by (simp add: raw_aop_ik_congruence_def cong_aop_sound)
  show "fits_ik k a ⟹ n ∈ numeric_domain_class.gamma a ⟹ n ∈ ik_range k"
    using cong_fits_sound[of k a n] by (simp add: fits_ik_congruence_def)
  show "n ∈ numeric_domain_class.gamma a ⟹ n ∈ gamma_answer (answer_ik a)"
    by (simp add: answer_ik_congruence_def gamma_int_dom_def top_int_dom_ext_def gamma_sign_top
        gamma_parity_top top_ivl_def gamma_ivl_top)
  show "(⋀k. sk = Some k ⟹ n ∈ ik_range k) ⟹ n ∈ gamma_answer q
      ⟹ n ∈ numeric_domain_class.gamma (of_answer_ik sk q :: congruence)" for sk q
    by (cases q)
      (simp_all add: of_answer_ik_congruence_def gamma_int_dom_def gamma_sign_top gamma_parity_top)
qed

end
