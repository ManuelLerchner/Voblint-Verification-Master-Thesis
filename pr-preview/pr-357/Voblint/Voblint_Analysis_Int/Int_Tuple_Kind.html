subsection ‹Lattice›

text ‹The order, join, bottom and top are taken componentwise.›

instantiation int_tuple :: (type) "{order_top, bounded_semilattice_sup_bot}"
begin

definition less_eq_int_tuple :: "'a int_tuple ⇒ 'a int_tuple ⇒ bool" where
  "a ≤ b ⟷ it_ivl a ≤ it_ivl b ∧ it_cong a ≤ it_cong b"

definition less_int_tuple :: "'a int_tuple ⇒ 'a int_tuple ⇒ bool" where
  "a < b ⟷ a ≤ b ∧ ¬ b ≤ (a :: 'a int_tuple)"

definition sup_int_tuple :: "'a int_tuple ⇒ 'a int_tuple ⇒ 'a int_tuple" where
  "a ⊔ b = IT (it_ivl a ⊔ it_ivl b) (it_cong a ⊔ it_cong b)"

definition bot_int_tuple :: "'a int_tuple" where "bot = IT bot bot"

definition top_int_tuple :: "'a int_tuple" where "top = IT top top"

instance proof
  fix a b c :: "'a int_tuple"
  show "a < b ⟷ a ≤ b ∧ ¬ b ≤ a" by (rule less_int_tuple_def)
  show "a ≤ a" by (simp add: less_eq_int_tuple_def)
  show "a ≤ b ⟹ b ≤ c ⟹ a ≤ c"
    by (auto simp: less_eq_int_tuple_def intro: order.trans)
  show "a ≤ b ⟹ b ≤ a ⟹ a = b"
    by (cases a; cases b) (auto simp: less_eq_int_tuple_def)
  show "a ≤ a ⊔ b" "b ≤ a ⊔ b" by (simp_all add: less_eq_int_tuple_def sup_int_tuple_def)
  show "b ≤ a ⟹ c ≤ a ⟹ b ⊔ c ≤ a"
    by (simp add: less_eq_int_tuple_def sup_int_tuple_def)
  show "bot ≤ a" by (simp add: less_eq_int_tuple_def bot_int_tuple_def)
  show "a ≤ top" by (simp add: less_eq_int_tuple_def top_int_tuple_def)
qed

end

lemma IT_le_IT [simp]: "IT i c ≤ IT j d ⟷ i ≤ j ∧ c ≤ d"
  by (simp add: less_eq_int_tuple_def)

subsection ‹Exact emptiness›

text ‹A class meets an interval when its least value above the lower bound, or its
  greatest value below the upper bound, lies inside. The test is exact, so emptiness of
  the product is decided exactly.›

lemma cong_greatest_below:
  fixes m c hi :: int
  assumes m: "0 < m"
  defines "r ≡ hi - (hi - c) mod m"
  shows "r ≤ hi" and "m dvd r - c"
    and "m dvd n - c ⟹ n ≤ hi ⟹ n ≤ r"
proof -
  show "r ≤ hi" using m by (simp add: r_def)
  have "r - c = (hi - c) - (hi - c) mod m" by (simp add: r_def)
  then show "m dvd r - c" by (simp only: dvd_minus_mod)
  assume d: "m dvd n - c" and hi: "n ≤ hi"
  have "(hi - n) mod m = (hi - c) mod m"
  proof -
    have "m dvd (hi - n) - (hi - c)" using d dvd_minus_iff[of m "n - c"] by simp
    then show ?thesis by (simp add: mod_eq_dvd_iff)
  qed
  moreover have "(hi - n) mod m ≤ hi - n"
    using hi m by (simp add: zmod_le_nonneg_dividend)
  ultimately show "n ≤ r" by (simp add: r_def)
qed

fun cong_meets :: "int × int ⇒ eint ⇒ eint ⇒ bool" where
  "cong_meets (c, m) l u =
     (if m = 0 then l ≤ Fin c ∧ Fin c ≤ u
      else case (l, u) of
        (Fin x, Fin y) ⇒ x + (c - x) mod m ≤ y
      | (Fin x, PlusInf) ⇒ True
      | (MinInf, Fin y) ⇒ True
      | (MinInf, PlusInf) ⇒ True
      | _ ⇒ False)"

lemma cong_meets_correct:
  assumes "0 ≤ m"
  shows "cong_meets (c, m) l u ⟷ (∃n. m dvd n - c ∧ l ≤ Fin n ∧ Fin n ≤ u)"
proof (cases "m = 0")
  case True
  then show ?thesis by auto
next
  case False
  with assms have m: "0 < m" by simp
  show ?thesis
  proof (cases l; cases u)
    fix x y assume l: "l = Fin x" and u: "u = Fin y"
    note least = cong_least_above[OF m, where lo = x and c = c]
    show ?thesis
      using least False by (auto simp: l u) force
  next
    fix x assume "l = Fin x" "u = PlusInf"
    then show ?thesis
      using cong_least_above[OF m, where lo = x and c = c] False
      by auto
  next
    fix y assume "l = MinInf" "u = Fin y"
    then show ?thesis
      using cong_greatest_below[OF m, where hi = y and c = c] False
      by auto
  next
    assume "l = MinInf" "u = PlusInf"
    then show ?thesis using False by (auto intro: exI[of _ c])
  qed (auto simp: False)
qed

definition it_empty :: "ivl ⇒ congruence ⇒ bool" where
  "it_empty i c =
     (case Rep_congruence c of
        None ⇒ True
      | Some cm ⇒ ¬ cong_meets cm (ivl_lower i) (ivl_upper i))"

lemma it_empty_correct: "it_empty i c ⟷ gamma_ivl i ∩ gamma_congruence c = {}"
proof (cases c rule: congruence_rep_cases)
  case Bot
  then show ?thesis by (simp add: it_empty_def gamma_congruence_def)
next
  case (Class r m)
  have "0 ≤ m" using normalized_Rep_congruence[of c] Class by (cases "m = 0") auto
  then show ?thesis
    using cong_meets_correct[of m r "ivl_lower i" "ivl_upper i"] Class
    by (cases i) (auto simp: it_empty_def gamma_congruence_Some)
qed

subsection ‹Mutual refinement›

text ‹
  Goblint's ▩‹Congruence.refine_with_interval›: a class meeting a finite interval
  in one value becomes that value, and one meeting it in none becomes bottom. The
  interval is then rounded to the class (\<^const>‹refine_ivl_with_congruence›, Goblint's
  ▩‹Interval.refine_with_congruence›). This is one round of
  ▩‹IntDomTuple.refine› in that order.
›

definition cong_refine_with_ivl :: "ivl ⇒ congruence ⇒ congruence" where
  "cong_refine_with_ivl i c =
     (case (i, Rep_congruence c) of
        (Ivl (Fin x) (Fin y), Some (r, m)) ⇒
          if m = 0 then (if x ≤ r ∧ r ≤ y then c else bot)
          else (let lo = x + (r - x) mod m; hi = y - (y - r) mod m
                in if hi < lo then bot else if lo = hi then congruence_of_int lo else c)
      | _ ⇒ c)"

lemma cong_refine_with_ivl_sound:
  assumes n: "n ∈ gamma_ivl i" "n ∈ gamma_congruence c"
  shows "n ∈ gamma_congruence (cong_refine_with_ivl i c)"
proof (cases "∃x y r m. i = Ivl (Fin x) (Fin y) ∧ Rep_congruence c = Some (r, m)")
  case True
  then obtain x y r m where i: "i = Ivl (Fin x) (Fin y)" and c: "Rep_congruence c = Some (r, m)"
    by blast
  have xy: "x ≤ n" "n ≤ y" using n(1) by (simp_all add: i)
  have d: "m dvd n - r" using n(2) c by (simp add: gamma_congruence_Some)
  show ?thesis
  proof (cases "m = 0")
    case True
    with d xy n(2) show ?thesis by (simp add: cong_refine_with_ivl_def i c)
  next
    case False
    then have m: "0 < m" using normalized_Rep_congruence[of c] c by auto
    have "x + (r - x) mod m ≤ n"
      using cong_least_above(3)[OF m d xy(1)] .
    moreover have "n ≤ y - (y - r) mod m"
      using cong_greatest_below(3)[OF m d xy(2)] .
    ultimately show ?thesis
      using False n(2) by (auto simp: cong_refine_with_ivl_def i c Let_def)
  qed
qed (use n in ‹auto simp: cong_refine_with_ivl_def split: ivl.splits eint.splits option.splits›)

definition it_refine :: "'m int_tuple ⇒ 'm int_tuple" where
  "it_refine a =
     (let c = cong_refine_with_ivl (it_ivl a) (it_cong a)
      in IT (refine_ivl_with_congruence c (it_ivl a)) c)"

lemma it_refine_sound: "n ∈ gamma_it a ⟹ n ∈ gamma_it (it_refine a)"
  using cong_refine_with_ivl_sound refine_ivl_with_congruence_cover
  by (cases a) (fastforce simp: it_refine_def Let_def)

text ‹‹Refine_Never› leaves the components apart, ‹Refine_Once› exchanges once, and
  ‹Refine_Fixpoint› exchanges until nothing changes. Like the value-level
  \<^const>‹refine_fix›, the fixpoint keeps its input when the loop does not stop.›

definition it_refine_mode :: "refine_mode ⇒ 'm int_tuple ⇒ 'm int_tuple" where
  "it_refine_mode md a =
     (case md of
        Refine_Never ⇒ a
      | Refine_Once ⇒ it_refine a
      | Refine_Fixpoint ⇒
          (case while_option (λx. it_refine x ≠ x) it_refine a of Some r ⇒ r | None ⇒ a))"

lemma it_refine_mode_sound: "n ∈ gamma_it a ⟹ n ∈ gamma_it (it_refine_mode md a)"
proof (cases md)
  case Refine_Fixpoint
  assume n: "n ∈ gamma_it a"
  have "n ∈ gamma_it r" if "while_option (λx. it_refine x ≠ x) it_refine a = Some r" for r
    using while_option_rule[where P = "λx. n ∈ gamma_it x", OF _ that] n it_refine_sound by blast
  with n Refine_Fixpoint show ?thesis by (auto simp: it_refine_mode_def split: option.splits)
qed (simp_all add: it_refine_mode_def it_refine_sound)

subsection ‹Domain instances›

text ‹Widening and narrowing also act componentwise; a value is empty when its congruence
  class has no member between its interval's bounds.›

instantiation int_tuple :: (type) numeric_domain
begin

definition widen_int_tuple :: "'a int_tuple ⇒ 'a int_tuple ⇒ 'a int_tuple" where
  "widen_int_tuple a b = IT (it_ivl a ∇ it_ivl b) (it_cong a ∇ it_cong b)"

definition narrow_int_tuple :: "'a int_tuple ⇒ 'a int_tuple ⇒ 'a int_tuple" where
  "narrow_int_tuple a b = IT (it_ivl a Δ it_ivl b) (it_cong a Δ it_cong b)"

definition is_empty_int_tuple :: "'a int_tuple ⇒ bool" where
  "is_empty_int_tuple a = it_empty (it_ivl a) (it_cong a)"

text ‹A value prints as bottom, as a single integer when a component pins one, and
  otherwise as each component.›

definition to_string_int_tuple :: "'a int_tuple ⇒ String.literal" where
  "to_string_int_tuple a =
     (if it_empty (it_ivl a) (it_cong a) then sym_bottom
      else case ivl_constant (it_ivl a) of
        Some n ⇒ string_of_int n
      | None ⇒ (case congruence_constant (it_cong a) of
          Some n ⇒ string_of_int n
        | None ⇒ STR ''intervals:'' + to_string (it_ivl a) + STR ''; congruences:''
                    + to_string (it_cong a)))"

definition gamma_int_tuple :: "'a int_tuple ⇒ int set" where
  "gamma_int_tuple = gamma_it"

instance proof
  fix a b :: "'a int_tuple"
  show "a ≤ a ∇ b" "b ≤ a ∇ b"
    by (simp_all add: widen_int_tuple_def less_eq_int_tuple_def widen_ge1 widen_ge2)
  show "b ≤ a ⟹ b ≤ a Δ b" "b ≤ a ⟹ a Δ b ≤ a"
    by (simp_all add: narrow_int_tuple_def less_eq_int_tuple_def narrow_ge narrow_le)
  show "numeric_domain_class.gamma (bot :: 'a int_tuple) = {}"
    by (simp add: gamma_int_tuple_def bot_int_tuple_def bot_congruence_def)
  show "numeric_domain_class.gamma (top :: 'a int_tuple) = UNIV"
    using gamma_top[where 'a = ivl]
    by (simp add: gamma_int_tuple_def top_int_tuple_def)
  show "a ≤ b ⟹ numeric_domain_class.gamma a ⊆ numeric_domain_class.gamma b"
    using gamma_ivl_mono[of "it_ivl a" "it_ivl b"]
      gamma_mono[of "it_cong a" "it_cong b"]
    by (auto simp: gamma_int_tuple_def gamma_it_def less_eq_int_tuple_def)
  show "is_empty a ⟷ numeric_domain_class.gamma a = {}"
    by (simp add: is_empty_int_tuple_def gamma_int_tuple_def gamma_it_def it_empty_correct)
qed

end

lemma gamma_int_tuple_IT [simp]:
  "numeric_domain_class.gamma (IT i c) = gamma_ivl i ∩ gamma_congruence c"
  by (simp add: gamma_int_tuple_def)

lemma it_refine_IT_sound:
  "n ∈ gamma_ivl i ⟹ n ∈ gamma_congruence c
     ⟹ n ∈ numeric_domain_class.gamma (it_refine_mode md (IT i c))"
  using it_refine_mode_sound[of n "IT i c"] by (simp add: gamma_int_tuple_def)

text ‹
  Arithmetic follows ▩‹IntDomTuple.map2ovc›: the interval computes first, and when
  its mathematical result fits the kind, no value overflows (▩‹no_ov›), so the
  congruence keeps its exact class. A conversion keeps the class on the same condition.
  Comparisons take the first component that decides, as ▩‹map2bool› takes the
  components' common answer. Every result is refined as the mode says.
›

instantiation int_tuple :: (int_refinement) kind_int_domain
begin

definition top_ik_int_tuple :: "ikind ⇒ 'a int_tuple" where
  "top_ik_int_tuple k = IT (top_ik k) (top_ik k)"

definition of_int_ik_int_tuple :: "ikind ⇒ int ⇒ 'a int_tuple" where
  "of_int_ik_int_tuple k n = IT (of_int_ik k n) (of_int_ik k n)"

definition cast_ik_int_tuple ::
    "signed_overflow_policy ⇒ ikind option ⇒ ikind ⇒ 'a int_tuple ⇒ 'a int_tuple" where
  "cast_ik_int_tuple ovp sk k a =
     it_refine_mode (int_mode TYPE('a)) (IT (cast_ik ovp sk k (it_ivl a))
                   (if fits_ik k (it_ivl a) then it_cong a else cast_ik ovp sk k (it_cong a)))"

text ‹The interval decides Goblint's ▩‹no_ov›: when the mathematical result fits, the
  congruence keeps its unnormalized class.›

definition arith_ik_int_tuple ::
    "signed_overflow_policy ⇒ ikind ⇒ aop ⇒ bool ⇒ 'a int_tuple ⇒ 'a int_tuple ⇒ 'a int_tuple" where
  "arith_ik_int_tuple ovp k op fits a b =
     (let r = raw_aop_ik op (it_ivl a) (it_ivl b)
      in it_refine_mode (int_mode TYPE('a)) (IT (cast_ik ovp None k r)
                       (if fits_ik k r then raw_aop_ik op (it_cong a) (it_cong b)
                        else arith_ik ovp k op fits (it_cong a) (it_cong b))))"

definition cmp_ik_int_tuple :: "cop ⇒ 'a int_tuple ⇒ 'a int_tuple ⇒ bool option" where
  "cmp_ik_int_tuple c a b =
     (case cmp_ik c (it_ivl a) (it_ivl b) of
        Some r ⇒ Some r
      | None ⇒ cmp_ik c (it_cong a) (it_cong b))"

definition meet_ik_int_tuple :: "'a int_tuple ⇒ 'a int_tuple ⇒ 'a int_tuple" where
  "meet_ik_int_tuple a b =
     it_refine_mode (int_mode TYPE('a)) (IT (meet_ik (it_ivl a) (it_ivl b)) (meet_ik (it_cong a) (it_cong b)))"

definition refine_ik_int_tuple :: "cop ⇒ 'a int_tuple ⇒ 'a int_tuple ⇒ 'a int_tuple × 'a int_tuple"
  where
  "refine_ik_int_tuple c a b =
     (let (i1, i2) = refine_ik c (it_ivl a) (it_ivl b);
          (c1, c2) = refine_ik c (it_cong a) (it_cong b)
      in (it_refine_mode (int_mode TYPE('a)) (IT i1 c1), it_refine_mode (int_mode TYPE('a)) (IT i2 c2)))"

definition raw_aop_ik_int_tuple :: "aop ⇒ 'a int_tuple ⇒ 'a int_tuple ⇒ 'a int_tuple" where
  "raw_aop_ik_int_tuple op a b =
     IT (raw_aop_ik op (it_ivl a) (it_ivl b)) (raw_aop_ik op (it_cong a) (it_cong b))"

definition fits_ik_int_tuple :: "ikind ⇒ 'a int_tuple ⇒ bool" where
  "fits_ik_int_tuple k a ⟷ fits_ik k (it_ivl a) ∨ fits_ik k (it_cong a)"

definition answer_ik_int_tuple :: "'a int_tuple ⇒ answer" where
  "answer_ik_int_tuple a = answer_ik (it_ivl a) ⊓ answer_ik (it_cong a)"

definition of_answer_ik_int_tuple :: "ikind option ⇒ answer ⇒ 'a int_tuple" where
  "of_answer_ik_int_tuple sk q = IT (of_answer_ik sk q) (of_answer_ik sk q)"

instance proof
  fix n m :: int and k :: ikind and a b :: "'a int_tuple" and op :: aop and c :: cop and r :: bool
  show "n ∈ ik_range k ⟹ n ∈ numeric_domain_class.gamma (top_ik k :: 'a int_tuple)"
    using top_ik_sound[of n k, where 'a = ivl] top_ik_sound[of n k, where 'a = congruence]
    by (simp add: top_ik_int_tuple_def)
  show "ik_norm k n ∈ numeric_domain_class.gamma (of_int_ik k n :: 'a int_tuple)"
    using of_int_ik_sound[of k n, where 'a = ivl] of_int_ik_sound[of k n, where 'a = congruence]
    by (simp add: of_int_ik_int_tuple_def)
  show "(⋀k'. sk = Some k' ⟹ n ∈ ik_range k') ⟹ n ∈ numeric_domain_class.gamma a
      ⟹ ik_norm k n ∈ numeric_domain_class.gamma (cast_ik ovp sk k a)" for ovp sk
  proof -
    assume src: "⋀k'. sk = Some k' ⟹ n ∈ ik_range k'" and n: "n ∈ numeric_domain_class.gamma a"
    obtain i cg where a: "a = IT i cg" by (cases a)
    have ni: "n ∈ gamma_ivl i" and nc: "n ∈ gamma_congruence cg" using n by (simp_all add: a)
    have "ik_norm k n ∈ gamma_ivl (cast_ik ovp sk k i)"
      using cast_ik_sound[where a = i and k = k and ovp = ovp, OF src] ni by simp
    moreover have "ik_norm k n ∈ gamma_congruence
        (if fits_ik k i then cg else cast_ik ovp sk k cg)"
      using fits_ik_sound[of k i n] cast_ik_sound[where a = cg and k = k and ovp = ovp, OF src] ni
        nc
      by (auto simp del: ik_range_iff)
    ultimately show ?thesis
      unfolding cast_ik_int_tuple_def a int_tuple.sel by (rule it_refine_IT_sound)
  qed
  show "(fits ⟹ m ∈ ik_range k ∧ n ∈ ik_range k) ⟹ m ∈ numeric_domain_class.gamma a
      ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ ik_norm k (aop_eval op m n) ∈ numeric_domain_class.gamma (arith_ik ovp k op fits a b)" for ovp fits
  proof -
    assume fit: "fits ⟹ m ∈ ik_range k ∧ n ∈ ik_range k"
      and ma: "m ∈ numeric_domain_class.gamma a" and nb: "n ∈ numeric_domain_class.gamma b"
    obtain i1 c1 i2 c2 where a: "a = IT i1 c1" and b: "b = IT i2 c2" by (cases a; cases b)
    let ?r = "raw_aop_ik op i1 i2"
    have rv: "aop_eval op m n ∈ gamma_ivl ?r"
      using raw_aop_ik_sound[of m i1 n i2] ma nb by (simp add: a b)
    have "ik_norm k (aop_eval op m n) ∈ gamma_ivl (cast_ik ovp None k ?r)"
      using cast_ik_None_sound[where a = ?r and k = k and ovp = ovp] rv by simp
    moreover have "ik_norm k (aop_eval op m n) ∈ gamma_congruence
        (if fits_ik k ?r then raw_aop_ik op c1 c2 else arith_ik ovp k op fits c1 c2)"
      using fits_ik_sound[of k ?r] rv raw_aop_ik_sound[of m c1 n c2]
        arith_ik_sound[where k = k and op = op and operands_fit = fits and ovp = ovp and a = c1 and
          b = c2,
          OF fit] ma nb
      by (auto simp: a b simp del: ik_range_iff)
    ultimately show ?thesis
      unfolding arith_ik_int_tuple_def a b int_tuple.sel Let_def by (rule it_refine_IT_sound)
  qed
  show "cmp_ik c a b = Some r ⟹ m ∈ numeric_domain_class.gamma a
      ⟹ n ∈ numeric_domain_class.gamma b ⟹ cop_eval c m n = r"
    by (cases a; cases b)
       (auto simp: cmp_ik_int_tuple_def split: option.splits dest: cmp_ik_sound)
  show "m ∈ numeric_domain_class.gamma a ⟹ m ∈ numeric_domain_class.gamma b
      ⟹ m ∈ numeric_domain_class.gamma (meet_ik a b)"
    using meet_ik_sound[of m "it_ivl a" "it_ivl b"] meet_ik_sound[of m "it_cong a" "it_cong b"]
    by (cases a; cases b) (simp add: meet_ik_int_tuple_def it_refine_IT_sound)
  show "cop_eval c m n ⟹ m ∈ numeric_domain_class.gamma a ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ m ∈ numeric_domain_class.gamma (fst (refine_ik c a b))
        ∧ n ∈ numeric_domain_class.gamma (snd (refine_ik c a b))"
    using refine_ik_sound[of c m n "it_ivl a" "it_ivl b"]
      refine_ik_sound[of c m n "it_cong a" "it_cong b"]
    by (cases a; cases b)
       (auto simp: refine_ik_int_tuple_def it_refine_IT_sound split: prod.splits)
  show "m ∈ numeric_domain_class.gamma a ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ aop_eval op m n ∈ numeric_domain_class.gamma (raw_aop_ik op a b)"
    using raw_aop_ik_sound[of m "it_ivl a" n "it_ivl b" op]
      raw_aop_ik_sound[of m "it_cong a" n "it_cong b" op]
    by (cases a; cases b) (simp add: raw_aop_ik_int_tuple_def)
  show "fits_ik k a ⟹ n ∈ numeric_domain_class.gamma a ⟹ n ∈ ik_range k"
    using fits_ik_sound[of k "it_ivl a" n] fits_ik_sound[of k "it_cong a" n]
    by (cases a) (auto simp: fits_ik_int_tuple_def simp del: ik_range_iff)
  show "n ∈ numeric_domain_class.gamma a ⟹ n ∈ gamma_answer (answer_ik a)"
    using answer_ik_sound[of n "it_ivl a"] answer_ik_sound[of n "it_cong a"]
    by (cases a) (simp add: answer_ik_int_tuple_def gamma_query_lift_inf[OF gamma_inf_int_dom])
  show "(⋀k. sk = Some k ⟹ n ∈ ik_range k) ⟹ n ∈ gamma_answer q
      ⟹ n ∈ numeric_domain_class.gamma (of_answer_ik sk q :: 'a int_tuple)" for sk q
    using of_answer_ik_sound[of sk n q, where 'a = ivl]
      of_answer_ik_sound[of sk n q, where 'a = congruence]
    by (simp add: of_answer_ik_int_tuple_def)
qed
