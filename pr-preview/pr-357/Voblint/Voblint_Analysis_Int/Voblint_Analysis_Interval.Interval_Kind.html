section ‹Intervals at a kind›

text ‹
  The kind-indexed operations of Goblint's ▩‹IntervalFunctor› without widening
  thresholds. Widening keeps a bound that did not grow and moves a growing bound to the
  kind's limit. Narrowing replaces only a bound that sits at the kind's limit.

  Goblint widens operands of the kind, whose bounds already lie in its range. The class
  laws must hold for every operand, so a growing bound moves to the kind's limit or to
  the new bound, whichever is further out; on operands in range this is the kind's
  limit.
›

text ‹Bounds are compared in the linear order of \<^typ>‹eint›, not by unfolding it.›

declare less_eq_eint_def [simp del]

definition ik_lo :: "ikind ⇒ eint" where "ik_lo k = Fin (ik_min k)"
definition ik_hi :: "ikind ⇒ eint" where "ik_hi k = Fin (ik_max k)"

fun widen_ivl_ik :: "ikind ⇒ ivl ⇒ ivl ⇒ ivl" where
  "widen_ivl_ik k (Ivl l0 u0) (Ivl l1 u1) =
     Ivl (if l0 ≤ l1 then l0 else min (ik_lo k) l1)
         (if u1 ≤ u0 then u0 else max (ik_hi k) u1)"

fun narrow_ivl_ik :: "ikind ⇒ ivl ⇒ ivl ⇒ ivl" where
  "narrow_ivl_ik k (Ivl l0 u0) (Ivl l1 u1) =
     Ivl (if l0 = ik_lo k then l1 else l0) (if u0 = ik_hi k then u1 else u0)"

text ‹The canonical bottom stays neutral, as for the kind-free widening.›

instantiation ivl :: ik_domain
begin

definition widen_ik_ivl :: "ikind ⇒ ivl ⇒ ivl ⇒ ivl" where
  "widen_ik_ivl k a b = (if a = bot then b else if b = bot then a else widen_ivl_ik k a b)"

definition narrow_ik_ivl :: "ikind ⇒ ivl ⇒ ivl ⇒ ivl" where
  "narrow_ik_ivl = narrow_ivl_ik"

definition empty_ik_ivl :: "ikind ⇒ ivl ⇒ bool" where
  "empty_ik_ivl k a = is_bottom_ivl (a ⊓ Ivl (ik_lo k) (ik_hi k))"

instance proof
  fix k :: ikind and a b :: ivl
  obtain l0 u0 l1 u1 where a: "a = Ivl l0 u0" and b: "b = Ivl l1 u1"
    by (meson ivl_exhaustE)
  have lo: "(if l0 ≤ l1 then l0 else min (ik_lo k) l1) ≤ l0"
       "(if l0 ≤ l1 then l0 else min (ik_lo k) l1) ≤ l1"
    using linear[of l0 l1] by (auto intro: order.trans[OF min.cobounded2])
  have hi: "u0 ≤ (if u1 ≤ u0 then u0 else max (ik_hi k) u1)"
       "u1 ≤ (if u1 ≤ u0 then u0 else max (ik_hi k) u1)"
    using linear[of u0 u1] by (auto intro: order.trans[OF _ max.cobounded2])
  have core1: "a ≤ widen_ivl_ik k a b" and core2: "b ≤ widen_ivl_ik k a b"
    using lo hi by (simp_all only: a b widen_ivl_ik.simps less_eq_ivl_def prod.case ivl.case)
  have bot_le: "bot ≤ c" for c :: ivl
    by simp
  show "a ≤ widen_ik k a b"
    using core1 bot_le by (auto simp: widen_ik_ivl_def)
  show "b ≤ widen_ik k a b"
    using core2 bot_le by (auto simp: widen_ik_ivl_def)
  show "b ≤ a ⟹ b ≤ narrow_ik k a b"
    by (cases a; cases b) (auto simp: narrow_ik_ivl_def less_eq_ivl_def)
  show "b ≤ a ⟹ narrow_ik k a b ≤ a"
    by (cases a; cases b) (auto simp: narrow_ik_ivl_def less_eq_ivl_def)
  have range: "gamma_ivl (Ivl (ik_lo k) (ik_hi k)) = ik_range k"
    by (auto simp: ik_lo_def ik_hi_def less_eq_eint_def)
  show "empty_ik k a ⟷ numeric_domain_class.gamma a ∩ ik_range k = {}"
    by (simp add: empty_ik_ivl_def is_bottom_ivl_correct flip: range)
qed

end

section ‹Intervals as a kind-aware integer domain›

text ‹
  An operation at kind ‹k› computes the mathematical interval and keeps it when it lies
  in ‹k›'s range. Otherwise it normalizes the interval as Goblint does: an unsigned kind
  wraps it when the wrapped interval is exact, and every other overflow answers
  ‹top_of k›.
›

definition ik_ivl :: "ikind ⇒ ivl" where "ik_ivl k = Ivl (ik_lo k) (ik_hi k)"

lemma gamma_ik_ivl [simp]: "gamma_ivl (ik_ivl k) = ik_range k"
  by (auto simp: ik_ivl_def ik_lo_def ik_hi_def less_eq_eint_def)

text ‹
  A kind that wraps under the overflow policy wraps a finite interval that spans less
  than its modulus and whose wrapped ends stay ordered; any other overflow gives the
  kind's whole range. This is Goblint's ▩‹norm› in ▩‹intervalDomain.ml›: under the
  default ▩‹sem.int.signed_overflow = assume_top› only an unsigned kind wraps, under
  ▩‹assume_wraparound› every kind does. A cast and an arithmetic result are treated
  alike; they differ only under ▩‹assume_none›, which assumes that no signed overflow
  occurs.
›

definition wrap_ivl_ik :: "ikind ⇒ int ⇒ int ⇒ ivl" where
  "wrap_ivl_ik k x y =
     (if y - x < ik_mod k ∧ ik_norm k x ≤ ik_norm k y
      then Ivl (Fin (ik_norm k x)) (Fin (ik_norm k y)) else ik_ivl k)"

definition norm_ivl_ik :: "signed_overflow_policy ⇒ ikind ⇒ ivl ⇒ ivl" where
  "norm_ivl_ik ovp k a =
     (if a ≤ ik_ivl k then a
      else case a of
        Ivl (Fin x) (Fin y) ⇒ if should_wrap ovp k then wrap_ivl_ik k x y else ik_ivl k
      | _ ⇒ ik_ivl k)"

lemma ik_max_min_mod: "ik_max k - ik_min k + 1 = ik_mod k"
  by (cases k) (simp_all add: ik_max_def ik_min_def ik_half_def ik_mod_def)

lemma wrap_ivl_ik_sound:
  assumes "x ≤ n" and "n ≤ y"
  shows "ik_norm k n ∈ gamma_ivl (wrap_ivl_ik k x y)"
proof (cases "y - x < ik_mod k ∧ ik_norm k x ≤ ik_norm k y")
  case True
  then have w: "y - x ≤ ik_max k - ik_min k" using ik_max_min_mod[of k] by linarith
  have "ik_norm k n = ik_norm k x + (n - x)"
    using True assms(1,2) ik_norm_interval_shift w by blast
  moreover have "ik_norm k y = ik_norm k x + (y - x)"
    using True assms(1,2) ik_norm_interval_shift order_trans w by blast
  ultimately show ?thesis
    using True assms by (simp add: wrap_ivl_ik_def less_eq_eint_def)
qed (auto simp: wrap_ivl_ik_def)

lemma norm_ivl_ik_sound:
  assumes "n ∈ gamma_ivl a"
  shows "ik_norm k n ∈ gamma_ivl (norm_ivl_ik ovp k a)"
proof (cases "a ≤ ik_ivl k")
  case True
  then have "n ∈ ik_range k" using gamma_ivl_mono[OF True] assms by auto
  with True assms show ?thesis by (simp add: norm_ivl_ik_def)
next
  case False
  obtain l u where a: "a = Ivl l u" by (cases a)
  show ?thesis
  proof (cases "∃x y. l = Fin x ∧ u = Fin y")
    case True
    then obtain x y where l: "l = Fin x" and u: "u = Fin y" by blast
    have "x ≤ n" "n ≤ y" using assms by (simp_all add: a l u less_eq_eint_def)
    with False show ?thesis by (simp add: norm_ivl_ik_def a l u wrap_ivl_ik_sound)
  next
    case nw: False
    then have "norm_ivl_ik ovp k a = ik_ivl k"
      using False by (cases l; cases u) (auto simp: norm_ivl_ik_def a)
    then show ?thesis by simp
  qed
qed

fun ivl_aop :: "aop ⇒ ivl ⇒ ivl ⇒ ivl" where
  "ivl_aop AAdd a b = a + b"
| "ivl_aop ASub a b = a - b"
| "ivl_aop AMul a b = a * b"
| "ivl_aop ADiv a b = ivl_div a b"
| "ivl_aop AMod a b = ivl_mod a b"

lemma ivl_aop_sound:
  "m ∈ gamma_ivl a ⟹ n ∈ gamma_ivl b ⟹ aop_eval op m n ∈ gamma_ivl (ivl_aop op a b)"
  by (cases op) (simp_all add: ivl_plus_sound ivl_minus_sound ivl_times_sound ivl_div_sound
      ivl_mod_sound)

fun ivl_lo :: "ivl ⇒ eint" where "ivl_lo (Ivl l u) = l"
fun ivl_hi :: "ivl ⇒ eint" where "ivl_hi (Ivl l u) = u"

lemma gamma_ivl_bounds:
  assumes "n ∈ gamma_ivl a"
  shows "ivl_lo a ≤ Fin n" and "Fin n ≤ ivl_hi a"
  using assms by (cases a; simp)+

text ‹The bounds a comparison known to hold puts on its operands: ‹m < n› bounds ‹m›
  by ‹n›'s upper bound less one and ‹n› by ‹m›'s lower bound plus one.›

definition ivl_le_bounds :: "ivl ⇒ ivl ⇒ ivl × ivl" where
  "ivl_le_bounds a b = (a ⊓ Ivl MinInf (ivl_hi b), b ⊓ Ivl (ivl_lo a) PlusInf)"

definition ivl_lt_bounds :: "ivl ⇒ ivl ⇒ ivl × ivl" where
  "ivl_lt_bounds a b =
     (a ⊓ Ivl MinInf (ivl_hi b - Fin 1), b ⊓ Ivl (ivl_lo a + Fin 1) PlusInf)"

lemma eint_le_minus_one: "Fin m < Fin n ⟹ Fin n ≤ u ⟹ Fin m ≤ u - Fin 1"
  by (cases u) (auto simp: less_eq_eint_def less_eint_def)

lemma eint_plus_one_le: "Fin m < Fin n ⟹ l ≤ Fin m ⟹ l + Fin 1 ≤ Fin n"
  by (cases l) (auto simp: less_eq_eint_def less_eint_def)

lemma ivl_le_bounds_sound:
  assumes "m ≤ n" "m ∈ gamma_ivl a" "n ∈ gamma_ivl b"
  shows "m ∈ gamma_ivl (fst (ivl_le_bounds a b)) ∧ n ∈ gamma_ivl (snd (ivl_le_bounds a b))"
proof -
  have mn: "eint_le (Fin m) (Fin n)" using assms(1) by simp
  have "Fin m ≤ ivl_hi b" "ivl_lo a ≤ Fin n"
    using eint_le_trans[OF mn gamma_ivl_bounds(2)[OF assms(3), unfolded less_eq_eint_def]]
      eint_le_trans[OF gamma_ivl_bounds(1)[OF assms(2), unfolded less_eq_eint_def] mn]
    by (simp_all add: less_eq_eint_def)
  then show ?thesis using assms
    by (auto simp: ivl_le_bounds_def less_eq_eint_def intro!: meet_ivl_gamma)
qed

lemma ivl_lt_bounds_sound:
  assumes "m < n" "m ∈ gamma_ivl a" "n ∈ gamma_ivl b"
  shows "m ∈ gamma_ivl (fst (ivl_lt_bounds a b)) ∧ n ∈ gamma_ivl (snd (ivl_lt_bounds a b))"
proof -
  have lt: "Fin m < Fin n" using assms(1) by (simp add: less_eint_def)
  have "Fin m ≤ ivl_hi b - Fin 1" "ivl_lo a + Fin 1 ≤ Fin n"
    using eint_le_minus_one[OF lt gamma_ivl_bounds(2)[OF assms(3)]]
      eint_plus_one_le[OF lt gamma_ivl_bounds(1)[OF assms(2)]] by blast+
  then show ?thesis using assms
    by (auto simp: ivl_lt_bounds_def less_eq_eint_def intro!: meet_ivl_gamma)
qed

text ‹
  A disequality with a known constant ‹x› moves a bound of the other side that equals
  ‹x› past it, as ▩‹inv_bin_int› excludes the interval bounds for ▩‹Ne›
  (▩‹baseInvariant.ml›, lines 395--416 at ▩‹0dc12d355›); a singleton equal to
  ‹x› becomes empty.
›

fun ivl_excl :: "ivl ⇒ ivl ⇒ ivl" where
  "ivl_excl (Ivl l u) (Ivl (Fin x) (Fin y)) =
     (if x = y then Ivl (if l = Fin x then Fin (x + 1) else l) (if u = Fin x then Fin (x - 1) else u)
      else Ivl l u)"
| "ivl_excl a b = a"

lemma ivl_excl_sound:
  "m ≠ n ⟹ m ∈ gamma_ivl a ⟹ n ∈ gamma_ivl b ⟹ m ∈ gamma_ivl (ivl_excl a b)"
  by (induction a b rule: ivl_excl.induct) (auto simp: less_eq_eint_def)

fun ivl_refine :: "cop ⇒ ivl ⇒ ivl ⇒ ivl × ivl" where
  "ivl_refine CLt a b = ivl_lt_bounds a b"
| "ivl_refine CLe a b = ivl_le_bounds a b"
| "ivl_refine CGt a b = prod.swap (ivl_lt_bounds b a)"
| "ivl_refine CGe a b = prod.swap (ivl_le_bounds b a)"
| "ivl_refine CEq a b = (a ⊓ b, a ⊓ b)"
| "ivl_refine CNe a b = (ivl_excl a b, ivl_excl b a)"

lemma ivl_refine_sound:
  "cop_eval c m n ⟹ m ∈ gamma_ivl a ⟹ n ∈ gamma_ivl b
     ⟹ m ∈ gamma_ivl (fst (ivl_refine c a b)) ∧ n ∈ gamma_ivl (snd (ivl_refine c a b))"
  by (cases c)
     (auto dest: ivl_le_bounds_sound ivl_lt_bounds_sound intro: meet_ivl_gamma ivl_excl_sound)

instantiation ivl :: kind_int_domain
begin

definition top_ik_ivl :: "ikind ⇒ ivl" where "top_ik_ivl = ik_ivl"

definition of_int_ik_ivl :: "ikind ⇒ int ⇒ ivl" where
  "of_int_ik_ivl k n = Ivl (Fin (ik_norm k n)) (Fin (ik_norm k n))"

definition cast_ik_ivl :: "signed_overflow_policy ⇒ ikind option ⇒ ikind ⇒ ivl ⇒ ivl" where
  "cast_ik_ivl ovp sk = norm_ivl_ik ovp"

definition arith_ik_ivl :: "signed_overflow_policy ⇒ ikind ⇒ aop ⇒ bool ⇒ ivl ⇒ ivl ⇒ ivl" where
  "arith_ik_ivl ovp k op fits a b = norm_ivl_ik ovp k (ivl_aop op a b)"

definition cmp_ik_ivl :: "cop ⇒ ivl ⇒ ivl ⇒ bool option" where
  "cmp_ik_ivl = cop_of interval_less interval_eq"

definition meet_ik_ivl :: "ivl ⇒ ivl ⇒ ivl" where "meet_ik_ivl a b = a ⊓ b"

definition refine_ik_ivl :: "cop ⇒ ivl ⇒ ivl ⇒ ivl × ivl" where "refine_ik_ivl = ivl_refine"

definition raw_aop_ik_ivl :: "aop ⇒ ivl ⇒ ivl ⇒ ivl" where "raw_aop_ik_ivl = ivl_aop"

definition fits_ik_ivl :: "ikind ⇒ ivl ⇒ bool" where "fits_ik_ivl k a ⟷ a ≤ ik_ivl k"

definition answer_ik_ivl :: "ivl ⇒ answer" where "answer_ik_ivl = answer_of_ivl"

definition of_answer_ik_ivl :: "ikind option ⇒ answer ⇒ ivl" where
  "of_answer_ik_ivl sk q =
     (let r = (case sk of Some k ⇒ ik_ivl k | None ⇒ ⊤)
      in case q of QBot ⇒ ⊥ | QTop ⇒ r | QLifted d ⇒ int_ivl d ⊓ r)"

instance proof
  fix n m :: int and k :: ikind and a b :: ivl and op :: aop and c :: cop and r :: bool
  show "n ∈ ik_range k ⟹ n ∈ numeric_domain_class.gamma (top_ik k :: ivl)"
    by (simp add: top_ik_ivl_def)
  show "ik_norm k n ∈ numeric_domain_class.gamma (of_int_ik k n :: ivl)"
    by (simp add: of_int_ik_ivl_def less_eq_eint_def)
  show "(⋀k'. sk = Some k' ⟹ n ∈ ik_range k') ⟹ n ∈ numeric_domain_class.gamma a
      ⟹ ik_norm k n ∈ numeric_domain_class.gamma (cast_ik ovp sk k a)" for ovp sk
    by (simp add: cast_ik_ivl_def norm_ivl_ik_sound)
  show "(fits ⟹ m ∈ ik_range k ∧ n ∈ ik_range k) ⟹ m ∈ numeric_domain_class.gamma a ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ ik_norm k (aop_eval op m n) ∈ numeric_domain_class.gamma (arith_ik ovp k op fits a b)" for ovp fits
    by (simp add: arith_ik_ivl_def norm_ivl_ik_sound ivl_aop_sound)
  show "cmp_ik c a b = Some r ⟹ m ∈ numeric_domain_class.gamma a ⟹ n ∈ numeric_domain_class.gamma b ⟹ cop_eval c m n = r"
    unfolding cmp_ik_ivl_def
    by (rule cop_of_sound[OF interval_numeric_queries.less_sound interval_numeric_queries.eq_sound])
  show "m ∈ numeric_domain_class.gamma a ⟹ m ∈ numeric_domain_class.gamma b
      ⟹ m ∈ numeric_domain_class.gamma (meet_ik a b)"
    by (simp add: meet_ik_ivl_def meet_ivl_gamma)
  show "cop_eval c m n ⟹ m ∈ numeric_domain_class.gamma a ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ m ∈ numeric_domain_class.gamma (fst (refine_ik c a b))
        ∧ n ∈ numeric_domain_class.gamma (snd (refine_ik c a b))"
    by (simp add: refine_ik_ivl_def ivl_refine_sound)
  show "m ∈ numeric_domain_class.gamma a ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ aop_eval op m n ∈ numeric_domain_class.gamma (raw_aop_ik op a b)"
    by (simp add: raw_aop_ik_ivl_def ivl_aop_sound)
  show "fits_ik k a ⟹ n ∈ numeric_domain_class.gamma a ⟹ n ∈ ik_range k"
    using gamma_ivl_mono[of a "ik_ivl k"] by (auto simp: fits_ik_ivl_def simp del: ik_range_iff)
  show "n ∈ numeric_domain_class.gamma a ⟹ n ∈ gamma_answer (answer_ik a)"
    by (simp add: answer_ik_ivl_def)
  show "(⋀k. sk = Some k ⟹ n ∈ ik_range k) ⟹ n ∈ gamma_answer q
      ⟹ n ∈ numeric_domain_class.gamma (of_answer_ik sk q :: ivl)" for sk q
    by (cases sk; cases q)
       (auto simp: of_answer_ik_ivl_def gamma_int_dom_def meet_ivl_gamma top_ivl_def gamma_ivl_top
         simp del: ik_range_iff)
qed

end
