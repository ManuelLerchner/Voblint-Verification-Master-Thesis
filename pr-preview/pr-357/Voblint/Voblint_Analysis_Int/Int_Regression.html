Theory Int_Regression

theory Int_Regression
  imports Int_Tuple_Kind "Voblint_Nonrelational.Nonrelational_Analysis" "Voblint_VIMP.VIMP_Notation"
begin

section ‹Regression: the typed Interval-Congruence product, executed›

text ‹
  Each case runs the certified solver on a typed program inside Isabelle and pins the
  published value of one variable. Termination is decided by evaluation.
›

abbreviation it_state_at :: "kind_decls ⇒ imp_prog ⇒ cfg_node ⇒ int_fixpoint int_tuple tabs" where
  "it_state_at ds p ≡ nonrel.state_at default_sem_conf ds 0 Globals_Join is_global p []"

abbreviation it_equations :: "kind_decls ⇒ imp_prog ⇒ cfg_node × cfg_node list ⇒
    (cfg_node × cfg_node list, (unit, cfg_node list) global_unknown,
     (int_fixpoint int_tuple tst lifted, int_fixpoint int_tuple tst lifted) dg_state) strategy_tree" where
  "it_equations ds p ≡ nonrel.equations default_sem_conf ds 0 is_global p"

abbreviation f_var :: "vname ⇒ var_id" where
  "f_var x ≡ ScopedId (STR ''f'') x"

abbreviation f_result :: "kind_decls ⇒ imp_prog ⇒ vname ⇒ int_fixpoint int_tuple kd" where
  "f_result ds p x ≡ it_state_at ds p (FunctionResult (STR ''f'')) (f_var x)"

subsection ‹No overflow keeps the class›

text ‹
  The interval bounds ‹x + 3› by ‹101›, so the addition cannot overflow
  (Goblint's ▩‹no_ov›) and the congruence keeps ‹3ℤ›. The exit guard
  ‹x ≥ 99› meets the interval ‹[99, 101]›, and refining with ‹3ℤ› leaves ‹99›.
  The Congruence analysis alone answers top here (‹Congruence_Regression›).
›

definition odd_step_prog :: imp_prog where
  "odd_step_prog = program {
     fun f() { x = 0; while (x < 99) { x = x + 3; } }
     fun main() { f(); } }"

lemma odd_step_terminates_c:
  "TD_side_rule_Interp_solve_c Globals_Join (it_equations [] odd_step_prog)
     (cfg_exit (elab_cfg odd_step_prog []), []) ≠ None"
  by eval

lemma odd_step_terminates:
  "nonrel.terminates TYPE(int_fixpoint int_tuple) default_sem_conf [] 0 Globals_Join is_global odd_step_prog"
  unfolding nonrel.terminates_code
  using TD_side_rule_Interp.solve_dom_of_solve_c[OF odd_step_terminates_c] by simp

lemma odd_step_value:
  "f_result [] odd_step_prog (STR ''x'') = KD I32 (IT (Ivl (Fin 99) (Fin 99)) (mk_congruence 99 0))"
  by eval

subsection ‹A guard the components refute together›

text ‹‹x› is ‹0› or ‹10›, so ‹x ∈ [0, 10] ∩ 10ℤ›. Neither component alone refutes
  ‹x == 5›; their meet is empty, so the branch is dead and ‹y› stays ‹0›.›

definition guard_prog :: imp_prog where
  "guard_prog = program {
     fun f(x) { y = 0; if (x == 5) { y = 1; } }
     fun main() { f(0); f(10); } }"

lemma guard_value:
  "f_result [] guard_prog (STR ''y'') = KD I32 (IT (Ivl (Fin 0) (Fin 0)) (mk_congruence 0 0))"
  by eval

subsection ‹Unsigned kinds›

text ‹
  ‹x› is a \<^const>‹U8› stepping by ‹10› below ‹250›. The class ‹10ℤ› rounds the
  guarded interval to ‹[0, 240]›, so ‹x + 10› stays within the kind and the class
  survives the conversion back; the exit is exactly ‹250›. A constant that leaves the
  kind wraps in both components.
›

definition wrap_loop_prog :: imp_prog where
  "wrap_loop_prog = program {
     fun f() { x = 0; while (x < 250) { x = x + 10; } }
     fun main() { f(); } }"

definition wrap_prog :: imp_prog where
  "wrap_prog = program { fun f() { x = 250; x = x + 10; } fun main() { f(); } }"

lemma unsigned_values:
  "f_result [(f_var (STR ''x''), U8)] wrap_loop_prog (STR ''x'')
     = KD U8 (IT (Ivl (Fin 250) (Fin 250)) (mk_congruence 250 0))"
  "f_result [(f_var (STR ''x''), U8)] wrap_prog (STR ''x'')
     = KD U8 (IT (Ivl (Fin 4) (Fin 4)) (mk_congruence 4 0))"
  by eval+

end
