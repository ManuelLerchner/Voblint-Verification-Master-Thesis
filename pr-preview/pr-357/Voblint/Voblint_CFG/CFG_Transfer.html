Theory CFG_Transfer

theory CFG_Transfer
  imports CFG_Def
begin

section ‹What each kind of edge does to the store›

text ‹
  Three transfers, one per thing an edge can do.  ‹edge_collect› runs an ordinary edge on a
  whole set of stores at once, by lifting the single-store \<^const>‹edge_step› pointwise ---
  derived from the concrete step rather than restated, so the two cannot drift apart.
  ‹call_enter› builds the callee's opening store: evaluate the actuals in the caller,
  reset the locals, bind the formals.  ‹combine_collect› does the reverse at a return:
  keep the callee's globals, restore the caller's locals, and write the callee's return
  value into the destination.

  All the payload comes from the edge itself, so none of these needs the procedure table.
›

definition edge_collect :: "kenv ⇒ (var_id, exp) gedge_action ⇒ store set ⇒ store set" where
  "edge_collect κ a S = {t. ∃s∈S. t ∈ edge_step κ a s}"

lemma edge_collect_simps [simp]:
  "edge_collect κ EA_Nop S = S"
  "edge_collect κ (EA_Assign x a) S = {s(x := kind_conv κ (Some x) (eval a s)) | s. s ∈ S}"
  "edge_collect κ (EA_Special sc x) S = {t. ∃s∈S. t ∈ special_step κ sc x s}"
  "edge_collect κ (EA_Assume b) S = {s. s ∈ S ∧ truthy (eval b s)}"
  "edge_collect κ (EA_AssumeNot b) S = {s. s ∈ S ∧ ¬ truthy (eval b s)}"
  "edge_collect κ (EA_Body p) S = S"
  "edge_collect κ (EA_Ret e p) S =
     {s(ReturnVar := (case e of None ⇒ s ReturnVar | Some a ⇒ eval a s)) | s. s ∈ S}"
  "edge_collect κ (EA_Check l c) S = S"
  unfolding edge_collect_def by (auto split: if_splits)

lemma edge_collect_single:
  "edge_collect κ a {s} = edge_step κ a s"
  by (cases a) (auto split: if_splits)

lemma edge_collect_empty_set [simp]: "edge_collect κ a {} = {}"
  by (cases a) auto

lemma edge_collect_mono:
  assumes "S ⊆ T"
  shows "edge_collect κ a S ⊆ edge_collect κ a T"
  using assms by (cases a) auto

subsection ‹Return-value transfer›

text ‹Return-value rehydration at the caller: write the callee's return slot into the
  destination over the combined store (callee globals, caller locals), converted to the
  destination.  It is fixed by the call's destination ‹dst›, which the ‹CallEdge› already
  records --- no side lookup.›

definition combine_collect ::
    "kenv ⇒ var_id option ⇒ store ⇒ store ⇒ store" where
  "combine_collect κ dst s t = combine_assign dst (kind_conv κ dst (t ReturnVar)) (combine_env is_global s t)"

subsection ‹Call-entry transfer›

text ‹Caller-side entry transfer at a call.  The actuals are evaluated in the caller store,
  the callee locals are reset (\<^const>‹enter_state›, globals preserved), and the resulting
  values are bound to the callee formals, each converted to its formal.  All payload
  comes from the ‹CallEdge›, so the transfer needs no procedure table.  This is exactly
  the entered store produced by the source \<^const>‹pstep› ‹Call› rule.›

definition call_enter :: "kenv ⇒ (var_id, exp) gcall_action ⇒ store ⇒ store" where
  "call_enter κ ca s =
     (case ca of CallEdge dst pars actuals ⇒
        bind_formals pars (map2 (λx. kind_conv κ (Some x)) pars (map (λe. eval e s) actuals))
          (enter_state is_global s))"

lemma call_enter_CallEdge:
  "call_enter κ (CallEdge dst pars actuals) s
     = bind_formals pars (map2 (λx. kind_conv κ (Some x)) pars (map (λe. eval e s) actuals))
         (enter_state is_global s)"
  by (simp add: call_enter_def)

text ‹A parameterless call is exactly \<^const>‹enter_state›: no actuals to evaluate and no
  formals to bind.›
lemma call_enter_Nil [simp]:
  "call_enter κ (CallEdge dst [] []) s = enter_state is_global s"
  by (simp add: call_enter_CallEdge)

subsection ‹What an edge may change›

text ‹An edge may change only the variable it assigns, and a return only its result slot.›

fun edge_writes :: "(var_id, exp) gedge_action ⇒ var_id list" where
  "edge_writes (EA_Assign x a) = [x]"
| "edge_writes (EA_Special sc x) = [x]"
| "edge_writes (EA_Ret e p) = [ReturnVar]"
| "edge_writes _ = []"

text ‹The frame each concrete step respects: a name outside its write set keeps
  its value. A call entry may only bind its formals, and a return takes every
  global from the callee except the destination it assigns.›

lemma edge_step_frame:
  "t ∈ edge_step κ a s ⟹ x ∉ set (edge_writes a) ⟹ t x = s x"
  by (cases a) (auto split: if_splits)

lemma call_enter_frame:
  "is_global x ⟹ x ∉ set pars ⟹ call_enter κ (CallEdge dst pars args) s x = s x"
  by (simp add: call_enter_CallEdge bind_formals_other enter_state_def)

lemma combine_collect_frame:
  "is_global x ⟹ dst ≠ Some x ⟹ combine_collect κ dst s t x = t x"
  by (cases dst) (auto simp: combine_collect_def)

text ‹The write set runs in generated code.›

lemmas [code] = edge_writes.simps

subsection ‹What an edge reads›

text ‹An edge reads the variables of its expressions.›

definition edge_reads :: "(var_id, exp) gedge_action ⇒ var_id list" where
  "edge_reads a = sorted_list_of_set (⋃ (exp_vars ` set (edge_expressions a)))"

subsection ‹Finite global footprints›

text ‹The declared globals among those names, as duplicate-free lists a
  transfer folds over to read or publish them.›

definition global_names_in :: "('v ⇒ bool) ⇒ 'v list ⇒ 'v list" where
  "global_names_in G xs = remdups (filter G xs)"

lemma set_global_names_in [simp]:
  "set (global_names_in G xs) = {x ∈ set xs. G x}"
  by (simp add: global_names_in_def)

lemma distinct_global_names_in [simp]: "distinct (global_names_in G xs)"
  by (simp add: global_names_in_def)

definition edge_global_reads :: "(var_id ⇒ bool) ⇒ (var_id, exp) gedge_action ⇒ var_id list" where
  "edge_global_reads G a = global_names_in G (edge_reads a)"

text ‹The writes are those of the concrete step, so they follow the return slot.›

definition edge_global_writes :: "(var_id ⇒ bool) ⇒ (var_id, exp) gedge_action ⇒ var_id list" where
  "edge_global_writes G a = global_names_in G (edge_writes a)"

definition call_global_reads :: "(var_id ⇒ bool) ⇒ exp list ⇒ var_id list" where
  "call_global_reads G es = global_names_in G (sorted_list_of_set (⋃ (exp_vars ` set es)))"

lemma set_edge_global_writes [simp]:
  "set (edge_global_writes G a) = {x ∈ set (edge_writes a). G x}"
  by (simp add: edge_global_writes_def)

lemma distinct_global_footprints [simp]:
  "distinct (edge_global_reads G a)"
  "distinct (edge_global_writes G a)"
  "distinct (call_global_reads G es)"
  by (simp_all add: edge_global_reads_def edge_global_writes_def call_global_reads_def)

end
