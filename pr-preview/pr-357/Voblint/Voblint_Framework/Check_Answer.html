Theory Check_Answer

theory Check_Answer
  imports Contextual_Check_Report
begin

section ‹A domain's answer to the value of an expression›

text ‹
  A domain that decides a comparison or logical operator ‹e› three-valuedly
  answers ‹EvalInt e› with ‹bool_answer›: such an expression evaluates to
  ‹0› or ‹1›, so a decided truth value is the exact integer, and an undecided
  one is the interval ‹[0, 1]›.
›

definition bool_answer :: "bool option ⇒ answer" where
  "bool_answer r =
     (case r of
        Some b ⇒ answer_of_int (if b then 1 else 0)
      | None ⇒ answer_of_ivl (Ivl (Fin 0) (Fin 1)))"

lemma classify_answer_gamma:
  "a ≠ QBot ⟹ classify_answer a =
     Decided (if 0 ∉ gamma_answer a then Check_Proved
              else if 1 ∉ gamma_answer a then Check_Refuted else Check_Unknown)"
  using answer_excludes_iff[of 0 a] answer_excludes_iff[of 1 a]
  by (cases a) (auto simp del: answer_excludes.simps)

section ‹Deciding a check from an answer›

text ‹
  The one check consumer: ask the state's handler the check's question and read
  the verdict off the answer, as Goblint's ‹assert› analysis reads
  ‹Queries.eval_bool›. \<^const>‹classify_answer› reads an answer that
  admits no value as \<^const>‹Dead›, but the consumer reports it as
  \<^const>‹Check_Unknown›: a check is reported dead only where the state
  itself is \<^const>‹Bot›, never from the answer to its query.
›

text ‹
  The question is the check's truth value, which a language spells with its own
  expressions: ‹tx c› evaluates to ‹1› where ‹c› holds and to ‹0› elsewhere.
›

definition answer_check ::
    "(exp ⇒ exp) ⇒ ('v ⇒ query ⇒ answer) ⇒ exp ⇒ 'v ⇒ check_result" where
  "answer_check tx q c v =
     (case classify_answer (q v (EvalInt (tx c))) of Decided r ⇒ r | _ ⇒ Check_Unknown)"

lemma answer_check_sound:
  assumes truth: "eval (tx c) s = (if truthy (eval c s) then 1 else 0)"
    and holds: "eval_holds (EvalInt (tx c)) (q v (EvalInt (tx c))) s"
  shows answer_check_proved: "answer_check tx q c v = Check_Proved ⟹ truthy (eval c s)"
    and answer_check_refuted: "answer_check tx q c v = Check_Refuted ⟹ ¬ truthy (eval c s)"
proof -
  let ?a = "q v (EvalInt (tx c))"
  have val: "(if truthy (eval c s) then 1 else 0) ∈ gamma_answer ?a"
    using holds truth by simp
  then have "?a ≠ QBot" by auto
  then show "answer_check tx q c v = Check_Proved ⟹ truthy (eval c s)"
    and "answer_check tx q c v = Check_Refuted ⟹ ¬ truthy (eval c s)"
    using val by (auto simp: answer_check_def classify_answer_gamma split: if_splits)
qed

end
