Theory DG_Reader_Transport

theory DG_Reader_Transport
  imports Routed_Context
begin

section ‹Reading a whole equation system through a pair of readers›

text ‹
  A ‹reader› here maps one carrier to another --- typically an implementation
  value to the mathematical value it stands for. This theory proves, once, that
  reading commutes with everything an equation is built from: a node's value,
  the side effects it emits, and the unknowns it queries. The only properties
  any proof in the chain uses are that a reader preserves ‹bot› and ‹(⊔)›, so
  they are the locale's two assumptions and the development is generic in both
  carriers. A concrete reader becomes a one-line interpretation.
›

subsection ‹Dependency commutation for the generator›

text ‹
  The generator trees are non-branching (‹QueryL› then ‹QueryG› then ‹Answer›/‹Side›),
  so the queried-unknown set is structural: independent of the analysis step values and
  the valuation.  Hence dependencies transport verbatim.
›

subsection ‹Carrier-generic whole-CFG commute›

text ‹
  The commute facts below only ever use that a reader preserves ‹bot› and ‹(⊔)›; no
  proof in the chain inspects ‹default_st_to_fun› or ‹gabs_state› itself.
  ‹dg_reader_commute_gen› factors that out: a pair of local/global readers ‹Floc›/‹Fglob›
  satisfying those two laws, from which every whole-tree and whole-equation-system commute
  fact in this chain is proved once.
›

text ‹
  Reading a pair componentwise is the datatype's own map,
  \<^const>‹map_dg_state›; its selector equations join the simplifier.
›

declare dg_state.map_sel [simp]

text ‹
  The locale assumes only that both readers preserve ‹bot› and joins;
  monotonicity and the commutation with ‹map_dg_state› follow.
›

locale dg_reader_commute_gen =
  fixes Floc :: "'a::bounded_semilattice_sup_bot ⇒ 'a2::bounded_semilattice_sup_bot"
    and Fglob :: "'b::bounded_semilattice_sup_bot ⇒ 'b2::bounded_semilattice_sup_bot"
  assumes Floc_bot: "Floc bot = bot"
      and Floc_sup: "⋀x y. Floc (x ⊔ y) = Floc x ⊔ Floc y"
      and Fglob_bot: "Fglob bot = bot"
      and Fglob_sup: "⋀x y. Fglob (x ⊔ y) = Fglob x ⊔ Fglob y"
begin

lemma Floc_mono: "x ≤ y ⟹ Floc x ≤ Floc y"
  by (metis Floc_sup le_iff_sup)

lemma Fglob_mono: "x ≤ y ⟹ Fglob x ≤ Fglob y"
  by (metis Fglob_sup le_iff_sup)

lemma map_dg_state_bot [simp]:
  "map_dg_state Floc Fglob (bot :: ('a,'b) dg_state) = bot"
  by (simp add: bot_dg_state_def Floc_bot Fglob_bot)

lemma map_dg_state_sup:
  "map_dg_state Floc Fglob (a ⊔ b :: ('a,'b) dg_state)
     = map_dg_state Floc Fglob a ⊔ map_dg_state Floc Fglob b"
  by (simp add: sup_dg_state_def Floc_sup Fglob_sup)

subsubsection ‹Bundled per-tree transport relation›

text ‹
  ‹dg_tree_st_commute› bundles the three facts one strategy tree must
  transport: its value, its side effects per key, and its dependencies.
›

definition dg_tree_st_commute ::
  "('u + 'k ⇒ ('a,'b) dg_state) ⇒ ('u, 'k, ('a,'b) dg_state) strategy_tree
    ⇒ ('u, 'k, ('a2,'b2) dg_state) strategy_tree ⇒ bool"
where
  "dg_tree_st_commute σ_st t_st t_abs ⟷
     map_dg_state Floc Fglob (traverse_rhs t_st σ_st) =
       traverse_rhs t_abs (map_dg_state Floc Fglob ∘ σ_st) ∧
     (∀k. map_dg_state Floc Fglob (sides_of_rhs t_st σ_st k) =
       sides_of_rhs t_abs (map_dg_state Floc Fglob ∘ σ_st) k) ∧
     dep_aux σ_st t_st =
       dep_aux (map_dg_state Floc Fglob ∘ σ_st) t_abs"

text ‹
  The same relation between two ∗‹programs›, bundled with the
  well-formedness each side needs for the fold's observations to be well posed.
  Carrying \<^const>‹sp_wf› inside the relation is what lets a list induction
  over two contribution lists keep it: a ‹list_all2› of this relation says every
  pair transports and every element runs its continuation once.
›

definition dg_prog_st_commute ::
  "('u + 'k ⇒ ('a,'b) dg_state)
   ⇒ ('u, 'k, ('a,'b) dg_state, ('a,'b) dg_state) strategy_program
   ⇒ ('u, 'k, ('a2,'b2) dg_state, ('a2,'b2) dg_state) strategy_program ⇒ bool"
where
  "dg_prog_st_commute σ_st p_st p_abs ⟷
     sp_wf p_st ∧ sp_wf p_abs
     ∧ dg_tree_st_commute σ_st (sp_compile p_st) (sp_compile p_abs)"

subsubsection ‹Routed heterogeneous CALL/COMB transport›

text ‹
  \<^const>‹routed_call_program›/\<^const>‹routed_entry_seed_programs›
  (\<^theory>‹Voblint_Framework.Routed_Call_Programs›) are the canonical
  heterogeneous routing shape: parametric only in a routing
  function ‹route› and a seed-key injection ‹seed›, with the seed payload
  carried on the ‹dg_local› half so ‹'D›/‹'G› stay independent. The lemmas
  below transport \<^const>‹routed_call_program›, so any context-sensitive analysis
  instantiating ‹cmb› at that constant discharges CALL/COMB transport once here
  rather than re-deriving its own tree-commute reasoning.

  The specification's own enter and combine appear inside the routed tree as
  compiled sub-trees, so their transport hypotheses are themselves tree commutes.
  Sequencing them is what ‹dg_tree_st_commute_seqcomp› does: a bind commutes when
  its head commutes and its continuation commutes at the head's answer.
›

lemma dg_tree_st_commute_seqcomp:
  assumes head: "dg_tree_st_commute σ_st t_st t_abs"
    and tail: "dg_tree_st_commute σ_st (k_st (traverse_rhs t_st σ_st))
                 (k_abs (traverse_rhs t_abs (map_dg_state Floc Fglob ∘ σ_st)))"
  shows "dg_tree_st_commute σ_st (sp_lift_tree t_st k_st) (sp_lift_tree t_abs k_abs)"
  using head tail
  unfolding dg_tree_st_commute_def
  by (simp add: map_dg_state_sup)

lemma dg_tree_st_commute_answer:
  "dg_tree_st_commute σ_st (Answer d) (Answer (map_dg_state Floc Fglob d))"
  by (simp add: dg_tree_st_commute_def)

text ‹A local-only transfer compiles to a single answer, so its commute is the
  reader equation on the pure function alone -- no tree reasoning, and no
  hypothesis about the global half beyond the reader's own ‹bot› law.›

lemma dg_tree_st_commute_local_transfer:
  assumes "Floc (f d) = F (Floc d)"
  shows "dg_tree_st_commute σ_st
           (sp_compile_with (λx. DG x bot) (local_transfer f (mk_dg_man d gk)))
           (sp_compile_with (λx. DG x bot) (local_transfer F (mk_dg_man (Floc d) gk)))"
  by (simp add: dg_tree_st_commute_def local_transfer_def mk_dg_man_def
      dg_state.map_sel bot_dg_state_def assms Floc_bot Fglob_bot)

text ‹
  What a compiled entry must satisfy to transport. Entry answers a list of
  caller-continuation/callee-entry pairs, which is not the solver's carrier, so the
  reader equation cannot be stated on the answer of a compiled tree the way the edge and
  combine transports are. It is stated on the program instead: run both entries
  against continuations that already agree on pairs mapped componentwise by the readers, and the
  two trees agree. This observes the entry program exactly as \<^const>‹enter_runs›
  does, and for the same reason.
›

definition dg_enter_st_commute ::
  "('u + 'k ⇒ ('a,'b) dg_state)
   ⇒ ('u,'k,('a,'b) dg_state,'a enter_result list) strategy_program
   ⇒ ('u,'k,('a2,'b2) dg_state,'a2 enter_result list) strategy_program ⇒ bool"
where
  "dg_enter_st_commute σ_st T_st T_abs ⟷
     (∀K_st K_abs.
        (∀ps. dg_tree_st_commute σ_st (K_st ps) (K_abs (map (map_prod Floc Floc) ps)))
          ⟶ dg_tree_st_commute σ_st (T_st K_st) (T_abs K_abs))"

end
end
