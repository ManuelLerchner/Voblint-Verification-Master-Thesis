Theory MCP_Registry

theory MCP_Registry
  imports MCP_Spec Check_Answer
begin

section ‹Registered analyses, run as one›

text ‹
  A registry assigns each analysis one field of a combined state: the component that
  runs on it, whether the field is live, what a solved field publishes, what a
  published value describes, whether it is empty, and how it answers a query. The
  registry runs any distinct, nonempty list of its analyses as one component
  (\<^const>‹mcp_combine›), decides checks from the meet of their answers, and is
  sound whenever each field is. Nothing here depends on the language: the language
  enters through its evaluator ‹ev›, its return slot ‹rv›, the conversion and the
  global predicate of each program, and ‹tx›, the expression that asks for a
  check's truth value.
›

locale mcp_registry =
  fixes κ :: kenv
    and tx :: "exp ⇒ exp"
    and field ::
      "(var_id ⇒ bool) ⇒ 'p ⇒ 'a ⇒ ('s::{semilattice_sup, order_bot} lifted) local_spec"
    and live :: "'a ⇒ 's ⇒ bool"
    and rd :: "(var_id ⇒ bool) ⇒ 's ⇒ 'v::bot"
    and val_gamma :: "'a ⇒ 'v ⇒ store set"
    and val_empty :: "'a ⇒ 'v ⇒ bool"
    and field_empty :: "'p ⇒ 'a ⇒ 's ⇒ bool"
    and answer :: "'a ⇒ 'v ⇒ channel ⇒ query ⇒ answer"
  assumes truth: "eval (tx c) s = (if truthy (eval c s) then 1 else 0)"
    and field_sound: "sound_local_spec κ
         (λx. gamma_lift (val_gamma a) (map_lift (rd is_global) x)) (field is_global p a)"
    and field_frame: "a ≠ b ⟹
         mcp_frame (field 𝒢 p a) (λx. gamma_lift (val_gamma b) (map_lift (rd 𝒢) x))"
    and single_entry_field: "single_entry (field 𝒢 p a)"
    and dead: "¬ live a r ⟹ val_gamma a (rd 𝒢 r) = {}"
    and field_empty_rd: "field_empty p a r = val_empty a (rd is_global r)"
    and val_empty_gamma: "val_empty a v ⟹ val_gamma a v = {}"
    and val_gamma_bot: "val_gamma a ⊥ = {}"
    and answer_sound: "s ∈ val_gamma a v ⟹ eval_query.channel_holds ch s
         ⟹ eval_holds q (answer a v ch q) s"
begin

abbreviation part_gamma :: "(var_id ⇒ bool) ⇒ 'a ⇒ 's lifted ⇒ store set" where
  "part_gamma 𝒢 a x ≡ gamma_lift (val_gamma a) (map_lift (rd 𝒢) x)"

definition act_gamma :: "'a list ⇒ 'v ⇒ store set" where
  "act_gamma as v = (⋂a ∈ set as. val_gamma a v)"

subsection ‹Each analysis answers what it knows›

text ‹
  A field answers a query as its published value does, so an analysis written
  without a query handler becomes a provider on the combined state
  (\<^const>‹with_qry›), as every Goblint analysis answers ‹EvalInt› from its domain.
  An answer may itself ask the channel, as Goblint's ▩‹base› asks about the operands
  of the expression it is asked about.
›

definition part_answer ::
    "(var_id ⇒ bool) ⇒ 'a ⇒ channel ⇒ 's lifted ⇒ query ⇒ answer" where
  "part_answer 𝒢 a ch x q = (case x of Bot ⇒ ⊤ | Lifted r ⇒ answer a (rd 𝒢 r) ch q)"

lemma part_answer_sound:
  "s ∈ part_gamma 𝒢 a x ⟹ eval_query.channel_holds ch s
     ⟹ eval_holds q (part_answer 𝒢 a ch x q) s"
  by (cases x) (auto simp: part_answer_def intro: answer_sound)

definition mcp_field :: "(var_id ⇒ bool) ⇒ 'p ⇒ 'a ⇒ 's lifted local_spec" where
  "mcp_field 𝒢 p a = with_qry (part_answer 𝒢 a) (field 𝒢 p a)"

lemma mcp_field_sound:
  "sound_local_spec κ (part_gamma is_global a)
     (mcp_field is_global p a)"
  unfolding mcp_field_def
  by (rule with_qry_sound[OF field_sound]) (rule part_answer_sound)

lemma mcp_field_frame: "a ≠ b ⟹ mcp_frame (mcp_field 𝒢 p a) (part_gamma 𝒢 b)"
  unfolding mcp_field_def by (rule with_qry_frame[OF field_frame])

subsection ‹The active analyses, run as one›

text ‹
  The combined state is unreachable as soon as one active analysis says its field
  is, as Goblint's ‹MCP› raises ‹Deadcode› when one component does. The
  normalization changes no concretization: a field that describes nothing empties
  the intersection already.
›

definition mcp_norm :: "'a list ⇒ 's lifted ⇒ 's lifted" where
  "mcp_norm as x =
     (case x of Bot ⇒ Bot | Lifted r ⇒ if list_all (λa. live a r) as then x else Bot)"

lemma mcp_gamma_norm:
  "mcp_gamma (map (part_gamma 𝒢) as) (mcp_norm as x) = mcp_gamma (map (part_gamma 𝒢) as) x"
  by (cases x) (auto simp: mcp_norm_def mcp_gamma_def list_all_iff dest: dead)

definition mcp_comp :: "'a list ⇒ (var_id ⇒ bool) ⇒ 'p ⇒ 's lifted local_spec" where
  "mcp_comp as 𝒢 p = map_local_spec (mcp_norm as) (mcp_combine (map (mcp_field 𝒢 p) as))"

theorem mcp_comp_sound:
  assumes "distinct as" and "as ≠ []"
  shows "sound_local_spec κ
           (mcp_gamma (map (part_gamma is_global) as)) (mcp_comp as is_global p)"
proof -
  let ?gcs = "map (λa. (part_gamma is_global a, mcp_field is_global p a)) as"
  have "sound_local_spec κ (mcp_gamma (map fst ?gcs)) (mcp_combine (map snd ?gcs))"
    by (rule mcp_combine_sound)
       (auto simp: mcp_field_sound assms intro!: mcp_independent_map mcp_field_frame)
  then show ?thesis
    unfolding mcp_comp_def
    by (intro map_local_spec_sound) (simp_all add: comp_def mcp_gamma_norm)
qed

lemma single_entry_mcp_comp: "as ≠ [] ⟹ single_entry (mcp_comp as 𝒢 p)"
  unfolding mcp_comp_def mcp_field_def
  by (intro single_entry_map_local_spec single_entry_mcp_combine)
     (auto simp: single_entry_field single_entry_with_qry)

subsection ‹What the combined state publishes›

text ‹The combined state denotes the stores every active analysis admits, and it is
  empty as soon as one active analysis reports its own field empty.›

lemma mcp_gamma_rd:
  "mcp_gamma (map (part_gamma 𝒢) as) x = gamma_lift (act_gamma as) (map_lift (rd 𝒢) x)"
  if "as ≠ []"
  using that by (cases x) (auto simp: mcp_gamma_def act_gamma_def)

lemma act_gamma_bot: "as ≠ [] ⟹ act_gamma as ⊥ = {}"
  by (auto simp: act_gamma_def val_gamma_bot)

definition mcp_emp :: "'a list ⇒ 'p ⇒ 's ⇒ bool" where
  "mcp_emp as p r = list_ex (λa. field_empty p a r) as"

definition mcp_empty_v :: "'a list ⇒ 'v ⇒ bool" where
  "mcp_empty_v as v = list_ex (λa. val_empty a v) as"

lemma mcp_emp_rd: "mcp_emp as p r = mcp_empty_v as (rd is_global r)"
  by (simp add: mcp_emp_def mcp_empty_v_def field_empty_rd)

lemma mcp_empty_v_gamma: "mcp_empty_v as v ⟹ act_gamma as v = {}"
  by (auto simp: mcp_empty_v_def act_gamma_def list_ex_iff dest: val_empty_gamma)

lemma sound_emptiness_mcp: "sound_emptiness (mcp_empty_v as) (act_gamma as)"
  by (rule sound_emptinessI) (rule mcp_empty_v_gamma)

subsection ‹Deciding a check from every active analysis›

text ‹
  The combined answer meets the active analyses' answers, as ‹MCP.query'› does,
  and a check is decided from it as Goblint's ‹assert› reads ‹Queries.eval_bool›. The
  question goes through the channel the combined answer closes over itself
  (\<^const>‹ask_rec›), so an answer that asks about a subexpression reaches every
  active analysis, and a query cycle answers ‹⊤›.
›

definition mcp_answer :: "'a list ⇒ channel ⇒ 'v ⇒ query ⇒ answer" where
  "mcp_answer as ch v q = fold (λa r. r ⊓ answer a v ch q) as ⊤"

lemma mcp_answer_fold_sound:
  "∀a ∈ set as. s ∈ val_gamma a v ⟹ eval_query.channel_holds ch s
   ⟹ eval_holds q r s ⟹ eval_holds q (fold (λa r. r ⊓ answer a v ch q) as r) s"
  by (induction as arbitrary: r) (auto intro: eval_query.inf_sound answer_sound)

lemma mcp_answer_sound:
  "s ∈ act_gamma as v ⟹ eval_query.channel_holds ch s
     ⟹ eval_holds q (mcp_answer as ch v q) s"
  unfolding mcp_answer_def act_gamma_def by (rule mcp_answer_fold_sound) auto

definition mcp_ask :: "'a list ⇒ 'v ⇒ channel" where
  "mcp_ask as v = ask_rec (mcp_answer as) query_depth {} v"

lemma mcp_ask_sound:
  assumes s: "s ∈ act_gamma as v"
  shows "eval_query.channel_holds (mcp_ask as v) s"
proof -
  show ?thesis
    unfolding mcp_ask_def by (rule ask_rec_sound) (rule mcp_answer_sound[OF s])
qed

definition mcp_classify :: "'a list ⇒ exp ⇒ 'v ⇒ check_result" where
  "mcp_classify as = answer_check tx (mcp_ask as)"

lemma mcp_classify_sound:
  assumes s: "s ∈ act_gamma as v"
  shows mcp_classify_proved: "mcp_classify as c v = Check_Proved ⟹ truthy (eval c s)"
    and mcp_classify_refuted: "mcp_classify as c v = Check_Refuted ⟹ ¬ truthy (eval c s)"
  using answer_check_sound[where tx = tx and q = "mcp_ask as" and c = c and v = v,
      OF truth eval_query.channel_holdsD[OF mcp_ask_sound[OF s]]]
  by (simp_all add: mcp_classify_def)

end

end
