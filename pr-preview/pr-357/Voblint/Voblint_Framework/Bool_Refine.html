Theory Bool_Refine

theory Bool_Refine
  imports Main
begin

section ‹Refining a state by a Boolean condition›

text ‹
  A guard narrows the state by the truth value its condition must have. The Boolean
  structure of the condition is the same for every language: a negation flips the
  required truth value; a conjunction that must hold and a disjunction that must fail
  narrow by both operands in turn; a disjunction that must hold, or a conjunction that
  must fail, holds through one of its operands, so each operand that can still have the
  required value narrows the state on its own and the results are joined, as Goblint's
  ▩‹BaseInvariant› joins the alternatives of a disjunction (▩‹baseInvariant.ml›,
  lines 724-731 at ▩‹0dc12d355›). An operand is dropped when its forward value rules the
  required truth value out. When neither operand can have it, no store satisfies the
  condition, and the language decides the result: bottom, as Goblint's ▩‹Deadcode›,
  or the state unchanged when its caller discards the branch by other means.

  A language supplies a view of its conditions as negation, conjunction, disjunction or
  atom, a size that decreases on operands, how an atom narrows a state, the
  feasibility test, the join and the infeasible result.
›

datatype 'e bool_view = BV_Not 'e | BV_And 'e 'e | BV_Or 'e 'e | BV_Atom

text ‹The parameters of that refinement. The size bound on operands makes the recursion
  on the condition well-founded.›

locale bool_refine =
  fixes view :: "'e ⇒ 'e bool_view" and sz :: "'e ⇒ nat"
    and atom :: "'e ⇒ bool ⇒ 's ⇒ 's" and feas :: "'e ⇒ bool ⇒ 's ⇒ bool"
    and join :: "'s ⇒ 's ⇒ 's" and dead :: "'s ⇒ 's"
  assumes sz_not: "view e = BV_Not a ⟹ sz a < sz e"
    and sz_and: "view e = BV_And a b ⟹ sz a < sz e ∧ sz b < sz e"
    and sz_or: "view e = BV_Or a b ⟹ sz a < sz e ∧ sz b < sz e"
begin

function brefine :: "'e ⇒ bool ⇒ 's ⇒ 's" where
  "brefine e pol σ =
     (case view e of
        BV_Not a ⇒ brefine a (¬ pol) σ
      | BV_And a b ⇒
          if pol then brefine b True (brefine a True σ)
          else (case (feas a False σ, feas b False σ) of
                  (True, True) ⇒ join (brefine a False σ) (brefine b False σ)
                | (True, False) ⇒ brefine a False σ
                | (False, True) ⇒ brefine b False σ
                | (False, False) ⇒ dead σ)
      | BV_Or a b ⇒
          if pol then (case (feas a True σ, feas b True σ) of
                  (True, True) ⇒ join (brefine a True σ) (brefine b True σ)
                | (True, False) ⇒ brefine a True σ
                | (False, True) ⇒ brefine b True σ
                | (False, False) ⇒ dead σ)
          else brefine b False (brefine a False σ)
      | BV_Atom ⇒ atom e pol σ)"
  by pat_completeness auto
termination
  by (relation "measure (λ(e, pol, σ). sz e)")
     (auto dest: sz_not sz_and sz_or)

declare brefine.simps [simp del]

text ‹
  A store that satisfies the condition with the required truth value stays described:
  the atom's narrowing, the feasibility test and the join are sound, and the view
  reflects how the language evaluates negation, conjunction and disjunction.
›

lemma brefine_sound:
  fixes ev :: "'e ⇒ 't ⇒ int" and γ :: "'s ⇒ 't set"
  assumes atom_sound: "⋀e pol σ t. view e = BV_Atom ⟹ t ∈ γ σ ⟹ (ev e t ≠ 0) = pol
        ⟹ t ∈ γ (atom e pol σ)"
    and ev_not: "⋀e a t. view e = BV_Not a ⟹ (ev e t ≠ 0) = (ev a t = 0)"
    and ev_and: "⋀e a b t. view e = BV_And a b ⟹ (ev e t ≠ 0) = (ev a t ≠ 0 ∧ ev b t ≠ 0)"
    and ev_or: "⋀e a b t. view e = BV_Or a b ⟹ (ev e t ≠ 0) = (ev a t ≠ 0 ∨ ev b t ≠ 0)"
    and feas_sound: "⋀e pol σ t. t ∈ γ σ ⟹ (ev e t ≠ 0) = pol ⟹ feas e pol σ"
    and join1: "⋀x y t. t ∈ γ x ⟹ t ∈ γ (join x y)"
    and join2: "⋀x y t. t ∈ γ y ⟹ t ∈ γ (join x y)"
  shows "t ∈ γ σ ⟹ (ev e t ≠ 0) = pol ⟹ t ∈ γ (brefine e pol σ)"
proof (induction e pol σ rule: brefine.induct)
  case (1 e pol σ)
  show ?case
  proof (cases "view e")
    case (BV_Not a)
    with 1 ev_not[OF BV_Not, of t] show ?thesis by (subst brefine.simps) simp
  next
    case (BV_And a b)
    show ?thesis
    proof (cases pol)
      case True
      with 1 ev_and[OF BV_And, of t] BV_And show ?thesis by (subst brefine.simps) simp
    next
      case False
      with 1 ev_and[OF BV_And, of t] BV_And feas_sound[of t σ a False] feas_sound[of t σ b False]
      show ?thesis
        by (subst brefine.simps) (auto split: bool.splits intro: join1 join2)
    qed
  next
    case (BV_Or a b)
    show ?thesis
    proof (cases pol)
      case True
      with 1 ev_or[OF BV_Or, of t] BV_Or feas_sound[of t σ a True] feas_sound[of t σ b True]
      show ?thesis
        by (subst brefine.simps) (auto split: bool.splits intro: join1 join2)
    next
      case False
      with 1 ev_or[OF BV_Or, of t] BV_Or show ?thesis by (subst brefine.simps) simp
    qed
  next
    case BV_Atom
    with 1 atom_sound show ?thesis by (subst brefine.simps) simp
  qed
qed

end

end
