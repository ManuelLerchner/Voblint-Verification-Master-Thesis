section ‹Reading one name of a combined state›

text ‹The two pointwise equations, so a proof about a single variable never has
  to unfold \<^const>‹combine_env›. The nesting law is what lets a combine step
  read a returned value straight out of the callee exit: every call site owns
  ‹ret_var› as its own compiler-internal name and never a user-declared global
  (‹reserved_ret_var›), so routing it through \<^const>‹combine_env› a second
  time would change nothing.›

lemma combine_env_local_eq [simp]:
  "¬ 𝒢 x ⟹ combine_env 𝒢 sc se x = sc x"
  by (simp add: combine_env_def)

lemma combine_env_global_eq [simp]:
  "𝒢 x ⟹ combine_env 𝒢 sc se x = se x"
  by (simp add: combine_env_def)

lemma combine_env_combine_env_left [simp]:
  "combine_env 𝒢 (combine_env 𝒢 dc g) (combine_env 𝒢 de g) = combine_env 𝒢 dc g"
  by (auto simp: combine_env_def)

end
