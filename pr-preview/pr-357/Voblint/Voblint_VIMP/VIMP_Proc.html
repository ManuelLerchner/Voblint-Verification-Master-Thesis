Theory VIMP_Proc

theory VIMP_Proc
  imports VIMP_Special "HOL-IMP.Star"
begin

section ‹Procedure commands and activation frames›

text ‹
  A call evaluates its actuals in the caller store, binds them in a fresh
  activation, and pushes a frame holding the caller store and the optional
  destination. ‹Restore› and ‹Unwind› never occur in source programs:
  ‹Restore› marks the activation boundary inside sequential syntax, and
  ‹Unwind› is the state after a ‹Return› has published its value through
  ‹ret_var›, discarding pending commands up to the nearest ‹Restore›.

  A check carries a label naming where it was written. The semantics never reads
  it; compilation copies it onto the check's edge, and the analysis result lists
  each check under it, so a report can be matched to its source check by label
  instead of by position in a list. The CLI parser writes the line and column of
  the check's ▩‹__voblint_check› keyword; a program written in Isabelle
  notation carries the placeholder ‹(0, 0)›.
›

type_synonym check_label = "nat × nat"

text ‹Every statement carries an annotation ‹'a›. Source programs written by hand or
  produced by the parser carry ‹()›; elaboration replaces it with a statement identity.
  The semantics never reads it: a step copies the annotation of the command it reduces
  onto what it leaves behind, so a residual still names the statement it came from. The
  variable and expression payloads are parameters too, so a later resolved or typed
  front end instantiates the same command syntax.›
datatype ('a, 'v, 'e) gcom =
    is_SKIP: SKIP 'a
  | Assign 'a (assign_var: 'v) (assign_rhs: 'e)
  | Check  'a check_label (check_cond: 'e)
  | Seq    (seq_first: "('a, 'v, 'e) gcom") (seq_second: "('a, 'v, 'e) gcom")
  | If     'a (if_cond: 'e) (if_then: "('a, 'v, 'e) gcom") (if_else: "('a, 'v, 'e) gcom")
  | While  'a (while_cond: 'e) (while_body: "('a, 'v, 'e) gcom")
  | Call   'a (call_dest: "'v option") (call_proc: pname) (call_args: "'e list")
  | Return 'a (return_val: "'e option")
  | is_Restore: Restore 'a
  | Unwind 'a

type_synonym 'a acom = "('a, vname, src_exp) gcom"
type_synonym src_com = "unit acom"

text ‹The expressions a command evaluates, as a list for the executable checks.›

fun com_exps :: "('a, 'v, 'e) gcom ⇒ 'e list" where
  "com_exps (SKIP _) = []"
| "com_exps (Assign _ _ e) = [e]"
| "com_exps (Check _ _ e) = [e]"
| "com_exps (Seq c1 c2) = com_exps c1 @ com_exps c2"
| "com_exps (If _ b c1 c2) = b # com_exps c1 @ com_exps c2"
| "com_exps (While _ b c) = b # com_exps c"
| "com_exps (Call _ _ _ es) = es"
| "com_exps (Return _ e) = (case e of Some e ⇒ [e] | None ⇒ [])"
| "com_exps (Restore _) = []"
| "com_exps (Unwind _) = []"

lemma set_com_exps [simp]: "set (com_exps c) = set3_gcom c"
  by (induction c) (auto split: option.splits)

text ‹The discriminators read as the equations they replace, so automation that used to
  substitute ‹c = SKIP› still finds the constructor.›
lemma is_SKIP_iff [simp]: "is_SKIP c ⟷ (∃i. c = SKIP i)"
  by (cases c) auto

lemma is_Restore_iff [simp]: "is_Restore c ⟷ (∃i. c = Restore i)"
  by (cases c) auto

text ‹Activation -- fresh locals and formal binding -- happens at the call site,
  so a declaration records only what a call site looks up by name.›
record ('a, 'v, 'e) gproc_decl =
  formals :: "'v list"
  body    :: "('a, 'v, 'e) gcom"

type_synonym proc_decl = "(unit, vname, src_exp) gproc_decl"

text ‹The reserved local ‹ret_var› carries a published result to the restore
  boundary. It is absent from source programs, so a value-less completion
  leaves it at ‹enter_state›'s zero.›
definition ret_var :: vname where
  "ret_var = STR ''#ret''"

type_synonym ('a, 'v, 'e) gptable = "pname ⇒ ('a, 'v, 'e) gproc_decl option"
type_synonym 'a ptable = "('a, vname, src_exp) gptable"
type_synonym proc_table = "unit ptable"

text ‹Formal binding is the same fold over the concrete store, over every abstract
  state, and over a typed store keyed by variable identities, so it is one
  polymorphic abbreviation.›
abbreviation bind_formals :: "'k list ⇒ 'v list ⇒ ('k ⇒ 'v) ⇒ 'k ⇒ 'v" where
  "bind_formals xs vs s ≡ fold (λ(x, v) st. st(x := v)) (zip xs vs) s"

lemma bind_formals_other:
  "x ∉ set xs ⟹ bind_formals xs vs s x = s x"
proof (induction xs arbitrary: vs s)
  case Nil then show ?case by simp
next
  case (Cons y ys)
  then show ?case by (cases vs) simp_all
qed

text ‹Whole procedure entry, for the same reason: reset every local to a chosen
  value, keep the globals, then bind the formals to the actuals evaluated in the
  caller's own state. Which state and which evaluator is all that separates the
  concrete entry from an abstract one -- the concrete semantics enters with its
  evaluator and the reset value ‹0›, an abstract domain with its own evaluator and its
  own ‹top› -- so it is one definition, not a concrete one with an ‹_abs› copy
  beside it.›
definition enter_binding ::
  "('k ⇒ bool) ⇒ 'a ⇒ ('e ⇒ ('k ⇒ 'a) ⇒ 'a)
   ⇒ 'k list ⇒ 'e list ⇒ ('k ⇒ 'a) ⇒ ('k ⇒ 'a)"
where
  "enter_binding 𝒢 reset_val ev xs es s =
     bind_formals xs (map (λe. ev e s) es) (enter_frame 𝒢 reset_val s)"

text ‹The return-slot write is a pure per-variable update, generic in the codomain:
  the concrete semantics uses it at a store of integers, and every abstract domain's
  state of abstract values reuses the same definition rather than restating an
  ‹_abs› copy of it.›
fun combine_assign :: "'k option ⇒ 'a ⇒ ('k ⇒ 'a) ⇒ ('k ⇒ 'a)" where
  "combine_assign None _ s = s"
| "combine_assign (Some x) v s = s(x := v)"

section ‹Source-program well-formedness›

text ‹
  Source syntax excludes the runtime markers \<^const>‹Restore› and
  \<^const>‹Unwind›. The reserved \<^const>‹ret_var› belongs to the
  call mechanism, so source expressions, assignments, destinations, and formal
  parameters cannot mention it.
›

subsection ‹Finite syntactic variable sets›

fun com_vnames :: "'a acom ⇒ vname set" where
  "com_vnames (SKIP _) = {}"
| "com_vnames (Assign _ x a) = insert x (exp_vnames a)"
| "com_vnames (Check _ l c) = exp_vnames c"
| "com_vnames (Seq c1 c2) = com_vnames c1 ∪ com_vnames c2"
| "com_vnames (If _ b c1 c2) =
    exp_vnames b ∪ com_vnames c1 ∪ com_vnames c2"
| "com_vnames (While _ b c) = exp_vnames b ∪ com_vnames c"
| "com_vnames (Call _ dst _ actuals) =
    (case dst of None ⇒ {} | Some x ⇒ {x}) ∪
    ⋃ (set (map exp_vnames actuals))"
| "com_vnames (Return _ e) = (case e of None ⇒ {} | Some a ⇒ exp_vnames a)"
| "com_vnames (Restore _) = {}"
| "com_vnames (Unwind _) = {}"

lemma finite_com_vnames [simp]: "finite (com_vnames c)"
  by (induction c) (auto split: option.splits)

fun source_com :: "('a, 'v, 'e) gcom ⇒ bool" where
  "source_com (SKIP _) = True"
| "source_com (Assign _ x a) = True"
| "source_com (Check _ l c) = True"
| "source_com (Seq c1 c2) = (source_com c1 ∧ source_com c2)"
| "source_com (If _ b c1 c2) = (source_com c1 ∧ source_com c2)"
| "source_com (While _ b c) = source_com c"
| "source_com (Call _ dst p actuals) = True"
| "source_com (Return _ e) = True"
| "source_com (Restore _) = False"
| "source_com (Unwind _) = False"

definition source_pi :: "('a, 'v, 'e) gptable ⇒ bool" where
  "source_pi Π = (∀p decl. Π p = Some decl ⟶ source_com (body decl))"

text ‹Well-formedness is stated once for any language: ‹vars› gives the variables an
  expression reads, and ‹rv› is the reserved return slot. The surface predicates are the
  instances at \<^const>‹exp_vnames› and \<^const>‹ret_var›.›

definition gsource_exp :: "('e ⇒ 'v set) ⇒ 'v ⇒ 'e ⇒ bool" where
  "gsource_exp vars rv a ⟷ rv ∉ vars a"

abbreviation source_exp :: "src_exp ⇒ bool" where
  "source_exp ≡ gsource_exp exp_vnames ret_var"

definition gvalid_formal :: "'v ⇒ ('v ⇒ bool) ⇒ 'v ⇒ bool" where
  "gvalid_formal rv 𝒢 x ⟷ ¬ 𝒢 x ∧ x ≠ rv"

abbreviation valid_formal :: "(vname ⇒ bool) ⇒ vname ⇒ bool" where
  "valid_formal ≡ gvalid_formal ret_var"

subsection ‹Procedure result contract›

text ‹
  Declarations carry no return-kind annotation. A body is a value provider when
  every syntactic way to finish the body reaches \<^const>‹Return› with a value.
  The analysis is conservative: a loop may fall through because its guard may be
  false, and a call resumes normally after its callee completes.
›

fun may_fallthrough :: "('a, 'v, 'e) gcom ⇒ bool" where
  "may_fallthrough (SKIP _) = True"
| "may_fallthrough (Assign _ _ _) = True"
| "may_fallthrough (Check _ _ _) = True"
| "may_fallthrough (Seq c1 c2) = (may_fallthrough c1 ∧ may_fallthrough c2)"
| "may_fallthrough (If _ _ c1 c2) = (may_fallthrough c1 ∨ may_fallthrough c2)"
| "may_fallthrough (While _ _ _) = True"
| "may_fallthrough (Call _ _ _ _) = True"
| "may_fallthrough (Return _ _) = False"
| "may_fallthrough (Restore _) = False"
| "may_fallthrough (Unwind _) = False"

fun may_return_none :: "('a, 'v, 'e) gcom ⇒ bool" where
  "may_return_none (Seq c1 c2) =
     (may_return_none c1 ∨ (may_fallthrough c1 ∧ may_return_none c2))"
| "may_return_none (If _ _ c1 c2) = (may_return_none c1 ∨ may_return_none c2)"
| "may_return_none (While _ _ c) = may_return_none c"
| "may_return_none (Return _ e) = (e = None)"
| "may_return_none _ = False"

fun may_return_value :: "('a, 'v, 'e) gcom ⇒ bool" where
  "may_return_value (Seq c1 c2) =
     (may_return_value c1 ∨ (may_fallthrough c1 ∧ may_return_value c2))"
| "may_return_value (If _ _ c1 c2) = (may_return_value c1 ∨ may_return_value c2)"
| "may_return_value (While _ _ c) = may_return_value c"
| "may_return_value (Return _ e) = (e ≠ None)"
| "may_return_value _ = False"

definition value_providing :: "('a, 'v, 'e) gcom ⇒ bool" where
  "value_providing c ⟷
     source_com c ∧ ¬ may_fallthrough c ∧
     ¬ may_return_none c ∧ may_return_value c"

subsection ‹Commands, declarations, and programs›

text ‹
  Every call names a declared procedure and supplies one actual per formal. A
  destination requires a value-providing callee; a destination-less call may
  discard a published value. Procedure formals are distinct local names, and the
  distinguished root command contains no return.
›

fun gwf_source_com ::
    "('e ⇒ 'v set) ⇒ 'v ⇒ ('a, 'v, 'e) gptable ⇒ ('a, 'v, 'e) gcom ⇒ bool" where
  "gwf_source_com vars rv Π (SKIP _) = True"
| "gwf_source_com vars rv Π (Assign _ x a) = (x ≠ rv ∧ gsource_exp vars rv a)"
| "gwf_source_com vars rv Π (Check _ l c) = gsource_exp vars rv c"
| "gwf_source_com vars rv Π (Seq c1 c2) = (gwf_source_com vars rv Π c1 ∧ gwf_source_com vars rv Π c2)"
| "gwf_source_com vars rv Π (If _ b c1 c2) =
     (gsource_exp vars rv b ∧ gwf_source_com vars rv Π c1 ∧ gwf_source_com vars rv Π c2)"
| "gwf_source_com vars rv Π (While _ b c) = (gsource_exp vars rv b ∧ gwf_source_com vars rv Π c)"
| "gwf_source_com vars rv Π (Call _ dst p actuals) =
     (case special_table p of
        Some desc ⇒
          classify_special desc actuals ≠ None ∧ list_all (gsource_exp vars rv) actuals ∧
          (case dst of None ⇒ False | Some x ⇒ x ≠ rv)
      | None ⇒
          (case Π p of
             None ⇒ False
           | Some decl ⇒
               length actuals = length (formals decl) ∧
               list_all (gsource_exp vars rv) actuals ∧
               (case dst of
                  None ⇒ True
                | Some x ⇒ x ≠ rv ∧ value_providing (body decl))))"
| "gwf_source_com vars rv Π (Return _ e) = (case e of None ⇒ True | Some a ⇒ gsource_exp vars rv a)"
| "gwf_source_com vars rv Π (Restore _) = False"
| "gwf_source_com vars rv Π (Unwind _) = False"

abbreviation wf_source_com :: "'a ptable ⇒ 'a acom ⇒ bool" where
  "wf_source_com ≡ gwf_source_com exp_vnames ret_var"

fun no_return :: "('a, 'v, 'e) gcom ⇒ bool" where
  "no_return (Seq c1 c2) = (no_return c1 ∧ no_return c2)"
| "no_return (If _ _ c1 c2) = (no_return c1 ∧ no_return c2)"
| "no_return (While _ _ c) = no_return c"
| "no_return (Return _ _) = False"
| "no_return _ = True"

definition gwf_proc_decl ::
    "('e ⇒ 'v set) ⇒ 'v ⇒ ('v ⇒ bool) ⇒ ('a, 'v, 'e) gptable
       ⇒ ('a, 'v, 'e) gproc_decl ⇒ bool" where
  "gwf_proc_decl vars rv 𝒢 Π decl ⟷
     distinct (formals decl) ∧
     list_all (gvalid_formal rv 𝒢) (formals decl) ∧
     gwf_source_com vars rv Π (body decl)"

abbreviation wf_proc_decl :: "(vname ⇒ bool) ⇒ 'a ptable ⇒ ('a, vname, src_exp) gproc_decl ⇒ bool"
  where
  "wf_proc_decl ≡ gwf_proc_decl exp_vnames ret_var"

definition greserved_ret_var :: "'v ⇒ ('v ⇒ bool) ⇒ bool" where
  "greserved_ret_var rv 𝒢 ⟷ ¬ 𝒢 rv"

abbreviation reserved_ret_var :: "(vname ⇒ bool) ⇒ bool" where
  "reserved_ret_var ≡ greserved_ret_var ret_var"

text ‹The entry procedure is fixed: a program has exactly one, it is called ‹main›, and the
  parser rejects formals on it.  Naming it here rather than threading it as a parameter means
  the compiler and its contract take a table and a callee list and nothing else.›
definition prog_main_name :: pname where
  "prog_main_name = STR ''main''"

text ‹The entry body, looked up rather than passed alongside the table.  The lookup cannot
  fail where ‹wf_source_program› holds --- its second conjunct is exactly that the entry is
  declared --- so an undeclared entry is an invariant violation, and aborts in generated code
  rather than compiling to a plausible empty program.›
definition main_body :: "('a, 'v, 'e) gptable ⇒ ('a, 'v, 'e) gcom" where
  "main_body Π =
     (case Π prog_main_name of
        Some decl ⇒ body decl
      | None ⇒ Code.abort (STR ''main_body: entry procedure not declared'') (λ_. SKIP undefined))"

text ‹Deliberately not ‹[simp]›: ‹wf_source_program›'s entry conjunct has the shape
  ‹Π prog_main_name = Some ⦇formals = [], body = main_body Π⦈›, against which
  this rule would rewrite ‹main_body Π› to itself.›
definition gwf_source_program ::
    "('e ⇒ 'v set) ⇒ 'v ⇒ ('v ⇒ bool) ⇒ ('a, 'v, 'e) gptable ⇒ bool" where
  "gwf_source_program vars rv 𝒢 Π ⟷
     greserved_ret_var rv 𝒢 ∧
     Π prog_main_name = Some (⦇formals = [], body = main_body Π⦈) ∧
     gwf_source_com vars rv Π (main_body Π) ∧ no_return (main_body Π) ∧
     (∀p decl. Π p = Some decl ⟶ gwf_proc_decl vars rv 𝒢 Π decl) ∧
     (∀p. Π p ≠ None ⟶ special_table p = None)"

abbreviation wf_source_program :: "(vname ⇒ bool) ⇒ 'a ptable ⇒ bool" where
  "wf_source_program ≡ gwf_source_program exp_vnames ret_var"

text ‹The compiler-input well-formedness defined in the compiler session unfolds
  through this pair before reaching \<^const>‹wf_source_com›; every site
  discharging that obligation shares the unfold skeleton, so it is collected
  here and the downstream definition adds itself to the same collection.›
named_theorems wf_compile_input_simps

declare
  gwf_proc_decl_def [wf_compile_input_simps]
  gwf_source_program_def [wf_compile_input_simps]

lemma wf_source_com_source_com:
  "gwf_source_com vars rv Π c ⟹ source_com c"
  by (induction c) (auto split: option.splits)

lemma wf_source_programD:
  assumes "gwf_source_program vars rv 𝒢 Π"
  shows "greserved_ret_var rv 𝒢"
    and "Π prog_main_name = Some ⦇formals = [], body = main_body Π⦈"
    and "gwf_source_com vars rv Π (main_body Π)"
    and "no_return (main_body Π)"
    and "Π p = Some decl ⟹ gwf_proc_decl vars rv 𝒢 Π decl"
    and "Π p = Some decl ⟹ special_table p = None"
    and "source_pi Π"
    and "source_com (main_body Π)"
  using assms wf_source_com_source_com [of vars rv Π]
  unfolding gwf_source_program_def source_pi_def gwf_proc_decl_def by blast+

end
