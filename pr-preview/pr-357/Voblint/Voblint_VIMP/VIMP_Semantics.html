Theory VIMP_Semantics

theory VIMP_Semantics
  imports Core_Com
begin

section ‹Typed source semantics›

text ‹
  A typed program runs on typed stores, which hold one value per variable identity. Every
  written value has the kind of the identity it is written to: elaboration already
  converts an assigned value, an actual argument and a returned value, and the semantics
  converts the two values it writes itself, a call result and a library result, to the
  destination's kind. This theory fixes the globals and the initial stores of that
  semantics and defines it.
›

text ‹The globals are the \<^const>‹GlobalId› identities. A call keeps them and resets
  every other identity, the shared return slot included, as \<^const>‹enter_state›
  keeps the declared globals.›

fun is_global :: "var_id ⇒ bool" where
  "is_global (GlobalId _) = True"
| "is_global _ = False"

text ‹A program starts with its globals at zero and every other identity holding some
  value of its kind.›

definition init_stores :: "kenv ⇒ store set" where
  "init_stores κ = {t. all_in_kinds κ t ∧ (∀x. t (GlobalId x) = 0)}"

section ‹Frame-stack small-step semantics›

text ‹
  A written value is converted to the kind of its destination, as C converts on
  assignment. ‹kind_conv› takes the destination as an option because a call without
  a destination converts nothing. A variable read carries no conversion: an identity
  holds a value of its own kind, since every write converts.
›

definition kind_conv :: "kenv ⇒ var_id option ⇒ int ⇒ int" where
  "kind_conv κ dst n = (case dst of None ⇒ n | Some v ⇒ ik_norm (κ v) n)"

text ‹Not ‹[simp]›: the semantics' rules are stated with the conversion folded, and
  unfolding it in a goal would keep them from matching.›

lemma kind_conv_simps:
  "kind_conv κ None n = n"
  "kind_conv κ (Some v) n = ik_norm (κ v) n"
  by (simp_all add: kind_conv_def)

text ‹A frame records the caller's store and the destination of the pending call.›

datatype frame = Frame (frame_store: store) (frame_dest: "var_id option")

text ‹The reference semantics every soundness claim is ultimately about.  A configuration is
  a command, a store, and a stack of caller frames; a call pushes one frame and continues in
  the callee body followed by ‹Restore›, a return writes the return slot and hands control to
  ‹Unwind›, which discards the rest of the callee body until that ‹Restore› pops the frame.
  ‹Restore› and ‹Unwind› exist only here, in configurations the semantics builds; no source
  program contains them.

  \<^const>‹eval› evaluates an expression, \<^const>‹is_global› classifies the variables a
  call keeps, \<^const>‹ReturnVar› is the return slot (Goblint's single
  ▩‹return_varinfo›), and \<^const>‹kind_conv› converts a value written to a
  destination, as Goblint converts a stored value to its target's declared type. A call
  converts each actual to its formal, as an assignment converts to its destination, so
  every entry store holds each formal at its own kind. Elaboration has already
  converted an actual it produced, as CIL does, and there the conversion changes
  nothing. A returned value is converted once, to the caller's destination.›

inductive pstep :: "kenv ⇒ ('a, var_id, exp) gptable
    ⇒ 'a com × store × frame list ⇒ 'a com × store × frame list ⇒ bool"
    ("(_,_ ⊢/ _ →⇩p/ _)" [51, 51, 51, 51] 50)
  for κ :: kenv and Π :: "('a, var_id, exp) gptable"
where
  Assign:  "κ,Π ⊢ (Assign i x a, s, frs) →⇩p (SKIP i, s(x := kind_conv κ (Some x) (eval a s)), frs)"
| Check:   "κ,Π ⊢ (Check i l c, s, frs) →⇩p (SKIP i, s, frs)"
| Seq1:    "κ,Π ⊢ (Seq (SKIP i) c2, s, frs) →⇩p (c2, s, frs)"
| Seq2:    "κ,Π ⊢ (c1, s, frs) →⇩p (c1', s', frs')
             ⟹ κ,Π ⊢ (Seq c1 c2, s, frs) →⇩p (Seq c1' c2, s', frs')"
| IfTrue:  "truthy (eval b s) ⟹ κ,Π ⊢ (If i b c1 c2, s, frs) →⇩p (c1, s, frs)"
| IfFalse: "¬ truthy (eval b s) ⟹ κ,Π ⊢ (If i b c1 c2, s, frs) →⇩p (c2, s, frs)"
| While:   "κ,Π ⊢ (While i b c, s, frs) →⇩p (If i b (Seq c (While i b c)) (SKIP i), s, frs)"
| Call:    "Π p = Some decl
             ⟹ length actuals = length (formals decl)
             ⟹ distinct (formals decl)
             ⟹ vals = map (λe. eval e s) actuals
             ⟹ callee = bind_formals (formals decl)
                           (map2 (λx. kind_conv κ (Some x)) (formals decl) vals)
                           (enter_state is_global s)
             ⟹ κ,Π ⊢ (Call i dst p actuals, s, frs)
                   →⇩p (Seq (body decl) (Restore i), callee, Frame s dst # frs)"
| Special: "special_table p = Some desc
             ⟹ classify_special desc actuals = Some sc
             ⟹ special_result eval sc s v
             ⟹ κ,Π ⊢ (Call i (Some x) p actuals, s, frs) →⇩p (SKIP i, s(x := kind_conv κ (Some x) v), frs)"
| RestoreStep:
    "κ,Π ⊢ (Restore i, s, Frame fr dst # frs)
       →⇩p (SKIP i, combine_assign dst (kind_conv κ dst (s ReturnVar)) (combine_env is_global fr s), frs)"
| ReturnSome:
    "κ,Π ⊢ (Return i (Some e), s, frs) →⇩p (Unwind i, s(ReturnVar := eval e s), frs)"
| ReturnNone:
    "κ,Π ⊢ (Return i None, s, frs) →⇩p (Unwind i, s, frs)"
| UnwindDead:
    "¬ is_Restore c2 ⟹ κ,Π ⊢ (Seq (Unwind i) c2, s, frs) →⇩p (Unwind i, s, frs)"
| UnwindAct:
    "κ,Π ⊢ (Seq (Unwind j) (Restore i), s, Frame fr dst # frs)
       →⇩p (SKIP i, combine_assign dst (kind_conv κ dst (s ReturnVar)) (combine_env is_global fr s), frs)"

abbreviation psteps :: "kenv ⇒ ('a, var_id, exp) gptable
    ⇒ 'a com × store × frame list ⇒ 'a com × store × frame list ⇒ bool"
    ("(_,_ ⊢/ _ →⇩p⇧*/ _)" [51, 51, 51, 51] 50) where
  "psteps κ Π x y ≡ star (pstep κ Π) x y"

inductive_cases SkipSE[elim!]: "κ,Π ⊢ (SKIP i, s, frs) →⇩p cfg"
inductive_cases AssignSE[elim!]: "κ,Π ⊢ (Assign i x a, s, frs) →⇩p cfg"
inductive_cases CheckSE[elim!]: "κ,Π ⊢ (Check i l c, s, frs) →⇩p cfg"
inductive_cases SeqSE[elim]: "κ,Π ⊢ (Seq c1 c2, s, frs) →⇩p cfg"
inductive_cases IfSE[elim!]: "κ,Π ⊢ (If i b c1 c2, s, frs) →⇩p cfg"
inductive_cases WhileSE[elim]: "κ,Π ⊢ (While i b c, s, frs) →⇩p cfg"
inductive_cases CallSE[elim]: "κ,Π ⊢ (Call i dst p actuals, s, frs) →⇩p cfg"
inductive_cases RestoreSE[elim!]: "κ,Π ⊢ (Restore i, s, frs) →⇩p cfg"
inductive_cases ReturnSE[elim!]: "κ,Π ⊢ (Return i e, s, frs) →⇩p cfg"
inductive_cases UnwindSE[elim!]: "κ,Π ⊢ (Unwind i, s, frs) →⇩p cfg"

declare pstep.intros [simp, intro]

lemma combine_env_rv_irrelevant [simp]:
  "combine_env is_global fr (t(ReturnVar := v)) = combine_env is_global fr t"
  by (rule ext) simp

lemma psteps_Seq2:
  "κ,Π ⊢ (c1, s, frs) →⇩p⇧* (c1', s', frs')
   ⟹ κ,Π ⊢ (Seq c1 c2, s, frs) →⇩p⇧* (Seq c1' c2, s', frs')"
  by (induction rule: star.induct[of "pstep κ Π", split_format(complete)])
     (auto intro: star.step)

subsection ‹Frame-stack extension›

text ‹A step never inspects the frames below the one it works on, so any run stays valid
  with more callers underneath.  This is what lets a proof about a procedure in isolation be
  reused at an arbitrary call depth.›

lemma pstep_frame_extend:
  "κ,Π ⊢ (c, s, frs) →⇩p (c', s', frs') ⟹
   κ,Π ⊢ (c, s, frs @ extra) →⇩p (c', s', frs' @ extra)"
  by (induction "(c, s, frs)" "(c', s', frs')"
        arbitrary: c s frs c' s' frs' rule: pstep.induct)
     auto

subsection ‹Call completion›

text ‹Running a whole call in one step of reasoning: a body that finishes under its own
  frame lets the caller resume with the frame popped and the return slot written.  Callers
  use these instead of replaying the ‹Restore› bookkeeping at every call site.›

lemma pstep_Call:
  assumes "Π p = Some decl"
      and "length actuals = length (formals decl)"
      and "distinct (formals decl)"
  shows "κ,Π ⊢ (Call i dst p actuals, s, frs) →⇩p (Seq (body decl) (Restore i),
            bind_formals (formals decl)
              (map2 (λx. kind_conv κ (Some x)) (formals decl) (map (λe. eval e s) actuals))
              (enter_state is_global s),
            Frame s dst # frs)"
  using assms by (rule pstep.Call) (rule refl)+

lemma pstep_Call_parameterless [intro]:
  assumes "Π p = Some (⦇formals = [], body = c⦈)"
  shows
    "κ,Π ⊢ (Call i dst p [], s, frs) →⇩p (Seq c (Restore i), enter_state is_global s, Frame s dst # frs)"
proof -
  have "κ,Π ⊢ (Call i dst p [], s, frs) →⇩p (Seq (body (⦇formals = [], body = c⦈)) (Restore i),
           bind_formals (formals (⦇formals = [], body = c⦈))
             (map2 (λx. kind_conv κ (Some x)) (formals (⦇formals = [], body = c⦈))
               (map (λe. eval e s) []))
             (enter_state is_global s),
           Frame s dst # frs)"
    using assms by (rule pstep_Call) simp_all
  then show ?thesis by simp
qed

subsection ‹A declaration without an initializer›

text ‹
  The frontend writes a local's declaration without an initializer as the library call
  ‹x = __voblint_nondet_int()› at the declaration's position, so it runs on every
  activation and every loop iteration that reaches it. That call may leave any value of
  ‹x›'s kind in ‹x› and changes nothing else. Its outcomes do not depend on what ‹x› held
  before: the ‹0› a call entry wrote, or the value of an earlier iteration or activation.
  That the frontend places every use of a local after its declaration is trusted.
›

lemma pstep_declare:
  assumes "Π special_pname_nondet_int = None"
  shows "κ,Π ⊢ (gcom.Call i (Some x) special_pname_nondet_int [], s, frs) →⇩p (c', s', frs')
           ⟷ c' = SKIP i ∧ frs' = frs ∧ (∃n ∈ ik_range (κ x). s' = s(x := n))"
proof
  assume "κ,Π ⊢ (gcom.Call i (Some x) special_pname_nondet_int [], s, frs) →⇩p (c', s', frs')"
  then show "c' = SKIP i ∧ frs' = frs ∧ (∃n ∈ ik_range (κ x). s' = s(x := n))"
    using assms by (auto elim!: CallSE simp: special_table_def kind_conv_simps)
next
  assume "c' = SKIP i ∧ frs' = frs ∧ (∃n ∈ ik_range (κ x). s' = s(x := n))"
  then obtain n where n: "n ∈ ik_range (κ x)" and eq: "c' = SKIP i" "frs' = frs" "s' = s(x := n)"
    by blast
  have "κ,Π ⊢ (gcom.Call i (Some x) special_pname_nondet_int [], s, frs) →⇩p (SKIP i, s(x := kind_conv κ (Some x) n), frs)"
    by (rule pstep.Special[where desc = SD_Nondet_Int and sc = "Nondet_Int :: exp gspecial_call"])
       (simp_all add: special_table_def)
  with n eq
  show "κ,Π ⊢ (gcom.Call i (Some x) special_pname_nondet_int [], s, frs) →⇩p (c', s', frs')"
    by (simp add: kind_conv_simps del: ik_range_iff)
qed

corollary pstep_declare_forgets:
  assumes "Π special_pname_nondet_int = None"
  shows "κ,Π ⊢ (gcom.Call i (Some x) special_pname_nondet_int [], s(x := m), frs) →⇩p cfg
           ⟷ κ,Π ⊢ (gcom.Call i (Some x) special_pname_nondet_int [], s, frs) →⇩p cfg"
  by (cases cfg) (simp add: pstep_declare[where Π = Π, OF assms])

section ‹The typed program›

text ‹A procedure's typed formals are the identities its formals resolve to, and its typed
  body is its elaboration.›

definition elab_ptable :: "imp_prog ⇒ kind_decls ⇒ (unit, var_id, exp) gptable" where
  "elab_ptable p ds q =
     map_option (λd. ⦇formals = map (resolve_var (declared_global p) q) (formals d),
                      body = elab_com (declared_global p) (prog_formals p) (kenv_of ds) q (body d)⦈)
       (prog_table p q)"

lemma elab_ptable_None_iff [simp]: "elab_ptable p ds q = None ⟷ prog_table p q = None"
  by (simp add: elab_ptable_def)

subsection ‹Elaboration keeps the control structure›

text ‹Elaboration rewrites expressions only, so every predicate on a command's control
  structure is the same before and after it.›

lemma elab_com_control [simp]:
  "source_com (elab_com 𝒢 fm κ q c) = source_com c"
  "may_fallthrough (elab_com 𝒢 fm κ q c) = may_fallthrough c"
  "may_return_none (elab_com 𝒢 fm κ q c) = may_return_none c"
  "may_return_value (elab_com 𝒢 fm κ q c) = may_return_value c"
  "no_return (elab_com 𝒢 fm κ q c) = no_return c"
  by (induction c) auto

lemma value_providing_elab_com [simp]:
  "value_providing (elab_com 𝒢 fm κ q c) = value_providing c"
  by (simp add: value_providing_def)

lemma classify_special_map_None_iff [simp]:
  "classify_special d (map f xs) = None ⟷ classify_special d xs = None"
  by (cases "(d, xs)" rule: classify_special.cases) auto

subsection ‹Well-formedness carries over›

text ‹Elaborated expressions never mention the return slot, so a well-formed source
  program elaborates to a well-formed typed program.›

lemma ReturnVar_notin_range_resolve_var [simp]: "ReturnVar ∉ range (resolve_var 𝒢 q)"
  by (auto simp: resolve_var_def)

lemma ReturnVar_notin_elaborate [simp]:
  "ReturnVar ∉ exp_vars (elaborate (resolve_var 𝒢 q) κ ik e)"
  using exp_vars_elaborate [of "resolve_var 𝒢 q" κ ik e]
    ReturnVar_notin_range_resolve_var [of 𝒢 q] by blast

lemma gsource_exp_elaborate [simp]:
  "gsource_exp exp_vars ReturnVar (elaborate (resolve_var 𝒢 q) κ ik e)"
  "gsource_exp exp_vars ReturnVar (elaborate_syn (resolve_var 𝒢 q) κ e)"
  by (simp_all add: gsource_exp_def elaborate_syn_def)

lemma gsource_exp_elab_args:
  "list_all (gsource_exp exp_vars ReturnVar) (elab_args (resolve_var 𝒢 q) κ fs es)"
  unfolding list_all_iff gsource_exp_def
  using exp_vars_elab_args ReturnVar_notin_range_resolve_var by blast

lemma gwf_source_com_elab_com:
  assumes wf: "wf_source_program (declared_global p) (prog_table p)"
    and c: "wf_source_com (prog_table p) c"
  shows "gwf_source_com exp_vars ReturnVar (elab_ptable p ds)
           (elab_com (declared_global p) (prog_formals p) (kenv_of ds) q c)"
  using c
proof (induction c)
  case (Call i dst r es)
  show ?case
  proof (cases "special_table r")
    case (Some desc)
    with wf have "prog_table p r = None"
      by (auto simp: gwf_source_program_def)
    then have "prog_formals p r = []" by (simp add: prog_formals_def)
    with Call Some show ?thesis
      by (auto simp: list_all_iff split: option.splits)
  next
    case None
    with Call show ?thesis
      by (auto simp: elab_ptable_def prog_formals_def gsource_exp_elab_args
               split: option.splits)
  qed
qed (auto split: option.splits)

lemma elab_ptable_main:
  assumes wf: "wf_source_program (declared_global p) (prog_table p)"
  shows "elab_ptable p ds prog_main_name
           = Some ⦇formals = [], body = main_body (elab_ptable p ds)⦈"
    and "main_body (elab_ptable p ds)
           = elab_com (declared_global p) (prog_formals p) (kenv_of ds) prog_main_name
               (main_body (prog_table p))"
proof -
  from wf have "prog_table p prog_main_name = Some ⦇formals = [], body = main_body (prog_table p)⦈"
    by (simp add: gwf_source_program_def)
  then have t: "elab_ptable p ds prog_main_name
      = Some ⦇formals = [], body = elab_com (declared_global p) (prog_formals p) (kenv_of ds)
                                      prog_main_name (main_body (prog_table p))⦈"
    by (simp add: elab_ptable_def)
  then show "main_body (elab_ptable p ds)
      = elab_com (declared_global p) (prog_formals p) (kenv_of ds) prog_main_name
          (main_body (prog_table p))"
    by (simp add: main_body_def)
  with t show "elab_ptable p ds prog_main_name
      = Some ⦇formals = [], body = main_body (elab_ptable p ds)⦈"
    by simp
qed

lemma gwf_proc_decl_elab:
  assumes wf: "wf_source_program (declared_global p) (prog_table p)"
    and d: "prog_table p q = Some d"
  shows "gwf_proc_decl exp_vars ReturnVar is_global (elab_ptable p ds)
           ⦇formals = map (resolve_var (declared_global p) q) (formals d),
            body = elab_com (declared_global p) (prog_formals p) (kenv_of ds) q (body d)⦈"
proof -
  from wf d have pd: "wf_proc_decl (declared_global p) (prog_table p) d"
    by (simp add: gwf_source_program_def)
  then have local: "∀x ∈ set (formals d). ¬ declared_global p x"
    by (auto simp: gwf_proc_decl_def gvalid_formal_def list_all_iff)
  then have "inj_on (resolve_var (declared_global p) q) (set (formals d))"
    by (auto simp: inj_on_def resolve_var_eq_iff)
  with pd have "distinct (map (resolve_var (declared_global p) q) (formals d))"
    by (simp add: gwf_proc_decl_def distinct_map)
  moreover have "list_all (gvalid_formal ReturnVar is_global)
                   (map (resolve_var (declared_global p) q) (formals d))"
    using local by (auto simp: list_all_iff gvalid_formal_def resolve_var_def)
  moreover have "gwf_source_com exp_vars ReturnVar (elab_ptable p ds)
                   (elab_com (declared_global p) (prog_formals p) (kenv_of ds) q (body d))"
    using pd by (intro gwf_source_com_elab_com [OF wf]) (simp add: gwf_proc_decl_def)
  ultimately show ?thesis by (simp add: gwf_proc_decl_def)
qed
