Theory Core_Com

theory Core_Com
  imports Core_Exp VIMP_Program
begin

section ‹Typed commands›

text ‹
  A typed command is the generic command type at identities and typed expressions,
  with the annotation left free. Elaboration resolves every name of a procedure and
  inserts the conversions C performs where a value is written: to the variable's kind
  at an assignment, to the formal's kind at an actual argument, and to the
  procedure's return kind at a return. A condition is evaluated at its own kind. The
  call destination is only resolved here, since converting the returned value to it
  happens when the frame is popped, which is the business of a typed semantics.
›

type_synonym 'a com = "('a, var_id, exp) gcom"

text ‹An actual argument converts to the kind of the formal it binds. The arguments
  without a formal, which only a library call has, convert to their common kind, as the
  operands of an arithmetic operator do: C has no integer ‹min› or ‹max›, and the
  macro ‹(a) < (b) ? (a) : (b)› that stands for one compares and selects at the usual
  arithmetic conversion of its arguments. \<^const>‹I32› is a neutral start: the
  common kind of \<^const>‹I32› and a kind is that kind promoted.›

definition args_kind :: "(vname ⇒ ikind) ⇒ src_exp list ⇒ ikind" where
  "args_kind Γ es = foldr (λe. usual_kind (esyn Γ e)) es I32"

lemma args_kind_pair: "args_kind Γ [a, b] = usual_kind (esyn Γ a) (esyn Γ b)"
  unfolding args_kind_def
  by (cases "esyn Γ a"; cases "esyn Γ b") (simp_all add: usual_kind_def)

fun elab_args :: "(vname ⇒ var_id) ⇒ kenv ⇒ var_id list ⇒ src_exp list ⇒ exp list" where
  "elab_args ρ κ (f # fs) (e # es) = elaborate ρ κ (κ f) e # elab_args ρ κ fs es"
| "elab_args ρ κ [] es = map (elaborate ρ κ (args_kind (κ ∘ ρ) es)) es"
| "elab_args ρ κ fs [] = []"

lemma exp_vars_elab_args: "e ∈ set (elab_args ρ κ fs es) ⟹ exp_vars e ⊆ range ρ"
  by (induction ρ κ fs es rule: elab_args.induct)
     (auto dest: subsetD [OF exp_vars_elaborate])

text ‹‹fm p› lists the formals of procedure ‹p›, empty for a library call, and
  ‹q› is the procedure whose body is elaborated.›

fun elab_com ::
  "(vname ⇒ bool) ⇒ (pname ⇒ vname list) ⇒ kenv ⇒ pname
     ⇒ ('a, vname, src_exp) gcom ⇒ 'a com" where
  "elab_com 𝒢 fm κ q (SKIP i) = SKIP i"
| "elab_com 𝒢 fm κ q (Assign i x e) =
     Assign i (resolve_var 𝒢 q x) (elaborate (resolve_var 𝒢 q) κ (κ (resolve_var 𝒢 q x)) e)"
| "elab_com 𝒢 fm κ q (Check i l c) = Check i l (elaborate_syn (resolve_var 𝒢 q) κ c)"
| "elab_com 𝒢 fm κ q (Seq c1 c2) = Seq (elab_com 𝒢 fm κ q c1) (elab_com 𝒢 fm κ q c2)"
| "elab_com 𝒢 fm κ q (If i b c1 c2) =
     If i (elaborate_syn (resolve_var 𝒢 q) κ b) (elab_com 𝒢 fm κ q c1) (elab_com 𝒢 fm κ q c2)"
| "elab_com 𝒢 fm κ q (While i b c) =
     While i (elaborate_syn (resolve_var 𝒢 q) κ b) (elab_com 𝒢 fm κ q c)"
| "elab_com 𝒢 fm κ q (Call i dst p es) =
     Call i (map_option (resolve_var 𝒢 q) dst) p
       (elab_args (resolve_var 𝒢 q) κ (map (resolve_var 𝒢 p) (fm p)) es)"
| "elab_com 𝒢 fm κ q (Return i e) =
     Return i (map_option (elaborate (resolve_var 𝒢 q) κ (κ (ReturnId q))) e)"
| "elab_com 𝒢 fm κ q (Restore i) = Restore i"
| "elab_com 𝒢 fm κ q (Unwind i) = Unwind i"

subsection ‹Well-typed commands›

text ‹A typed command is well-typed when its expressions are, and every written
  value already has the kind of the place it is written to. ‹fids p› lists the
  identities of the formals of ‹p›.›

fun args_ok :: "kenv ⇒ var_id list ⇒ exp list ⇒ bool" where
  "args_ok κ (f # fs) (e # es) ⟷ exp_kind e = κ f ∧ args_ok κ fs es"
| "args_ok κ _ _ ⟷ True"

fun wt_com :: "kenv ⇒ (pname ⇒ var_id list) ⇒ pname ⇒ 'a com ⇒ bool" where
  "wt_com κ fids q (SKIP i) ⟷ True"
| "wt_com κ fids q (Assign i v e) ⟷ wt_exp κ e ∧ exp_kind e = κ v"
| "wt_com κ fids q (Check i l c) ⟷ wt_exp κ c"
| "wt_com κ fids q (Seq c1 c2) ⟷ wt_com κ fids q c1 ∧ wt_com κ fids q c2"
| "wt_com κ fids q (If i b c1 c2) ⟷
     wt_exp κ b ∧ wt_com κ fids q c1 ∧ wt_com κ fids q c2"
| "wt_com κ fids q (While i b c) ⟷ wt_exp κ b ∧ wt_com κ fids q c"
| "wt_com κ fids q (Call i dst p es) ⟷ list_all (wt_exp κ) es ∧ args_ok κ (fids p) es"
| "wt_com κ fids q (Return i e) ⟷
     pred_option (λa. wt_exp κ a ∧ exp_kind a = κ (ReturnId q)) e"
| "wt_com κ fids q (Restore i) ⟷ True"
| "wt_com κ fids q (Unwind i) ⟷ True"

lemma elab_args_wt:
  "list_all (wt_exp κ) (elab_args ρ κ fs es) ∧ args_ok κ fs (elab_args ρ κ fs es)"
  by (induction ρ κ fs es rule: elab_args.induct) (simp_all add: wt_exp_elaborate list_all_iff)

theorem wt_com_elab_com:
  "wt_com κ (λp. map (resolve_var 𝒢 p) (fm p)) q (elab_com 𝒢 fm κ q c)"
  by (induction c)
     (simp_all add: wt_exp_elaborate elaborate_syn_def elab_args_wt option.pred_map comp_def
       option.pred_True)

subsection ‹Erasure and shape›

text ‹Erasing a typed command names every identity by its source name and drops kinds
  and conversions. It returns the command elaboration started from. The statement is
  syntactic: a typed command and its erasure can execute differently wherever a kind
  wraps, so it says nothing about preserving behaviour.›

abbreviation com_erase :: "'a com ⇒ ('a, vname, src_exp) gcom" where
  "com_erase ≡ map_gcom id var_name exp_erase"

lemma map_exp_erase_elab_args [simp]:
  assumes "⋀x. var_name (ρ x) = x" and "list_all cast_free es"
  shows "map exp_erase (elab_args ρ κ fs es) = es"
  using assms(2)
proof (induction es arbitrary: fs)
  case Nil
  show ?case by (cases fs) simp_all
next
  case (Cons e es)
  have syn: "list_all cast_free xs ⟹ map (λa. exp_erase (elaborate ρ κ k a)) xs = xs"
    for xs k
    by (induction xs) (simp_all add: exp_erase_elaborate [OF assms(1)])
  show ?case
    using Cons syn
    by (cases fs) (simp_all add: exp_erase_elaborate [OF assms(1)] comp_def)
qed

theorem com_erase_elab_com:
  assumes "∀e ∈ set3_gcom c. cast_free e"
  shows "com_erase (elab_com 𝒢 fm κ q c) = c"
  using assms
proof (induction c)
  case (Return i e)
  then show ?case by (cases e) (simp_all add: exp_erase_elaborate [OF var_name_resolve_var])
qed (auto simp: exp_erase_elaborate [OF var_name_resolve_var] elaborate_syn_def
       map_exp_erase_elab_args [OF var_name_resolve_var] option.map_comp comp_def
       option.map_ident list_all_iff)

text ‹
  Elaboration keeps a command's skeleton: the constructors, the annotations and the
  number of arguments of every call. Only the variables and expressions at its leaves
  change.
›

lemma length_elab_args [simp]: "length (elab_args ρ κ fs es) = length es"
  by (induction ρ κ fs es rule: elab_args.induct) simp_all

theorem skeleton_elab_com:
  "map_gcom id (λ_. ()) (λ_. ()) (elab_com 𝒢 fm κ q c) = map_gcom id (λ_. ()) (λ_. ()) c"
  by (induction c) (simp_all add: option.map_comp comp_def map_replicate_const)

subsection ‹Programs›

text ‹Kinds are declared per identity. The parser declares every variable of a program,
  so the default ‹I32› only makes the environment total, for identities such as the
  return slot that no declaration names.›

type_synonym kind_decls = "(var_id × ikind) list"

definition kenv_of :: "kind_decls ⇒ kenv" where
  "kenv_of ds v = (case map_of ds v of Some k ⇒ k | None ⇒ I32)"

definition prog_formals :: "imp_prog ⇒ pname ⇒ vname list" where
  "prog_formals p q = (case prog_table p q of Some d ⇒ formals d | None ⇒ [])"

end
