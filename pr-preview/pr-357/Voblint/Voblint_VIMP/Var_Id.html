Theory Var_Id

theory Var_Id
  imports VIMP_Proc
begin

section ‹Static variable identities›

text ‹
  A name is not an identity: two procedures may both call a formal ‹n›, and the two
  are different variables that may carry different kinds. A ‹var_id› names the scope
  a binding belongs to, in the role of CIL's ‹varinfo›. ‹ReturnId› is the
  return slot of one procedure. It is a constructor rather than a spelling, so no
  source program can name it, and each procedure's slot can carry its own kind.
›

datatype var_id =
    GlobalId vname
  | ScopedId pname vname
  | ReturnId pname
  | ReturnVar

derive linorder var_id

text ‹‹ReturnVar› is the one slot a return writes and a call result is read from,
  Goblint's ▩‹return_varinfo›: activation-local, belonging to no single procedure.
  ‹ReturnId p› names the kind procedure ‹p› returns, which elaboration converts a
  returned value to.›

fun var_scope :: "var_id ⇒ pname option" where
  "var_scope (GlobalId _) = None"
| "var_scope (ScopedId q _) = Some q"
| "var_scope (ReturnId q) = Some q"
| "var_scope ReturnVar = None"

text ‹The source name of an identity, which a report shows and erasure restores.
  The return slot keeps the reserved source name \<^const>‹ret_var›.›

fun var_name :: "var_id ⇒ vname" where
  "var_name (GlobalId x) = x"
| "var_name (ScopedId _ x) = x"
| "var_name (ReturnId _) = ret_var"
| "var_name ReturnVar = ret_var"

subsection ‹Resolution›

text ‹
  VIMP has no block scopes and no shadowing: a name the program declares global is
  that global in every procedure, and every other name is local to the procedure
  that mentions it (@{const enter_state} resets exactly those at a call). Resolution
  reads that rule off the globals classifier ‹𝒢›.
›

definition resolve_var :: "(vname ⇒ bool) ⇒ pname ⇒ vname ⇒ var_id" where
  "resolve_var 𝒢 q x = (if 𝒢 x then GlobalId x else ScopedId q x)"

lemma var_name_resolve_var [simp]: "var_name (resolve_var 𝒢 q x) = x"
  by (simp add: resolve_var_def)

lemma var_scope_resolve_var [simp]:
  "var_scope (resolve_var 𝒢 q x) = (if 𝒢 x then None else Some q)"
  by (simp add: resolve_var_def)

lemma resolve_var_not_ReturnId [simp]: "resolve_var 𝒢 q x ≠ ReturnId r"
  by (simp add: resolve_var_def)

lemma resolve_var_not_ReturnVar [simp]: "resolve_var 𝒢 q x ≠ ReturnVar"
  by (simp add: resolve_var_def)

text ‹Resolution is unique: within one procedure distinct names get distinct
  identities, and two procedures share an identity only for a global.›

theorem resolve_var_eq_iff:
  "resolve_var 𝒢 q x = resolve_var 𝒢 r y ⟷ x = y ∧ (𝒢 x ∨ q = r)"
  by (auto simp: resolve_var_def)

subsection ‹Kinds and typed stores›

text ‹
  A kind environment answers a kind for every identity. VIMP's grammar declares no
  kinds yet, so an environment is an input to elaboration; ‹int_kenv› is the one a
  program without declarations means, every variable a C ‹int›.
›

type_synonym kenv = "var_id ⇒ ikind"
type_synonym store = "var_id ⇒ int"

definition int_kenv :: kenv where
  "int_kenv = (λ_. I32)"

text ‹A store is typed when every identity holds a value of its kind.›

definition all_in_kinds :: "kenv ⇒ store ⇒ bool" where
  "all_in_kinds κ t ⟷ (∀v. t v ∈ ik_range (κ v))"

lemma all_in_kindsD: "all_in_kinds κ t ⟹ t v ∈ ik_range (κ v)"
  by (simp add: all_in_kinds_def)

lemma all_in_kinds_zero [simp, intro]: "all_in_kinds κ (λ_. 0)"
  by (simp add: all_in_kinds_def del: ik_range_iff)

lemma all_in_kinds_update [simp, intro]:
  "all_in_kinds κ t ⟹ all_in_kinds κ (t(v := ik_norm (κ v) n))"
  by (simp add: all_in_kinds_def del: ik_range_iff)

end
