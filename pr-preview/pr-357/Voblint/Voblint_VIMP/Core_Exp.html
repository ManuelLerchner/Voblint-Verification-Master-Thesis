Theory Core_Exp

theory Core_Exp
  imports Var_Id
begin

section ‹Typed expressions›

text ‹
  Under machine kinds an expression's value depends on the kind each operation computes at:
  ‹200 + 100 < 250› holds over \<^typ>‹int› but fails when the sum wraps at an
  8-bit kind. This theory states that meaning twice. ‹src_eval› reads it off a source
  expression and a kind for every name, in the shape the C standard describes it.
  ‹eval› evaluates a typed expression whose nodes already carry their kinds, which
  is what an analyzer consumes. Elaboration translates the first into the second, and
  @{text eval_elaborate} shows that it preserves the meaning.
›

subsection ‹Operators›

datatype aop = AAdd | ASub | AMul | ADiv | AMod
datatype cop = CLt | CLe | CGt | CGe | CEq | CNe

text ‹Division and remainder keep VIMP's total conventions for a zero divisor
  (@{const c_div}, @{const c_mod}). On a nonzero divisor they truncate toward zero, as
  C does, and on the nonnegative values of an unsigned kind they are floor division.›

fun aop_eval :: "aop ⇒ int ⇒ int ⇒ int" where
  "aop_eval AAdd a b = a + b"
| "aop_eval ASub a b = a - b"
| "aop_eval AMul a b = a * b"
| "aop_eval ADiv a b = c_div a b"
| "aop_eval AMod a b = c_mod a b"

fun cop_eval :: "cop ⇒ int ⇒ int ⇒ bool" where
  "cop_eval CLt a b = (a < b)"
| "cop_eval CLe a b = (a ≤ b)"
| "cop_eval CGt a b = (a > b)"
| "cop_eval CGe a b = (a ≥ b)"
| "cop_eval CEq a b = (a = b)"
| "cop_eval CNe a b = (a ≠ b)"

lemma ik_norm_bool [simp]: "ik_norm ik (if P then 1 else 0) = (if P then 1 else 0)"
  by simp

subsection ‹The source-level typed semantics›

text ‹
  An unsuffixed decimal constant takes the first of \<^const>‹I32› and
  \<^const>‹I64› that represents its magnitude (ISO/IEC 9899 6.4.4.1p5); the frontend
  folds a sign into the literal, and C types the unsigned token. A literal outside
  \<^const>‹I64› has no type in that list, and well-formedness rejects it
  (\<^const>‹lits_fit›). ‹esyn› is the kind an expression has of its own accord: a
  variable's kind, a literal's kind, the usual arithmetic conversion of an arithmetic
  operator's operands, and \<^const>‹I32› for the truth value of a comparison or a
  logical operator.
›

definition ik_of_lit :: "int ⇒ ikind" where
  "ik_of_lit n = (if ¦n¦ ≤ ik_max I32 then I32 else I64)"

fun esyn :: "(vname ⇒ ikind) ⇒ src_exp ⇒ ikind" where
  "esyn Γ (N n) = ik_of_lit n"
| "esyn Γ (V x) = Γ x"
| "esyn Γ (Plus a b) = usual_kind (esyn Γ a) (esyn Γ b)"
| "esyn Γ (Minus a b) = usual_kind (esyn Γ a) (esyn Γ b)"
| "esyn Γ (Times a b) = usual_kind (esyn Γ a) (esyn Γ b)"
| "esyn Γ (Div a b) = usual_kind (esyn Γ a) (esyn Γ b)"
| "esyn Γ (Mod a b) = usual_kind (esyn Γ a) (esyn Γ b)"
| "esyn Γ (Less a b) = I32"
| "esyn Γ (LessEq a b) = I32"
| "esyn Γ (Greater a b) = I32"
| "esyn Γ (GreaterEq a b) = I32"
| "esyn Γ (NotEq a b) = I32"
| "esyn Γ (Eq a b) = I32"
| "esyn Γ (Not a) = I32"
| "esyn Γ (And a b) = I32"
| "esyn Γ (Or a b) = I32"
| "esyn Γ (Cast ik a) = ik"

text ‹
  ‹src_eval Γ ik e s› is the value of ‹e› converted to kind ‹ik›. An arithmetic
  operator converts both operands to their common kind ‹k›, computes there with
  wraparound, and only then converts to ‹ik›. A comparison converts its operands
  the same way and yields ‹0› or ‹1›. A logical operator tests each operand at its
  own kind. Computing an inner operation at the context's kind instead would give
  ‹4294967295 < u + 1› the wrong value for an unsigned 32-bit ‹u›: the comparison
  agrees on 64 bits, but the sum wraps at 32 bits before the comparison sees it.
›

definition arith_at :: "ikind ⇒ ikind ⇒ aop ⇒ int ⇒ int ⇒ int" where
  "arith_at ik k op a b = ik_norm ik (ik_norm k (aop_eval op a b))"

fun src_eval :: "(vname ⇒ ikind) ⇒ ikind ⇒ src_exp ⇒ src_store ⇒ int" where
  "src_eval Γ ik (N n) s = ik_norm ik n"
| "src_eval Γ ik (V x) s = ik_norm ik (ik_norm (Γ x) (s x))"
| "src_eval Γ ik (Plus a b) s = (let k = esyn Γ (Plus a b) in
     arith_at ik k AAdd (src_eval Γ k a s) (src_eval Γ k b s))"
| "src_eval Γ ik (Minus a b) s = (let k = esyn Γ (Minus a b) in
     arith_at ik k ASub (src_eval Γ k a s) (src_eval Γ k b s))"
| "src_eval Γ ik (Times a b) s = (let k = esyn Γ (Times a b) in
     arith_at ik k AMul (src_eval Γ k a s) (src_eval Γ k b s))"
| "src_eval Γ ik (Div a b) s = (let k = esyn Γ (Div a b) in
     arith_at ik k ADiv (src_eval Γ k a s) (src_eval Γ k b s))"
| "src_eval Γ ik (Mod a b) s = (let k = esyn Γ (Mod a b) in
     arith_at ik k AMod (src_eval Γ k a s) (src_eval Γ k b s))"
| "src_eval Γ ik (Less a b) s = (let k = usual_kind (esyn Γ a) (esyn Γ b) in
     if cop_eval CLt (src_eval Γ k a s) (src_eval Γ k b s) then 1 else 0)"
| "src_eval Γ ik (LessEq a b) s = (let k = usual_kind (esyn Γ a) (esyn Γ b) in
     if cop_eval CLe (src_eval Γ k a s) (src_eval Γ k b s) then 1 else 0)"
| "src_eval Γ ik (Greater a b) s = (let k = usual_kind (esyn Γ a) (esyn Γ b) in
     if cop_eval CGt (src_eval Γ k a s) (src_eval Γ k b s) then 1 else 0)"
| "src_eval Γ ik (GreaterEq a b) s = (let k = usual_kind (esyn Γ a) (esyn Γ b) in
     if cop_eval CGe (src_eval Γ k a s) (src_eval Γ k b s) then 1 else 0)"
| "src_eval Γ ik (NotEq a b) s = (let k = usual_kind (esyn Γ a) (esyn Γ b) in
     if cop_eval CNe (src_eval Γ k a s) (src_eval Γ k b s) then 1 else 0)"
| "src_eval Γ ik (Eq a b) s = (let k = usual_kind (esyn Γ a) (esyn Γ b) in
     if cop_eval CEq (src_eval Γ k a s) (src_eval Γ k b s) then 1 else 0)"
| "src_eval Γ ik (Not a) s = (if truthy (src_eval Γ (esyn Γ a) a s) then 0 else 1)"
| "src_eval Γ ik (And a b) s =
     (if truthy (src_eval Γ (esyn Γ a) a s) ∧ truthy (src_eval Γ (esyn Γ b) b s) then 1 else 0)"
| "src_eval Γ ik (Or a b) s =
     (if truthy (src_eval Γ (esyn Γ a) a s) ∨ truthy (src_eval Γ (esyn Γ b) b s) then 1 else 0)"
| "src_eval Γ ik (Cast k a) s = ik_norm ik (ik_norm k (src_eval Γ (esyn Γ a) a s))"

subsection ‹Typed expressions and their evaluator›

text ‹
  A typed expression carries the kind each arithmetic node computes at and names
  variables by identity. ‹TCast›, the one node without a source counterpart, is a
  conversion, in the role of CIL's ‹CastE›. ‹eval› needs no kind environment: every
  kind it uses is on the node. A variable read does not convert, because a typed store
  already holds a value of the variable's kind.
›

datatype exp =
    TN ikind int
  | TVar ikind var_id
  | TArith ikind aop exp exp
  | TCast ikind exp
  | TCmp cop exp exp
  | TNot exp
  | TAnd exp exp
  | TOr exp exp

derive linorder aop
derive linorder cop
derive linorder exp

fun eval :: "exp ⇒ store ⇒ int" where
  "eval (TN ik n) t = ik_norm ik n"
| "eval (TVar ik v) t = t v"
| "eval (TArith ik op a b) t = ik_norm ik (aop_eval op (eval a t) (eval b t))"
| "eval (TCast ik a) t = ik_norm ik (eval a t)"
| "eval (TCmp op a b) t = (if cop_eval op (eval a t) (eval b t) then 1 else 0)"
| "eval (TNot a) t = (if truthy (eval a t) then 0 else 1)"
| "eval (TAnd a b) t = (if truthy (eval a t) ∧ truthy (eval b t) then 1 else 0)"
| "eval (TOr a b) t = (if truthy (eval a t) ∨ truthy (eval b t) then 1 else 0)"

text ‹The kind a node produces: the one an arithmetic or literal node carries, the
  target of a conversion, and \<^const>‹I32› for a truth value.›

fun exp_kind :: "exp ⇒ ikind" where
  "exp_kind (TN ik _) = ik"
| "exp_kind (TVar ik _) = ik"
| "exp_kind (TArith ik _ _ _) = ik"
| "exp_kind (TCast ik _) = ik"
| "exp_kind (TCmp _ _ _) = I32"
| "exp_kind (TNot _) = I32"
| "exp_kind (TAnd _ _) = I32"
| "exp_kind (TOr _ _) = I32"

text ‹A typed expression is well-typed when its kinds agree with each other and with
  the environment: an arithmetic node's operands carry its kind, a comparison's two
  operands share one, and a variable read carries the variable's kind. Abstract
  transfers that read an operand's kind off the operand rely on exactly this.›

fun wt_exp :: "kenv ⇒ exp ⇒ bool" where
  "wt_exp κ (TN ik n) ⟷ True"
| "wt_exp κ (TVar ik v) ⟷ κ v = ik"
| "wt_exp κ (TArith ik op a b) ⟷
     exp_kind a = ik ∧ exp_kind b = ik ∧ wt_exp κ a ∧ wt_exp κ b"
| "wt_exp κ (TCast ik a) ⟷ wt_exp κ a"
| "wt_exp κ (TCmp op a b) ⟷ exp_kind a = exp_kind b ∧ wt_exp κ a ∧ wt_exp κ b"
| "wt_exp κ (TNot a) ⟷ wt_exp κ a"
| "wt_exp κ (TAnd a b) ⟷ wt_exp κ a ∧ wt_exp κ b"
| "wt_exp κ (TOr a b) ⟷ wt_exp κ a ∧ wt_exp κ b"

theorem eval_in_range:
  assumes "wt_exp κ e" and "all_in_kinds κ t"
  shows "eval e t ∈ ik_range (exp_kind e)"
  using assms by (cases e) (auto dest: all_in_kindsD simp del: ik_range_iff)

subsection ‹Elaboration›

text ‹
  ‹elaborate ρ κ ik e› translates a source expression of one procedure, whose names
  ‹ρ› resolves, into a typed expression whose value has kind ‹ik›. It follows
  \<^const>‹src_eval› node for node and inserts a conversion only where a value reaches a
  context at another kind. A conversion to the kind a value already has is left out:
  it would change no value, but a domain without magnitudes, such as Sign, could only
  answer top for it.
›

definition cast_to :: "ikind ⇒ ikind ⇒ exp ⇒ exp" where
  "cast_to ik k a = (if ik = k then a else TCast ik a)"

fun elaborate :: "(vname ⇒ var_id) ⇒ kenv ⇒ ikind ⇒ src_exp ⇒ exp" where
  "elaborate ρ κ ik (N n) = TN ik n"
| "elaborate ρ κ ik (V x) = cast_to ik (κ (ρ x)) (TVar (κ (ρ x)) (ρ x))"
| "elaborate ρ κ ik (Plus a b) = (let k = esyn (κ ∘ ρ) (Plus a b) in
     cast_to ik k (TArith k AAdd (elaborate ρ κ k a) (elaborate ρ κ k b)))"
| "elaborate ρ κ ik (Minus a b) = (let k = esyn (κ ∘ ρ) (Minus a b) in
     cast_to ik k (TArith k ASub (elaborate ρ κ k a) (elaborate ρ κ k b)))"
| "elaborate ρ κ ik (Times a b) = (let k = esyn (κ ∘ ρ) (Times a b) in
     cast_to ik k (TArith k AMul (elaborate ρ κ k a) (elaborate ρ κ k b)))"
| "elaborate ρ κ ik (Div a b) = (let k = esyn (κ ∘ ρ) (Div a b) in
     cast_to ik k (TArith k ADiv (elaborate ρ κ k a) (elaborate ρ κ k b)))"
| "elaborate ρ κ ik (Mod a b) = (let k = esyn (κ ∘ ρ) (Mod a b) in
     cast_to ik k (TArith k AMod (elaborate ρ κ k a) (elaborate ρ κ k b)))"
| "elaborate ρ κ ik (Less a b) = (let k = usual_kind (esyn (κ ∘ ρ) a) (esyn (κ ∘ ρ) b) in
     cast_to ik I32 (TCmp CLt (elaborate ρ κ k a) (elaborate ρ κ k b)))"
| "elaborate ρ κ ik (LessEq a b) = (let k = usual_kind (esyn (κ ∘ ρ) a) (esyn (κ ∘ ρ) b) in
     cast_to ik I32 (TCmp CLe (elaborate ρ κ k a) (elaborate ρ κ k b)))"
| "elaborate ρ κ ik (Greater a b) = (let k = usual_kind (esyn (κ ∘ ρ) a) (esyn (κ ∘ ρ) b) in
     cast_to ik I32 (TCmp CGt (elaborate ρ κ k a) (elaborate ρ κ k b)))"
| "elaborate ρ κ ik (GreaterEq a b) = (let k = usual_kind (esyn (κ ∘ ρ) a) (esyn (κ ∘ ρ) b) in
     cast_to ik I32 (TCmp CGe (elaborate ρ κ k a) (elaborate ρ κ k b)))"
| "elaborate ρ κ ik (NotEq a b) = (let k = usual_kind (esyn (κ ∘ ρ) a) (esyn (κ ∘ ρ) b) in
     cast_to ik I32 (TCmp CNe (elaborate ρ κ k a) (elaborate ρ κ k b)))"
| "elaborate ρ κ ik (Eq a b) = (let k = usual_kind (esyn (κ ∘ ρ) a) (esyn (κ ∘ ρ) b) in
     cast_to ik I32 (TCmp CEq (elaborate ρ κ k a) (elaborate ρ κ k b)))"
| "elaborate ρ κ ik (Not a) =
     cast_to ik I32 (TNot (elaborate ρ κ (esyn (κ ∘ ρ) a) a))"
| "elaborate ρ κ ik (And a b) = cast_to ik I32
     (TAnd (elaborate ρ κ (esyn (κ ∘ ρ) a) a) (elaborate ρ κ (esyn (κ ∘ ρ) b) b))"
| "elaborate ρ κ ik (Or a b) = cast_to ik I32
     (TOr (elaborate ρ κ (esyn (κ ∘ ρ) a) a) (elaborate ρ κ (esyn (κ ∘ ρ) b) b))"
| "elaborate ρ κ ik (Cast k a) =
     cast_to ik k (TCast k (elaborate ρ κ (esyn (κ ∘ ρ) a) a))"

text ‹A context without a kind of its own, such as a branch condition, evaluates an
  expression at the kind it synthesizes.›

definition elaborate_syn :: "(vname ⇒ var_id) ⇒ kenv ⇒ src_exp ⇒ exp" where
  "elaborate_syn ρ κ e = elaborate ρ κ (esyn (κ ∘ ρ) e) e"

lemma exp_kind_cast_to [simp]:
  "exp_kind (cast_to ik k a) = (if ik = k then exp_kind a else ik)"
  by (simp add: cast_to_def)

lemma exp_kind_elaborate [simp]: "exp_kind (elaborate ρ κ ik e) = ik"
  by (cases e) (simp_all add: Let_def)

theorem wt_exp_elaborate: "wt_exp κ (elaborate ρ κ ik e)"
  by (induction e arbitrary: ik) (simp_all add: Let_def cast_to_def)

text ‹
  Elaboration preserves the meaning. The typed store ‹t› holds a value for every
  identity, and the procedure sees it through its resolution, as ‹t ∘ ρ›. The premise
  that ‹t› is typed is used at one place: \<^const>‹src_eval› reads a variable at its own
  kind, \<^const>‹eval› reads the stored value directly, and the two agree because the
  stored value already lies in that kind's range.
›

theorem eval_elaborate:
  assumes "all_in_kinds κ t"
  shows "eval (elaborate ρ κ ik e) t = src_eval (κ ∘ ρ) ik e (t ∘ ρ)"
proof -
  have var: "ik_norm (κ v) (t v) = t v" for v
    by (rule ik_norm_id [OF all_in_kindsD [OF assms]])
  show ?thesis
    by (induction e arbitrary: ik) (simp_all add: Let_def cast_to_def arith_at_def var)
qed

corollary eval_elaborate_syn:
  assumes "all_in_kinds κ t"
  shows "eval (elaborate_syn ρ κ e) t = src_eval (κ ∘ ρ) (esyn (κ ∘ ρ) e) e (t ∘ ρ)"
  using eval_elaborate [OF assms] by (simp add: elaborate_syn_def)

text ‹The identities a typed expression reads.›

fun exp_vars :: "exp ⇒ var_id set" where
  "exp_vars (TN _ _) = {}"
| "exp_vars (TVar _ v) = {v}"
| "exp_vars (TArith _ _ a b) = exp_vars a ∪ exp_vars b"
| "exp_vars (TCast _ a) = exp_vars a"
| "exp_vars (TCmp _ a b) = exp_vars a ∪ exp_vars b"
| "exp_vars (TNot a) = exp_vars a"
| "exp_vars (TAnd a b) = exp_vars a ∪ exp_vars b"
| "exp_vars (TOr a b) = exp_vars a ∪ exp_vars b"

lemma exp_vars_elaborate: "exp_vars (elaborate ρ κ ik e) ⊆ range ρ"
  by (induction e arbitrary: ik) (auto simp: Let_def cast_to_def)

text ‹A conversion to a kind whose range contains the source kind's range changes no
  value.›

definition ik_widens :: "ikind ⇒ ikind ⇒ bool" where
  "ik_widens k ik ⟷ ik_min ik ≤ ik_min k ∧ ik_max k ≤ ik_max ik"

subsection ‹Erasure›

text ‹
  Erasure drops the kinds and the conversions and names every identity by its source
  name. It recovers the expression elaboration started from. It belongs to no
  semantics: the erased expression has lost the kinds that \<^const>‹eval› wraps at, so
  read over unbounded integers it can mean something other than the original.
›

fun aop_exp :: "aop ⇒ src_exp ⇒ src_exp ⇒ src_exp" where
  "aop_exp AAdd = Plus" | "aop_exp ASub = Minus" | "aop_exp AMul = Times"
| "aop_exp ADiv = Div" | "aop_exp AMod = Mod"

fun cop_exp :: "cop ⇒ src_exp ⇒ src_exp ⇒ src_exp" where
  "cop_exp CLt = Less" | "cop_exp CLe = LessEq" | "cop_exp CGt = Greater"
| "cop_exp CGe = GreaterEq" | "cop_exp CEq = Eq" | "cop_exp CNe = NotEq"

fun exp_erase :: "exp ⇒ src_exp" where
  "exp_erase (TN _ n) = N n"
| "exp_erase (TVar _ v) = V (var_name v)"
| "exp_erase (TArith _ op a b) = aop_exp op (exp_erase a) (exp_erase b)"
| "exp_erase (TCast _ a) = exp_erase a"
| "exp_erase (TCmp op a b) = cop_exp op (exp_erase a) (exp_erase b)"
| "exp_erase (TNot a) = Not (exp_erase a)"
| "exp_erase (TAnd a b) = And (exp_erase a) (exp_erase b)"
| "exp_erase (TOr a b) = Or (exp_erase a) (exp_erase b)"

text ‹Erasure drops every conversion, including one the source wrote, so it recovers
  an expression without explicit casts.›

fun cast_free :: "src_exp ⇒ bool" where
  "cast_free (Cast _ _) ⟷ False"
| "cast_free (Plus a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free (Minus a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free (Times a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free (Div a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free (Mod a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free (Less a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free (LessEq a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free (Greater a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free (GreaterEq a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free (NotEq a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free (Eq a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free (Not a) ⟷ cast_free a"
| "cast_free (And a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free (Or a b) ⟷ cast_free a ∧ cast_free b"
| "cast_free _ ⟷ True"

theorem exp_erase_elaborate:
  assumes "⋀x. var_name (ρ x) = x" and "cast_free e"
  shows "exp_erase (elaborate ρ κ ik e) = e"
  using assms(2) by (induction e arbitrary: ik) (simp_all add: Let_def cast_to_def assms(1))
