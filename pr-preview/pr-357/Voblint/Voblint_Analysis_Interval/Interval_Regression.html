Theory Interval_Regression

theory Interval_Regression
  imports Interval_Kind "Voblint_Nonrelational.Nonrelational_Analysis" "Voblint_VIMP.VIMP_Notation"
begin

section ‹Regression: the typed Interval analysis, executed›

text ‹
  Each case runs the certified solver on a typed program inside Isabelle and pins the
  published value of one variable. Termination is decided by evaluation.
›


abbreviation ivl_equations :: "kind_decls ⇒ imp_prog ⇒ cfg_node × cfg_node list ⇒
    (cfg_node × cfg_node list, (unit, cfg_node list) global_unknown,
     (ivl tst lifted, ivl tst lifted) dg_state) strategy_tree" where
  "ivl_equations ds p ≡ nonrel.equations default_sem_conf ds 0 is_global p"

abbreviation ivl_state_at :: "kind_decls ⇒ imp_prog ⇒ cfg_node ⇒ ivl tabs" where
  "ivl_state_at ds p ≡ nonrel.state_at default_sem_conf ds 0 Globals_Join is_global p []"

abbreviation main_var :: "vname ⇒ var_id" where
  "main_var x ≡ ScopedId (STR ''main'') x"

subsection ‹A counting loop at \<^const>‹I32››

text ‹The guard bounds the body and the loop exit.›

definition loop_prog :: imp_prog where
  "loop_prog = program { fun main() { x = 0; while (x < 20) { x = x + 1; } } }"

lemma loop_terminates_c:
  "TD_side_rule_Interp_solve_c Globals_Join (ivl_equations [] loop_prog)
     (cfg_exit (elab_cfg loop_prog []), []) ≠ None"
  by eval

lemma loop_terminates:
  "nonrel.terminates TYPE(ivl) default_sem_conf [] 0 Globals_Join is_global loop_prog"
  unfolding nonrel.terminates_code
  using TD_side_rule_Interp.solve_dom_of_solve_c[OF loop_terminates_c] by simp

abbreviation loop_at :: "pp ⇒ ivl tabs" where
  "loop_at ≡ ivl_state_at [] loop_prog"

lemma loop_values:
  "loop_at (Statement 1) (main_var (STR ''x'')) = KD I32 (Ivl (Fin 0) (Fin 20))"
  "loop_at (Statement 2) (main_var (STR ''x'')) = KD I32 (Ivl (Fin 0) (Fin 19))"
  "loop_at (Statement 3) (main_var (STR ''x'')) = KD I32 (Ivl (Fin 20) (Fin 20))"
  by eval+

subsection ‹An unsigned char that wraps›

text ‹‹x› is a \<^const>‹U8›; ‹x + 10› is computed at \<^const>‹I32› and converted back.
  The value ‹260› leaves the range of \<^const>‹U8› and wraps to ‹4›, as in Goblint.›

definition wrap_prog :: imp_prog where
  "wrap_prog = program { fun main() { x = 250; x = x + 10; } }"

definition wrap_kinds :: kind_decls where
  "wrap_kinds = [(main_var (STR ''x''), U8)]"

lemma wrap_terminates_c:
  "TD_side_rule_Interp_solve_c Globals_Join (ivl_equations wrap_kinds wrap_prog)
     (cfg_exit (elab_cfg wrap_prog wrap_kinds), []) ≠ None"
  by eval

abbreviation wrap_at :: "pp ⇒ ivl tabs" where
  "wrap_at ≡ ivl_state_at wrap_kinds wrap_prog"

lemma wrap_values:
  "wrap_at (Statement 1) (main_var (STR ''x'')) = KD U8 (Ivl (Fin 250) (Fin 250))"
  "wrap_at (Statement 2) (main_var (STR ''x'')) = KD U8 (Ivl (Fin 4) (Fin 4))"
  by eval+


subsection ‹A signed char that overflows›

text ‹A signed kind does not wrap under Goblint's default overflow policy: converting
  ‹130› to \<^const>‹I8› answers the whole range.›

definition swrap_prog :: imp_prog where
  "swrap_prog = program { fun main() { x = 120; x = x + 10; } }"

definition swrap_kinds :: kind_decls where
  "swrap_kinds = [(main_var (STR ''x''), I8)]"

abbreviation swrap_at :: "pp ⇒ ivl tabs" where
  "swrap_at ≡ ivl_state_at swrap_kinds swrap_prog"

lemma swrap_values:
  "swrap_at (Statement 2) (main_var (STR ''x'')) = KD I8 (Ivl (Fin (-128)) (Fin 127))"
  by eval


subsection ‹Backward refinement through arithmetic and conversions›

text ‹Two calls give ‹f›'s formal ‹x› the range ‹[0,99]›. Then ‹x + 1› cannot
  overflow, so ‹x + 1 < 10› bounds ‹x› by ‹8›; without the inversion ‹y› would be
  ‹[0,99]› when ‹f› returns.›

abbreviation f_var :: "vname ⇒ var_id" where
  "f_var x ≡ ScopedId (STR ''f'') x"

definition arith_prog :: imp_prog where
  "arith_prog = program {
     fun f(x) { y = 0; if (x + 1 < 10) { y = x; } }
     fun main() { f(0); f(99); } }"

lemma arith_refined:
  "ivl_state_at [] arith_prog
            (FunctionResult (STR ''f'')) (f_var (STR ''y''))
     = KD I32 (Ivl (Fin 0) (Fin 8))"
  by eval

text ‹Comparing a signed ‹x› with an unsigned ‹y› converts ‹x› to unsigned. When
  ‹x› is known non-negative the conversion keeps its value, so ‹x < y› bounds ‹x›; when
  ‹x› may be negative the conversion is not injective on its values and refines
  nothing.›

definition mixed_kinds :: kind_decls where
  "mixed_kinds = [(f_var (STR ''y''), U32)]"

definition mixed_prog :: imp_prog where
  "mixed_prog = program {
     fun f(x) { y = 5; z = 0; if (x < y) { z = x; } }
     fun main() { f(0); f(9); } }"

definition mixed_neg_prog :: imp_prog where
  "mixed_neg_prog = program {
     fun f(x) { y = 5; z = 0; if (x < y) { z = x; } }
     fun main() { f(0 - 10); f(9); } }"

lemma mixed_refined:
  "ivl_state_at mixed_kinds mixed_prog
            (FunctionResult (STR ''f'')) (f_var (STR ''z''))
     = KD I32 (Ivl (Fin 0) (Fin 4))"
  "ivl_state_at mixed_kinds mixed_neg_prog
            (FunctionResult (STR ''f'')) (f_var (STR ''z''))
     = KD I32 (Ivl (Fin (-10)) (Fin 9))"
  by eval+

end
