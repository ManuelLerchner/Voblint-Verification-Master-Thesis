Theory Congruence_Regression

theory Congruence_Regression
  imports Congruence_Kind "Voblint_Nonrelational.Nonrelational_Analysis"
    "Voblint_VIMP.VIMP_Notation"
begin

section ‹Regression: the typed Congruence analysis, executed›

text ‹
  Each case runs the certified solver on a typed program inside Isabelle and pins the
  published value of one variable. Termination is decided by evaluation.
›

abbreviation cong_state_at :: "kind_decls ⇒ imp_prog ⇒ cfg_node ⇒ congruence tabs" where
  "cong_state_at ds p ≡ nonrel.state_at default_sem_conf ds 0 Globals_Join is_global p []"

abbreviation f_var :: "vname ⇒ var_id" where
  "f_var x ≡ ScopedId (STR ''f'') x"

abbreviation f_result :: "kind_decls ⇒ imp_prog ⇒ vname ⇒ congruence kd" where
  "f_result ds p x ≡ cong_state_at ds p (FunctionResult (STR ''f'')) (f_var x)"

abbreviation cong_equations :: "kind_decls ⇒ imp_prog ⇒ cfg_node × cfg_node list ⇒
    (cfg_node × cfg_node list, (unit, cfg_node list) global_unknown,
     (congruence tst lifted, congruence tst lifted) dg_state) strategy_tree" where
  "cong_equations ds p ≡ nonrel.equations default_sem_conf ds 0 is_global p"

subsection ‹Constants›

text ‹
  Under Goblint's default ▩‹assume_top›, standalone Congruence answers top for signed
  addition and multiplication, constants included; only a difference of constants that
  stays in range is kept. An unsigned kind computes constants modulo its range. A
  \<^const>‹U8› operand is promoted to the signed \<^const>‹I32› first, so the examples
  use \<^const>‹U32›.
›

definition const_prog :: imp_prog where
  "const_prog = program { fun f() { x = 3; x = x * 5; } fun main() { f(); } }"

definition sub_prog :: imp_prog where
  "sub_prog = program { fun f() { x = 3; x = x - 5; } fun main() { f(); } }"

definition u32_x :: kind_decls where
  "u32_x = [(f_var (STR ''x''), U32)]"

lemma const_value:
  "f_result [] const_prog (STR ''x'') = KD I32 top"
  "f_result u32_x const_prog (STR ''x'') = KD U32 (mk_congruence 15 0)"
  "f_result [] sub_prog (STR ''x'') = KD I32 (mk_congruence (-2) 0)"
  by eval+

subsection ‹Classes through a loop›

text ‹
  A step of ‹4› keeps an unsigned ‹x› in ‹4ℤ›: wrapping shifts a value by a multiple of
  ‹2 ^ 32›, which ‹4› divides (▩‹handle_overflow›). A step of ‹3› loses the class,
  since ‹3› does not divide ‹2 ^ 32›. At the signed \<^const>‹I32› the step answers top
  under ▩‹assume_top›; the interval product's ▩‹no_ov› recovers the class there.
›

definition pow2_prog :: imp_prog where
  "pow2_prog = program {
     fun f() { x = 0; while (x < 100) { x = x + 4; } }
     fun main() { f(); } }"

definition odd_step_prog :: imp_prog where
  "odd_step_prog = program {
     fun f() { x = 0; while (x < 99) { x = x + 3; } }
     fun main() { f(); } }"

lemma pow2_terminates_c:
  "TD_side_rule_Interp_solve_c Globals_Join (cong_equations [] pow2_prog)
     (cfg_exit (elab_cfg pow2_prog []), []) ≠ None"
  by eval

lemma pow2_terminates:
  "nonrel.terminates TYPE(congruence) default_sem_conf [] 0 Globals_Join is_global pow2_prog"
  unfolding nonrel.terminates_code
  using TD_side_rule_Interp.solve_dom_of_solve_c[OF pow2_terminates_c] by simp

lemma loop_values:
  "f_result u32_x pow2_prog (STR ''x'') = KD U32 (mk_congruence 0 4)"
  "f_result u32_x odd_step_prog (STR ''x'') = KD U32 top"
  "f_result [] pow2_prog (STR ''x'') = KD I32 top"
  by eval+

subsection ‹Overflow›

text ‹‹-6 + 10› wraps to ‹4› at \<^const>‹U32›, as in Goblint. A signed difference of
  constants that leaves the range answers top, Goblint's signed overflow under
  ▩‹assume_top›.›

definition wrap_prog :: imp_prog where
  "wrap_prog = program { fun f() { x = 0 - 6; x = x + 10; } fun main() { f(); } }"

definition swrap_prog :: imp_prog where
  "swrap_prog = program { fun f() { x = 0 - 2147483647; x = x - 10; } fun main() { f(); } }"

lemma wrap_values:
  "f_result u32_x wrap_prog (STR ''x'') = KD U32 (mk_congruence 4 0)"
  "f_result [] swrap_prog (STR ''x'') = KD I32 top"
  by eval+

subsection ‹Conversions›

text ‹
  ‹x› is a \<^const>‹U32› stepping by ‹8›. Widening it to the \<^const>‹I64› ‹y› keeps
  its class, as Goblint's ▩‹cast_to› does for a contained ▩‹from_ik›. Narrowing the
  class ‹6ℤ› from \<^const>‹I32› to \<^const>‹U8› answers top, as ▩‹cast_to› does for
  a non-constant; narrowing the constant ‹300› wraps it to ‹44›.
›

definition conv_prog :: imp_prog where
  "conv_prog = program {
     fun f() { x = 0; while (x < 200) { x = x + 8; } y = x; }
     fun main() { f(); } }"

definition conv_kinds :: kind_decls where
  "conv_kinds = [(f_var (STR ''x''), U32), (f_var (STR ''y''), I64)]"

definition narrow_prog :: imp_prog where
  "narrow_prog = program {
     fun f(c) { y = 0; if (c < 1) { y = 6; } z = y; w = 300; v = w; }
     fun main() { f(0); f(5); } }"

definition narrow_kinds :: kind_decls where
  "narrow_kinds = [(f_var (STR ''z''), U8), (f_var (STR ''v''), U8)]"

lemma conv_values:
  "f_result conv_kinds conv_prog (STR ''x'') = KD U32 (mk_congruence 0 8)"
  "f_result conv_kinds conv_prog (STR ''y'') = KD I64 (mk_congruence 0 8)"
  "f_result narrow_kinds narrow_prog (STR ''y'') = KD I32 (mk_congruence 0 6)"
  "f_result narrow_kinds narrow_prog (STR ''z'') = KD U8 top"
  "f_result narrow_kinds narrow_prog (STR ''v'') = KD U8 (mk_congruence 44 0)"
  by eval+

end
