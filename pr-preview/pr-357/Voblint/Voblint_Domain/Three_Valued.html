Theory Three_Valued

theory Three_Valued
  imports Main
begin

section ‹Three-valued Boolean combinators›

text ‹
  ‹bin_log f ann› lifts a Boolean connective ‹f› to ‹bool option› with the
  short-circuit reading ‹f› already has: one definite operand equal to the
  annihilator ‹ann› decides the result regardless of the other side, two
  definite operands are combined by ‹f›, and anything else is ‹None›.
  Goblint's ‹id_binary_log› (‹base.ml›) has the same shape; ‹and_opt› and
  ‹or_opt› are its two instances.

  Both expression evaluation and check classification combine three-valued
  answers this way, so the combinators sit below either of them.
›

definition bin_log ::
  "(bool ⇒ bool ⇒ bool) ⇒ bool ⇒ bool option ⇒ bool option ⇒ bool option" where
  "bin_log f ann x y =
     (if x = Some ann ∨ y = Some ann then Some ann
      else case (x, y) of (Some a, Some b) ⇒ Some (f a b) | _ ⇒ None)"

abbreviation and_opt :: "bool option ⇒ bool option ⇒ bool option" where
  "and_opt ≡ bin_log (∧) False"

abbreviation or_opt :: "bool option ⇒ bool option ⇒ bool option" where
  "or_opt ≡ bin_log (∨) True"

end
