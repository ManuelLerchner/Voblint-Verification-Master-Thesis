Theory Sign_Kind

theory Sign_Kind
  imports "Voblint_Analysis_Sign.Sign_Backward" "Voblint_Nonrelational.Nonrelational_Analysis"
    "Voblint_VIMP.VIMP_Notation"
begin

section ‹Signs at a kind›

text ‹
  A sign carries no magnitude, so it cannot rule out an overflow: normalizing a
  nonzero sign to a kind answers the kind's top, every sign at a signed kind and the
  nonnegative ones at an unsigned kind (\<^const>‹sign_cast›). Zero survives every
  normalization. A conversion from a kind whose range lies inside the target's
  changes no value and keeps the sign. Sign has no Goblint counterpart, and it is
  much weaker at machine kinds than a sign analysis over integers that do not wrap.
›

lemma minus_one_in_signed_range: "ik_signed k ⟹ - 1 ∈ ik_range k"
  by (cases k) (simp_all add: ik_min_def ik_max_def ik_half_def)

lemma unsigned_range_nonneg: "¬ ik_signed k ⟹ n ∈ ik_range k ⟹ 0 ≤ n"
  by (simp add: ik_min_def)

instantiation sign :: ik_domain
begin

definition widen_ik_sign :: "ikind ⇒ sign ⇒ sign ⇒ sign" where
  "widen_ik_sign k a b = a ⊔ b"

definition narrow_ik_sign :: "ikind ⇒ sign ⇒ sign ⇒ sign" where
  "narrow_ik_sign k a b = a ⊓ b"

definition empty_ik_sign :: "ikind ⇒ sign ⇒ bool" where
  "empty_ik_sign k a ⟷ a = SBot ∨ (¬ ik_signed k ∧ a = SNeg)"

instance proof
  fix k :: ikind and a b :: sign
  show "a ≤ widen_ik k a b" "b ≤ widen_ik k a b" by (simp_all add: widen_ik_sign_def)
  show "b ≤ a ⟹ b ≤ narrow_ik k a b" "b ≤ a ⟹ narrow_ik k a b ≤ a"
    by (simp_all add: narrow_ik_sign_def inf_absorb2)
  show "empty_ik k a ⟷ numeric_domain_class.gamma a ∩ ik_range k = {}"
  proof
    assume "empty_ik k a"
    then show "numeric_domain_class.gamma a ∩ ik_range k = {}"
      using unsigned_range_nonneg[of k]
      by (fastforce simp: empty_ik_sign_def simp del: ik_range_iff)
  next
    assume e: "numeric_domain_class.gamma a ∩ ik_range k = {}"
    have "0 ∉ gamma_sign a" "1 ∉ gamma_sign a" "ik_signed k ⟹ - 1 ∉ gamma_sign a"
      using e minus_one_in_signed_range[of k] zero_in_ik_range[of k] one_in_ik_range[of k]
      by (auto simp del: ik_range_iff zero_in_ik_range one_in_ik_range)
    then show "empty_ik k a" by (cases a) (auto simp: empty_ik_sign_def)
  qed
qed

end

fun sign_aop :: "aop ⇒ sign ⇒ sign ⇒ sign" where
  "sign_aop AAdd a b = a + b"
| "sign_aop ASub a b = a - b"
| "sign_aop AMul a b = a * b"
| "sign_aop ADiv a b = sign_div a b"
| "sign_aop AMod a b = sign_mod a b"

lemma sign_aop_sound:
  "m ∈ gamma_sign a ⟹ n ∈ gamma_sign b ⟹ aop_eval op m n ∈ gamma_sign (sign_aop op a b)"
  by (cases op) (simp_all add: sign_plus_sound sign_minus_sound sign_times_sound
      sign_div_sound sign_mod_sound)


instantiation sign :: kind_int_domain
begin

definition top_ik_sign :: "ikind ⇒ sign" where
  "top_ik_sign k = (if ik_signed k then STop else SNonNeg)"

definition of_int_ik_sign :: "ikind ⇒ int ⇒ sign" where
  "of_int_ik_sign k n = sign_of_int (ik_norm k n)"

definition cast_ik_sign :: "signed_overflow_policy ⇒ ikind option ⇒ ikind ⇒ sign ⇒ sign" where
  "cast_ik_sign ovp sk k a =
     (case sk of
        Some s ⇒ if ik_widens s k then a else sign_cast k a
      | None ⇒ sign_cast k a)"

text ‹Some operations on in-range operands cannot leave a signed kind's range, whatever
  their magnitudes: a sum of opposite signs, a difference of equal signs or with a
  negative minuend, a product with zero, a quotient unless both operands may be negative
  (only the minimum divided by ‹-1› overflows), and every remainder. Their sign needs no
  normalization.›

fun sign_safe :: "aop ⇒ sign ⇒ sign ⇒ bool" where
  "sign_safe AAdd a b ⟷ a ≤ SZero ∨ b ≤ SZero
     ∨ (a ≤ SNonNeg ∧ b ≤ SNonPos) ∨ (a ≤ SNonPos ∧ b ≤ SNonNeg)"
| "sign_safe ASub a b ⟷ b ≤ SZero ∨ (a ≤ SNonNeg ∧ b ≤ SNonNeg) ∨ (a ≤ SNeg ∧ b ≤ SNonPos)"
| "sign_safe AMul a b ⟷ a ≤ SZero ∨ b ≤ SZero"
| "sign_safe ADiv a b ⟷ a ≤ SNonNeg ∨ b ≤ SNonNeg"
| "sign_safe AMod a b ⟷ True"


lemma c_div_between:
  assumes "0 ≤ a ∨ 0 ≤ b"
  shows "min 0 a - (if 0 ≤ a then a else 0) ≤ c_div a b ∧ c_div a b ≤ max 0 a"
proof (cases "b = 0")
  case False
  have q1: "0 ≤ ¦a¦ div ¦b¦" using False by (simp add: pos_imp_zdiv_nonneg_iff)
  have q2: "¦a¦ div ¦b¦ ≤ ¦a¦"
    by (cases "a = 0") (simp_all add: int_div_le_self)
  note q = q1 q2
  then show ?thesis
    using False assms by (cases "0 ≤ a"; cases "0 < b") (auto simp: c_div_def sgn_if)
qed (simp add: c_div_def)

lemma sign_safe_range:
  assumes s: "ik_signed k" and m: "m ∈ ik_range k" and n: "n ∈ ik_range k"
    and ma: "m ∈ gamma_sign a" and nb: "n ∈ gamma_sign b" and safe: "sign_safe op a b"
  shows "aop_eval op m n ∈ ik_range k"
proof -
  have lo: "ik_min k = - ik_max k - 1"
    using s by (simp add: ik_min_def ik_max_def)
  have mk: "ik_min k ≤ m" "m ≤ ik_max k" and nk: "ik_min k ≤ n" "n ≤ ik_max k"
    using m n by simp_all
  have sg: "⋀x c. x ∈ gamma_sign c ⟹ c ≤ SZero ⟹ x = 0"
    "⋀x c. x ∈ gamma_sign c ⟹ c ≤ SNonNeg ⟹ 0 ≤ x"
    "⋀x c. x ∈ gamma_sign c ⟹ c ≤ SNonPos ⟹ x ≤ 0"
    "⋀x c. x ∈ gamma_sign c ⟹ c ≤ SNeg ⟹ x < 0"
    by (case_tac [!] c) (auto simp: less_eq_sign_def)
  show ?thesis
  proof (cases op)
    case ADiv
    have "0 ≤ m ∨ 0 ≤ n" using safe ADiv sg(2)[OF ma] sg(2)[OF nb] by auto
    with c_div_between[of m n] mk nk lo ADiv show ?thesis by (auto split: if_splits)
  next
    case AMod
    have z: "ik_min k ≤ 0" "0 ≤ ik_max k" by simp_all
    with c_mod_between[of m n] mk AMod show ?thesis
      by (cases "0 ≤ m") (auto simp: min_def max_def simp del: ik_min_le_zero zero_le_ik_max)
  qed (use safe mk nk lo sg ma nb in ‹auto dest: sg›)
qed

definition arith_ik_sign :: "signed_overflow_policy ⇒ ikind ⇒ aop ⇒ bool ⇒ sign ⇒ sign ⇒ sign" where
  "arith_ik_sign ovp k op fits a b =
     (if fits ∧ ik_signed k ∧ sign_safe op a b then sign_aop op a b
      else sign_cast k (sign_aop op a b))"

definition cmp_ik_sign :: "cop ⇒ sign ⇒ sign ⇒ bool option" where
  "cmp_ik_sign = cop_of sign_lt sign_eqb"

definition meet_ik_sign :: "sign ⇒ sign ⇒ sign" where
  "meet_ik_sign a b = a ⊓ b"

text ‹A disequality with zero excludes zero from the other side, the sign counterpart
  of the bound exclusion of ▩‹inv_bin_int› for ▩‹Ne›.›

fun sign_excl :: "sign ⇒ sign ⇒ sign" where
  "sign_excl SNonNeg SZero = SPos"
| "sign_excl SNonPos SZero = SNeg"
| "sign_excl SZero SZero = SBot"
| "sign_excl a b = a"

lemma sign_excl_sound:
  "m ≠ n ⟹ m ∈ gamma_sign a ⟹ n ∈ gamma_sign b ⟹ m ∈ gamma_sign (sign_excl a b)"
  by (induction a b rule: sign_excl.induct) auto

text ‹An order between two values refines each by the other's sign through the
  value-level \<^const>‹inv_less_sign›: a comparison reads values the kind already holds, so no
  wraparound is involved.›

definition refine_ik_sign :: "cop ⇒ sign ⇒ sign ⇒ sign × sign" where
  "refine_ik_sign c a b =
     (case c of
        CEq ⇒ (a ⊓ b, a ⊓ b)
      | CNe ⇒ (sign_excl a b, sign_excl b a)
      | CLt ⇒ inv_less_sign True a b
      | CGe ⇒ inv_less_sign False a b
      | CGt ⇒ prod.swap (inv_less_sign True b a)
      | CLe ⇒ prod.swap (inv_less_sign False b a))"

definition raw_aop_ik_sign :: "aop ⇒ sign ⇒ sign ⇒ sign" where
  "raw_aop_ik_sign = sign_aop"

definition fits_ik_sign :: "ikind ⇒ sign ⇒ bool" where
  "fits_ik_sign k a ⟷ a = SBot ∨ a = SZero"

definition answer_ik_sign :: "sign ⇒ answer" where
  "answer_ik_sign a = answer_with (int_sign_update (λ_. a))"

definition of_answer_ik_sign :: "ikind option ⇒ answer ⇒ sign" where
  "of_answer_ik_sign sk q = (case q of QBot ⇒ ⊥ | QTop ⇒ ⊤ | QLifted d ⇒ int_sign d)"

instance proof
  fix n m :: int and k :: ikind and a b :: sign and op :: aop and c :: cop and r :: bool
  show "n ∈ ik_range k ⟹ n ∈ numeric_domain_class.gamma (top_ik k :: sign)"
    using unsigned_range_nonneg[of k n] by (auto simp: top_ik_sign_def simp del: ik_range_iff)
  show "ik_norm k n ∈ numeric_domain_class.gamma (of_int_ik k n :: sign)"
    by (simp add: of_int_ik_sign_def sign_of_int_gamma)
  show "(⋀k'. sk = Some k' ⟹ n ∈ ik_range k') ⟹ n ∈ numeric_domain_class.gamma a
      ⟹ ik_norm k n ∈ numeric_domain_class.gamma (cast_ik ovp sk k a)" for ovp sk
    by (cases sk) (auto simp: cast_ik_sign_def ik_widens_def sign_cast_sound split: if_splits)
  show "(fits ⟹ m ∈ ik_range k ∧ n ∈ ik_range k) ⟹ m ∈ numeric_domain_class.gamma a
      ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ ik_norm k (aop_eval op m n) ∈ numeric_domain_class.gamma (arith_ik ovp k op fits a b)" for ovp fits
    using sign_safe_range[of k m n a b op] sign_aop_sound[of m a n b op]
    by (auto simp: arith_ik_sign_def sign_cast_sound simp del: ik_range_iff)
  show "cmp_ik c a b = Some r ⟹ m ∈ numeric_domain_class.gamma a
      ⟹ n ∈ numeric_domain_class.gamma b ⟹ cop_eval c m n = r"
    unfolding cmp_ik_sign_def by simp (rule cop_of_sound[OF sign_lt_sound sign_eqb_sound])
  show "m ∈ numeric_domain_class.gamma a ⟹ m ∈ numeric_domain_class.gamma b
      ⟹ m ∈ numeric_domain_class.gamma (meet_ik a b)"
    by (simp add: meet_ik_sign_def)
  show "cop_eval c m n ⟹ m ∈ numeric_domain_class.gamma a ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ m ∈ numeric_domain_class.gamma (fst (refine_ik c a b))
        ∧ n ∈ numeric_domain_class.gamma (snd (refine_ik c a b))"
    using inv_less_sign_sound[of m a n b True] inv_less_sign_sound[of m a n b False]
      inv_less_sign_sound[of n b m a True] inv_less_sign_sound[of n b m a False]
      sign_excl_sound[of m n a b] sign_excl_sound[of n m b a]
    by (cases c) (auto simp: refine_ik_sign_def not_less)
  show "m ∈ numeric_domain_class.gamma a ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ aop_eval op m n ∈ numeric_domain_class.gamma (raw_aop_ik op a b)"
    by (simp add: raw_aop_ik_sign_def sign_aop_sound)
  show "fits_ik k a ⟹ n ∈ numeric_domain_class.gamma a ⟹ n ∈ ik_range k"
    by (auto simp: fits_ik_sign_def)
  show "n ∈ numeric_domain_class.gamma a ⟹ n ∈ gamma_answer (answer_ik a)"
    by (simp add: answer_ik_sign_def gamma_int_dom_def top_int_dom_ext_def gamma_sign_top
        gamma_parity_top top_ivl_def gamma_ivl_top)
  show "(⋀k. sk = Some k ⟹ n ∈ ik_range k) ⟹ n ∈ gamma_answer q
      ⟹ n ∈ numeric_domain_class.gamma (of_answer_ik sk q :: sign)" for sk q
    by (cases q)
      (simp_all add: of_answer_ik_sign_def gamma_int_dom_def gamma_sign_top gamma_parity_top)
qed

end

text ‹An unsigned ‹x› is nonnegative whatever its arithmetic did.›

definition sign_prog :: imp_prog where
  "sign_prog = program { fun f() { x = 0; x = x - 5; } fun main() { f(); } }"

lemma sign_value:
  "nonrel.state_at default_sem_conf [(ScopedId (STR ''f'') (STR ''x''), U8)] 0 Globals_Join is_global sign_prog []
     (FunctionResult (STR ''f'')) (ScopedId (STR ''f'') (STR ''x''))
     = KD U8 SNonNeg"
