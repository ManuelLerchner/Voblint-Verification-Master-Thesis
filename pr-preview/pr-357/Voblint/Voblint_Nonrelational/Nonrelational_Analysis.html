section ‹The typed non-relational analysis›

text ‹
  The executable typed component at any kind-aware integer domain, run by the shared
  D/G pipeline over the typed CFG. The kind declarations ‹ds› are a parameter of the
  instance: they fix the kind environment the concretization, the transfer and the
  initial state read, and the program varies under them. The domain is a type
  parameter, so every constant of the instance takes the domain's ‹TYPE› first.
›

subsection ‹Verdicts›

definition nonrel_classify :: "sem_conf ⇒ kenv ⇒ exp ⇒ 'a::kind_int_domain tabs ⇒ check_result"
  where
  "nonrel_classify sem κ c d =
     (case truth_ik (aeval sem κ (λ_. ⊤) c d) of
        Some True ⇒ Check_Proved
      | Some False ⇒ Check_Refuted
      | None ⇒ Check_Unknown)"

lemma nonrel_classify_sound:
  assumes s: "s ∈ gamma_tabs κ d"
  shows nonrel_classify_proved: "nonrel_classify sem κ c d = Check_Proved ⟹ truthy (eval c s)"
    and nonrel_classify_refuted: "nonrel_classify sem κ c d = Check_Refuted ⟹ ¬ truthy (eval c s)"
proof -
  have truth: "truthy (eval c s) = b"
    if "truth_ik (aeval sem κ (λ_. ⊤) c d) = Some b" for b
    using s by (rule truth_ik_sound[OF that aeval_sound]) simp
  show "nonrel_classify sem κ c d = Check_Proved ⟹ truthy (eval c s)"
    using truth by (auto simp: nonrel_classify_def split: option.splits bool.splits)
  show "nonrel_classify sem κ c d = Check_Refuted ⟹ ¬ truthy (eval c s)"
    using truth by (auto simp: nonrel_classify_def split: option.splits bool.splits)
qed

subsection ‹The initial state›

text ‹Every global starts at zero at its declared kind; every local is unconstrained.›

definition nonrel_init_st :: "kind_decls ⇒ 'a::kind_int_domain tst" where
  "nonrel_init_st ds =
     ⟪(KTop, []), (KD I32 (of_int_ik I32 0), map (λ(v, k). (v, KD k (of_int_ik k 0))) ds)⟫"

lemma nonrel_rd_init_st:
  "nonrel_rd (nonrel_init_st ds :: 'a::kind_int_domain tst)
     = (λv. if is_global v then KD (kenv_of ds v) (of_int_ik (kenv_of ds v) 0) else KTop)"
  by (auto simp: nonrel_init_st_def default_st_to_fun_def location_of_def kenv_of_def map_of_map
      fun_eq_iff split: option.splits)

lemma nonrel_init_sound:
  "init_stores (kenv_of ds)
     ⊆ gamma_tabs (kenv_of ds) (nonrel_rd (nonrel_init_st ds :: 'a::kind_int_domain tst))"
proof
  fix t assume t: "t ∈ init_stores (kenv_of ds)"
  have rng: "t v ∈ ik_range (kenv_of ds v)" for v
    using t by (simp add: init_stores_def all_in_kindsD del: ik_range_iff)
  show "t ∈ gamma_tabs (kenv_of ds) (nonrel_rd (nonrel_init_st ds))"
    unfolding nonrel_rd_init_st
  proof (rule gamma_tabsI[OF init_stores_in_kinds[OF t]], goal_cases)
    case (1 v)
    show ?case
    proof (cases v)
      case (GlobalId x)
      then have "t v = 0" using t by (simp add: init_stores_def)
      then show ?thesis
        using of_int_ik_sound[of "kenv_of ds v" "t v"] rng[of v] GlobalId
        by (simp del: ik_range_iff)
    qed simp_all
  qed
qed

subsection ‹The pipeline instance›

text ‹
  Call-string contexts of bound ‹k›, after Goblint's ▩‹callstring.ml›: a call pushes its
  call site and keeps the ‹k› most recent. At ‹k = 0› every activation shares the empty
  string, the context-insensitive analysis. The joined global unknown and the certified
  solver run at any update rule ‹r›. The component reads no global, so every variable
  lives in a point's own state.
›

lemma elab_cfg_calls_source_unique: "calls_source_unique (elab_cfg p ds)"
  unfolding calls_source_unique_def elab_cfg_def using compile_prog_calls_source_unique by blast

lemma nonrel_exec_spec_sound_rd:
  "sound_local_spec κ
     (λd. gamma_lift (gamma_tabs κ) (map_lift nonrel_rd d))
     (nonrel_exec_spec sem κ :: 'a::kind_int_domain tst lifted local_spec)"
  using nonrel_exec_spec_sound[of κ sem, where 'a = 'a] by (simp add: gamma_exec_def[abs_def])

text ‹What the component owes the pipeline does not depend on the context policy, so it
  is proved once, for any route and root context.›

lemma nonrel_routed:
  "dg_analysis
    (kenv_of ds) (λ_. init_stores (kenv_of ds))
    (λ_ _. nonrel_exec_spec sem (kenv_of ds) :: 'a::kind_int_domain tst lifted local_spec)
    (λ_. st_is_bot) (λ𝒢. ρ⇘𝒢⇙) (nonrel_init_st ds)
    (Analysis_Global ()) Analysis_Global Activation_Seed
    (TD_side_rule_Interp_solve r)
    (TD_side_rule_Interp.solve_dom TYPE((unit, 'c) global_unknown)
       TYPE(('a tst lifted, 'a tst lifted) dg_state) r)
    ⊥ (nonrel_classify sem (kenv_of ds)) (gamma_tabs (kenv_of ds)) t_empty
    (TD_side_rule_Interp_solve_c r)
    (λ_ c. dg_spec_of c) (λp d e. d) (λ_ d. d) (λ_ d. Bot) (λ_ p. []) (λp ci d e. d)
    (λp. elab_cfg p ds) (λ_. FunctionEntry prog_main_name)"
proof (rule dg_analysis_whole_stateI[OF td_certified_solver], goal_cases CompSound EnterSingle
    EmptyRd EmptyV SeedNe ClProved ClRefuted BotState InitSound SeedNeGlobal PEntry FinIntra
    FinCalls CallsUnique)
  case (CompSound p) show ?case by (rule nonrel_exec_spec_sound_rd)
next
  case (EnterSingle p ci d) show ?case
    by (simp add: nonrel_exec_spec_def dg_pipeline.comp_entry_def)
next
  case (EmptyRd p s) show ?case by (simp add: st_is_bot_iff t_empty_def)
next
  case EmptyV show ?case by (rule sound_emptinessI) (rule t_empty_sound)
next
  case (SeedNe v ctx) show ?case by simp
next
  case (ClProved c d s) then show ?case by (intro nonrel_classify_proved)
next
  case (ClRefuted c d s) then show ?case by (intro nonrel_classify_refuted)
next
  case BotState show ?case by simp
next
  case (InitSound p) show ?case by (rule nonrel_init_sound)
next
  case (SeedNeGlobal v ctx n) show ?case by simp
next
  case (PEntry p) show ?case by (simp add: elab_cfg_def)
next
  case (FinIntra p) show ?case by (simp add: elab_cfg_def compile_prog_finite)
next
  case (FinCalls p) show ?case by (simp add: elab_cfg_def compile_prog_finite)
next
  case (CallsUnique p) show ?case by (rule elab_cfg_calls_source_unique)
qed

global_interpretation nonrel: dg_analysis
    "kenv_of ds" "λ_. init_stores (kenv_of ds)"
    "λ_ _. nonrel_exec_spec sem (kenv_of ds) :: 'a::kind_int_domain tst lifted local_spec"
    "λ_. st_is_bot" "λ𝒢. ρ⇘𝒢⇙" "nonrel_init_st ds"
    "Analysis_Global ()" Analysis_Global Activation_Seed "λ_. cs_route k" "[]"
    "TD_side_rule_Interp_solve r"
    "TD_side_rule_Interp.solve_dom TYPE((unit, cfg_node list) global_unknown)
       TYPE(('a tst lifted, 'a tst lifted) dg_state) r"
    ⊥ "nonrel_classify sem (kenv_of ds)" "gamma_tabs (kenv_of ds)" t_empty
    "TD_side_rule_Interp_solve_c r"
    "λ_ c. dg_spec_of c" "λp d e. d" "λ_ d. d" "λ_ d. Bot" "λ_ p. []" "λp ci d e. d"
    "λp. elab_cfg p ds" "λ_. FunctionEntry prog_main_name"
  for sem ds r k
  by (rule nonrel_routed)

subsection ‹Entry-state contexts›

text ‹
  A call's context is the abstract values of the callee's formals in the state the call
  enters. Goblint's default context is the whole entry state
  (▩‹DefaultSpec.context›); the formals are the part of it a call determines, since
  the callee's other locals start unconstrained.
›

definition nonrel_es_ctx ::
    "(var_id ⇒ bool) ⇒ 'a::kind_int_domain tst lifted ⇒ (var_id, exp) gcall_action
       ⇒ 'a kd list" where
  "nonrel_es_ctx 𝒢 d ca =
     (case ca of CallEdge dst pars args ⇒
        (case d of Lifted st ⇒ map (ρ⇘𝒢⇙ st) pars
         | Bot ⇒ map (λ_. ⊥) pars))"

global_interpretation nonrel_es: dg_analysis
    "kenv_of ds" "λ_. init_stores (kenv_of ds)"
    "λ_ _. nonrel_exec_spec sem (kenv_of ds) :: 'a::kind_int_domain tst lifted local_spec"
    "λ_. st_is_bot" "λ𝒢. ρ⇘𝒢⇙" "nonrel_init_st ds"
    "Analysis_Global ()" Analysis_Global Activation_Seed "λ𝒢 u ctx d ca. nonrel_es_ctx 𝒢 d ca" "[]"
    "TD_side_rule_Interp_solve r"
    "TD_side_rule_Interp.solve_dom TYPE((unit, 'a kd list) global_unknown)
       TYPE(('a tst lifted, 'a tst lifted) dg_state) r"
    ⊥ "nonrel_classify sem (kenv_of ds)" "gamma_tabs (kenv_of ds)" t_empty
    "TD_side_rule_Interp_solve_c r"
    "λ_ c. dg_spec_of c" "λp d e. d" "λ_ d. d" "λ_ d. Bot" "λ_ p. []" "λp ci d e. d"
    "λp. elab_cfg p ds" "λ_. FunctionEntry prog_main_name"
  for sem ds r
  by (rule nonrel_routed)
