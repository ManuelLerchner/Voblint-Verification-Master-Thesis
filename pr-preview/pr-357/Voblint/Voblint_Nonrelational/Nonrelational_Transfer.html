section ‹The typed non-relational transfer›

text ‹
  What one CFG edge does to a typed abstract state, after Goblint's ▩‹base›
  analysis. An assignment evaluates its right-hand side and converts the value to the
  destination's kind (▩‹cast_to›); a library call writes its result at the
  destination's kind; a guard is dropped when its condition is decided against it; a
  return writes the returned value into the return slot; a call resets the callee's
  locals to their kinds' whole ranges and binds the formals; and the return combine
  takes the callee's globals and writes the converted return value.
›

fun neg_cop :: "cop ⇒ cop" where
  "neg_cop CLt = CGe" | "neg_cop CLe = CGt" | "neg_cop CGt = CLe" | "neg_cop CGe = CLt"
| "neg_cop CEq = CNe" | "neg_cop CNe = CEq"

lemma cop_eval_neg_cop [simp]: "cop_eval (neg_cop c) m n ⟷ ¬ cop_eval c m n"
  by (cases c) auto

text ‹The Boolean structure of a typed condition, for \<^locale>‹bool_refine›.›

fun nonrel_view :: "exp ⇒ exp bool_view" where
  "nonrel_view (TNot a) = BV_Not a"
| "nonrel_view (TAnd a b) = BV_And a b"
| "nonrel_view (TOr a b) = BV_Or a b"
| "nonrel_view _ = BV_Atom"

lemma eval_in_range_kind:
  assumes "¬ (∃ik v. e = TVar ik v)"
  shows "eval e t ∈ ik_range (exp_kind e)"
  using assms by (cases e) auto

text ‹Normalizing an intermediate result changes no later sum or difference at the same
  kind, so an operation that wrapped is inverted exactly by the wrapping inverse.›

lemma ik_norm_mod_eq: "ik_norm k m mod ik_mod k = m mod ik_mod k"
  by (simp add: ik_norm_mod mod_add_left_eq)

lemma ik_norm_cong: "x mod ik_mod k = y mod ik_mod k ⟹ ik_norm k x = ik_norm k y"
  by (metis ik_norm_mod mod_diff_cong)

lemma ik_norm_norm_add: "ik_norm k (ik_norm k m + n) = ik_norm k (m + n)"
  by (intro ik_norm_cong mod_add_cong ik_norm_mod_eq refl)

lemma ik_norm_norm_diff: "ik_norm k (ik_norm k m - n) = ik_norm k (m - n)"
  by (intro ik_norm_cong mod_diff_cong ik_norm_mod_eq refl)

lemma ik_norm_diff_norm: "ik_norm k (n - ik_norm k m) = ik_norm k (n - m)"
  by (intro ik_norm_cong mod_diff_cong ik_norm_mod_eq refl)

text ‹The smaller of two values is the left one where the left is at most the right,
  and the right one elsewhere; each case refines its operand by the comparison that
  selects it, so constants or disjoint ranges decide the result.›

definition min_ik :: "'a::kind_int_domain ⇒ 'a ⇒ 'a" where
  "min_ik a b = fst (refine_ik CLe a b) ⊔ snd (refine_ik CGt a b)"

definition max_ik :: "'a::kind_int_domain ⇒ 'a ⇒ 'a" where
  "max_ik a b = snd (refine_ik CLe a b) ⊔ fst (refine_ik CGt a b)"

lemma min_ik_sound:
  assumes "m ∈ numeric_domain_class.gamma a" and "n ∈ numeric_domain_class.gamma b"
  shows "min m n ∈ numeric_domain_class.gamma (min_ik a b :: 'a::kind_int_domain)"
proof (cases "m ≤ n")
  case True
  then have "m ∈ numeric_domain_class.gamma (fst (refine_ik CLe a b))"
    using refine_ik_sound[of CLe m n a b] assms by simp
  with True show ?thesis using gamma_mono[OF sup.cobounded1] by (fastforce simp: min_ik_def)
next
  case False
  then have "n ∈ numeric_domain_class.gamma (snd (refine_ik CGt a b))"
    using refine_ik_sound[of CGt m n a b] assms by simp
  with False show ?thesis using gamma_mono[OF sup.cobounded2] by (fastforce simp: min_ik_def)
qed

lemma max_ik_sound:
  assumes "m ∈ numeric_domain_class.gamma a" and "n ∈ numeric_domain_class.gamma b"
  shows "max m n ∈ numeric_domain_class.gamma (max_ik a b :: 'a::kind_int_domain)"
proof (cases "m ≤ n")
  case True
  then have "n ∈ numeric_domain_class.gamma (snd (refine_ik CLe a b))"
    using refine_ik_sound[of CLe m n a b] assms by simp
  with True show ?thesis using gamma_mono[OF sup.cobounded1] by (fastforce simp: max_ik_def)
next
  case False
  then have "m ∈ numeric_domain_class.gamma (fst (refine_ik CGt a b))"
    using refine_ik_sound[of CGt m n a b] assms by simp
  with False show ?thesis using gamma_mono[OF sup.cobounded2] by (fastforce simp: max_ik_def)
qed

text ‹The minimum or maximum of two operands of one kind lies in that kind's range.›

definition pair_kind :: "kenv ⇒ exp ⇒ exp ⇒ ikind option" where
  "pair_kind κ a b = (if src_kind κ a = src_kind κ b then src_kind κ a else None)"

context
  fixes sem :: sem_conf and κ :: kenv and ch :: "channel"
begin

definition nonrel_assign :: "var_id ⇒ exp ⇒ 'a::kind_int_domain tabs ⇒ 'a tabs" where
  "nonrel_assign x e σ = σ(x := KD (κ x) (cast_ik (sem_int_signed_overflow sem) (src_kind κ e) (κ x) (aeval_ask sem κ ch e σ)))"

definition special_val :: "exp gspecial_call ⇒ var_id ⇒ 'a::kind_int_domain tabs ⇒ 'a" where
  "special_val sc x σ =
     (case sc of Nondet_Int ⇒ top_ik (κ x)
      | Min a b ⇒ cast_ik (sem_int_signed_overflow sem) (pair_kind κ a b) (κ x) (min_ik (aeval_ask sem κ ch a σ) (aeval_ask sem κ ch b σ))
      | Max a b ⇒ cast_ik (sem_int_signed_overflow sem) (pair_kind κ a b) (κ x) (max_ik (aeval_ask sem κ ch a σ) (aeval_ask sem κ ch b σ)))"

definition nonrel_special :: "exp gspecial_call ⇒ var_id ⇒ 'a::kind_int_domain tabs ⇒ 'a tabs" where
  "nonrel_special sc x σ = σ(x := KD (κ x) (special_val sc x σ))"

text ‹
  A guard narrows the variables it compares, as Goblint's ▩‹baseInvariant›
  does. A comparison refines each operand by the value it must have and passes that
  value down: a conversion whose operand already fits the target kind keeps every value,
  and a sum or difference that cannot overflow is inverted on mathematical integers. One
  that may overflow is inverted at its kind, with wraparound, for each operand of that
  kind, as Goblint's ▩‹inv_bin_int› does: every operand value in the kind's range
  is the wrapped difference of the result and the other operand. Any other node stops
  the refinement. A negation flips the
  polarity, a conjunction that holds and a disjunction that fails narrow by both sides.
›

definition narrow_var :: "var_id ⇒ 'a::kind_int_domain ⇒ 'a tabs ⇒ 'a tabs" where
  "narrow_var v a σ =
     (if v = ReturnVar then σ else σ(v := KD (κ v) (meet_ik a (read_var κ σ v))))"

fun refine_var :: "exp ⇒ 'a::kind_int_domain ⇒ 'a tabs ⇒ 'a tabs" where
  "refine_var (TVar ik v) a σ = narrow_var v a σ"
| "refine_var (TCast ik e) a σ =
     (if fits_ik ik (aeval_ask sem κ ch e σ) then refine_var e a σ else σ)"
| "refine_var (TArith ik op e⇩1 e⇩2) a σ =
     (let a⇩1 = aeval_ask sem κ ch e⇩1 σ; a⇩2 = aeval_ask sem κ ch e⇩2 σ
      in if fits_ik ik (raw_aop_ik op a⇩1 a⇩2) then
           (case op of
             AAdd ⇒ refine_var e⇩2 (raw_aop_ik ASub a a⇩1) (refine_var e⇩1 (raw_aop_ik ASub a a⇩2) σ)
           | ASub ⇒ refine_var e⇩2 (raw_aop_ik ASub a⇩1 a) (refine_var e⇩1 (raw_aop_ik AAdd a a⇩2) σ)
           | _ ⇒ σ)
         else if op ∉ {AAdd, ASub} then σ
         else let σ' = (if src_kind κ e⇩1 = Some ik
                         then refine_var e⇩1 (arith_ik (sem_int_signed_overflow sem) ik (if op = AAdd then ASub else AAdd) False a a⇩2) σ else σ)
              in if src_kind κ e⇩2 = Some ik
                 then refine_var e⇩2 (if op = AAdd then arith_ik (sem_int_signed_overflow sem) ik ASub False a a⇩1 else arith_ik (sem_int_signed_overflow sem) ik ASub False a⇩1 a) σ'
                 else σ')"
| "refine_var e a σ = σ"

text ‹
  The Boolean structure of a guard is the generic one of \<^locale>‹bool_refine›: a
  comparison is the atom, narrowed by \<^const>‹refine_ik›; an operand of a disjunction
  that must hold, or of a conjunction that must fail, is kept only while its forward value
  admits the required truth value, and the kept alternatives are joined. When none is
  kept no store satisfies the guard, and the result is bottom, as Goblint's
  ▩‹Deadcode›. A branch keeps the refinement only at the variables its guard reads,
  where the executable state stores it; a variable refined to bottom marks the branch
  unreachable.
›

definition nonrel_narrow :: "exp ⇒ bool ⇒ 'a::kind_int_domain tabs ⇒ 'a tabs" where
  "nonrel_narrow e pol σ =
     (case e of
        TCmp c l r ⇒
          (let (al, ar) = refine_ik (if pol then c else neg_cop c)
                            (aeval_ask sem κ ch l σ) (aeval_ask sem κ ch r σ)
           in refine_var l al (refine_var r ar σ))
      | _ ⇒ refine_var e (fst (refine_ik (if pol then CNe else CEq)
                         (aeval_ask sem κ ch e σ) (of_int_ik (exp_kind e) 0))) σ)"

text ‹
  An atom that leaves one of its variables without a value describes no store, and is
  bottom as a whole: joined with the other alternative of a disjunction, it then adds
  nothing, where the variables it did not narrow would otherwise bring the incoming
  state back. Only the atom's variables change, so checking them suffices.
›

definition nonrel_atom :: "exp ⇒ bool ⇒ 'a::kind_int_domain tabs ⇒ 'a tabs" where
  "nonrel_atom e pol σ =
     (let σ' = nonrel_narrow e pol σ in if ∃v ∈ exp_vars e. is_empty (σ' v) then ⊥ else σ')"

definition nonrel_feas :: "exp ⇒ bool ⇒ 'a::kind_int_domain tabs ⇒ bool" where
  "nonrel_feas e pol σ = (truth_ik (aeval_ask sem κ ch e σ) ≠ Some (¬ pol))"

lemma nonrel_bool_locale: "bool_refine nonrel_view size"
  by unfold_locales (auto elim: nonrel_view.elims)

definition nonrel_refine :: "exp ⇒ bool ⇒ 'a::kind_int_domain tabs ⇒ 'a tabs" where
  "nonrel_refine = bool_refine.brefine nonrel_view nonrel_atom nonrel_feas sup (λ_. ⊥)"

definition nonrel_branch :: "exp ⇒ bool ⇒ 'a::kind_int_domain tabs ⇒ 'a tabs" where
  "nonrel_branch b pol σ =
     (let refined = (λv. if v ∈ exp_vars b then nonrel_refine b pol σ v else σ v)
      in case truth_ik (aeval_ask sem κ ch b σ) of
           Some r ⇒ if r = pol then refined else ⊥
         | None ⇒ refined)"

text ‹The returned value is tagged with a kind whose range it lies in: a variable's
  declared kind, or the kind an operation normalized it to.›

definition ret_kd :: "exp ⇒ 'a::kind_int_domain tabs ⇒ 'a kd" where
  "ret_kd e σ =
     (case e of TVar ik v ⇒ if v = ReturnVar then KTop else KD (κ v) (read_var κ σ v)
      | _ ⇒ KD (exp_kind e) (aeval_ask sem κ ch e σ))"

definition nonrel_return :: "exp option ⇒ 'a::kind_int_domain tabs ⇒ 'a tabs" where
  "nonrel_return r σ = (case r of None ⇒ σ | Some e ⇒ σ(ReturnVar := ret_kd e σ))"

definition nonrel_enter_frame :: "'a::kind_int_domain tabs ⇒ 'a tabs" where
  "nonrel_enter_frame σ = enter_frame is_global KTop σ"

definition nonrel_enter_state ::
    "var_id list ⇒ exp list ⇒ 'a::kind_int_domain tabs ⇒ 'a tabs" where
  "nonrel_enter_state ps es σ =
     bind_formals ps (map2 (λp e. KD (κ p) (cast_ik (sem_int_signed_overflow sem) (src_kind κ e) (κ p) (aeval_ask sem κ ch e σ))) ps es)
       (nonrel_enter_frame σ)"

definition nonrel_combine_assign :: "var_id option ⇒ 'a::kind_int_domain tabs ⇒ 'a tabs ⇒ 'a tabs"
  where
  "nonrel_combine_assign dst σ σ⇩e =
     (case dst of None ⇒ σ
      | Some x ⇒ σ(x := KD (κ x)
          (cast_ik (sem_int_signed_overflow sem) (kd_kind (σ⇩e ReturnVar)) (κ x) (kd_val (σ⇩e ReturnVar)))))"

subsection ‹Soundness of each operation›

text ‹Each operation's result describes the store its edge produces from any typed
  store the input state describes.›

lemma in_kinds_upd:
  "in_kinds κ t ⟹ (x ≠ ReturnVar ⟹ n ∈ ik_range (κ x)) ⟹ in_kinds κ (t(x := n))"
  by (auto simp: in_kinds_def)

lemma gamma_tabs_upd:
  fixes d :: "'a::kind_int_domain kd"
  assumes "t ∈ gamma_tabs κ σ"
    and "n ∈ numeric_domain_class.gamma d" and "x ≠ ReturnVar ⟹ n ∈ ik_range (κ x)"
  shows "t(x := n) ∈ gamma_tabs κ (σ(x := d))"
  using gamma_tabsD[OF assms(1)] assms(2,3)
  by (intro gamma_tabsI in_kinds_upd) auto

lemma nonrel_assign_sound:
  assumes "t ∈ gamma_tabs κ σ" and ch: "eval_query.channel_holds ch t"
  shows "t(x := ik_norm (κ x) (eval e t)) ∈ gamma_tabs κ (nonrel_assign x e σ)"
proof -
  have "ik_norm (κ x) (eval e t) ∈ numeric_domain_class.gamma (cast_ik (sem_int_signed_overflow sem) (src_kind κ e) (κ x) (aeval_ask sem κ ch e σ))"
    by (rule cast_ik_src_sound[OF gamma_tabsD(1)[OF assms(1)] aeval_ask_sound[OF assms]])
  then show ?thesis unfolding nonrel_assign_def by (simp add: gamma_tabs_upd[OF assms(1)])
qed

lemma special_val_sound:
  assumes "t ∈ gamma_tabs κ σ" and ch: "eval_query.channel_holds ch t" and
    "special_result eval sc t v"
  shows "ik_norm (κ x) v ∈ numeric_domain_class.gamma (special_val sc x σ)"
proof (cases sc)
  case Nondet_Int then show ?thesis by (simp add: special_val_def top_ik_sound)
next
  case (Min a b)
  then have v:
    "v ∈ numeric_domain_class.gamma (min_ik (aeval_ask sem κ ch a σ) (aeval_ask sem κ ch b σ))"
    using assms(3) min_ik_sound[OF aeval_ask_sound[OF assms(1,2)] aeval_ask_sound[OF assms(1,2)]]
    by simp
  have rng: "v ∈ ik_range k" if "pair_kind κ a b = Some k" for k
    using that assms(3) Min src_kind_sound[OF gamma_tabsD(1)[OF assms(1)], of a k]
      src_kind_sound[OF gamma_tabsD(1)[OF assms(1)], of b k]
    by (auto simp: pair_kind_def min_def split: if_splits)
  from Min show ?thesis
    by (simp only: special_val_def gspecial_call.case) (rule cast_ik_sound[OF rng v])
next
  case (Max a b)
  then have v:
    "v ∈ numeric_domain_class.gamma (max_ik (aeval_ask sem κ ch a σ) (aeval_ask sem κ ch b σ))"
    using assms(3) max_ik_sound[OF aeval_ask_sound[OF assms(1,2)] aeval_ask_sound[OF assms(1,2)]]
    by simp
  have rng: "v ∈ ik_range k" if "pair_kind κ a b = Some k" for k
    using that assms(3) Max src_kind_sound[OF gamma_tabsD(1)[OF assms(1)], of a k]
      src_kind_sound[OF gamma_tabsD(1)[OF assms(1)], of b k]
    by (auto simp: pair_kind_def max_def split: if_splits)
  from Max show ?thesis
    by (simp only: special_val_def gspecial_call.case) (rule cast_ik_sound[OF rng v])
qed

lemma narrow_var_sound:
  assumes "t ∈ gamma_tabs κ σ" and "t v ∈ numeric_domain_class.gamma a"
  shows "t ∈ gamma_tabs κ (narrow_var v a σ)"
proof (cases "v = ReturnVar")
  case False
  have rng: "t v ∈ ik_range (κ v)"
    using gamma_tabsD(1)[OF assms(1)] False by (simp add: in_kinds_def)
  have "t v ∈ numeric_domain_class.gamma (meet_ik a (read_var κ σ v))"
    by (rule meet_ik_sound[OF assms(2) read_var_sound[OF assms(1)]])
  with rng have "t(v := t v) ∈ gamma_tabs κ (σ(v := KD (κ v) (meet_ik a (read_var κ σ v))))"
    by (intro gamma_tabs_upd[OF assms(1)]) simp_all
  with False show ?thesis by (simp add: narrow_var_def)
qed (use assms in ‹simp add: narrow_var_def›)

lemma refine_var_sound:
  assumes "t ∈ gamma_tabs κ σ" and ch: "eval_query.channel_holds ch t" and
    "eval e t ∈ numeric_domain_class.gamma a"
  shows "t ∈ gamma_tabs κ (refine_var e a σ)"
  using assms
proof (induction e arbitrary: a σ)
  case (TVar ik v) then show ?case by (simp add: narrow_var_sound)
next
  case (TCast ik e)
  show ?case
  proof (cases "fits_ik ik (aeval_ask sem κ ch e σ)")
    case True
    have "eval e t ∈ ik_range ik"
      by (rule fits_ik_sound[OF True aeval_ask_sound[OF TCast.prems(1,2)]])
    then have "eval e t ∈ numeric_domain_class.gamma a"
      using TCast.prems(3) by (simp del: ik_range_iff)
    with True TCast.IH[OF TCast.prems(1,2)] show ?thesis by simp
  qed (use TCast.prems in simp)
next
  case (TArith ik op e⇩1 e⇩2)
  define a⇩1 where "a⇩1 = aeval_ask sem κ ch e⇩1 σ"
  define a⇩2 where "a⇩2 = aeval_ask sem κ ch e⇩2 σ"
  have m: "eval e⇩1 t ∈ numeric_domain_class.gamma a⇩1"
    and n: "eval e⇩2 t ∈ numeric_domain_class.gamma a⇩2"
    using aeval_ask_sound[OF TArith.prems(1,2)] by (simp_all add: a⇩1_def a⇩2_def)
  show ?case
  proof (cases "fits_ik ik (raw_aop_ik op a⇩1 a⇩2)")
    case fits: True
    have "aop_eval op (eval e⇩1 t) (eval e⇩2 t) ∈ ik_range ik"
      by (rule fits_ik_sound[OF fits raw_aop_ik_sound[OF m n]])
    then have r: "aop_eval op (eval e⇩1 t) (eval e⇩2 t) ∈ numeric_domain_class.gamma a"
      using TArith.prems(3) by (simp del: ik_range_iff)
    show ?thesis
    proof (cases op)
      case AAdd
      have m': "eval e⇩1 t ∈ numeric_domain_class.gamma (raw_aop_ik ASub a a⇩2)"
        using raw_aop_ik_sound[OF r n, of ASub] AAdd by simp
      have n': "eval e⇩2 t ∈ numeric_domain_class.gamma (raw_aop_ik ASub a a⇩1)"
        using raw_aop_ik_sound[OF r m, of ASub] AAdd by simp
      have "refine_var (TArith ik op e⇩1 e⇩2) a σ
          = refine_var e⇩2 (raw_aop_ik ASub a a⇩1) (refine_var e⇩1 (raw_aop_ik ASub a a⇩2) σ)"
        using fits AAdd by (simp add: a⇩1_def a⇩2_def Let_def)
      with TArith.IH(2)[OF TArith.IH(1)[OF TArith.prems(1,2) m'] TArith.prems(2) n'] show ?thesis
        by simp
    next
      case ASub
      have m': "eval e⇩1 t ∈ numeric_domain_class.gamma (raw_aop_ik AAdd a a⇩2)"
        using raw_aop_ik_sound[OF r n, of AAdd] ASub by simp
      have n': "eval e⇩2 t ∈ numeric_domain_class.gamma (raw_aop_ik ASub a⇩1 a)"
        using raw_aop_ik_sound[OF m r, of ASub] ASub by simp
      have "refine_var (TArith ik op e⇩1 e⇩2) a σ
          = refine_var e⇩2 (raw_aop_ik ASub a⇩1 a) (refine_var e⇩1 (raw_aop_ik AAdd a a⇩2) σ)"
        using fits ASub by (simp add: a⇩1_def a⇩2_def Let_def)
      with TArith.IH(2)[OF TArith.IH(1)[OF TArith.prems(1,2) m'] TArith.prems(2) n'] show ?thesis
        by simp
    qed (use fits TArith.prems in ‹simp_all add: a⇩1_def a⇩2_def Let_def›)
  next
    case nofit: False
    show ?thesis
    proof (cases "op ∈ {AAdd, ASub}")
      case True
      have vimp: "in_kinds κ t" by (rule gamma_tabsD(1)[OF TArith.prems(1)])
      have r: "ik_norm ik (aop_eval op (eval e⇩1 t) (eval e⇩2 t)) ∈ numeric_domain_class.gamma a"
        using TArith.prems(3) by simp
      define σ' where "σ' = (if src_kind κ e⇩1 = Some ik
          then refine_var e⇩1 (arith_ik (sem_int_signed_overflow sem) ik (if op = AAdd then ASub else AAdd) False a a⇩2) σ else σ)"
      have s1: "t ∈ gamma_tabs κ σ'"
      proof (cases "src_kind κ e⇩1 = Some ik")
        case k: True
        have "eval e⇩1 t ∈ ik_range ik" by (rule src_kind_sound[OF vimp k])
        then have "eval e⇩1 t ∈ numeric_domain_class.gamma (arith_ik (sem_int_signed_overflow sem) ik (if op = AAdd then ASub else AAdd) False a a⇩2)"
          using arith_ik_sound[where k = ik and op = "if op = AAdd then ASub else AAdd", OF FalseE r
            n]
            True
          by (auto simp: ik_norm_norm_add ik_norm_norm_diff simp del: ik_range_iff)
        with k show ?thesis
          unfolding σ'_def using TArith.IH(1)[OF TArith.prems(1,2)] by simp
      qed (simp add: σ'_def TArith.prems(1))
      have eq: "refine_var (TArith ik op e⇩1 e⇩2) a σ = (if src_kind κ e⇩2 = Some ik
          then refine_var e⇩2 (if op = AAdd then arith_ik (sem_int_signed_overflow sem) ik ASub False a a⇩1 else arith_ik (sem_int_signed_overflow sem) ik ASub False a⇩1 a) σ' else σ')"
        using nofit True by (simp add: a⇩1_def a⇩2_def Let_def σ'_def)
      have "t ∈ gamma_tabs κ (if src_kind κ e⇩2 = Some ik
          then refine_var e⇩2 (if op = AAdd then arith_ik (sem_int_signed_overflow sem) ik ASub False a a⇩1 else arith_ik (sem_int_signed_overflow sem) ik ASub False a⇩1 a) σ' else σ')"
      proof (cases "src_kind κ e⇩2 = Some ik")
        case k: True
        have rng: "eval e⇩2 t ∈ ik_range ik" by (rule src_kind_sound[OF vimp k])
        have "eval e⇩2 t ∈ numeric_domain_class.gamma (if op = AAdd then arith_ik (sem_int_signed_overflow sem) ik ASub False a a⇩1 else arith_ik (sem_int_signed_overflow sem) ik ASub False a⇩1 a)"
        proof (cases "op = AAdd")
          case True
          then show ?thesis
            using arith_ik_sound[where k = ik and op = ASub, OF FalseE r m] rng
            by (simp add: ik_norm_norm_diff del: ik_range_iff)
        next
          case False
          with ‹op ∈ {AAdd, ASub}› have "op = ASub" by simp
          then show ?thesis
            using arith_ik_sound[where k = ik and op = ASub, OF FalseE m r] rng
            by (simp add: ik_norm_diff_norm del: ik_range_iff)
        qed
        with k show ?thesis using TArith.IH(2)[OF s1 TArith.prems(2)] by simp
      qed (simp add: s1)
      with eq show ?thesis by simp
    qed (use nofit TArith.prems in ‹auto simp: a⇩1_def a⇩2_def Let_def›)
  qed
qed simp_all

lemma nonrel_refine_sound:
  assumes "t ∈ gamma_tabs κ σ" and ch: "eval_query.channel_holds ch t" and
    "truthy (eval b t) = pol"
  shows "t ∈ gamma_tabs κ (nonrel_refine b pol σ)"
proof -
  let ?γ = "λσ. gamma_tabs κ σ ∩ Collect (eval_query.channel_holds ch)"
  have bare: "t ∈ ?γ (refine_var e (fst (refine_ik (if pol then CNe else CEq)
                 (aeval_ask sem κ ch e σ) (of_int_ik (exp_kind e) 0))) σ)"
    if "t ∈ ?γ σ" and "(eval e t ≠ 0) = pol" for e pol σ t
  proof -
    have g: "t ∈ gamma_tabs κ σ" and h: "eval_query.channel_holds ch t" using that(1) by auto
    have holds: "cop_eval (if pol then CNe else CEq) (eval e t) 0" using that(2) by auto
    have "eval e t ∈ numeric_domain_class.gamma
            (fst (refine_ik (if pol then CNe else CEq) (aeval_ask sem κ ch e σ) (of_int_ik (exp_kind e) 0)))"
      using refine_ik_sound[OF holds aeval_ask_sound[OF g h, of e]
              of_int_ik_sound[of "exp_kind e" 0, unfolded ik_norm_zero]] by simp
    then show ?thesis using h by (simp add: refine_var_sound[OF g h])
  qed
  have narrow: "t ∈ ?γ (nonrel_narrow e pol σ)"
    if "t ∈ ?γ σ" and "(eval e t ≠ 0) = pol" for e pol σ t
  proof (cases e)
    case (TCmp c l r)
    let ?c = "if pol then c else neg_cop c"
    have g: "t ∈ gamma_tabs κ σ" and h: "eval_query.channel_holds ch t" using that(1) by auto
    have holds: "cop_eval ?c (eval l t) (eval r t)"
      using that(2) TCmp by (auto split: if_splits)
    obtain al ar where ref:
      "refine_ik ?c (aeval_ask sem κ ch l σ) (aeval_ask sem κ ch r σ) = (al, ar)"
      by (cases "refine_ik ?c (aeval_ask sem κ ch l σ) (aeval_ask sem κ ch r σ)")
    have "eval l t ∈ numeric_domain_class.gamma al" "eval r t ∈ numeric_domain_class.gamma ar"
      using refine_ik_sound[OF holds aeval_ask_sound[OF g h, where sem = sem]
        aeval_ask_sound[OF g h, where sem = sem]]
        ref
      by simp_all
    then show ?thesis using ref g h TCmp by (simp add: nonrel_narrow_def refine_var_sound)
  qed (use bare[OF that] in ‹simp_all add: nonrel_narrow_def›)
  have atom: "t ∈ ?γ (nonrel_atom e pol σ)"
    if "t ∈ ?γ σ" and "(eval e t ≠ 0) = pol" for e pol σ t
  proof -
    have n: "t ∈ ?γ (nonrel_narrow e pol σ)" by (rule narrow[OF that])
    then have "¬ is_empty (nonrel_narrow e pol σ v)" for v
      using gamma_tabsD(2)[of t κ "nonrel_narrow e pol σ" v] is_empty_correct by blast
    with n show ?thesis by (simp add: nonrel_atom_def Let_def)
  qed
  have feas: "nonrel_feas e pol σ" if "t ∈ ?γ σ" and "(eval e t ≠ 0) = pol" for e pol σ t
  proof -
    have g: "t ∈ gamma_tabs κ σ" and h: "eval_query.channel_holds ch t" using that(1) by auto
    have "eval e t ∈ numeric_domain_class.gamma (aeval_ask sem κ ch e σ)"
      by (rule aeval_ask_sound[OF g h])
    then show ?thesis
      using that(2) truth_ik_sound[of "aeval_ask sem κ ch e σ" "¬ pol" "eval e t"]
      by (auto simp: nonrel_feas_def truthy_def)
  qed
  have join1: "t ∈ ?γ (x ⊔ y)" if "t ∈ ?γ x" for x y :: "'a tabs" and t
    using that gamma_tabs_mono[OF sup_ge1[of x y]] by auto
  have join2: "t ∈ ?γ (x ⊔ y)" if "t ∈ ?γ y" for x y :: "'a tabs" and t
    using that gamma_tabs_mono[OF sup_ge2[of y x]] by auto
  have "t ∈ ?γ (bool_refine.brefine nonrel_view nonrel_atom nonrel_feas sup (λ_. ⊥) b pol σ)"
    apply (rule bool_refine.brefine_sound[OF nonrel_bool_locale, where ev = eval and γ = ?γ])
    subgoal by (rule atom)
    subgoal for e a t by (cases e rule: nonrel_view.cases) simp_all
    subgoal for e a c t by (cases e rule: nonrel_view.cases) simp_all
    subgoal for e a c t by (cases e rule: nonrel_view.cases) simp_all
    subgoal by (rule feas)
    subgoal by (rule join1)
    subgoal by (rule join2)
    subgoal using assms(1,2) by simp
    subgoal using assms(3) by (simp add: truthy_def)
    done
  then show ?thesis by (simp add: nonrel_refine_def)
qed

subsection ‹The typed component›

text ‹The component answers ‹EvalInt› by evaluating without asking the expression
  itself back, as Goblint's ▩‹base› answers it with ▩‹eval_rv_no_ask_evalint›.

  alternative, the caller's state paired with the entered one.›

fun nonrel_query :: "sem_conf ⇒ kenv ⇒ channel ⇒ 'a::kind_int_domain tabs ⇒ channel" where
  "nonrel_query sem κ ch σ (EvalInt e) = answer_ik (aeval sem κ ch e σ)"

definition nonrel_spec ::
    "sem_conf ⇒ kenv ⇒ 'a::kind_int_domain tabs local_spec" where
  "nonrel_spec sem κ =
     ⦇ ls_query = nonrel_query sem κ,
       ls_skip = (λch d. d),
       ls_assign = (λch x e. nonrel_assign sem κ ch x e),
       ls_special = (λch sc x. nonrel_special sem κ ch sc x),
       ls_branch = (λch b pol. nonrel_branch sem κ ch b pol),
       ls_body = (λch p d. d),
       ls_return = (λch r p. nonrel_return sem κ ch r),
       ls_event = (λch evt d. d),
       ls_enter = (λch ci p.
         [(fst p, nonrel_enter_state sem κ ch (ci_formals ci) (ci_args ci) (fst p))]),
       ls_combine_env = (λch ch' ci. combine_env is_global),
       ls_combine_assign = (λch ci. nonrel_combine_assign sem κ (ci_dst ci)) ⦈"

theorem nonrel_spec_sound:
  "sound_local_spec κ (gamma_tabs κ)
     (nonrel_spec sem κ :: 'a::kind_int_domain tabs local_spec)"

  show ?thesis
  proof (rule sound_local_specI)
    show "⋀x y. x ≤ y ⟹ gamma_tabs κ x ⊆ gamma_tabs κ y"
      by (rule gamma_tabs_mono)
    show "sound_query (gamma_tabs κ) (ls_query (nonrel_spec sem κ))"
      unfolding sound_query_def
      apply (intro allI impI)
      subgoal for ch s x q
        by (cases q) (simp_all add: nonrel_spec_def answer_ik_sound aeval_sound)
      done
    show "sound_skip (gamma_tabs κ) (ls_skip (nonrel_spec sem κ))"
      by (simp add: nonrel_spec_def sound_skip_def)
    show "sound_assign κ (gamma_tabs κ) (ls_assign (nonrel_spec sem κ))"
      by (auto simp: nonrel_spec_def sound_assign_def intro!: nonrel_assign_sound)
    show "sound_special κ (gamma_tabs κ) (ls_special (nonrel_spec sem κ))"
      by (auto simp: nonrel_spec_def sound_special_def intro!: nonrel_special_sound)
    show "sound_branch (gamma_tabs κ) (ls_branch (nonrel_spec sem κ))"
      by (auto simp: nonrel_spec_def sound_branch_def intro!: nonrel_branch_sound)
