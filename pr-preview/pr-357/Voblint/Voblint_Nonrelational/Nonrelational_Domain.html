section ‹Comparisons from two queries›

text ‹A domain that decides ‹<› and ‹=› decides every comparison: the others swap or
  negate those two answers.›

fun cop_of :: "('a ⇒ 'a ⇒ bool option) ⇒ ('a ⇒ 'a ⇒ bool option) ⇒ cop ⇒ 'a ⇒ 'a ⇒ bool option"
where
  "cop_of lt eq CLt a b = lt a b"
| "cop_of lt eq CGt a b = lt b a"
| "cop_of lt eq CLe a b = map_option HOL.Not (lt b a)"
| "cop_of lt eq CGe a b = map_option HOL.Not (lt a b)"
| "cop_of lt eq CEq a b = eq a b"
| "cop_of lt eq CNe a b = map_option HOL.Not (eq a b)"

lemma cop_of_sound:
  fixes conc :: "'a ⇒ int set"
  assumes lt: "⋀a b r i j. lt a b = Some r ⟹ i ∈ conc a ⟹ j ∈ conc b ⟹ (i < j) = r"
    and eq: "⋀a b r i j. eq a b = Some r ⟹ i ∈ conc a ⟹ j ∈ conc b ⟹ (i = j) = r"
  shows "cop_of lt eq c a b = Some r ⟹ m ∈ conc a ⟹ n ∈ conc b ⟹ cop_eval c m n = r"
  by (cases c) (auto dest: lt eq)

section ‹Kind-aware integer domains›

text ‹
  The concrete semantics wraps every result, signed or not. How an abstract domain treats
  a possible signed overflow is a separate choice, Goblint's
  ▩‹sem.int.signed_overflow›: ‹Assume_Top›, its default, answers the kind's top,
  and ‹Assume_Wraparound› wraps as the concrete semantics does. Both are sound here. An
  unsigned kind always wraps (▩‹should_wrap› in ▩‹intDomain0.ml›).
›

datatype signed_overflow_policy = Assume_Top | Assume_Wraparound

definition should_wrap :: "signed_overflow_policy ⇒ ikind ⇒ bool" where
  "should_wrap ovp k ⟷ ¬ ik_signed k ∨ ovp = Assume_Wraparound"

text ‹
  The options of Goblint's ▩‹sem.int› group that the domains read. Goblint reads
  them from its global configuration at every use; here one value is a parameter of
  the evaluator and the transfers, and every soundness theorem holds for each value.
  A further option is a further field.
›

record sem_conf =
  sem_int_signed_overflow :: signed_overflow_policy

definition default_sem_conf :: sem_conf where
  "default_sem_conf = ⦇ sem_int_signed_overflow = Assume_Top ⦈"

text ‹A remainder of in-range operands, and a quotient unless the divisor is ‹-1›, stays
  in the kind's range: no policy applies to them.›

lemma c_mod_between: "min 0 a ≤ c_mod a b ∧ c_mod a b ≤ max 0 a"
proof (cases "b = 0")
  case False
  have "c_mod a b = sgn a * (¦a¦ mod ¦b¦)"
    using False by (simp add: c_mod_def c_div_def algebra_simps abs_mult_sgn
        minus_div_mult_eq_mod[symmetric] sgn_mult_abs; metis abs_sgn)
  moreover have "0 ≤ ¦a¦ mod ¦b¦" "¦a¦ mod ¦b¦ ≤ ¦a¦"
    using False by (simp_all add: zmod_le_nonneg_dividend)
  ultimately show ?thesis by (cases "0 ≤ a") (auto simp: sgn_if)
qed (simp add: c_mod_def)

lemma c_mod_in_range: "m ∈ ik_range k ⟹ c_mod m n ∈ ik_range k"
  using c_mod_between[of m n] ik_min_le_zero[of k] zero_le_ik_max[of k]
  by (cases "0 ≤ m") (auto simp: min_def max_def simp del: ik_min_le_zero zero_le_ik_max)

lemma c_div_in_range:
  assumes m: "m ∈ ik_range k" and n: "n ∈ ik_range k" and n1: "n ≠ -1"
  shows "c_div m n ∈ ik_range k"
proof (cases "n = 0")
  case False
  define q where "q = ¦m¦ div ¦n¦"
  have q0: "0 ≤ q" using False by (simp add: q_def pos_imp_zdiv_nonneg_iff)
  have "q * ¦n¦ = ¦m¦ - ¦m¦ mod ¦n¦" by (simp add: q_def minus_mod_eq_div_mult)
  moreover have "0 ≤ ¦m¦ mod ¦n¦" using False by simp
  ultimately have qn: "q * ¦n¦ ≤ ¦m¦" by linarith
  have cd: "c_div m n = sgn m * sgn n * q" using False by (simp add: c_div_def q_def)
  have lo: "ik_min k = (if ik_signed k then - ik_max k - 1 else 0)"
    by (simp add: ik_min_def ik_max_def)
  have one: "1 ≤ ik_max k" by (cases k) (simp_all add: ik_max_def ik_half_def ik_mod_def)
  have mk: "ik_min k ≤ m" "m ≤ ik_max k" "ik_min k ≤ n" "n ≤ ik_max k" using m n by simp_all
  show ?thesis
  proof (cases "n = 1")
    case True
    then show ?thesis using m by simp
  next
    case n2: False
    with False n1 have "2 ≤ ¦n¦" by linarith
    then have "2 * q ≤ ¦m¦" using qn mult_left_mono[of 2 "¦n¦" q] q0 by (simp add: mult.commute)
    then show ?thesis using cd lo one mk q0 by (auto simp: sgn_if split: if_splits)
  qed
qed simp

text ‹
  The operations a typed transfer needs from an integer domain, after Goblint's
  ▩‹IntDomain.S›: every operation that produces a value takes the kind it computes
  at. ‹top_ik k› is Goblint's ‹top_of ik›, ‹of_int_ik› its ‹of_int ik›, ‹cast_ik ovp sk k› its
  ‹cast_to ?from_ik:sk k›, where ‹sk› is the operand's kind when it is known, and
  ‹arith_ik ovp k op operands_fit› the binary operation at kind ‹k›, whose concrete result
  is normalized to ‹k›; ‹operands_fit› says that both operands lie in ‹k›'s range, as
  they do when each operand has kind ‹k›. It does not say that the result does. A
  comparison answers ‹Some b› only when every pair of
  operands agrees on ‹b›. ‹refine_ik c a b› narrows both operands of a comparison
  known to hold, as ▩‹inv_bin_int› in Goblint's ▩‹baseInvariant.ml› does for a guard. ‹raw_aop_ik› is
  the operation on mathematical integers, which a backward step inverts, and
  ‹fits_ik k a› says that every value of ‹a› lies in ‹k›'s range, so that
  normalizing to ‹k› changes none of them.

  ‹answer_ik› and ‹of_answer_ik› translate between a value and an answer to Goblint's
  ‹EvalInt› query, as Goblint's ▩‹IntDomTuple› is both the base value and the
  query result. ‹of_answer_ik sk› may use the kind ‹sk› of the queried expression,
  when it is known, to bound an answer that claims nothing.
›

class kind_int_domain = ik_domain +
  fixes top_ik :: "ikind ⇒ 'a::{bounded_semilattice_sup_bot, order_top}"
    and of_int_ik :: "ikind ⇒ int ⇒ 'a"
    and cast_ik :: "signed_overflow_policy ⇒ ikind option ⇒ ikind ⇒ 'a ⇒ 'a"
    and arith_ik :: "signed_overflow_policy ⇒ ikind ⇒ aop ⇒ bool ⇒ 'a ⇒ 'a ⇒ 'a"
    and cmp_ik :: "cop ⇒ 'a ⇒ 'a ⇒ bool option"
    and meet_ik :: "'a ⇒ 'a ⇒ 'a"
    and refine_ik :: "cop ⇒ 'a ⇒ 'a ⇒ 'a × 'a"
    and raw_aop_ik :: "aop ⇒ 'a ⇒ 'a ⇒ 'a"
    and fits_ik :: "ikind ⇒ 'a ⇒ bool"
    and answer_ik :: "'a ⇒ answer"
    and of_answer_ik :: "ikind option ⇒ answer ⇒ 'a"
  assumes top_ik_sound: "n ∈ ik_range k ⟹ n ∈ γ (top_ik k)"
    and of_int_ik_sound: "ik_norm k n ∈ γ (of_int_ik k n)"
    and cast_ik_sound:
      "(⋀k'. sk = Some k' ⟹ n ∈ ik_range k') ⟹ n ∈ γ a
         ⟹ ik_norm k n ∈ γ (cast_ik ovp sk k a)"
    and arith_ik_sound:
      "(operands_fit ⟹ m ∈ ik_range k ∧ n ∈ ik_range k) ⟹ m ∈ γ a ⟹ n ∈ γ b
         ⟹ ik_norm k (aop_eval op m n) ∈ γ (arith_ik ovp k op operands_fit a b)"
    and cmp_ik_sound:
      "cmp_ik c a b = Some r ⟹ m ∈ γ a ⟹ n ∈ γ b ⟹ cop_eval c m n = r"
    and meet_ik_sound: "m ∈ γ a ⟹ m ∈ γ b ⟹ m ∈ γ (meet_ik a b)"
    and refine_ik_sound:
      "cop_eval c m n ⟹ m ∈ γ a ⟹ n ∈ γ b
         ⟹ m ∈ γ (fst (refine_ik c a b)) ∧ n ∈ γ (snd (refine_ik c a b))"
    and raw_aop_ik_sound: "m ∈ γ a ⟹ n ∈ γ b ⟹ aop_eval op m n ∈ γ (raw_aop_ik op a b)"
    and fits_ik_sound: "fits_ik k a ⟹ n ∈ γ a ⟹ n ∈ ik_range k"
    and answer_ik_sound: "n ∈ γ a ⟹ n ∈ gamma_answer (answer_ik a)"
    and of_answer_ik_sound:
      "(⋀k. sk = Some k ⟹ n ∈ ik_range k) ⟹ n ∈ gamma_answer q
         ⟹ n ∈ γ (of_answer_ik sk q)"

text ‹A single-component domain answers with its component of \<^typ>‹int_dom› and
  leaves the others at top, so its answer claims exactly what its value does.›

abbreviation answer_with :: "(int_dom ⇒ int_dom) ⇒ answer" where
  "answer_with f ≡ QLifted (f ⊤)"

section ‹Typed abstract states›

text ‹
  A typed abstract state maps every variable identity to a kind-tagged value. It
  describes the typed stores whose values lie in the state's values: \<^const>‹in_kinds›
  is part of the description, so a value the state does not constrain still stays in
  its variable's range.
›

type_synonym 'a tabs = "var_id ⇒ 'a kd"

definition gamma_tabs :: "kenv ⇒ 'a::ik_domain tabs ⇒ store set" where
  "gamma_tabs κ σ = {t. in_kinds κ t ∧ (∀v. t v ∈ γ (σ v))}"

lemma gamma_tabsD:
  assumes "t ∈ gamma_tabs κ σ"
  shows "in_kinds κ t" and "t v ∈ γ (σ v)"
  using assms by (simp_all add: gamma_tabs_def)

lemma gamma_tabsI:
  "in_kinds κ t ⟹ (⋀v. t v ∈ γ (σ v)) ⟹ t ∈ gamma_tabs κ σ"
  by (simp add: gamma_tabs_def)

lemma gamma_tabs_mono:
  assumes "σ⇩1 ≤ σ⇩2"
  shows "gamma_tabs κ σ⇩1 ⊆ gamma_tabs κ σ⇩2"
proof -
  have "x v ∈ γ (σ⇩2 v)" if "x v ∈ γ (σ⇩1 v)" for x v
    using that gamma_mono[OF le_funD[OF assms, of v]] by blast
  then show ?thesis by (auto simp: gamma_tabs_def simp del: gamma_kd.simps)
qed

lemma gamma_tabs_bot [simp]: "gamma_tabs κ ⊥ = {}"
  by (simp add: gamma_tabs_def bot_fun_def bot_kd_def)

section ‹Reading and evaluating›

text ‹
  Reading a variable gives its value without the tag. An untagged top is the
  variable's whole range, as Goblint reads an unconstrained local at ‹top_of ik›; the
  return slot has no declared kind and reads as top.
›

fun kd_val :: "'a::kind_int_domain kd ⇒ 'a" where
  "kd_val KBot = ⊥"
| "kd_val (KD k a) = a"
| "kd_val KTop = ⊤"

definition read_var :: "kenv ⇒ 'a::kind_int_domain tabs ⇒ var_id ⇒ 'a" where
  "read_var κ σ v =
     (case σ v of KTop ⇒ if v = ReturnVar then ⊤ else top_ik (κ v) | d ⇒ kd_val d)"

lemma kd_val_sound: "n ∈ γ d ⟹ n ∈ γ (kd_val d)"
  by (cases d) simp_all

text ‹A tagged value lies in its tag's range, so a conversion from it knows its source.›

definition kd_kind :: "'a kd ⇒ ikind option" where
  "kd_kind d = (case d of KD k a ⇒ Some k | _ ⇒ None)"

lemma kd_kind_range: "n ∈ γ d ⟹ kd_kind d = Some k ⟹ n ∈ ik_range k"
  by (cases d) (auto simp: kd_kind_def)

lemma read_var_sound:
  assumes "t ∈ gamma_tabs κ σ"
  shows "t v ∈ γ (read_var κ σ v)"
proof -
  have ty: "in_kinds κ t" and tv: "t v ∈ γ (σ v)"
    using gamma_tabsD[OF assms] by blast+
  show ?thesis
  proof (cases "σ v")
    case KTop
    show ?thesis
    proof (cases "v = ReturnVar")
      case False
      then have "t v ∈ ik_range (κ v)" using ty by (simp add: in_kinds_def)
      with KTop False show ?thesis by (simp add: read_var_def top_ik_sound)
    qed (use KTop in ‹simp add: read_var_def›)
  qed (use tv in ‹simp_all add: read_var_def›)
qed

text ‹A truth value is ‹0› or ‹1›; an undecided one is both.›

definition bool_val :: "bool option ⇒ 'a::kind_int_domain" where
  "bool_val r =
     (case r of Some True ⇒ of_int_ik I32 1 | Some False ⇒ of_int_ik I32 0
      | None ⇒ of_int_ik I32 0 ⊔ of_int_ik I32 1)"

definition truth_ik :: "'a::kind_int_domain ⇒ bool option" where
  "truth_ik a = cmp_ik CNe a (of_int_ik I32 0)"

lemma of_int_ik_I32_01:
  "0 ∈ γ (of_int_ik I32 0 :: 'a::kind_int_domain)"
  "1 ∈ γ (of_int_ik I32 1 :: 'a::kind_int_domain)"
  using of_int_ik_sound[of I32 0] of_int_ik_sound[of I32 1]
  by (simp_all add: ik_norm_def ik_min_def ik_max_def)

lemma bool_val_sound:
  assumes "⋀b. r = Some b ⟹ P = b"
  shows "(if P then 1 else 0) ∈ γ (bool_val r :: 'a::kind_int_domain)"
proof (cases r)
  case None
  have "0 ∈ γ (of_int_ik I32 0 ⊔ of_int_ik I32 1 :: 'a)"
    "1 ∈ γ (of_int_ik I32 0 ⊔ of_int_ik I32 1 :: 'a)"
    using of_int_ik_I32_01 gamma_mono[OF sup.cobounded1] gamma_mono[OF sup.cobounded2]
    by blast+
  with None show ?thesis by (simp add: bool_val_def)
next
  case (Some b)
  with assms show ?thesis by (cases b) (simp_all add: bool_val_def of_int_ik_I32_01)
qed

lemma truth_ik_sound:
  "truth_ik a = Some b ⟹ n ∈ γ a ⟹ truthy n = b"
  unfolding truth_ik_def
  using cmp_ik_sound[of CNe a "of_int_ik I32 0" b n 0] of_int_ik_I32_01(1)
  by (simp add: truthy_def)

text ‹A conversion knows its operand's kind when the operand produces values of that
  kind: an operation always does, and a variable does when it is read at its declared
  kind.›

definition src_kind :: "kenv ⇒ exp ⇒ ikind option" where
  "src_kind κ e =
     (case e of TVar ik v ⇒ if v ≠ ReturnVar ∧ ik = κ v then Some ik else None
      | _ ⇒ Some (exp_kind e))"

lemma src_kind_sound:
  assumes "in_kinds κ t" and "src_kind κ e = Some k"
  shows "eval e t ∈ ik_range k"
  using assms by (cases e) (auto simp: src_kind_def in_kinds_def split: if_splits)

lemma cast_ik_None_sound: "n ∈ γ a ⟹ ik_norm k n ∈ γ (cast_ik ovp None k a)"
  by (rule cast_ik_sound) simp_all

text ‹A conversion may name the kind of the expression it converts, as Goblint's
  ▩‹cast_to ?from_ik› does: a typed store gives that expression a value in its kind's range.›

lemma cast_ik_src_sound:
  assumes "in_kinds κ t" and "eval e t ∈ γ a"
  shows "ik_norm k (eval e t) ∈ γ (cast_ik ovp (src_kind κ e) k a)"
  using assms(2) src_kind_sound[OF assms(1)] by (intro cast_ik_sound) auto

text ‹
  Evaluation consults the other analyses, as Goblint's ▩‹base› does
  (▩‹eval_rv_ask_evalint› in ▩‹src/analyses/base.ml› at ▩‹0dc12d355›): every
  operand that is not a constant is asked about through the channel ‹ch›, and its
  answer is met with the analysis's own evaluation. Goblint takes the answer alone:
  there the evaluated state is the one the channel answers about, since CIL splits a
  conjunction into one guard per conjunct. A VIMP guard refines a whole conjunction in
  one state, so the second conjunct is evaluated in a state the first has already
  narrowed, finer than the channel's. ‹aeval› asks
  about the operands but not about the expression itself, which is how the analysis
  answers a query without asking it back; ‹aeval_ask› also asks about the
  expression. An answer that claims nothing, such as the one a query cycle gets,
  falls back to the analysis's own evaluation, where Goblint takes the kind's top.
  Under the channel that answers every query with ‹⊤›, both are the plain
  evaluation.
›

context
  fixes sem :: sem_conf
begin

fun aeval :: "kenv ⇒ (query ⇒ answer) ⇒ exp ⇒ 'a::kind_int_domain tabs ⇒ 'a"
  and aeval_ask :: "kenv ⇒ (query ⇒ answer) ⇒ exp ⇒ 'a::kind_int_domain tabs ⇒ 'a" where
  "aeval κ ch (TN ik n) σ = of_int_ik ik n"
| "aeval κ ch (TVar ik v) σ = read_var κ σ v"
| "aeval κ ch (TArith ik op a b) σ =
     arith_ik (sem_int_signed_overflow sem) ik op (src_kind κ a = Some ik ∧ src_kind κ b = Some ik)
       (aeval_ask κ ch a σ) (aeval_ask κ ch b σ)"
| "aeval κ ch (TCast ik a) σ =
     cast_ik (sem_int_signed_overflow sem) (src_kind κ a) ik (aeval_ask κ ch a σ)"
| "aeval κ ch (TCmp op a b) σ =
     bool_val (cmp_ik op (aeval_ask κ ch a σ) (aeval_ask κ ch b σ))"
| "aeval κ ch (TNot a) σ = bool_val (map_option HOL.Not (truth_ik (aeval_ask κ ch a σ)))"
| "aeval κ ch (TAnd a b) σ =
     bool_val (case (truth_ik (aeval_ask κ ch a σ), truth_ik (aeval_ask κ ch b σ)) of
                 (Some False, _) ⇒ Some False | (_, Some False) ⇒ Some False
               | (Some True, Some True) ⇒ Some True | _ ⇒ None)"
| "aeval κ ch (TOr a b) σ =
     bool_val (case (truth_ik (aeval_ask κ ch a σ), truth_ik (aeval_ask κ ch b σ)) of
                 (Some True, _) ⇒ Some True | (_, Some True) ⇒ Some True
               | (Some False, Some False) ⇒ Some False | _ ⇒ None)"
| "aeval_ask κ ch e σ =
     (case e of TN ik n ⇒ of_int_ik ik n
      | _ ⇒ (case ch (EvalInt e) of QTop ⇒ aeval κ ch e σ
              | q ⇒ meet_ik (aeval κ ch e σ) (of_answer_ik (src_kind κ e) q)))"

lemma aeval_ask_of_sound:
  assumes "t ∈ gamma_tabs κ σ" and "eval_query.channel_holds ch t"
    and "eval e t ∈ γ (aeval κ ch e σ)"
  shows "eval e t ∈ γ (aeval_ask κ ch e σ)"
proof -
  have ans: "eval e t ∈ gamma_answer (ch (EvalInt e))"
    using eval_query.channel_holdsD[OF assms(2), of "EvalInt e"] by simp
  have rng: "⋀k. src_kind κ e = Some k ⟹ eval e t ∈ ik_range k"
    using src_kind_sound[OF gamma_tabsD(1)[OF assms(1)]] .
  show ?thesis
    using assms(3) of_answer_ik_sound[OF rng ans] meet_ik_sound
    by (cases e; cases "ch (EvalInt e)") auto
qed

theorem aeval_sound:
  assumes "t ∈ gamma_tabs κ σ" and "eval_query.channel_holds ch t"
  shows "eval e t ∈ γ (aeval κ ch e σ)"
proof (induction e)
  case (TN ik n) show ?case by (simp add: of_int_ik_sound)
next
  case (TVar ik v) show ?case by (simp add: read_var_sound[OF assms(1)])
next
  case (TArith ik op a b)
  have "src_kind κ e = Some ik ⟹ eval e t ∈ ik_range ik" for e
    using src_kind_sound[OF gamma_tabsD(1)[OF assms(1)]] .
  with TArith show ?case
    using aeval_ask_of_sound[OF assms] by (simp add: arith_ik_sound)
next
  case (TCast ik a)
  have "eval a t ∈ ik_range k" if "src_kind κ a = Some k" for k
    using src_kind_sound[OF gamma_tabsD(1)[OF assms(1)] that] .
  with TCast aeval_ask_of_sound[OF assms] show ?case by (simp add: cast_ik_sound)
next
  case (TCmp op a b) then show ?case
    using aeval_ask_of_sound[OF assms] by (simp add: bool_val_sound cmp_ik_sound)
next
  case (TNot a)
  have a: "eval a t ∈ γ (aeval_ask κ ch a σ)"
    by (rule aeval_ask_of_sound[OF assms TNot])
  have "(if ¬ truthy (eval a t) then 1 else 0)
      ∈ γ (bool_val (map_option HOL.Not (truth_ik (aeval_ask κ ch a σ))) :: 'a)"
    by (rule bool_val_sound) (use a truth_ik_sound in auto)
  moreover have "eval (TNot a) t = (if ¬ truthy (eval a t) then 1 else 0)"
    by simp
  ultimately show ?case by (simp only: aeval.simps)
next
  case (TAnd a b)
  have a: "eval a t ∈ γ (aeval_ask κ ch a σ)" and b: "eval b t ∈ γ (aeval_ask κ ch b σ)"
    using aeval_ask_of_sound[OF assms] TAnd by blast+
  show ?case
    unfolding eval.simps aeval.simps
    by (rule bool_val_sound)
       (use a b truth_ik_sound in ‹auto split: option.splits bool.splits›)
next
  case (TOr a b)
  have a: "eval a t ∈ γ (aeval_ask κ ch a σ)" and b: "eval b t ∈ γ (aeval_ask κ ch b σ)"
    using aeval_ask_of_sound[OF assms] TOr by blast+
  show ?case
    unfolding eval.simps aeval.simps
    by (rule bool_val_sound)
       (use a b truth_ik_sound in ‹auto split: option.splits bool.splits›)
qed

corollary aeval_ask_sound:
  assumes "t ∈ gamma_tabs κ σ" and "eval_query.channel_holds ch t"
  shows "eval e t ∈ γ (aeval_ask κ ch e σ)"
  by (rule aeval_ask_of_sound[OF assms aeval_sound[OF assms]])

end
