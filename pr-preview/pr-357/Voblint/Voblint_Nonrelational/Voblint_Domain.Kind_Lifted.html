Theory Voblint_Domain.Kind_Lifted

theory Kind_Lifted
  imports Abstract_Domain "Voblint_VIMP.Machine_Kind"
begin

section ‹Kind-tagged values›

text ‹
  Goblint's ▩‹IntDomLifter› stores an integer domain element together with its
  ▩‹ikind›, so that operations without a kind argument, the lattice join and the
  solver's widening among them, can still act at the right kind. The value domain
  around it adds a bottom and a top, and turns a join of elements of different kinds
  into top. ‹'a kd› is that construction: ‹KD k a› is the element ‹a› at kind
  ‹k›, and it denotes only integers in the range of ‹k›.

  The component domain supplies the kind-indexed operations of Goblint's
  ▩‹IntDomain.S› that the lifter needs: widening and narrowing at a kind, and an
  emptiness test against a kind's range.
›

class ik_domain = numeric_domain +
  fixes widen_ik :: "ikind ⇒ 'a::{bounded_semilattice_sup_bot, order_top} ⇒ 'a ⇒ 'a"
    and narrow_ik :: "ikind ⇒ 'a ⇒ 'a ⇒ 'a"
    and empty_ik :: "ikind ⇒ 'a ⇒ bool"
  assumes widen_ik_ge1: "a ≤ widen_ik k a b"
    and widen_ik_ge2: "b ≤ widen_ik k a b"
    and narrow_ik_ge: "b ≤ a ⟹ b ≤ narrow_ik k a b"
    and narrow_ik_le: "b ≤ a ⟹ narrow_ik k a b ≤ a"
    and empty_ik_correct: "empty_ik k a ⟷ γ a ∩ ik_range k = {}"

datatype 'a kd = KBot | KD ikind 'a | KTop

subsection ‹Order and join›

text ‹Elements are comparable only at the same kind. A join across kinds is top, as
  in Goblint's value domain, which catches the lifter's ▩‹IncompatibleIKinds›.›

instantiation kd :: (ik_domain) order_bot
begin

fun less_eq_kd :: "'a kd ⇒ 'a kd ⇒ bool" where
  "less_eq_kd KBot _ ⟷ True"
| "less_eq_kd _ KTop ⟷ True"
| "less_eq_kd (KD k a) (KD l b) ⟷ k = l ∧ a ≤ b"
| "less_eq_kd _ _ ⟷ False"

definition less_kd :: "'a kd ⇒ 'a kd ⇒ bool" where
  "less_kd a b ⟷ a ≤ b ∧ ¬ b ≤ a"

definition bot_kd :: "'a kd" where "bot_kd = KBot"

instance proof
  fix x y z :: "'a kd"
  show "x < y ⟷ x ≤ y ∧ ¬ y ≤ x" by (simp add: less_kd_def)
  show "x ≤ x" by (cases x) simp_all
  show "x ≤ y ⟹ y ≤ z ⟹ x ≤ z"
    by (cases x; cases y; cases z) (auto intro: order_trans)
  show "x ≤ y ⟹ y ≤ x ⟹ x = y"
    by (cases x; cases y) (auto intro: antisym)
  show "bot ≤ x" by (simp add: bot_kd_def)
qed

end

instantiation kd :: (ik_domain) order_top
begin

definition top_kd :: "'a kd" where "top_kd = KTop"

instance proof
  fix x :: "'a kd"
  show "x ≤ top" by (cases x) (simp_all add: top_kd_def)
qed

end

instantiation kd :: (ik_domain) semilattice_sup
begin

fun sup_kd :: "'a kd ⇒ 'a kd ⇒ 'a kd" where
  "sup_kd KBot b = b"
| "sup_kd a KBot = a"
| "sup_kd (KD k a) (KD l b) = (if k = l then KD k (a ⊔ b) else KTop)"
| "sup_kd _ _ = KTop"

instance proof
  fix x y z :: "'a kd"
  show "x ≤ x ⊔ y" by (cases x; cases y) simp_all
  show "y ≤ x ⊔ y" by (cases x; cases y) simp_all
  show "y ≤ x ⟹ z ≤ x ⟹ y ⊔ z ≤ x"
    by (cases x; cases y; cases z) auto
qed

end

instance kd :: (ik_domain) bounded_semilattice_sup_bot ..

subsection ‹Widening and narrowing at the operands' kind›

text ‹Two operands of one kind are widened or narrowed by the component domain at that
  kind. Widening operands of different kinds gives \<^const>‹KTop›.›

instantiation kd :: (ik_domain) warrowing
begin

fun widen_kd :: "'a kd ⇒ 'a kd ⇒ 'a kd" where
  "widen_kd KBot b = b"
| "widen_kd a KBot = a"
| "widen_kd (KD k a) (KD l b) = (if k = l then KD k (widen_ik k a b) else KTop)"
| "widen_kd _ _ = KTop"

fun narrow_kd :: "'a kd ⇒ 'a kd ⇒ 'a kd" where
  "narrow_kd KTop b = b"
| "narrow_kd (KD k a) (KD l b) = (if k = l then KD k (narrow_ik k a b) else KD k a)"
| "narrow_kd a _ = KBot"

instance proof
  fix a b :: "'a kd"
  show "a ≤ a ∇ b" by (cases a; cases b) (simp_all add: widen_ik_ge1)
  show "b ≤ a ∇ b" by (cases a; cases b) (simp_all add: widen_ik_ge2)
  show "b ≤ a ⟹ b ≤ a Δ b"
    by (cases a; cases b) (auto simp: narrow_ik_ge)
  show "b ≤ a ⟹ a Δ b ≤ a"
    by (cases a; cases b) (auto simp: narrow_ik_le)
qed

end

subsection ‹Concretization›

text ‹A kind-tagged element denotes the integers of its component that its kind can
  represent.›

fun gamma_kd :: "'a::ik_domain kd ⇒ int set" where
  "gamma_kd KBot = {}"
| "gamma_kd (KD k a) = γ a ∩ ik_range k"
| "gamma_kd KTop = UNIV"

instantiation kd :: (ik_domain) numeric_domain
begin

fun is_empty_kd :: "'a kd ⇒ bool" where
  "is_empty_kd KBot ⟷ True"
| "is_empty_kd (KD k a) ⟷ empty_ik k a"
| "is_empty_kd KTop ⟷ False"

fun to_string_kd :: "'a kd ⇒ String.literal" where
  "to_string_kd KBot = to_string (bot :: 'a)"
| "to_string_kd (KD k a) = to_string a"
| "to_string_kd KTop = sym_top"

definition gamma_kd_def [simp]: "numeric_domain_class.gamma (a :: 'a kd) = gamma_kd a"

instance proof
  fix a b :: "'a kd"
  show "γ (bot :: 'a kd) = {}" by (simp add: bot_kd_def)
  show "γ (top :: 'a kd) = UNIV" by (simp add: top_kd_def)
  show "a ≤ b ⟹ γ a ⊆ γ b"
    by (cases a; cases b) (auto dest: gamma_mono)
  show "is_empty a ⟷ γ a = {}"
    by (cases a) (simp_all add: empty_ik_correct)
qed

end

end
