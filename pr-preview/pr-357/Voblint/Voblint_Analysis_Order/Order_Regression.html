Theory Order_Regression

theory Order_Regression
  imports Order_Analysis "Voblint_VIMP.VIMP_Notation"
begin

section ‹Regression: the typed order analysis, executed›

text ‹
  Each case runs the certified solver on a typed program inside Isabelle and pins the
  order facts published at the end of ‹f›. Termination is decided by evaluation.
›

abbreviation f_var :: "vname ⇒ var_id" where
  "f_var x ≡ ScopedId (STR ''f'') x"

abbreviation ord_result :: "kind_decls ⇒ imp_prog ⇒ var_id order_rel" where
  "ord_result ds p ≡ order_analysis.state_at ds Globals_Join is_global p () (FunctionResult (STR ''f''))"

text ‹The loop exits when ‹x < y› fails, so ‹y ≤ x› holds at the exit. The body
  assigns ‹x›, which forgets every fact about it, so no fact survives into the loop.›

definition loop_prog :: imp_prog where
  "loop_prog = program {
     fun f(x, y) { while (x < y) { x = x + 1; } }
     fun main() { f(0, 5); } }"

lemma loop_terminates_c:
  "TD_side_rule_Interp_solve_c Globals_Join (order_analysis.equations [] is_global loop_prog)
     (cfg_exit (elab_cfg loop_prog []), ()) ≠ None"
  by eval

lemma loop_terminates: "order_analysis.terminates [] Globals_Join is_global loop_prog"
  unfolding order_analysis.terminates_code
  using TD_side_rule_Interp.solve_dom_of_solve_c[OF loop_terminates_c] by simp

lemma loop_facts: "ord_result [] loop_prog = OFacts True {(f_var (STR ''y''), f_var (STR ''x''))}"
  by eval

text ‹The facts are about stored values, which already lie in their kinds: a \<^const>‹U8›
  ‹x› and an \<^const>‹I32› ‹y› compared through a conversion yield no fact, since
  the comparison does not see the variables' values unconverted.›

definition mixed_prog :: imp_prog where
  "mixed_prog = program {
     fun f(x, y) { while (x < y) { y = y - 1; } }
     fun main() { f(0, 5); } }"

lemma mixed_facts:
  "ord_result [(f_var (STR ''x''), U8)] mixed_prog = OFacts True {}"
  by eval

subsection ‹Rendering›

text ‹‹to_string› renders bottom and top by their symbols and a set of facts as a sorted
  conjunction of ‹≤› facts, whatever the flag.›

lemma to_string_order_rel_regression:
  "to_string (OBot :: vname order_rel) = sym_bottom"
  "to_string (OFacts True {} :: vname order_rel) = sym_top"
  "to_string (OFacts False {(STR ''y'', STR ''z''), (STR ''x'', STR ''y'')}) =
     STR ''{x<le>y <and> y<le>z}''"
  by eval+

end
