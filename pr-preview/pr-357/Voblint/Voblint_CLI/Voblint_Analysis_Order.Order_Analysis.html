section ‹The order domain at typed variables›

text ‹
  A relational carrier over any variable type: a finite set of pairs ‹(x, y)›, each
  meaning that ‹x›'s value is at most ‹y›'s. Stored values are already normalized to
  their kinds, so the facts compare the values a run holds; a guard adds a fact only
  when it compares two variable reads, whose values the comparison sees unconverted.
  The flag records that every variable other than the return slot holds a value of its
  kind (@{const in_kinds}): a copy between two variables of one kind then passes the
  conversion unchanged.
›

datatype 'v order_rel = OBot | OFacts bool "('v × 'v) set"

instantiation order_rel :: (type) "{bounded_semilattice_sup_bot, order_top}"
begin

fun less_eq_order_rel :: "'v order_rel ⇒ 'v order_rel ⇒ bool" where
  "less_eq_order_rel OBot _ = True"
| "less_eq_order_rel (OFacts _ _) OBot = False"
| "less_eq_order_rel (OFacts t a) (OFacts u b) = (b ⊆ a ∧ (u ⟶ t))"

definition less_order_rel :: "'v order_rel ⇒ 'v order_rel ⇒ bool" where
  "less_order_rel a b ⟷ a ≤ b ∧ ¬ b ≤ a"

fun sup_order_rel :: "'v order_rel ⇒ 'v order_rel ⇒ 'v order_rel" where
  "sup_order_rel OBot b = b"
| "sup_order_rel a OBot = a"
| "sup_order_rel (OFacts t a) (OFacts u b) = OFacts (t ∧ u) (a ∩ b)"

definition bot_order_rel :: "'v order_rel" where "bot_order_rel = OBot"

definition top_order_rel :: "'v order_rel" where "top_order_rel = OFacts False {}"

instance proof
  fix x y z :: "'v order_rel"
  show "x < y ⟷ x ≤ y ∧ ¬ y ≤ x" by (simp add: less_order_rel_def)
  show "x ≤ x" by (cases x) simp_all
  show "x ≤ y ⟹ y ≤ z ⟹ x ≤ z" by (cases x; cases y; cases z) auto
  show "x ≤ y ⟹ y ≤ x ⟹ x = y" by (cases x; cases y) auto
  show "x ≤ x ⊔ y" "y ≤ x ⊔ y" by (cases x; cases y; auto)+
  show "y ≤ x ⟹ z ≤ x ⟹ y ⊔ z ≤ x" by (cases x; cases y; cases z) auto
  show "bot ≤ x" by (simp add: bot_order_rel_def)
  show "x ≤ top" by (cases x) (simp_all add: top_order_rel_def)
qed

end

instantiation order_rel :: (type) warrowing
begin

definition widen_order_rel :: "'v order_rel ⇒ 'v order_rel ⇒ 'v order_rel" where
  "widen_order_rel a b = a ⊔ b"

definition narrow_order_rel :: "'v order_rel ⇒ 'v order_rel ⇒ 'v order_rel" where
  "narrow_order_rel a b = b"

instance proof
  fix x y :: "'v order_rel"
  show "x ≤ x ∇ y" "y ≤ x ∇ y" by (simp_all add: widen_order_rel_def)
  show "y ≤ x ⟹ y ≤ x Δ y" "y ≤ x ⟹ x Δ y ≤ x"
    by (simp_all add: narrow_order_rel_def)
qed

end

text ‹
  An order value prints in the notation of Goblint's two-variable equality domain,
  which prints a conjunction ‹{x=y ∧ z=y}› and its extremes as ‹⊥› and ‹⊤›;
  here each conjunct is an ordering ‹x≤y›. Printing needs a name for each variable
  and a linear order on them, so that \<^const>‹sorted_list_of_set› enumerates the
  facts deterministically (pairs are ordered lexicographically). The flag is not
  printed.
›

class order_name = linorder +
  fixes order_name :: "'a ⇒ String.literal"

instantiation String.literal :: order_name
begin

definition order_name_literal :: "String.literal ⇒ String.literal" where
  "order_name_literal x = x"

instance ..

end

fun string_of_facts :: "('v::order_name × 'v) list ⇒ String.literal" where
  "string_of_facts [] = STR ''''"
| "string_of_facts [(x, y)] = order_name x + sym_le + order_name y"
| "string_of_facts ((x, y) # p # ps) =
     order_name x + sym_le + order_name y + STR '' '' + sym_and + STR '' ''
     + string_of_facts (p # ps)"

instantiation order_rel :: (order_name) executable_domain
begin

definition is_empty_order_rel :: "'v::order_name order_rel ⇒ bool" where
  "is_empty_order_rel d ⟷ d = OBot"

definition to_string_order_rel :: "'v::order_name order_rel ⇒ String.literal" where
  "to_string_order_rel d =
     (case d of
        OBot ⇒ sym_bottom
      | OFacts _ ps ⇒
          (if ps = {} then sym_top
           else STR ''{'' + string_of_facts (sorted_list_of_set ps) + STR ''}''))"

instance ..

end

fun gamma_ord :: "kenv ⇒ var_id order_rel ⇒ store set" where
  "gamma_ord κ OBot = {}"
| "gamma_ord κ (OFacts t ps) = {s. (t ⟶ in_kinds κ s) ∧ (∀(x, y) ∈ ps. s x ≤ s y)}"

lemma gamma_ord_mono: "a ≤ b ⟹ gamma_ord κ a ⊆ gamma_ord κ b"
  by (cases a; cases b) auto

lemma gamma_ord_top [simp]: "gamma_ord κ top = UNIV"
  by (simp add: top_order_rel_def)

fun ord_forget :: "'v ⇒ 'v order_rel ⇒ 'v order_rel" where
  "ord_forget x OBot = OBot"
| "ord_forget x (OFacts t ps) = OFacts t {(a, b) ∈ ps. a ≠ x ∧ b ≠ x}"

text ‹Every write converts to its destination's kind except the return slot's, which no
  kind constrains, so a forgotten destination keeps the stores typed.›

lemma ord_forget_sound:
  "s ∈ gamma_ord κ d ⟹ v = ik_norm (κ x) n ∨ x = ReturnVar
     ⟹ s(x := v) ∈ gamma_ord κ (ord_forget x d)"
  by (cases d) auto

subsection ‹Guards›

text ‹The orders a comparison between ‹x› and ‹y› fixes when it has the truth value
  ‹pol›. A disequality fixes none: the carrier holds only weak orders.›

fun cmp_facts :: "cop ⇒ bool ⇒ 'v ⇒ 'v ⇒ ('v × 'v) set" where
  "cmp_facts CLt True x y = {(x, y)}"
| "cmp_facts CLe True x y = {(x, y)}"
| "cmp_facts CGt True x y = {(y, x)}"
| "cmp_facts CGe True x y = {(y, x)}"
| "cmp_facts CEq True x y = {(x, y), (y, x)}"
| "cmp_facts CNe True x y = {}"
| "cmp_facts CLt False x y = {(y, x)}"
| "cmp_facts CLe False x y = {(y, x)}"
| "cmp_facts CGt False x y = {(x, y)}"
| "cmp_facts CGe False x y = {(x, y)}"
| "cmp_facts CEq False x y = {}"
| "cmp_facts CNe False x y = {(x, y), (y, x)}"

lemma cmp_facts_sound:
  "cop_eval c (s x) (s y) = pol ⟹ (a, b) ∈ cmp_facts c pol x y ⟹ s a ≤ s b"
  by (cases c; cases pol) auto

fun ord_branch :: "exp ⇒ bool ⇒ var_id order_rel ⇒ var_id order_rel" where
  "ord_branch (TCmp c (TVar k x) (TVar k' y)) pol (OFacts t ps) = OFacts t (ps ∪ cmp_facts c pol x y)"
| "ord_branch (TNot b) pol d = ord_branch b (¬ pol) d"
| "ord_branch _ _ d = d"

lemma ord_branch_sound:
  "s ∈ gamma_ord κ d ⟹ truthy (eval b s) = pol ⟹ s ∈ gamma_ord κ (ord_branch b pol d)"
proof (induction b pol d rule: ord_branch.induct)
  case (1 c k x k' y pol t ps)
  then show ?case using cmp_facts_sound[of c s x y pol] by (auto split: if_splits)
next
  case (2 b pol d)
  then show ?case by (auto split: if_splits)
qed simp_all

subsection ‹Verdicts›

text ‹A pair ‹(x, y)› decides ‹x ≤ y› as true and ‹y < x› as false; with the reverse
  pair it also decides equality.›

definition ord_le :: "var_id order_rel ⇒ var_id ⇒ var_id ⇒ bool" where
  "ord_le d x y = (case d of OBot ⇒ False | OFacts _ ps ⇒ x = y ∨ (x, y) ∈ ps)"

lemma ord_le_sound: "ord_le d x y ⟹ s ∈ gamma_ord κ d ⟹ s x ≤ s y"
  by (cases d) (auto simp: ord_le_def)

fun ord_decide :: "cop ⇒ var_id order_rel ⇒ var_id ⇒ var_id ⇒ bool option" where
  "ord_decide CLe d x y = (if ord_le d x y then Some True else None)"
| "ord_decide CGe d x y = (if ord_le d y x then Some True else None)"
| "ord_decide CLt d x y = (if ord_le d y x then Some False else None)"
| "ord_decide CGt d x y = (if ord_le d x y then Some False else None)"
| "ord_decide CEq d x y = (if ord_le d x y ∧ ord_le d y x then Some True else None)"
| "ord_decide CNe d x y = (if ord_le d x y ∧ ord_le d y x then Some False else None)"

lemma ord_decide_sound:
  "ord_decide c d x y = Some r ⟹ s ∈ gamma_ord κ d ⟹ cop_eval c (s x) (s y) = r"
  using ord_le_sound[of d x y s κ] ord_le_sound[of d y x s κ]
  by (cases c) (auto split: if_splits)

fun ord_classify :: "exp ⇒ var_id order_rel ⇒ check_result" where
  "ord_classify (TCmp c (TVar k x) (TVar k' y)) d =
     (case ord_decide c d x y of
        Some True ⇒ Check_Proved
      | Some False ⇒ Check_Refuted
      | None ⇒ Check_Unknown)"
| "ord_classify _ d = Check_Unknown"

lemma ord_classify_sound:
  assumes "s ∈ gamma_ord κ d"
  shows ord_classify_proved: "ord_classify c d = Check_Proved ⟹ truthy (eval c s)"
    and ord_classify_refuted: "ord_classify c d = Check_Refuted ⟹ ¬ truthy (eval c s)"
  using assms ord_decide_sound
  by (induction c d rule: ord_classify.induct;
      fastforce split: option.splits bool.splits if_splits)+

subsection ‹The query›

text ‹
  The component answers ‹EvalInt› for a comparison between two variables from its own
  facts, as Goblint's ▩‹varEq› and ▩‹expRelation› do
  (▩‹varEq.ml› line 595, ▩‹expRelation.ml› lines 53 to 70 at ▩‹0dc12d355›):
  a decided comparison is the exact truth value, anything else is ‹⊤›. It asks
  nobody, as those analyses do not.
›

fun ord_query :: "channel ⇒ var_id order_rel lifted ⇒ channel" where
  "ord_query ch (Lifted d) (EvalInt (TCmp c (TVar k x) (TVar k' y))) =
     (case ord_decide c d x y of Some b ⇒ answer_of_int (if b then 1 else 0) | None ⇒ ⊤)"
| "ord_query ch d q = ⊤"

lemma ord_query_sound:
  "s ∈ gamma_ord κ d ⟹ eval_holds q (ord_query ch (Lifted d) q) s"
  using ord_decide_sound
  by (cases "(ch, Lifted d, q)" rule: ord_query.cases) (auto split: option.splits)

subsection ‹Assignments›

text ‹
  At ‹x = e› the component asks about the state before the assignment, where
  ‹(κ x) e› evaluates to the new value of ‹x› and every other variable keeps
  its value. The exact answer ‹1› to ‹(κ x) e ≤ y› makes ‹(x, y)› true
  afterwards, and ‹y ≤ (κ x) e› makes ‹(y, x)› true. The candidate partners are
  the variables in ‹ys›. A copy from a variable of the same kind records both orders
  when the stores are typed and gives the destination the source's facts, as Goblint's
  ▩‹varEq› records an equality for an assignment between variables of one type
  (▩‹varEq.ml› at ▩‹0dc12d355›).
›

definition ord_copy ::
    "kenv ⇒ var_id ⇒ exp ⇒ (var_id × var_id) set ⇒ (var_id × var_id) set" where
  "ord_copy κ x e ps = (case e of
     TVar k y ⇒ if y ≠ x ∧ y ≠ ReturnVar ∧ κ y = κ x
       then {(x, y), (y, x)} ∪ (λ(a, b). (x, b)) ` Set.filter (λ(a, b). a = y) ps
            ∪ (λ(a, b). (a, x)) ` Set.filter (λ(a, b). b = y) ps
       else {}
   | _ ⇒ {})"

definition ord_learn ::
    "kenv ⇒ var_id list ⇒ channel ⇒ var_id ⇒ exp ⇒ var_id order_rel ⇒ var_id order_rel" where
  "ord_learn κ ys ch x e d =
     (case d of
        OBot ⇒ OBot
      | OFacts t ps ⇒ OFacts t (ps
          ∪ (if t then ord_copy κ x e ps else {})
          ∪ set (map (λy. (x, y)) (filter (λy. y ≠ x
                ∧ answer_const (ch (EvalInt (TCmp CLe (TCast (κ x) e) (TVar (κ y) y)))) = Some 1) ys))
          ∪ set (map (λy. (y, x)) (filter (λy. y ≠ x
                ∧ answer_const (ch (EvalInt (TCmp CLe (TVar (κ y) y) (TCast (κ x) e)))) = Some 1) ys))))"

lemma eval_holds_constD:
  "eval_holds (EvalInt e) a s ⟹ answer_const a = Some n ⟹ eval e s = n"
  using gamma_int_dom_constant_subset by (cases a rule: answer_const.cases) auto

lemma ord_learn_sound:
  assumes post: "s(x := ik_norm (κ x) (eval e s)) ∈ gamma_ord κ d"
    and ch: "eval_query.channel_holds ch s"
  shows "s(x := ik_norm (κ x) (eval e s)) ∈ gamma_ord κ (ord_learn κ ys ch x e d)"
proof (cases d)
  case OBot with post show ?thesis by simp
next
  case (OFacts t ps)
  define s' where "s' = s(x := ik_norm (κ x) (eval e s))"
  have up: "s' x ≤ s' y"
    if "y ≠ x" "answer_const (ch (EvalInt (TCmp CLe (TCast (κ x) e) (TVar (κ y) y)))) = Some 1" for
      y
    using that eval_holds_constD[OF eval_query.channel_holdsD[OF ch] that(2)]
    by (simp add: s'_def split: if_splits)
  have down: "s' y ≤ s' x"
    if "y ≠ x" "answer_const (ch (EvalInt (TCmp CLe (TVar (κ y) y) (TCast (κ x) e)))) = Some 1" for
      y
    using that eval_holds_constD[OF eval_query.channel_holdsD[OF ch] that(2)]
    by (simp add: s'_def split: if_splits)
  have mem: "s' ∈ gamma_ord κ (OFacts t ps)" using post OFacts by (simp add: s'_def)
  have copy: "s' a ≤ s' b" if t: "t" and ab: "(a, b) ∈ ord_copy κ x e ps" for a b
  proof -
    from ab obtain k y where e: "e = TVar k y" and y: "y ≠ x" "y ≠ ReturnVar" "κ y = κ x"
      and ab': "(a, b) = (x, y) ∨ (a, b) = (y, x) ∨ (a = x ∧ (y, b) ∈ ps) ∨ (b = x ∧ (a, y) ∈ ps)"
      by (auto simp: ord_copy_def split: exp.splits if_splits)
    from post OFacts t have "in_kinds κ s'" by (simp add: s'_def)
    then have "s' y ∈ ik_range (κ x)" using y unfolding in_kinds_def by metis
    then have "s' x = s' y" using y by (simp add: s'_def e del: ik_range_iff)
    then show ?thesis using ab' mem by auto
  qed
  show ?thesis
    using mem up down copy unfolding s'_def[symmetric] by (auto simp: ord_learn_def OFacts)
qed

subsection ‹Calls›

text ‹The callee starts from no facts and the caller learns none back. The callee's
  stores are typed when the caller's are, since a call converts each actual to its
  formal's kind; the caller's after the call when both sides' are, as the result
  converts to its destination's kind.›

definition ord_in_kinds :: "var_id order_rel lifted ⇒ bool" where
  "ord_in_kinds d = (case d of Lifted (OFacts t _) ⇒ t | _ ⇒ False)"

definition ord_enter :: "var_id list ⇒ exp list ⇒ var_id order_rel lifted ⇒ var_id order_rel" where
  "ord_enter ps es d = OFacts (ord_in_kinds d) {}"

subsection ‹The component›

text ‹
  Assignments forget their destination and learn, library calls forget it, and a
  return forgets the return slot. The caller learns nothing back from a call: a callee
  may change globals, so no fact survives it.
›

definition ord_spec :: "kenv ⇒ var_id list ⇒ var_id order_rel lifted local_spec"
  where
  "ord_spec κ ys =
     ⦇ ls_query = ord_query,
       ls_skip = (λch d. d),
       ls_assign = (λch x e. map_lift (ord_learn κ ys ch x e ∘ ord_forget x)),
       ls_special = (λch sc x. map_lift (ord_forget x)),
       ls_branch = (λch b pol. map_lift (ord_branch b pol)),
       ls_body = (λch p d. d),
       ls_return = (λch r p d. case r of None ⇒ d | Some a ⇒ map_lift (ord_forget ReturnVar) d),
       ls_event = (λch evt d. d),
       ls_enter = (λch ci p. [(fst p, Lifted (ord_enter (ci_formals ci) (ci_args ci) (fst p)))]),
       ls_combine_env = combine_env_identity,
       ls_combine_assign = (λch ci d de. Lifted (OFacts (ord_in_kinds d ∧ ord_in_kinds de) {})) ⦈"

abbreviation gamma_ord_lift :: "kenv ⇒ var_id order_rel lifted ⇒ store set" where
  "gamma_ord_lift κ d ≡ gamma_lift (gamma_ord κ) (map_lift (λd. d) d)"

lemma gamma_ord_lift_simps [simp]:
  "gamma_ord_lift κ Bot = {}"
  "gamma_ord_lift κ (Lifted x) = gamma_ord κ x"
  "gamma_ord_lift κ (map_lift f (Lifted x)) = gamma_ord κ (f x)"
  by (simp_all add: map_lift_def)

lemma gamma_ord_lift_mono: "x ≤ y ⟹ gamma_ord_lift κ x ⊆ gamma_ord_lift κ y"
  by (cases x; cases y) (auto dest: gamma_ord_mono)

lemma ord_enter_sound:
  assumes "s ∈ gamma_ord_lift κ d"
  shows "bind_formals ps (map2 (λx. ik_norm (κ x)) ps (map (λe. eval e s) es))
           (enter_state is_global s)
           ∈ gamma_ord κ (ord_enter ps es d)"
proof -
  have "in_kinds κ (call_enter κ (CallEdge dst ps es) s)" if "ord_in_kinds d" for dst
    using assms that by (intro in_kinds_call_enter)
      (auto simp: ord_in_kinds_def split: lifted.splits order_rel.splits)
  then show ?thesis
    by (auto simp: ord_enter_def call_enter_CallEdge)
qed

lemma ord_in_kinds_in_kinds: "s ∈ gamma_ord_lift κ d ⟹ ord_in_kinds d ⟹ in_kinds κ s"
  by (auto simp: ord_in_kinds_def split: lifted.splits order_rel.splits)

lemma in_kinds_gcombine:
  "in_kinds κ s ⟹ in_kinds κ t
     ⟹ in_kinds κ (combine_assign dst (kind_conv κ dst (t ReturnVar)) (combine_env is_global s t))"
  by (cases dst) (auto simp: in_kinds_combine_env)

theorem ord_spec_sound:
  "sound_local_spec κ (gamma_ord_lift κ) (ord_spec κ ys)"
proof -
  have L: "∃x. d = Lifted x ∧ s ∈ gamma_ord κ x" if "s ∈ gamma_ord_lift κ d" for s d
    using that by (cases d) simp_all
  show ?thesis
  proof (rule sound_local_specI)
    show "⋀x y. x ≤ y ⟹ gamma_ord_lift κ x ⊆ gamma_ord_lift κ y" by (rule gamma_ord_lift_mono)
    show "sound_query (gamma_ord_lift κ) (ls_query (ord_spec κ ys))"
      by (auto simp: ord_spec_def sound_query_def ord_query_sound dest!: L)
    show "sound_skip (gamma_ord_lift κ) (ls_skip (ord_spec κ ys))"
      by (simp add: ord_spec_def sound_skip_def)
    show "sound_body (gamma_ord_lift κ) (ls_body (ord_spec κ ys))"
      by (simp add: ord_spec_def sound_body_def)
    show "sound_event (gamma_ord_lift κ) (ls_event (ord_spec κ ys))"
      by (simp add: ord_spec_def sound_event_def)
    show "sound_assign κ (gamma_ord_lift κ) (ls_assign (ord_spec κ ys))"
      by (auto dest!: L simp: ord_spec_def sound_assign_def
          intro!: ord_learn_sound ord_forget_sound)
    show "sound_special κ (gamma_ord_lift κ) (ls_special (ord_spec κ ys))"
      by (auto simp: ord_spec_def sound_special_def dest!: L intro: ord_forget_sound)
    show "sound_branch (gamma_ord_lift κ) (ls_branch (ord_spec κ ys))"
      using ord_branch_sound by (fastforce simp: ord_spec_def sound_branch_def dest!: L)
    show "sound_return (gamma_ord_lift κ) (ls_return (ord_spec κ ys))"
      by (auto simp: ord_spec_def sound_return_def dest!: L intro: ord_forget_sound
          split: option.splits)
    show "sound_enter κ (gamma_ord_lift κ) (ls_enter (ord_spec κ ys))"
      by (auto simp: ord_spec_def sound_enter_def call_enter_CallEdge
          intro: ord_enter_sound)
    show "sound_combine κ (gamma_ord_lift κ) (ls_combine_env (ord_spec κ ys))
        (ls_combine_assign (ord_spec κ ys))"
      by (auto simp: ord_spec_def sound_combine_def combine_env_identity_def
          combine_collect_def intro!: in_kinds_gcombine dest:
            ord_in_kinds_in_kinds)
  qed
qed

text ‹The candidate partners: every variable some procedure of the program declares
  (@{const program_vars}).›

definition ord_vars :: "imp_prog ⇒ var_id list" where
  "ord_vars p = remdups (concat (map (λq. map (resolve_var (declared_global p) q)
     (filter (λx. x ≠ ret_var) (scope_vnames_list p q))) (prog_main_name # prog_procs p)))"

subsection ‹The pipeline instance›

text ‹The same unit-context instance as the non-relational typed analyses, with the
  order component in place of a per-variable state. The concretization does not depend
  on the kind declarations, but the program's runs and initial stores do.›

global_interpretation order_analysis: dg_analysis
    "kenv_of ds" "λ_. init_stores (kenv_of ds)"
    "λ_ p. ord_spec (kenv_of ds) (ord_vars p)" "λ_ d. d = OBot" "λ_ d. d" "OFacts True {}"
    "Analysis_Global ()" Analysis_Global Activation_Seed "λ_ _ _ _ _. ()" "()"
    "TD_side_rule_Interp_solve r"
    "TD_side_rule_Interp.solve_dom TYPE((unit, unit) global_unknown)
       TYPE((var_id order_rel lifted, var_id order_rel lifted) dg_state) r"
    OBot ord_classify "gamma_ord (kenv_of ds)" "λd. d = OBot"
    "TD_side_rule_Interp_solve_c r"
    "λ_ c. dg_spec_of c" "λp d e. d" "λ_ d. d" "λ_ d. Bot" "λ_ p. []" "λp ci d e. d"
    "λp. elab_cfg p ds" "λ_. FunctionEntry prog_main_name"
  for ds r
proof (rule dg_analysis_whole_stateI[OF td_certified_solver], goal_cases CompSound EnterSingle
    EmptyRd EmptyV SeedNe ClProved ClRefuted BotState InitSound SeedNeGlobal PEntry FinIntra
    FinCalls CallsUnique)
  case (CompSound p) show ?case by (rule ord_spec_sound)
next
  case (EnterSingle p ci d) show ?case
    by (simp add: ord_spec_def dg_pipeline.comp_entry_def)
next
  case (EmptyRd p s) show ?case by simp
next
  case EmptyV show ?case by (rule sound_emptinessI) simp
next
  case (SeedNe v ctx) show ?case by simp
next
  case (ClProved c d s) then show ?case by (intro ord_classify_proved)
next
  case (ClRefuted c d s) then show ?case by (intro ord_classify_refuted)
next
  case BotState show ?case by simp
next
