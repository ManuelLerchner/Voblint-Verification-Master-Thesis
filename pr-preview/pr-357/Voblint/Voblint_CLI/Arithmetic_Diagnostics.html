Theory Arithmetic_Diagnostics

theory Arithmetic_Diagnostics
  imports "Voblint_Framework.Contextual_Check_Report"
begin

section ‹Arithmetic obligations›

text ‹Diagnostics inspect all expression operands, matching VIMP's total,
  pure expression model. They neither change execution nor assume short-circuit
  evaluation for logical operators. Lists retain repeated syntactically equal occurrences.›

text ‹
  An obligation names an operation, the hazard it may run into and the operand the
  hazard concerns. A division or remainder must not divide by zero; its operand is the
  divisor. A signed arithmetic operation must compute a result its kind represents,
  as C requires (ISO/IEC 9899 6.5p5); its operand is the operation itself. A signed
  remainder must have a quotient its kind represents, since C leaves the remainder
  undefined when the quotient is (6.5.5p6); its operand is that quotient. A conversion
  must convert a value the target kind represents, the categories Goblint separates as
  arithmetic and cast overflow (▩‹Checks.Category› at ‹0dc12d355›); its operand is
  the converted expression. The result of a library ‹min› or ‹max› is converted as
  well, but has no expression of its own: its obligation names the conversion of the
  first argument and, as operand, the second. Which hazards a language raises, and when
  one is safe, is the language's; the report treats them alike.
›

datatype arithmetic_hazard = Division_By_Zero | Signed_Overflow | Remainder_Overflow
  | Conversion_Overflow | Min_Conversion_Overflow | Max_Conversion_Overflow

datatype 'e arithmetic_obligation = Arithmetic_Obligation
  (arithmetic_hazard: arithmetic_hazard)
  (arithmetic_operation: 'e)
  (arithmetic_operand: 'e)

section ‹Sites and their solved classifications›

text ‹
  ‹arithmetic_expression_sites› collects the expressions each CFG edge and call
  evaluates; ‹obligation_sites› turns them into the obligations the solved table then
  classifies, given how an expression of the language yields them.
›

text ‹Only suppress a negative guard edge when the same point has a matching
  positive guard edge. All other expression-bearing edges remain in the report.›

definition arithmetic_expression_sites ::
    "('v::linorder, 'e::linorder) gcfg ⇒ (pp × 'e list) list" where
  "arithmetic_expression_sites g =
     (let edges = cfg_intra_list g
      in map (λ(u, a, v). (u, edge_expressions a))
           (filter (λ(u, a, v). case a of EA_AssumeNot e ⇒
             ¬ list_ex (λ(u', a', v'). u = u' ∧ a' = EA_Assume e) edges
             | _ ⇒ True) edges)) @
       map (λ(u, ca, ce, k). (u, ce_args ca)) (cfg_calls_list g)"

definition obligation_sites ::
    "('e ⇒ 'e arithmetic_obligation list) ⇒ ('v::linorder, 'e::linorder) gcfg
       ⇒ (pp × 'e arithmetic_obligation list) list" where
  "obligation_sites obls g =
     map (λ(v, es). (v, concat (map obls es))) (arithmetic_expression_sites g)"

text ‹A diagnostic names an obligation of a site list, read through a classifier of
  obligations. The site list and the classifier depend on the expression language and
  the analysis; the classification does not.›

datatype 'e arithmetic_diagnostic = Arithmetic_Diagnostic
  (diagnostic_point: pp)
  (diagnostic_occurrence: nat)
  (diagnostic_obligation: "'e arithmetic_obligation")
  (diagnostic_verdict: check_result)

fun arithmetic_diagnostic_of ::
    "pp ⇒ nat ⇒ 'e arithmetic_obligation ⇒ contextual_verdict
       ⇒ 'e arithmetic_diagnostic list" where
  "arithmetic_diagnostic_of v i obligation (Lifted Check_Refuted) =
     [Arithmetic_Diagnostic v i obligation Check_Refuted]"
| "arithmetic_diagnostic_of v i obligation (Lifted Check_Unknown) =
     [Arithmetic_Diagnostic v i obligation Check_Unknown]"
| "arithmetic_diagnostic_of v i obligation _ = []"

definition arithmetic_diagnostics ::
    "(pp × 'e arithmetic_obligation list) list ⇒ ('ctx, 'a) solved_table
       ⇒ ('e arithmetic_obligation ⇒ 'a ⇒ check_result)
       ⇒ 'e arithmetic_diagnostic list" where
  "arithmetic_diagnostics sites r classify =
     concat (map (λ(v, obligations).
       concat (map (λ(i, obligation).
         arithmetic_diagnostic_of v i obligation (point_verdict r classify v obligation))
         (zip [0..<length obligations] obligations))) sites)"

lemma arithmetic_expression_sites_intra:
  assumes "finite (intra g)" and "(v, a, w) ∈ intra g"
  shows "(v, edge_expressions a) ∈ set (arithmetic_expression_sites g)"
proof -
  let ?edges = "cfg_intra_list g"
  have mem: "(v, a, w) ∈ set ?edges" using assms by simp
  have kept: "⋀b t. (v, b, t) ∈ set ?edges ⟹
      (case b of EA_AssumeNot e ⇒
        ¬ list_ex (λ(u', a', v'). v = u' ∧ a' = EA_Assume e) ?edges
        | _ ⇒ True) ⟹
      (v, edge_expressions b) ∈ set (arithmetic_expression_sites g)"
    unfolding arithmetic_expression_sites_def
    by (force simp: Let_def image_iff)
  show ?thesis
  proof (cases "∃e. a = EA_AssumeNot e")
    case False
    with mem show ?thesis by (intro kept[of a w]) (auto split: gedge_action.splits)
  next
    case True
    then obtain e where a: "a = EA_AssumeNot e" by blast
    show ?thesis
    proof (cases "list_ex (λ(u', a', v'). v = u' ∧ a' = EA_Assume e) ?edges")
      case False
      from kept[OF mem] False a show ?thesis by simp
    next
      case True
      then obtain t where pos: "(v, EA_Assume e, t) ∈ set ?edges"
        by (auto simp: list_ex_iff)
      from kept[OF pos] a show ?thesis by simp
    qed
  qed
qed

lemma arithmetic_expression_sites_call:
  assumes "finite (calls g)" and "(v, ca, ce, k) ∈ calls g"
  shows "(v, ce_args ca) ∈ set (arithmetic_expression_sites g)"
  using assms unfolding arithmetic_expression_sites_def
  by (force simp: image_iff)

lemma arithmetic_diagnostics_absent:
  assumes site: "(v, obligations) ∈ set sites"
    and ob: "obligation ∈ set obligations"
    and absent: "∀d ∈ set (arithmetic_diagnostics sites r classify). diagnostic_point d ≠ v"
  shows "point_verdict r classify v obligation = Bot ∨
    point_verdict r classify v obligation = Lifted Check_Proved"
proof -
  have "obligation ∈ set (map snd (zip [0..<length obligations] obligations))"
    using ob by simp
  then obtain pair where pair: "pair ∈ set (zip [0..<length obligations] obligations)"
    and snd: "snd pair = obligation"
    unfolding set_map by blast
  have indexed: "(fst pair, obligation) ∈ set (zip [0..<length obligations] obligations)"
    using pair snd by (cases pair) auto
  define i where "i = fst pair"
  have indexed: "(i, obligation) ∈ set (zip [0..<length obligations] obligations)"
    using indexed unfolding i_def .
  let ?verdict = "point_verdict r classify v obligation"
  have included: "set (arithmetic_diagnostic_of v i obligation ?verdict)
      ⊆ set (arithmetic_diagnostics sites r classify)"
    using site indexed
    unfolding arithmetic_diagnostics_def
    by (force simp only: set_concat set_map image_iff case_prod_beta)
  have point: "⋀d. d ∈ set (arithmetic_diagnostic_of v i obligation ?verdict) ⟹
      diagnostic_point d = v"
  proof -
    fix d
    assume d: "d ∈ set (arithmetic_diagnostic_of v i obligation ?verdict)"
    show "diagnostic_point d = v"
    proof (cases ?verdict)
      case Bot
      with d show ?thesis by simp
    next
      case (Lifted verdict)
      with d show ?thesis by (cases verdict) auto
    qed
  qed
  have empty: "arithmetic_diagnostic_of v i obligation ?verdict = []"
    using included point absent
    by (meson empty_subsetI set_empty2 subset_antisym subset_iff)
  show ?thesis
  proof (cases ?verdict)
    case Bot
    then show ?thesis by simp
  next
    case (Lifted verdict)
    with empty show ?thesis by (cases verdict) auto
  qed
qed

text ‹An error names a point whose obligation the solved table refutes.›

lemma arithmetic_diagnostic_of_refuted:
  assumes "d ∈ set (arithmetic_diagnostic_of v i obligation verdict)"
    and "diagnostic_verdict d = Check_Refuted"
  shows "verdict = Lifted Check_Refuted ∧ diagnostic_point d = v
    ∧ diagnostic_obligation d = obligation"
  using assms by (cases "(v, i, obligation, verdict)" rule: arithmetic_diagnostic_of.cases) auto

lemma arithmetic_diagnostics_refuted:
  assumes "d ∈ set (arithmetic_diagnostics sites r classify)"
    and "diagnostic_verdict d = Check_Refuted"
  shows "point_verdict r classify (diagnostic_point d) (diagnostic_obligation d)
      = Lifted Check_Refuted"
  using assms unfolding arithmetic_diagnostics_def
  by (auto dest!: arithmetic_diagnostic_of_refuted)

end
