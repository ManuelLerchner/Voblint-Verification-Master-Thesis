section ‹Parity at a kind›

text ‹
  Normalizing to a kind changes a value by a multiple of ‹2 ^ w›, which is even, so
  every operation keeps its parity across a wraparound (‹parity_cast_sound›). The
  typed operations are therefore the mathematical ones. Every kind's range holds an
  even and an odd value, so a parity meets the range exactly when it is not bottom, and
  no parity other than bottom bounds a value to a kind. Parity has no Goblint
  counterpart.
›

instantiation parity :: ik_domain
begin

definition widen_ik_parity :: "ikind ⇒ parity ⇒ parity ⇒ parity" where
  "widen_ik_parity k a b = a ⊔ b"

definition narrow_ik_parity :: "ikind ⇒ parity ⇒ parity ⇒ parity" where
  "narrow_ik_parity k a b = a ⊓ b"

definition empty_ik_parity :: "ikind ⇒ parity ⇒ bool" where
  "empty_ik_parity k a ⟷ a = PBot"

instance proof
  fix k :: ikind and a b :: parity
  show "a ≤ widen_ik k a b" "b ≤ widen_ik k a b" by (simp_all add: widen_ik_parity_def)
  show "b ≤ a ⟹ b ≤ narrow_ik k a b" "b ≤ a ⟹ narrow_ik k a b ≤ a"
    by (simp_all add: narrow_ik_parity_def inf_absorb2)
  show "empty_ik k a ⟷ numeric_domain_class.gamma a ∩ ik_range k = {}"
  proof (cases a)
    case PEven then show ?thesis
      by (simp add: empty_ik_parity_def disjoint_iff) (rule exI[of _ 0], simp)
  next
    case POdd then show ?thesis
      using one_in_ik_range[of k]
      by (simp add: empty_ik_parity_def disjoint_iff del: one_in_ik_range) (rule exI[of _ 1], simp)
  next
    case PTop then show ?thesis
      by (simp add: empty_ik_parity_def) (metis zero_in_ik_range empty_iff)
  qed (simp add: empty_ik_parity_def)
qed

end

fun parity_aop :: "aop ⇒ parity ⇒ parity ⇒ parity" where
  "parity_aop AAdd a b = a + b"
| "parity_aop ASub a b = a - b"
| "parity_aop AMul a b = a * b"
| "parity_aop ADiv a b = parity_div a b"
| "parity_aop AMod a b = parity_mod a b"

lemma parity_aop_sound:
  "m ∈ gamma_parity a ⟹ n ∈ gamma_parity b ⟹ aop_eval op m n ∈ gamma_parity (parity_aop op a b)"
  by (cases op) (simp_all add: parity_plus_sound parity_minus_sound parity_times_sound
      parity_div_sound parity_mod_sound)


instantiation parity :: kind_int_domain
begin

definition top_ik_parity :: "ikind ⇒ parity" where "top_ik_parity k = PTop"

definition of_int_ik_parity :: "ikind ⇒ int ⇒ parity" where
  "of_int_ik_parity k n = parity_of_int (ik_norm k n)"

definition cast_ik_parity :: "signed_overflow_policy ⇒ ikind option ⇒ ikind ⇒ parity ⇒ parity" where
  "cast_ik_parity ovp sk k a = a"

definition arith_ik_parity ::
  "signed_overflow_policy ⇒ ikind ⇒ aop ⇒ bool ⇒ parity ⇒ parity ⇒ parity" where
  "arith_ik_parity ovp k op fits = parity_aop op"

definition cmp_ik_parity :: "cop ⇒ parity ⇒ parity ⇒ bool option" where
  "cmp_ik_parity = cop_of parity_lt parity_eqb"

definition meet_ik_parity :: "parity ⇒ parity ⇒ parity" where
  "meet_ik_parity a b = a ⊓ b"

definition refine_ik_parity :: "cop ⇒ parity ⇒ parity ⇒ parity × parity" where
  "refine_ik_parity c a b = (if c = CEq then (a ⊓ b, a ⊓ b) else (a, b))"

definition raw_aop_ik_parity :: "aop ⇒ parity ⇒ parity ⇒ parity" where
  "raw_aop_ik_parity = parity_aop"

definition fits_ik_parity :: "ikind ⇒ parity ⇒ bool" where
  "fits_ik_parity k a ⟷ a = PBot"

definition answer_ik_parity :: "parity ⇒ answer" where
  "answer_ik_parity a = answer_with (int_parity_update (λ_. a))"

definition of_answer_ik_parity :: "ikind option ⇒ answer ⇒ parity" where
  "of_answer_ik_parity sk q = (case q of QBot ⇒ ⊥ | QTop ⇒ ⊤ | QLifted d ⇒ int_parity d)"

instance proof
  fix n m :: int and k :: ikind and a b :: parity and op :: aop and c :: cop and r :: bool
  show "n ∈ ik_range k ⟹ n ∈ numeric_domain_class.gamma (top_ik k :: parity)"
    by (simp add: top_ik_parity_def)
  show "ik_norm k n ∈ numeric_domain_class.gamma (of_int_ik k n :: parity)"
    by (simp add: of_int_ik_parity_def parity_of_int_gamma)
  show "(⋀k'. sk = Some k' ⟹ n ∈ ik_range k') ⟹ n ∈ numeric_domain_class.gamma a
      ⟹ ik_norm k n ∈ numeric_domain_class.gamma (cast_ik ovp sk k a)" for ovp sk
    by (simp add: cast_ik_parity_def parity_cast_sound)
  show "(fits ⟹ m ∈ ik_range k ∧ n ∈ ik_range k) ⟹ m ∈ numeric_domain_class.gamma a
      ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ ik_norm k (aop_eval op m n) ∈ numeric_domain_class.gamma (arith_ik ovp k op fits a b)" for ovp fits
    by (simp add: arith_ik_parity_def parity_cast_sound parity_aop_sound)
  show "cmp_ik c a b = Some r ⟹ m ∈ numeric_domain_class.gamma a
      ⟹ n ∈ numeric_domain_class.gamma b ⟹ cop_eval c m n = r"
    unfolding cmp_ik_parity_def by simp (rule cop_of_sound[OF parity_lt_sound parity_eqb_sound])
  show "m ∈ numeric_domain_class.gamma a ⟹ m ∈ numeric_domain_class.gamma b
      ⟹ m ∈ numeric_domain_class.gamma (meet_ik a b)"
    by (simp add: meet_ik_parity_def)
  show "cop_eval c m n ⟹ m ∈ numeric_domain_class.gamma a ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ m ∈ numeric_domain_class.gamma (fst (refine_ik c a b))
        ∧ n ∈ numeric_domain_class.gamma (snd (refine_ik c a b))"
    by (cases c) (simp_all add: refine_ik_parity_def)
  show "m ∈ numeric_domain_class.gamma a ⟹ n ∈ numeric_domain_class.gamma b
      ⟹ aop_eval op m n ∈ numeric_domain_class.gamma (raw_aop_ik op a b)"
    by (simp add: raw_aop_ik_parity_def parity_aop_sound)
  show "fits_ik k a ⟹ n ∈ numeric_domain_class.gamma a ⟹ n ∈ ik_range k"
    by (simp add: fits_ik_parity_def)
  show "n ∈ numeric_domain_class.gamma a ⟹ n ∈ gamma_answer (answer_ik a)"
    by (simp add: answer_ik_parity_def gamma_int_dom_def top_int_dom_ext_def gamma_sign_top
        gamma_parity_top top_ivl_def gamma_ivl_top)
  show "(⋀k. sk = Some k ⟹ n ∈ ik_range k) ⟹ n ∈ gamma_answer q
      ⟹ n ∈ numeric_domain_class.gamma (of_answer_ik sk q :: parity)" for sk q
    by (cases q)
      (simp_all add: of_answer_ik_parity_def gamma_int_dom_def gamma_sign_top gamma_parity_top)
qed

end

declare [[code abort: gamma_parity]]

subsection ‹Executed›

text ‹A step of ‹2› from ‹0› keeps ‹x› even at \<^const>‹U8›, across every wraparound.›

definition parity_prog :: imp_prog where
  "parity_prog = program {
     fun f() { x = 0; while (x != 7) { x = x + 2; } }
     fun main() { f(); } }"

lemma parity_value:
  "nonrel.state_at default_sem_conf [(ScopedId (STR ''f'') (STR ''x''), U8)] 0 Globals_Join is_global parity_prog []
     (Statement 1) (ScopedId (STR ''f'') (STR ''x''))
     = KD U8 PEven"
  by eval
