Theory Run_Voblint

theory Run_Voblint
  imports Run_Report
begin

section ‹The public analysis operation›

text ‹
  ‹run_voblint› analyzes a program under its kind declarations with the options of a
  \<^typ>‹voblint_conf›: the activated analyses, a context policy, a global update
  rule, where program globals live, and the options the domains read, as Goblint's
  configuration tree selects them (▩‹ana.activated›, ▩‹sem.int.signed_overflow›). It
  answers the report \<^const>‹report_of› builds from the solve. An activation is valid
  when it names at least one analysis and none twice.
›

definition valid_activation :: "analysis list ⇒ bool" where
  "valid_activation as ⟷ as ≠ [] ∧ distinct as"

definition run_voblint :: "voblint_conf ⇒ kind_decls ⇒ imp_prog ⇒ report analysis_answer" where
  "run_voblint conf ds p =
     (if ¬ valid_activation (conf_analyses conf) then Invalid_Activation
      else if ¬ wf_program_compile_input_exec p then Malformed_Program
      else case report_run conf ds p of
             None ⇒ No_Answer
           | Some res ⇒ Analysed res)"

subsection ‹What an answer guarantees›

text ‹
  The headline theorem, about the report a caller receives. Its stores are the typed
  stores that reach a node from a store whose globals are zero at their kinds. Every
  such store lies in one of the states the report lists at the node, read through the
  concretization of the activated analyses; a proved check holds at it and a refuted one
  fails. Neither verdict says that the node is reached, and an unknown verdict says
  nothing. The theorem holds for every configuration. It assumes only that the run
  answered: termination of the solve is not a premise, because the executable solve
  answers only when it has terminated.
›

theorem run_voblint_sound:
  assumes run: "run_voblint conf ds p = Analysed res"
  shows run_voblint_states:
      "⋀v s. s ∈ reaching ds p v
         ⟹ ∃d. Lifted d ∈ report_states_at res v
               ∧ s ∈ mcp.act_gamma ds (activation_of (conf_analyses conf)) d"
    and run_voblint_proved:
      "⋀chk s. chk ∈ set (report_checks res) ⟹ check_verdict chk = Decided Check_Proved
         ⟹ s ∈ reaching ds p (check_point chk) ⟹ truthy (eval (check_exp chk) s)"
    and run_voblint_refuted:
      "⋀chk s. chk ∈ set (report_checks res) ⟹ check_verdict chk = Decided Check_Refuted
         ⟹ s ∈ reaching ds p (check_point chk) ⟹ ¬ truthy (eval (check_exp chk) s)"
proof -
  have exec: "wf_program_compile_input_exec p"
    and rep: "report_run conf ds p = Some res"
    using run by (auto simp: run_voblint_def split: if_splits option.splits)
  note wf = wf_program_compile_input_exec_sound[OF exec]
  show "⋀v s. s ∈ reaching ds p v
          ⟹ ∃d. Lifted d ∈ report_states_at res v
                ∧ s ∈ mcp.act_gamma ds (activation_of (conf_analyses conf)) d"
    by (erule report_state_sound[OF wf rep]) blast
  show "⋀chk s. chk ∈ set (report_checks res) ⟹ check_verdict chk = Decided Check_Proved
          ⟹ s ∈ reaching ds p (check_point chk) ⟹ truthy (eval (check_exp chk) s)"
    using report_check_sound(1)[OF wf rep] by blast
  show "⋀chk s. chk ∈ set (report_checks res) ⟹ check_verdict chk = Decided Check_Refuted
          ⟹ s ∈ reaching ds p (check_point chk) ⟹ ¬ truthy (eval (check_exp chk) s)"
    using report_check_sound(2)[OF wf rep] by blast
qed

text ‹An answer was given for a valid activation and a well-formed program, and its
  report records the configuration it was run with.›

lemma run_voblint_AnalysedE:
  assumes "run_voblint conf ds p = Analysed res"
  obtains "valid_activation (conf_analyses conf)" and "wf_program_compile_input_exec p"
    and "report_run conf ds p = Some res"
  using assms unfolding run_voblint_def by (auto split: if_splits option.splits)

lemma report_run_shape:
  assumes "report_run conf ds p = Some res"
  shows "report_config res = conf" and "report_cfg res = elab_cfg p ds"
  using assms
  by (auto simp: report_run_def cs_run_def es_run_def prog_report_of_def
      report_of_def Let_def split: context_policy.splits program_globals.splits)

text ‹
  The report contract in one statement: an analysed answer was given for a valid
  activation and a well-formed program, and is a report of exactly that
  configuration and the program's elaborated graph. The theorems that follow read it.
›

theorem run_voblint_report_contract:
  assumes "run_voblint conf ds p = Analysed res"
  shows "valid_activation (conf_analyses conf)" and "wf_program_compile_input_exec p"
    and "report_config res = conf" and "report_cfg res = elab_cfg p ds"
  using assms by (auto elim!: run_voblint_AnalysedE dest: report_run_shape)

subsection ‹What the arithmetic diagnostics guarantee›

text ‹
  A point the report's diagnostics do not name is safe: at every store that reaches it,
  every obligation of every expression an edge or a call there evaluates holds, every
  value returned there fits the kind each caller converts it to, and every ‹min› or
  ‹max› computed there fits the kind of its destination. No divisor is zero, no
  signed operation computes a result outside its kind, no signed remainder has a quotient
  outside its kind, and no conversion narrower than its source kind converts a value
  outside its target. A diagnostic says nothing on its own unless it is refuted: then its
  obligation fails at every store that reaches its point, so a run that reaches the point
  violates it. The run is one of core VIMP, which evaluates both operands of ‹&&› and
  ‹||›: a C program that short-circuits need not evaluate the operand whose obligation
  is refuted. Both hold under either overflow policy, which changes only how precisely
  the analyses bound a wrapped value.
›

theorem run_voblint_arithmetic_safe:
  assumes run: "run_voblint conf ds p = Analysed res"
    and absent: "SAFE res v"
    and s: "s ∈ reaching ds p v"
  shows run_voblint_arithmetic_safe_intra:
      "(v, a, w) ∈ intra (elab_cfg p ds) ⟹ e ∈ set (edge_expressions a)
         ⟹ ob ∈ set (arith_obligations e) ⟹ s ⊨ ob"
    and run_voblint_arithmetic_safe_call:
      "(v, ca, ce, k) ∈ calls (elab_cfg p ds) ⟹ e ∈ set (ce_args ca)
         ⟹ ob ∈ set (arith_obligations e) ⟹ s ⊨ ob"
    and run_voblint_arithmetic_safe_return:
      "(u, ca, ce, k) ∈ calls (elab_cfg p ds) ⟹ ce_dst ca = Some x
         ⟹ (v, EA_Ret (Some e) (callee_of_entry ce), w) ∈ intra (elab_cfg p ds)
         ⟹ ¬ ik_widens (exp_kind e) (kenv_of ds x)
         ⟹ eval e s ∈ ik_range (kenv_of ds x)"
    and run_voblint_arithmetic_safe_min:
      "(v, EA_Special (Min e1 e2) x, w) ∈ intra (elab_cfg p ds)
         ⟹ ¬ (ik_widens (exp_kind e1) (kenv_of ds x) ∧ ik_widens (exp_kind e2) (kenv_of ds x))
         ⟹ min (eval e1 s) (eval e2 s) ∈ ik_range (kenv_of ds x)"
    and run_voblint_arithmetic_safe_max:
      "(v, EA_Special (Max e1 e2) x, w) ∈ intra (elab_cfg p ds)
         ⟹ ¬ (ik_widens (exp_kind e1) (kenv_of ds x) ∧ ik_widens (exp_kind e2) (kenv_of ds x))
         ⟹ max (eval e1 s) (eval e2 s) ∈ ik_range (kenv_of ds x)"
proof -
  from run have wf: "wf_program_compile_input p"
    and rep: "report_run conf ds p = Some res"
    by (auto elim!: run_voblint_AnalysedE intro: wf_program_compile_input_exec_sound)
  note safe = report_obligation_safe[OF wf rep s _ _ absent]
  show "s ⊨ ob"
    if "(v, a, w) ∈ intra (elab_cfg p ds)" "e ∈ set (edge_expressions a)"
      "ob ∈ set (arith_obligations e)"
    using prog_obligation_site_intra[OF that] safe by blast
  show "s ⊨ ob"
    if "(v, ca, ce, k) ∈ calls (elab_cfg p ds)" "e ∈ set (ce_args ca)"
      "ob ∈ set (arith_obligations e)"
    using prog_obligation_site_call[OF that] safe by blast
  show "eval e s ∈ ik_range (kenv_of ds x)"
    if "(u, ca, ce, k) ∈ calls (elab_cfg p ds)" "ce_dst ca = Some x"
      "(v, EA_Ret (Some e) (callee_of_entry ce), w) ∈ intra (elab_cfg p ds)"
      "¬ ik_widens (exp_kind e) (kenv_of ds x)"
    using safe[OF prog_obligation_site_return[OF that]] by simp
  show "min (eval e1 s) (eval e2 s) ∈ ik_range (kenv_of ds x)"
    if edge: "(v, EA_Special (Min e1 e2) x, w) ∈ intra (elab_cfg p ds)"
      and narrows:
        "¬ (ik_widens (exp_kind e1) (kenv_of ds x) ∧ ik_widens (exp_kind e2) (kenv_of ds x))"
  proof -
    have ob: "Arithmetic_Obligation Min_Conversion_Overflow (TCast (kenv_of ds x) e1) e2
        ∈ set (special_obligations (kenv_of ds x) (Min e1 e2))"
      using narrows by simp
    show ?thesis using safe[OF prog_obligation_site_special[OF edge ob] ob] by simp
  qed
  show "max (eval e1 s) (eval e2 s) ∈ ik_range (kenv_of ds x)"
    if edge: "(v, EA_Special (Max e1 e2) x, w) ∈ intra (elab_cfg p ds)"
      and narrows:
        "¬ (ik_widens (exp_kind e1) (kenv_of ds x) ∧ ik_widens (exp_kind e2) (kenv_of ds x))"
  proof -
    have ob: "Arithmetic_Obligation Max_Conversion_Overflow (TCast (kenv_of ds x) e1) e2
        ∈ set (special_obligations (kenv_of ds x) (Max e1 e2))"
      using narrows by simp
    show ?thesis using safe[OF prog_obligation_site_special[OF edge ob] ob] by simp
  qed
qed

theorem run_voblint_arithmetic_refuted:
  assumes run: "run_voblint conf ds p = Analysed res"
    and d: "d ∈ set (report_diagnostics res)" and r: "diagnostic_verdict d = Check_Refuted"
    and s: "s ∈ reaching ds p (diagnostic_point d)"
  shows "¬ s ⊨ diagnostic_obligation d"
proof -
  from run have wf: "wf_program_compile_input p"
    and rep: "report_run conf ds p = Some res"
    by (auto elim!: run_voblint_AnalysedE intro: wf_program_compile_input_exec_sound)
  show ?thesis by (rule report_obligation_refuted[OF wf rep s d r refl])
qed

text ‹
  For signed overflow in particular: at a point no diagnostic names, every signed
  addition, subtraction, multiplication and division nested in an expression evaluated
  there computes, from the operand values of any reaching store, a mathematical result
  its kind represents. So no execution that reaches the point overflows there, whichever
  value wraparound would have produced.
›

corollary run_voblint_no_signed_overflow:
  assumes run: "run_voblint conf ds p = Analysed res"
    and absent: "SAFE res v"
    and s: "s ∈ reaching ds p v"
    and evaluated: "((v, a, w) ∈ intra (elab_cfg p ds) ∧ e ∈ set (edge_expressions a))
                    ∨ ((v, ca, ce, k) ∈ calls (elab_cfg p ds) ∧ e ∈ set (ce_args ca))"
    and sub: "TArith ik op x y ∈ subexps e" and signed: "ik_signed ik" and op: "op ≠ AMod"
  shows "aop_eval op (eval x s) (eval y s) ∈ ik_range ik"
  using evaluated arith_obligations_overflow[OF sub signed op]
    run_voblint_arithmetic_safe[OF run absent s]
  by (metis obligation_holds_overflow)

text ‹
  For signed remainders: at a point no diagnostic names, every signed remainder nested
  in an expression evaluated there divides, at any reaching store, with a truncated
  quotient its kind represents, so none is ‹INT_MIN % -1›.
›

corollary run_voblint_no_remainder_overflow:
  assumes run: "run_voblint conf ds p = Analysed res"
    and absent: "SAFE res v"
    and s: "s ∈ reaching ds p v"
    and evaluated: "((v, a, w) ∈ intra (elab_cfg p ds) ∧ e ∈ set (edge_expressions a))
                    ∨ ((v, ca, ce, k) ∈ calls (elab_cfg p ds) ∧ e ∈ set (ce_args ca))"
    and sub: "TArith ik AMod x y ∈ subexps e" and signed: "ik_signed ik"
  shows "c_div (eval x s) (eval y s) ∈ ik_range ik"
  using evaluated arith_obligations_remainder[OF sub signed]
    run_voblint_arithmetic_safe[OF run absent s]
  by (metis obligation_holds_remainder)

text ‹
  For conversions: at a point no diagnostic names, every conversion nested in an
  expression evaluated there whose source kind does not widen into its target converts
  a value the target represents, so no conversion changes the value it converts.
›

corollary run_voblint_no_conversion_overflow:
  assumes run: "run_voblint conf ds p = Analysed res"
    and absent: "SAFE res v"
    and s: "s ∈ reaching ds p v"
    and evaluated: "((v, a, w) ∈ intra (elab_cfg p ds) ∧ e ∈ set (edge_expressions a))
                    ∨ ((v, ca, ce, k) ∈ calls (elab_cfg p ds) ∧ e ∈ set (ce_args ca))"
    and sub: "TCast ik x ∈ subexps e" and narrows: "¬ ik_widens (exp_kind x) ik"
  shows "eval x s ∈ ik_range ik"
  using evaluated arith_obligations_conversion[OF sub narrows]
    run_voblint_arithmetic_safe[OF run absent s]
  by (metis obligation_holds_conversion(1))

subsection ‹The report's vocabulary›

text ‹
  The guarantees above, read through the predicates every report shares:
  \<^const>‹PROVED›, \<^const>‹REFUTED›, \<^const>‹DEAD› and \<^const>‹SAFE›, and
  ‹s ⊨ ob› for a store meeting an obligation. ‹reaching ds p v› are the stores that
  reach ‹v›. A proved check holds at each of them and a refuted one fails; a dead
  point has none; at a safe point every obligation the report decides there holds.
  None of them says that ‹v› is reached.

  ‹SAFE› is only as strong as the obligations it covers: those of every expression
  an edge or a call at ‹v› evaluates and of every value returned there that a
  caller converts to a narrower kind, and of every library ‹min› or ‹max› result
  written there to a kind that does not hold both arguments' kinds
  (‹run_voblint_obligations_cover›). It says nothing about a hazard Voblint does not
  check.
›

theorem run_voblint_report_sound:
  assumes run: "run_voblint conf ds p = Analysed res" and s: "s ∈ reaching ds p v"
  shows run_voblint_PROVED: "PROVED res v c ⟹ truthy (eval c s)"
    and run_voblint_REFUTED: "REFUTED res v c ⟹ ¬ truthy (eval c s)"
    and run_voblint_DEAD: "DEAD res v ⟹ False"
    and run_voblint_SAFE: "SAFE res v ⟹ ob ∈ obligations_at ds p v ⟹ s ⊨ ob"
proof -
  show "PROVED res v c ⟹ truthy (eval c s)"
    using run_voblint_proved[OF run] s by (auto simp: HAS_VERDICT_def)
  show "REFUTED res v c ⟹ ¬ truthy (eval c s)"
    using run_voblint_refuted[OF run] s by (auto simp: HAS_VERDICT_def)
  show "DEAD res v ⟹ False"
    using run_voblint_states[OF run s] by (auto simp: DEAD_def)
  show "SAFE res v ⟹ ob ∈ obligations_at ds p v ⟹ s ⊨ ob"
  proof -
    assume safe: "SAFE res v" and ob: "ob ∈ obligations_at ds p v"
    from run have wf: "wf_program_compile_input p"
      and rep: "report_run conf ds p = Some res"
      by (auto elim!: run_voblint_AnalysedE intro: wf_program_compile_input_exec_sound)
    from ob obtain obs where "(v, obs) ∈ set (prog_obligation_sites ds p)" and "ob ∈ set obs"
      by (auto simp: mem_obligations_at)
    from report_obligation_safe[OF wf rep s this] safe show "s ⊨ ob" .
  qed
qed

corollary run_voblint_dead_unreached:
  "run_voblint conf ds p = Analysed res ⟹ DEAD res v ⟹ reaching ds p v = {}"
  using run_voblint_DEAD by blast

subsection ‹What an answer guarantees about a run of the program›

text ‹
  The same guarantees, read at a run of the typed program itself rather than at a
  node of its graph. Run the program from a store whose globals are zero at their
  kinds and stop anywhere: the configuration reached is simulated at a node ‹v› of
  the elaborated graph, the store reaches ‹v›, the report holds a state at ‹v› describing
  it, ‹v› is not dead, every check the report decides at ‹v› has its verdict at the
  store, and if ‹v› is safe every obligation decided there holds of the store.
›

theorem run_voblint_source_sound:
  assumes run: "run_voblint conf ds p = Analysed res"
    and s0: "s0 ∈ init_stores (kenv_of ds)"
    and exec:
      "(kenv_of ds),(elab_ptable p ds) ⊢ (main_body (elab_ptable p ds), s0, []) →⇩p⇧* (residual, s, frs)"
  shows "∃v stk. elab_ptable p ds, elab_cfg p ds, prog_main_name
                    ⊢ (residual, s, frs) ≈ (v, s, stk)
           ∧ s ∈ reaching ds p v
           ∧ (∃d. Lifted d ∈ report_states_at res v
                  ∧ s ∈ mcp.act_gamma ds (activation_of (conf_analyses conf)) d)
           ∧ ¬ DEAD res v
           ∧ (∀c. PROVED res v c ⟶ truthy (eval c s))
           ∧ (∀c. REFUTED res v c ⟶ ¬ truthy (eval c s))
           ∧ (SAFE res v ⟶ (∀ob ∈ obligations_at ds p v. s ⊨ ob))"
proof -
  from run have wf: "wf_program_compile_input p"
    by (auto elim!: run_voblint_AnalysedE intro: wf_program_compile_input_exec_sound)
  from source_reaches_node_collect[OF wf s0 exec] obtain v stk
    where sim: "elab_ptable p ds, elab_cfg p ds, prog_main_name ⊢ (residual, s, frs) ≈ (v, s, stk)"
      and s: "s ∈ reaching ds p v"
    by blast
  show ?thesis
    using sim s run_voblint_states[OF run s] run_voblint_report_sound[OF run s] by blast
qed

text ‹
  Read at a check: a run about to execute ‹Check l e› finds a check labelled ‹l› for
  ‹e› in the report, listed at a node this very store reaches. Its verdict is not
  ‹Dead›, and a decided verdict holds of the store.
›

theorem run_voblint_check_sound:
  assumes run: "run_voblint conf ds p = Analysed res"
    and s0: "s0 ∈ init_stores (kenv_of ds)"
    and exec:
      "(kenv_of ds),(elab_ptable p ds) ⊢ (main_body (elab_ptable p ds), s0, []) →⇩p⇧* (residual, s, frs)"
    and chk: "next_check residual = Some (l, e)"
  shows "∃c ∈ set (report_checks res). check_label c = l ∧ check_exp c = e
           ∧ s ∈ reaching ds p (check_point c)
           ∧ check_verdict c ≠ Dead
           ∧ (check_verdict c = Decided Check_Proved ⟶ truthy (eval e s))
           ∧ (check_verdict c = Decided Check_Refuted ⟶ ¬ truthy (eval e s))"
proof -
  from run have wf: "wf_program_compile_input p"
    and rep: "report_run conf ds p = Some res"
    by (auto elim!: run_voblint_AnalysedE intro: wf_program_compile_input_exec_sound)
  from source_reaches_node_collect[OF wf s0 exec] obtain v stk
    where sim: "elab_ptable p ds, elab_cfg p ds, prog_main_name ⊢ (residual, s, frs) ≈ (v, s, stk)"
      and s: "s ∈ reaching ds p v"
    by blast
  from csim_next_check_edge[OF sim chk]
  obtain w where "(v, EA_Check l e, w) ∈ intra (elab_cfg p ds)"
    by blast
  from report_check_row[OF wf rep this] obtain c where c: "c ∈ set (report_checks res)"
    and pt: "check_point c = v" and lbl: "check_label c = l" and ex: "check_exp c = e"
    by blast
  have "check_verdict c ≠ Dead" by (rule report_check_live[OF wf rep s c pt])
  with c pt lbl ex s run_voblint_proved[OF run c] run_voblint_refuted[OF run c] show ?thesis
    by force
qed

text ‹
  Read by label, as a consumer finds a source check's row: when the report's labels
  are distinct, which a consumer can decide on the report, the row labelled ‹l› is the
  one for the check about to run, and its verdict holds of the store. That the parser
  wrote each check's own source position into its label stays outside the theorem.
›

corollary run_voblint_labelled_check_sound:
  assumes run: "run_voblint conf ds p = Analysed res"
    and s0: "s0 ∈ init_stores (kenv_of ds)"
    and exec:
      "(kenv_of ds),(elab_ptable p ds) ⊢ (main_body (elab_ptable p ds), s0, []) →⇩p⇧* (residual, s, frs)"
    and chk: "next_check residual = Some (l, e)"
    and distinct: "distinct (map check_label (report_checks res))"
  shows "∃c ∈ set (report_checks res). check_label c = l"
    and "∀c ∈ set (report_checks res). check_label c = l ⟶
           check_exp c = e
           ∧ check_verdict c ≠ Dead
           ∧ (check_verdict c = Decided Check_Proved ⟶ truthy (eval e s))
           ∧ (check_verdict c = Decided Check_Refuted ⟶ ¬ truthy (eval e s))"
proof -
  from run_voblint_check_sound[OF run s0 exec chk]
  obtain c0 where c0: "c0 ∈ set (report_checks res)" "check_label c0 = l" "check_exp c0 = e"
    and v0: "check_verdict c0 ≠ Dead"
      "check_verdict c0 = Decided Check_Proved ⟶ truthy (eval e s)"
      "check_verdict c0 = Decided Check_Refuted ⟶ ¬ truthy (eval e s)"
    by blast
  from c0 show "∃c ∈ set (report_checks res). check_label c = l" by blast
  have inj: "inj_on check_label (set (report_checks res))"
    using distinct by (simp add: distinct_map)
  show "∀c ∈ set (report_checks res). check_label c = l ⟶
           check_exp c = e
           ∧ check_verdict c ≠ Dead
           ∧ (check_verdict c = Decided Check_Proved ⟶ truthy (eval e s))
           ∧ (check_verdict c = Decided Check_Refuted ⟶ ¬ truthy (eval e s))"
  proof (intro ballI impI)
    fix c assume "c ∈ set (report_checks res)" and "check_label c = l"
    with c0 inj have "c = c0" by (auto dest: inj_onD)
    with c0 v0 show "check_exp c = e ∧ check_verdict c ≠ Dead
           ∧ (check_verdict c = Decided Check_Proved ⟶ truthy (eval e s))
           ∧ (check_verdict c = Decided Check_Refuted ⟶ ¬ truthy (eval e s))" by simp
  qed
qed

text ‹
  The whole chain for any context policy, given what the policy owes: every valid
  activation trace of the elaborated graph carries some context, and the stores filed
  under the policy's contexts at a node are exactly the stores that reach it. A run
  of the typed program is then simulated at a node, by a valid activation trace with
  a context, whose store lies in that context's bucket; the buckets make up the
  stores reaching the node, the report holds a state there describing the store, and
  the report's checks hold of it.
›

theorem run_voblint_spine:
  fixes p :: imp_prog and ds :: kind_decls and adm c⇩0
  defines "g ≡ elab_cfg p ds" and "S ≡ init_stores (kenv_of ds)"
  assumes run: "run_voblint conf ds p = Analysed res"
    and s0: "s0 ∈ S"
    and exec:
      "(kenv_of ds),(elab_ptable p ds) ⊢ (main_body (elab_ptable p ds), s0, []) →⇩p⇧* (residual, s, frs)"
    and has_ctx: "⋀t. t ∈ 𝒯⇘kenv_of ds,g,S⇙
                   ⟹ ∃c. activation_context_rel (kenv_of ds) adm c⇩0 g t c"
    and buckets: "⋀v. (⋃c'. 𝒜⇘kenv_of ds,adm,c⇩0,g,S⇙ v c')
                   = reaching ds p v"
  shows "∃v stk t c. elab_ptable p ds, g, prog_main_name ⊢ (residual, s, frs) ≈ (v, s, stk)
           ∧ activation_trace_repr (kenv_of ds) g S (v, s, stk) t
           ∧ activation_context_rel (kenv_of ds) adm c⇩0 g t c
           ∧ s ∈ 𝒜⇘kenv_of ds,adm,c⇩0,g,S⇙ v c
           ∧ (⋃c'. 𝒜⇘kenv_of ds,adm,c⇩0,g,S⇙ v c') = reaching ds p v
           ∧ (∃d. Lifted d ∈ report_states_at res v
                  ∧ s ∈ mcp.act_gamma ds (activation_of (conf_analyses conf)) d)
           ∧ (∀c. PROVED res v c ⟶ truthy (eval c s))
           ∧ (∀c. REFUTED res v c ⟶ ¬ truthy (eval c s))"
proof -
  from run have wf: "wf_program_compile_input p"
    by (auto elim!: run_voblint_AnalysedE intro: wf_program_compile_input_exec_sound)
  have cfg: "g = compile_prog (elab_ptable p ds) (prog_procs p)"
    unfolding g_def by (rule elab_cfg_def)
  from source_store_in_activation_collect
         [OF elab_wf_compile_input[OF wf] s0 exec has_ctx[unfolded cfg]]
  obtain v stk t c
    where sim: "elab_ptable p ds, g, prog_main_name ⊢ (residual, s, frs) ≈ (v, s, stk)"
      and rep: "activation_trace_repr (kenv_of ds) g S (v, s, stk) t"
      and rel: "activation_context_rel (kenv_of ds) adm c⇩0 g t c"
      and mem: "s ∈ 𝒜⇘kenv_of ds,adm,c⇩0,g,S⇙ v c"
    unfolding cfg by blast
  have s: "s ∈ reaching ds p v" using mem buckets[of v] by blast
  show ?thesis
    using sim rep rel mem buckets[of v] run_voblint_states[OF run s]
      run_voblint_report_sound[OF run s]
    by blast
qed

text ‹
  Each context policy owes the spine's two facts. A call string is a function of the
  trace, so every trace carries one and the buckets cover the stores outright. An
  entry-state context is read off the solve, and the answer was given only because
  the solve terminated, which is what makes the admitted contexts cover every trace.
›

theorem run_voblint_call_string_chain:
  fixes p :: imp_prog and ds :: kind_decls and k :: nat
  defines "g ≡ elab_cfg p ds" and "S ≡ init_stores (kenv_of ds)"
    and "adm ≡ context_policy_of_fun (cs_context k)"
  assumes run: "run_voblint conf ds p = Analysed res"
    and cx: "conf_context conf = Call_String k"
    and s0: "s0 ∈ S"
    and exec:
      "(kenv_of ds),(elab_ptable p ds) ⊢ (main_body (elab_ptable p ds), s0, []) →⇩p⇧* (residual, s, frs)"
  shows "∃v stk t c. elab_ptable p ds, g, prog_main_name ⊢ (residual, s, frs) ≈ (v, s, stk)
           ∧ activation_trace_repr (kenv_of ds) g S (v, s, stk) t
           ∧ activation_context_rel (kenv_of ds) adm [] g t c
           ∧ s ∈ 𝒜⇘kenv_of ds,adm,[],g,S⇙ v c
           ∧ (⋃c'. 𝒜⇘kenv_of ds,adm,[],g,S⇙ v c') = reaching ds p v
           ∧ (∃d. Lifted d ∈ report_states_at res v
                  ∧ s ∈ mcp.act_gamma ds (activation_of (conf_analyses conf)) d)
           ∧ (∀c. PROVED res v c ⟶ truthy (eval c s))
           ∧ (∀c. REFUTED res v c ⟶ ¬ truthy (eval c s))"
proof -
  have has_ctx: "∃c. activation_context_rel (kenv_of ds) adm [] g t c"
    if "t ∈ 𝒯⇘kenv_of ds,g,S⇙" for t
    unfolding adm_def
    by (rule exI, subst activation_context_rel_of_fun_iff[OF that]) (rule refl)
  have buckets: "(⋃c'. 𝒜⇘kenv_of ds,adm,[],g,S⇙ v c') = reaching ds p v"
    for v
    unfolding adm_def g_def S_def
    by (rule mcp_cs.fun_route_node_collect_eq_Union[where ctx_fun = "cs_context k", symmetric])
  show ?thesis
    unfolding g_def S_def
    by (rule run_voblint_spine[OF run s0[unfolded S_def] exec has_ctx[unfolded g_def S_def]
          buckets[unfolded g_def S_def]])
qed

text ‹An entry-state run admits the contexts its placement's registration reads off the
  solve.›

theorem run_voblint_entry_state_chain:
  fixes p :: imp_prog and ds :: kind_decls and conf :: voblint_conf
  defines "g ≡ elab_cfg p ds" and "S ≡ init_stores (kenv_of ds)"
    and "adm ≡ (case conf_program_globals conf of
                 Program_Globals_Flow_Sensitive ⇒
                   mcp_es.entry_context_rel (conf_sem conf) ds (conf_analyses conf)
                     (conf_globals conf) p
               | Program_Globals_Flow_Insensitive ⇒
                   mcp_split_es.entry_context_rel (conf_sem conf) ds (conf_analyses conf)
                     (conf_globals conf) (declared_globals p) p)"
  assumes run: "run_voblint conf ds p = Analysed res"
    and cx: "conf_context conf = Entry_State"
    and s0: "s0 ∈ S"
    and exec:
      "(kenv_of ds),(elab_ptable p ds) ⊢ (main_body (elab_ptable p ds), s0, []) →⇩p⇧* (residual, s, frs)"
  shows "∃v stk t c. elab_ptable p ds, g, prog_main_name ⊢ (residual, s, frs) ≈ (v, s, stk)
           ∧ activation_trace_repr (kenv_of ds) g S (v, s, stk) t
           ∧ activation_context_rel (kenv_of ds) adm mcp_root_ctx g t c
           ∧ s ∈ 𝒜⇘kenv_of ds,adm,mcp_root_ctx,g,S⇙ v c
           ∧ (⋃c'. 𝒜⇘kenv_of ds,adm,mcp_root_ctx,g,S⇙ v c')
               = reaching ds p v
           ∧ (∃d. Lifted d ∈ report_states_at res v
                  ∧ s ∈ mcp.act_gamma ds (activation_of (conf_analyses conf)) d)
           ∧ (∀c. PROVED res v c ⟶ truthy (eval c s))
           ∧ (∀c. REFUTED res v c ⟶ ¬ truthy (eval c s))"
proof -
  from run have wf: "wf_program_compile_input p"
    and rep: "report_run conf ds p = Some res"
    by (auto elim!: run_voblint_AnalysedE intro: wf_program_compile_input_exec_sound)
  have chain: "(∀t. t ∈ 𝒯⇘kenv_of ds,g,S⇙
                  ⟶ (∃c. activation_context_rel (kenv_of ds) adm mcp_root_ctx g t c))
    ∧ (∀v. (⋃c'. 𝒜⇘kenv_of ds,adm,mcp_root_ctx,g,S⇙ v c')
             = reaching ds p v)"
  proof (cases "conf_program_globals conf")
    case Program_Globals_Flow_Sensitive
    from rep cx Program_Globals_Flow_Sensitive obtain sol where
      "TD_side_rule_Interp_solve_c (conf_globals conf)
         (mcp_es.equations (conf_sem conf) ds (conf_analyses conf) is_global p)
         (mcp_es.root_query ds p) = Some sol"
      by (auto simp: report_run_def es_run_def Let_def dg_pipeline.root_query_def)
    note solves = mcp_es.solve_c_run(1)[OF this]
    show ?thesis
      using mcp_es.entry_state_has_context_of_terminates
          [OF elab_cfg_def elab_wf_compile_input[OF wf] solves]
        mcp_es.entry_state_node_collect_eq_Union_of_terminates
          [OF elab_cfg_def elab_wf_compile_input[OF wf] solves]
      unfolding adm_def g_def S_def Program_Globals_Flow_Sensitive by simp
  next
    case Program_Globals_Flow_Insensitive
    from rep cx Program_Globals_Flow_Insensitive obtain sol where
      "TD_side_rule_Interp_solve_c (conf_globals conf)
         (mcp_split_es.equations (conf_sem conf) ds (conf_analyses conf) (declared_globals p)
            is_global p)
         (mcp_split_es.root_query ds p) = Some sol"
      by (auto simp: report_run_def es_run_def Let_def dg_pipeline.root_query_def)
    note solves = mcp_split_es.solve_c_run(1)[OF this]
    show ?thesis
      using mcp_split_es.entry_state_has_context_of_terminates
          [OF elab_cfg_def elab_wf_compile_input[OF wf] solves]
        mcp_split_es.entry_state_node_collect_eq_Union_of_terminates
          [OF elab_cfg_def elab_wf_compile_input[OF wf] solves]
      unfolding adm_def g_def S_def Program_Globals_Flow_Insensitive by simp
  qed
  show ?thesis
    unfolding g_def S_def
    by (rule run_voblint_spine[OF run s0[unfolded S_def] exec
          conjunct1[OF chain, unfolded g_def S_def, rule_format]
          conjunct2[OF chain, unfolded g_def S_def, rule_format]])
qed

text ‹Every obligation of an expression evaluated at ‹v›, every narrowing
  conversion of a value returned at ‹v›, and every narrowing conversion of a library
  result written at ‹v› is one the report decides there.›

lemma run_voblint_obligations_cover:
  shows "(v, a, w) ∈ intra (elab_cfg p ds) ⟹ e ∈ set (edge_expressions a)
           ⟹ set (arith_obligations e) ⊆ obligations_at ds p v"
    and "(v, ca, ce, k) ∈ calls (elab_cfg p ds) ⟹ e ∈ set (ce_args ca)
           ⟹ set (arith_obligations e) ⊆ obligations_at ds p v"
    and "(u, ca, ce, k) ∈ calls (elab_cfg p ds) ⟹ ce_dst ca = Some x
           ⟹ (v, EA_Ret (Some e) (callee_of_entry ce), w) ∈ intra (elab_cfg p ds)
           ⟹ ¬ ik_widens (exp_kind e) (kenv_of ds x)
           ⟹ Arithmetic_Obligation Conversion_Overflow (TCast (kenv_of ds x) e) e
                 ∈ obligations_at ds p v"
    and "(v, EA_Special sc x, w) ∈ intra (elab_cfg p ds)
           ⟹ set (special_obligations (kenv_of ds x) sc) ⊆ obligations_at ds p v"
     apply (auto simp: mem_obligations_at dest: prog_obligation_site_intra prog_obligation_site_call
       intro: prog_obligation_site_return)[3]
  using prog_obligation_site_special by (fastforce simp: mem_obligations_at)
