Theory MCP

theory MCP
  imports MCP_Carrier
begin

section ‹The active typed analyses, run through the pipeline›

text ‹
  A caller activates a list of typed analyses, as Goblint's ▩‹ana.activated›. The
  registry runs the distinct ones as one component; the pipeline solves it on the typed
  CFG at call-string contexts of bound ‹k›, exactly as it solves one analysis.
›

definition activation_of :: "analysis list ⇒ analysis list" where
  "activation_of as = (if as = [] then [Interval] else remdups as)"

lemma activation_of_ne [simp]: "activation_of as ≠ []"
  by (simp add: activation_of_def)

lemma distinct_activation_of [simp]: "distinct (activation_of as)"
  by (simp add: activation_of_def)

text ‹Every active field starts at its analysis's own entry state, every other at bottom.›

definition mcp_init :: "kind_decls ⇒ analysis list ⇒ mcp_st" where
  "mcp_init ds as =
     Product (if Interval ∈ set as then Lifted (nonrel_init_st ds) else Bot)
     (Product (if Congruence ∈ set as then Lifted (nonrel_init_st ds) else Bot)
     (Product (Product (if Ana_Int Refine_Never ∈ set as then Lifted (nonrel_init_st ds) else Bot)
        (Product (if Ana_Int Refine_Once ∈ set as then Lifted (nonrel_init_st ds) else Bot)
                 (if Ana_Int Refine_Fixpoint ∈ set as then Lifted (nonrel_init_st ds) else Bot)))
     (Product (if Sign ∈ set as then Lifted (nonrel_init_st ds) else Bot)
     (Product (if Parity ∈ set as then Lifted (nonrel_init_st ds) else Bot)
              (if Order ∈ set as then Lifted (OFacts True {}) else Bot)))))"

lemma mcp_init_sound:
  "init_stores (kenv_of ds) ⊆ mcp.act_gamma ds as (mcp_rd is_global (mcp_init ds as))"
proof
  fix t assume t: "t ∈ init_stores (kenv_of ds)"
  have "t ∈ val_gamma (kenv_of ds) a (mcp_rd is_global (mcp_init ds as))" if m: "a ∈ set as" for a
  proof (cases a rule: analysis_cases)
    case Order with m show ?thesis
      by (simp add: mcp_init_def mcp_rd_def init_stores_in_kinds[OF t])
  qed (use m in ‹simp_all add: mcp_init_def mcp_rd_def subsetD[OF nonrel_init_sound t]›)
  then show "t ∈ mcp.act_gamma ds as (mcp_rd is_global (mcp_init ds as))"
    by (simp add: mcp.act_gamma_def)
qed

text ‹What the combination owes the pipeline does not depend on the context policy, so
  it is proved once, for any route and root context.›

lemma mcp_cs_routed:
  "dg_analysis
    (kenv_of ds) (λ_. init_stores (kenv_of ds))
    (mcp_comp sem ds (activation_of as)) (mcp_emp (activation_of as)) mcp_rd
    (mcp_init ds (activation_of as))
    Analysis_Buffer Analysis_Global Activation_Seed
    (TD_side_rule_Interp_solve r)
    (TD_side_rule_Interp.solve_dom TYPE((var_id, 'c) global_unknown)
       TYPE((mcp_st lifted, mcp_st lifted) dg_state) r)
    ⊥ (mcp_classify sem ds (activation_of as)) (mcp.act_gamma ds (activation_of as))
    (mcp_empty_v (activation_of as))
    (TD_side_rule_Interp_solve_c r)
    (λ_ c. dg_spec_of c) (λp d e. d) (λ_ d. d) (λ_ d. Bot) (λ_ p. []) (λp ci d e. d)
    (λp. elab_cfg p ds) (λ_. FunctionEntry prog_main_name)"
proof (rule dg_analysis_whole_stateI[OF td_certified_solver], goal_cases CompSound EnterSingle
    EmptyRd EmptyV SeedNe ClProved ClRefuted BotState InitSound SeedNeGlobal PEntry FinIntra
    FinCalls CallsUnique)
  case (CompSound p)
  have eq: "mcp_gamma (map (mcp.part_gamma ds is_global) (activation_of as))
      = (λd. gamma_lift (mcp.act_gamma ds (activation_of as)) (map_lift (mcp_rd is_global) d))"
    by (rule ext) (rule mcp.mcp_gamma_rd[OF activation_of_ne])
  show ?case
    using mcp.mcp_comp_sound[OF distinct_activation_of activation_of_ne, where sem = sem and ds = ds
      and p = p,
        of as]
    unfolding eq by simp
next
  case (EnterSingle p ci d)
  show ?case
    using single_entryD[OF mcp.single_entry_mcp_comp[OF activation_of_ne],
        of sem ds as is_global p
          "ls_channel (mcp_comp sem ds (activation_of as) is_global p) d" ci "(d, d)"]
    by (simp add: dg_pipeline.comp_entry_def)
next
  case (EmptyRd p s) show ?case by (rule mcp.mcp_emp_rd)
next
  case EmptyV show ?case by (rule mcp.sound_emptiness_mcp)
next
  case (SeedNe v ctx) show ?case by simp
next
  case (ClProved c d s) then show ?case by (intro mcp.mcp_classify_proved)
next
  case (ClRefuted c d s) then show ?case by (intro mcp.mcp_classify_refuted)
next
  case BotState show ?case by (rule mcp.act_gamma_bot[OF activation_of_ne])
next
  case (InitSound p) show ?case by (rule mcp_init_sound)
next
  case (SeedNeGlobal v ctx n) show ?case by simp
next
  case (PEntry p) show ?case by (simp add: elab_cfg_def)
next
  case (FinIntra p) show ?case by (simp add: elab_cfg_def compile_prog_finite)
next
  case (FinCalls p) show ?case by (simp add: elab_cfg_def compile_prog_finite)
next
  case (CallsUnique p) show ?case by (rule elab_cfg_calls_source_unique)
qed

subsection ‹At call-string contexts›

text ‹The combination run with ‹k›-bounded call strings as contexts, rooted at the
  empty call string.›

global_interpretation mcp_cs: dg_analysis
    "kenv_of ds" "λ_. init_stores (kenv_of ds)"
    "mcp_comp sem ds (activation_of as)" "mcp_emp (activation_of as)" mcp_rd
    "mcp_init ds (activation_of as)"
    Analysis_Buffer Analysis_Global Activation_Seed "λ_. cs_route k" "[]"
    "TD_side_rule_Interp_solve r"
    "TD_side_rule_Interp.solve_dom TYPE((var_id, cfg_node list) global_unknown)
       TYPE((mcp_st lifted, mcp_st lifted) dg_state) r"
    ⊥ "mcp_classify sem ds (activation_of as)" "mcp.act_gamma ds (activation_of as)"
    "mcp_empty_v (activation_of as)"
    "TD_side_rule_Interp_solve_c r"
    "λ_ c. dg_spec_of c" "λp d e. d" "λ_ d. d" "λ_ d. Bot" "λ_ p. []" "λp ci d e. d"
    "λp. elab_cfg p ds" "λ_. FunctionEntry prog_main_name"
  for sem ds as r k
  by (rule mcp_cs_routed)

subsection ‹At entry-state contexts›

text ‹
  A call is keyed by what each active field gives the callee's formals on entry, as the
  entry-state analyses key a single domain (\<^const>‹nonrel_es_ctx›); a field no active analysis runs keys
  nothing, and the order field keys nothing.
›

type_synonym mcp_ctx =
  "(ivl kd list, (congruence kd list,
    ((int_never int_tuple kd list, (int_once int_tuple kd list, int_fixpoint int_tuple kd list)
      analysis_product) analysis_product,
    (sign kd list, (parity kd list, unit list) analysis_product) analysis_product)
    analysis_product) analysis_product) analysis_product"

definition mcp_route ::
    "analysis list ⇒ (var_id ⇒ bool) ⇒ cfg_node ⇒ mcp_ctx ⇒ mcp_st lifted
       ⇒ (var_id, exp) gcall_action ⇒ mcp_ctx" where
  "mcp_route as 𝒢 u ctx d ca =
     Product (if Interval ∈ set as then nonrel_es_ctx 𝒢 (lift_get slot1 d) ca else [])
     (Product (if Congruence ∈ set as then nonrel_es_ctx 𝒢 (lift_get slot2 d) ca else [])
     (Product (Product (if Ana_Int Refine_Never ∈ set as then nonrel_es_ctx 𝒢 (lift_get slot3n d) ca else [])
        (Product (if Ana_Int Refine_Once ∈ set as then nonrel_es_ctx 𝒢 (lift_get slot3o d) ca else [])
                 (if Ana_Int Refine_Fixpoint ∈ set as then nonrel_es_ctx 𝒢 (lift_get slot3f d) ca else [])))
     (Product (if Sign ∈ set as then nonrel_es_ctx 𝒢 (lift_get slot4 d) ca else [])
     (Product (if Parity ∈ set as then nonrel_es_ctx 𝒢 (lift_get slot5 d) ca else []) []))))"

definition mcp_root_ctx :: mcp_ctx where
  "mcp_root_ctx = Product [] (Product [] (Product (Product [] (Product [] []))
     (Product [] (Product [] []))))"

global_interpretation mcp_es: dg_analysis
    "kenv_of ds" "λ_. init_stores (kenv_of ds)"
    "mcp_comp sem ds (activation_of as)" "mcp_emp (activation_of as)" mcp_rd
    "mcp_init ds (activation_of as)"
    Analysis_Buffer Analysis_Global Activation_Seed "mcp_route (activation_of as)" mcp_root_ctx
    "TD_side_rule_Interp_solve r"
    "TD_side_rule_Interp.solve_dom TYPE((var_id, mcp_ctx) global_unknown)
       TYPE((mcp_st lifted, mcp_st lifted) dg_state) r"
    ⊥ "mcp_classify sem ds (activation_of as)" "mcp.act_gamma ds (activation_of as)"
    "mcp_empty_v (activation_of as)"
    "TD_side_rule_Interp_solve_c r"
    "λ_ c. dg_spec_of c" "λp d e. d" "λ_ d. d" "λ_ d. Bot" "λ_ p. []" "λp ci d e. d"
    "λp. elab_cfg p ds" "λ_. FunctionEntry prog_main_name"
  for sem ds as r
  by (rule mcp_cs_routed)

subsection ‹End-to-end soundness›

text ‹Under either context policy, every typed store reaching a node lies in the
  combined state the solved table holds there at some context.›

context
  fixes sem :: sem_conf and ds :: kind_decls and as :: "analysis list" and r :: globals_rule and k
    :: nat
begin

lemma mcp_cs_node_sound:
  assumes wf: "wf_program_compile_input p"
    and solves: "mcp_cs.terminates sem ds as k r is_global p"
  shows "(reaching ds p v)
           ⊆ (⋃ctx. mcp.act_gamma ds (activation_of as) (mcp_cs.state_at sem ds as k r is_global p ctx v))"
  using mcp_cs.fun_route_result_node_sound
      [OF elab_cfg_def _ elab_wf_compile_input[OF wf] solves, of "cs_context k" v]
  by (simp add: cs_route_def cs_context_def)

lemma mcp_es_node_sound:
  assumes wf: "wf_program_compile_input p"
    and solves: "mcp_es.terminates sem ds as r is_global p"
  shows "(reaching ds p v)
           ⊆ (⋃ctx. mcp.act_gamma ds (activation_of as) (mcp_es.state_at sem ds as r is_global p ctx v))"
  using mcp_es.entry_state_result_node_sound
      [OF elab_cfg_def elab_wf_compile_input[OF wf] solves, of v]
  by simp

end
