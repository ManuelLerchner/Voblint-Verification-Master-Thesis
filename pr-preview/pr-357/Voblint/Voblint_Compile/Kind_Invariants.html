Theory Kind_Invariants

theory Kind_Invariants
  imports Source_To_Trace
begin

section ‹Typed stores stay typed›

text ‹
  Every identity other than the return slot holds a value of its kind in every store a
  valid activation trace of the typed graph reaches. Writes to a variable convert to its
  kind (\<^const>‹kind_conv›), and a call entry resets locals to ‹0› and binds each formal to
  its actual converted to the formal's kind. The return slot
  is shared by procedures returning different kinds, so it is excluded: it holds a value
  of the returning procedure's kind, which the destination conversion re-establishes at
  the caller.
›

definition in_kinds :: "kenv ⇒ store ⇒ bool" where
  "in_kinds κ t ⟷ (∀v. v ≠ ReturnVar ⟶ t v ∈ ik_range (κ v))"

lemma in_kinds_update [simp]:
  "in_kinds κ t ⟹ in_kinds κ (t(v := ik_norm (κ v) n))"
  "in_kinds κ t ⟹ in_kinds κ (t(ReturnVar := n))"
  by (auto simp: in_kinds_def simp del: ik_range_iff)

lemma in_kinds_bind_formals:
  "in_kinds κ t ⟹ (⋀x n. (x, n) ∈ set (zip xs ns) ⟹ n ∈ ik_range (κ x))
   ⟹ in_kinds κ (bind_formals xs ns t)"
proof (induction xs arbitrary: ns t)
  case (Cons x xs)
  then show ?case
    by (cases ns) (auto simp: in_kinds_def simp del: ik_range_iff)
qed simp

lemma in_kinds_enter_state: "in_kinds κ t ⟹ in_kinds κ (enter_state is_global t)"
  by (auto simp: in_kinds_def enter_state_def enter_frame_def simp del: ik_range_iff)

lemma in_kinds_combine_env:
  "in_kinds κ s ⟹ in_kinds κ t ⟹ in_kinds κ (combine_env g s t)"
  by (auto simp: in_kinds_def simp del: ik_range_iff)

subsection ‹Graphs with typed calls›

text ‹A call edge passes typed actuals when each actual is well-typed, does not read the
  return slot, and has the kind of the formal it binds.›

definition calls_in_kinds :: "kenv ⇒ (var_id, exp) gcfg ⇒ bool" where
  "calls_in_kinds κ g ⟷
     (∀u dst pars es v w. (u, CallEdge dst pars es, v, w) ∈ calls g ⟶
        list_all (λe. wt_exp κ e ∧ ReturnVar ∉ exp_vars e) es ∧ args_ok κ pars es)"

lemma zip_map2_self: "zip xs (map2 f xs ys) = map (λ(x, y). (x, f x y)) (zip xs ys)"
proof (induction xs arbitrary: ys)
  case (Cons x xs) then show ?case by (cases ys) simp_all
qed simp

lemma in_kinds_call_enter:
  assumes t: "in_kinds κ t"
  shows "in_kinds κ (call_enter κ (CallEdge dst pars es) t)"
proof -
  have "n ∈ ik_range (κ x)"
    if "(x, n) ∈ set (zip pars (map2 (λx. kind_conv κ (Some x)) pars (map (λe. eval e t) es)))"
    for x n
    using that by (auto simp: zip_map2_self kind_conv_simps)
  then show ?thesis
    unfolding call_enter_CallEdge
    by (rule in_kinds_bind_formals[OF in_kinds_enter_state[OF t]])
qed

text ‹A return resumes one of the suspended activations below a trace
  (\<^const>‹ancestors›), so the invariant is carried for all of them at once.›

lemma valid_activation_trace_callers_in_kinds:
  assumes vimp: "calls_in_kinds κ g" and S: "⋀s. s ∈ S ⟹ in_kinds κ s"
    and t: "t ∈ 𝒯⇘κ,g,S⇙"
  shows "in_kinds κ (sink_store t) ∧ (∀c ∈ ancestors t. in_kinds κ (sink_store c))"
  using t
proof (induction t rule: valid_activation_trace.induct)
  case (root s)
  then show ?case by (simp add: S)
next
  case (intra t a v s')
  then show ?case
    by (cases a)
      (auto simp: in_kinds_def kind_conv_simps split: option.splits if_splits simp del:
        ik_range_iff)
next
  case (call caller dst pars args p cont)
  then show ?case by (simp add: in_kinds_call_enter)
next
  case (ret callee caller p dst pars args cont)
  from ret.IH ancestors_caller [OF ret.hyps(2)]
  have "in_kinds κ (sink_store caller)" "∀c ∈ ancestors caller. in_kinds κ (sink_store c)"
    and callee: "in_kinds κ (sink_store callee)"
    by auto
  moreover have "in_kinds κ (combine_collect κ dst (sink_store caller) (sink_store callee))"
    using calculation
    by (cases dst) (auto simp: combine_collect_def in_kinds_combine_env kind_conv_simps)
  ultimately show ?case by (simp add: sink_store_def)
qed

theorem valid_activation_trace_in_kinds:
  assumes "calls_in_kinds κ g" and "⋀s. s ∈ S ⟹ in_kinds κ s"
    and "t ∈ 𝒯⇘κ,g,S⇙"
  shows "in_kinds κ (sink_store t)"
  using valid_activation_trace_callers_in_kinds [OF assms] by blast

theorem node_collect_in_kinds:
  assumes "calls_in_kinds κ g" and "⋀s. s ∈ S ⟹ in_kinds κ s"
    and "s ∈ 𝒞⇘κ,g,S⇙ v"
  shows "in_kinds κ s"
  using assms(3) valid_activation_trace_in_kinds [OF assms(1,2)]
  by (auto simp: node_collect_def)

subsection ‹The typed graph has typed calls›

text ‹Every call edge the compiler emits comes from a \<^const>‹Call› in a compiled body,
  with that call's destination and actuals and the callee's formals.›

fun com_calls :: "('a, 'v, 'e) gcom ⇒ ('v option × pname × 'e list) set" where
  "com_calls (gcom.Call _ dst q es) = {(dst, q, es)}"
| "com_calls (Seq c1 c2) = com_calls c1 ∪ com_calls c2"
| "com_calls (If _ _ c1 c2) = com_calls c1 ∪ com_calls c2"
| "com_calls (While _ _ c) = com_calls c"
| "com_calls _ = {}"

lemma compile_calls_from:
  "compile Π p c k n = (n', en, E, K) ⟹ (u, CallEdge dst pars es, v, w) ∈ K
   ⟹ ∃q. (dst, q, es) ∈ com_calls c ∧ pars = call_formals Π q"
proof (induction c arbitrary: k n n' en E K)
  case (Seq c1 c2)
  from Seq.prems(1) obtain n1 E1 K1 n2 E2 K2 where
      c1: "compile Π p c1 (Statement (n + csize c1)) n = (n1, Statement n, E1, K1)"
    and c2: "compile Π p c2 k (n + csize c1) = (n2, Statement (n + csize c1), E2, K2)"
    and K: "K = K1 ∪ K2"
    by (rule compile_SeqE)
  show ?case using Seq.IH(1)[OF c1] Seq.IH(2)[OF c2] Seq.prems(2) K by auto
next
  case (If i b c1 c2)
  from If.prems(1) obtain n1 E1 K1 n2 E2 K2 where
      c1: "compile Π p c1 k (Suc n) = (n1, Statement (Suc n), E1, K1)"
    and c2: "compile Π p c2 k (Suc n + csize c1) = (n2, Statement (Suc n + csize c1), E2, K2)"
    and K: "K = K1 ∪ K2"
    by (rule compile_IfE)
  show ?case using If.IH(1)[OF c1] If.IH(2)[OF c2] If.prems(2) K by auto
next
  case (While i b c)
  from While.prems(1) obtain n1 E1 K1 where
      c1: "compile Π p c (Statement n) (Suc n) = (n1, Statement (Suc n), E1, K1)"
    and K: "K = K1"
    by (rule compile_WhileE)
  show ?case using While.IH[OF c1] While.prems(2) K by auto
qed (auto split: option.splits simp del: call_formals_def)

lemma compile_prog_calls_from:
  assumes main: "Π prog_main_name = Some ⦇formals = [], body = main_body Π⦈"
    and e: "(u, CallEdge dst pars es, v, w) ∈ calls (compile_prog Π ps)"
  shows "∃r d q. Π r = Some d ∧ (dst, q, es) ∈ com_calls (body d) ∧ pars = call_formals Π q"
proof -
  obtain n1 Eprocs Kprocs n2 Emain Kmain where
      procs: "compile_procs Π ps 0 = (n1, Eprocs, Kprocs)"
    and mainc: "compile_proc Π prog_main_name ⦇formals = [], body = main_body Π⦈ n1
                  = (n2, Emain, Kmain)"
    and KC: "calls (compile_prog Π ps) = Kprocs ∪ Kmain"
    by (rule compile_prog_intra_split)
  from e KC consider (P) "(u, CallEdge dst pars es, v, w) ∈ Kprocs"
    | (M) "(u, CallEdge dst pars es, v, w) ∈ Kmain"
    by auto
  then show ?thesis
  proof cases
    case P
    from compile_procs_calls_origin [OF procs P] obtain r d m m' Ep Kp where
        d: "Π r = Some d" and cp: "compile_proc Π r d m = (m', Ep, Kp)"
      and eK: "(u, CallEdge dst pars es, v, w) ∈ Kp"
      by blast
    from cp obtain Eb where
        cb: "compile Π r (body d) (Statement (m + csize (body d))) m
               = (m + csize (body d), Statement m, Eb, Kp)"
      by (rule compile_procE)
    from compile_calls_from [OF cb eK] d show ?thesis by blast
  next
    case M
    from mainc obtain Eb where
        cb: "compile Π prog_main_name (main_body Π) (Statement (n1 + csize (main_body Π))) n1
               = (n1 + csize (main_body Π), Statement n1, Eb, Kmain)"
      by (rule compile_procE) simp
    from compile_calls_from [OF cb M] main show ?thesis
      by (metis gproc_decl.select_convs(2))
  qed
qed

lemma init_stores_in_kinds: "t ∈ init_stores κ ⟹ in_kinds κ t"
  by (auto simp: init_stores_def in_kinds_def all_in_kinds_def simp del: ik_range_iff)

subsection ‹Runs of the elaborated program stay within their kinds›

text ‹Elaboration puts a conversion on every argument, so the call edges of the elaborated
  graph pass well-typed arguments to formals of the right kinds.›

lemma com_calls_elab_com:
  "com_calls (elab_com 𝒢 fm κ r c)
     = (λ(dst, q, es). (map_option (resolve_var 𝒢 r) dst, q,
                         elab_args (resolve_var 𝒢 r) κ (map (resolve_var 𝒢 q) (fm q)) es))
       ` com_calls c"
  by (induction c) (auto simp: image_Un)

lemma call_formals_elab_ptable:
  "call_formals (elab_ptable p ds) q = map (resolve_var (declared_global p) q) (prog_formals p q)"
  by (cases "prog_table p q") (auto simp: elab_ptable_def prog_formals_def)

theorem calls_in_kinds_elab_cfg:
  assumes wf: "wf_program_compile_input p"
  shows "calls_in_kinds (kenv_of ds) (elab_cfg p ds)"
  unfolding calls_in_kinds_def
proof (intro allI impI)
  fix u dst pars es v w
  assume e: "(u, CallEdge dst pars es, v, w) ∈ calls (elab_cfg p ds)"
  from wf have src: "wf_source_program (declared_global p) (prog_table p)"
    by (simp add: gwf_compile_input_def)
  from compile_prog_calls_from [OF elab_ptable_main(1) [OF src] e [unfolded elab_cfg_def]]
  obtain r d q where d: "elab_ptable p ds r = Some d" and c: "(dst, q, es) ∈ com_calls (body d)"
    and pars: "pars = call_formals (elab_ptable p ds) q"
    by blast
  let ?ρ = "resolve_var (declared_global p) r"
  let ?fs = "map (resolve_var (declared_global p) q) (prog_formals p q)"
  from d c obtain es0 where es: "es = elab_args ?ρ (kenv_of ds) ?fs es0"
    by (auto simp: elab_ptable_def com_calls_elab_com)
  have "list_all (λe. wt_exp (kenv_of ds) e ∧ ReturnVar ∉ exp_vars e) es"
    using elab_args_wt [of "kenv_of ds" ?ρ ?fs es0] exp_vars_elab_args
      [of _ ?ρ "kenv_of ds" ?fs es0]
      ReturnVar_notin_range_resolve_var [of "declared_global p" r]
    unfolding es list_all_iff by blast
  moreover have "args_ok (kenv_of ds) pars es"
    using elab_args_wt [of "kenv_of ds" ?ρ ?fs es0]
    by (simp add: es pars call_formals_elab_ptable del: call_formals_def)
  ultimately show "list_all (λe. wt_exp (kenv_of ds) e ∧ ReturnVar ∉ exp_vars e) es
                     ∧ args_ok (kenv_of ds) pars es" ..
qed

text ‹Hence a run of the elaborated program that starts from a store within its kinds keeps
  every identity other than the return slot at a value of its kind, through assignments,
  declarations, calls, returns and recursion. The premise on the initial store is explicit;
  the initial stores of the soundness theorems satisfy it (@{thm init_stores_in_kinds}).›

theorem run_in_kinds:
  assumes wf: "wf_program_compile_input p"
    and s0: "in_kinds (kenv_of ds) s0"
    and run:
      "(kenv_of ds),(elab_ptable p ds) ⊢ (main_body (elab_ptable p ds), s0, []) →⇩p⇧* (residual, s, frs)"
  shows "in_kinds (kenv_of ds) s"
proof -
  from source_reaches_node_collect [OF wf _ run, of "{s0}"]
  obtain v where "s ∈ 𝒞⇘kenv_of ds,elab_cfg p ds,{s0}⇙ v" by blast
  then show ?thesis
    using node_collect_in_kinds [OF calls_in_kinds_elab_cfg [OF wf]] s0 by blast
qed

corollary run_from_init_in_kinds:
  assumes "wf_program_compile_input p"
    and "s0 ∈ init_stores (kenv_of ds)"
    and
      "(kenv_of ds),(elab_ptable p ds) ⊢ (main_body (elab_ptable p ds), s0, []) →⇩p⇧* (residual, s, frs)"
  shows "in_kinds (kenv_of ds) s"
  using run_in_kinds [OF assms(1) init_stores_in_kinds [OF assms(2)] assms(3)] .
