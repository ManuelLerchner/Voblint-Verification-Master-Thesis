section ‹One concrete call, two overlapping entry alternatives, two contexts›

text ‹
  A call's entry transfer answers a ∗‹list› of (continuation, callee entry)
  alternatives, and the entry-state route selects one context per alternative. When two
  alternatives both cover the same concrete call, that call is represented under two
  contexts at once. This theory builds exactly that situation on the stock executable Sign
  specification by overriding \<^const>‹dgs_enter› alone: alternative 1 enters the callee
  exactly (the formal ‹a› is \<^const>‹SPos›, the continuation's ‹x› untouched), alternative 2
  enters it with every formal forgotten to \<^const>‹STop› ∗‹and› forgets the continuation's
  own ‹x› to \<^const>‹STop› too. The concrete call with ‹x = 1› lies in both entries and both
  continuations, and the route sends the two entries to ‹[SPos]› and ‹[STop]›.

  Reusing ‹x› itself (rather than an artificial marker) keeps the pairing of a continuation
  with its entry observable per alternative --- alternative 1 answers ‹(x, y) = (SPos, SPos)›,
  alternative 2 answers ‹(STop, STop)› --- while also making each alternative's continuation
  a genuine, sound description of the caller's state: ‹STop› covers whatever ‹x› actually is,
  so both alternatives cover the one real caller store, and the routed context locale's
  totality obligation holds honestly rather than vacuously.

  The executable half pins the solved table: two alternatives stay distinguishable through
  the whole call/combine machinery, and each is read back and joined correctly. The
  soundness half shows the same thing relationally: the existing generic routed soundness
  bridge (\<^theory>‹Voblint_Exec.Routed_Exec_Refinement›) already supports list-valued,
  non-deterministic entry without any change to it --- this example is what supplies and
  discharges the corresponding non-deterministic entry obligations for the first time, and
  goes on to show the one concrete caller activation genuinely admitted under both
  ‹[SPos]› and ‹[STop]› via ‹ov_R›.
›

subsection ‹The overriding entry transfer›

text ‹
  ‹ov_enter› returns two overlapping alternatives per call, built from
  ‹forget_formals› and ‹forget_var›.
›

definition forget_formals ::
  "(vname ⇒ bool) ⇒ call_info ⇒ sign default_st ⇒ sign default_st" where
  "forget_formals 𝒢 ci s =
     foldl (λs f. s⟨location_of 𝒢 f := STop⟩) s (ci_formals ci)"

definition forget_var ::
  "(vname ⇒ bool) ⇒ vname ⇒ sign ⇒ sign default_st lifted ⇒ sign default_st lifted" where
  "forget_var 𝒢 v w = map_lift (λs. s⟨location_of 𝒢 v := w⟩)"

text ‹Alternative 1 is the stock singleton entry, continuation untouched. Alternative 2
  forgets the formals on entry ∗‹and› forgets the caller's own ‹x› in its continuation to
  \<^const>‹STop›: a real, sound weakening (‹STop› covers whatever ‹x› actually was), not an
  artificial tag on a name the program never uses. Every store in alternative 1's entry is
  in alternative 2's, so the two overlap; every store covered by alternative 1's
  continuation is covered by alternative 2's too (‹STop› is the top of the lattice), so both
  continuations cover the one real caller state --- the totality this override must satisfy
  is not vacuous.›

definition ov_enter ::
  "(vname ⇒ bool) ⇒ (sign default_st ⇒ bool) ⇒ call_info
   ⇒ sign default_st lifted ⇒ sign default_st lifted enter_result list" where
  "ov_enter 𝒢 ep ci d =
     (let entered = transfer_lift ep (generic_enter_st_for sign_ops 𝒢 ci) d
      in [(d, entered),
          (forget_var 𝒢 (STR ''x'') STop d, map_lift (forget_formals 𝒢 ci) entered)])"

definition ov_spec ::
  "(vname ⇒ bool) ⇒ (sign default_st ⇒ bool)
   ⇒ (pp × sign list, (unit, sign list) global_unknown, unit,
        sign default_st lifted, sign default_st lifted) dg_spec" where
  "ov_spec 𝒢 ep = (sign_tf.spec_exec 𝒢 ep)
     ⦇ dgs_enter := (λci. local_enter_transfer (ov_enter 𝒢 ep ci)) ⦈"


lemmas [code] = ov_spec_def[unfolded sign_tf.spec_exec_def]

text ‹Only the entry differs; every other field is \<^const>‹sign_tf.spec_exec›'s.›

lemma dg_spec_step_ov_spec [simp]:
  "dg_spec_step (ov_spec 𝒢 ep) a = dg_spec_step (sign_tf.spec_exec 𝒢 ep) a"
  by (cases a) (simp_all add: ov_spec_def)

lemma dgs_combine_env_ov_spec [simp]:
  "combine_env⇧♯ (ov_spec 𝒢 ep) = combine_env⇧♯ (sign_tf.spec_exec 𝒢 ep)"
  by (simp add: ov_spec_def)

lemma dgs_combine_assign_ov_spec [simp]:
  "combine_assign⇧♯ (ov_spec 𝒢 ep) = combine_assign⇧♯ (sign_tf.spec_exec 𝒢 ep)"
  by (simp add: ov_spec_def)

lemma dgs_enter_ov_spec [simp]:
  "enter⇧♯ (ov_spec 𝒢 ep) ci = local_enter_transfer (ov_enter 𝒢 ep ci)"
  by (simp add: ov_spec_def)

lemma dgs_query_ov_spec [simp]:
  "dgs_query (ov_spec 𝒢 ep) = dgs_query (sign_tf.spec_exec 𝒢 ep)"
  by (simp add: ov_spec_def)

text ‹The override runs its continuation once wherever the stock specification
  does: only the entry changed, and it answers its alternatives outright.›

lemma dg_spec_wf_ov_spec [intro, simp]: "dg_spec_wf (ov_spec 𝒢 ep)"
proof (unfold dg_spec_wf_def, intro conjI allI impI)
  show "sp_wf (dg_spec_step (ov_spec 𝒢 ep) a ((mk_dg_man d unknown_of)⦇man_ask := A⦈))"
    if "∀q. sp_wf (A q)" for a d unknown_of A
    using that by (simp add: dg_spec_wf_step_ask[OF sign_tf.dg_spec_wf_spec_exec])
next
  show "sp_wf (dgs_query (ov_spec 𝒢 ep) ((mk_dg_man d unknown_of)⦇man_ask := A⦈) q)"
    if "∀q. sp_wf (A q)" for d unknown_of A q
    using that by (simp add: dg_spec_wf_query[OF sign_tf.dg_spec_wf_spec_exec])
next
  fix ci d unknown_of
  show "sp_wf (enter⇧♯ (ov_spec 𝒢 ep) ci (mk_dg_man d unknown_of))"
    by (simp add: local_enter_transfer_def)
next
  fix ci d unknown_of ex
  show "sp_wf (dg_spec_combine_transfer (ov_spec 𝒢 ep) ci (mk_dg_man d unknown_of) ex)"
    unfolding dg_spec_combine_transfer_def dgs_combine_env_ov_spec dgs_combine_assign_ov_spec
    by (simp add: sign_tf.spec_exec_def exec_dg_spec_def
        local_combine_transfer_def)
qed

text ‹Entry is absent from \<^locale>‹analysis_contract›, so the override inherits the
  stock specification's contract outright.›

lemma analysis_contract_ov_spec:
  assumes exact: "⋀s. ep s = is_empty_state (ρ⇘𝒢⇙ s)"
  shows "analysis_contract (ov_spec 𝒢 ep) (λd _. γ (ρ⇘𝒢⇙ d)) 𝒢"
proof -
  interpret stock: analysis_contract "sign_tf.spec_exec 𝒢 ep" "(λd _. γ (ρ⇘𝒢⇙ d))" 𝒢
    by (rule sign_tf.sound_exec[OF exact])
  show ?thesis
  proof (unfold_locales, goal_cases)
    case 1 show ?case by (rule dg_spec_wf_ov_spec)
  next
    case (2 d d' g g') then show ?case by (rule stock.gammaDG_mono)
  next
    case (3 a τ src key)
    show ?case using stock.step_sound by (simp add: dg_spec_edge_program_def)
  next
    case (4 s dc key τ t de ci)
    show ?case using stock.combine_sound[where ci = ci and q = dc and r = de
          and τ = τ and key = key, OF 4(1) 4(2)]
      by (simp add: dg_spec_combine_transfer_def)
  qed
qed

subsection ‹The routed equation system, solved›

text ‹The entry-state registration's construction of \<^const>‹dg_pipeline.equations›
  at the overriding specification: same generator, same route, same buffered seed
  protocol, same plain-join solver.›

definition ov_eqs ::
  "(pp × sign list, (unit, sign list) global_unknown,
    (sign default_st lifted, sign default_st lifted) dg_state) eqsT" where
  "ov_eqs =
     routed_node_rhs_buffered intra_predecessor_addr_list call_site_list (λ_. Analysis_Global ())
       (exec_formals_route ov_gs)
       (λctx' src a. dg_spec_edge_program (ov_spec ov_gs ov_ep) a src (λ_. Analysis_Global ()))
       (routed_call_program (ov_spec ov_gs ov_ep) (λ_. Analysis_Global ()) Activation_Seed
          (static_resolve ov_cfg) (λd. d = Bot))
       (routed_entry_seed_programs Activation_Seed)
       ov_cfg Bot (Lifted cinit_sign_st) Bot"

definition ov_sol ::
  "(pp × sign list) set ×
   (pp × sign list + (unit, sign list) global_unknown
      ⇒ (sign default_st lifted, sign default_st lifted) dg_state)" where
  "ov_sol = TD_side_always_join_Interp_solve ov_eqs (cfg_exit ov_cfg, [])"

definition ov_result :: "(sign list, sign abs_state) solved_table" where
  "ov_result = Solved_Table (fst ov_sol)
     (λv ctx. ρ⇘ov_gs⇙ (canonicalize_lift ov_ep (dg_local (snd ov_sol (Inl (v, ctx))))))"

abbreviation ov_read :: "cfg_node ⇒ sign list ⇒ vname ⇒ sign lifted" where
  "ov_read v ctx x ≡ map_lift (λst. st x) (lookup_table ov_result v ctx)"

abbreviation ov_seed :: "sign list ⇒ vname ⇒ sign lifted" where
  "ov_seed ctx x ≡ map_lift (λs. ρ⇘ov_gs⇙ s x)
                     (dg_local (snd ov_sol (Inr (Activation_Seed p_entry ctx))))"

lemmas ov_unfold = ov_result_def ov_sol_def ov_eqs_def ov_cfg_def ov_pi_def ov_procs_def
  ov_ep_def ov_program_def

definition ov_alt_answer :: "nat ⇒ sign default_st lifted" where
  "ov_alt_answer i =
     (let caller = dg_local (snd ov_sol (Inl (Statement 3, [])));
          alts = ov_enter ov_gs ov_ep (call_info_of ov_ca (STR ''p'')) caller
      in dg_local (traverse_program
           (routed_call_alternative_program (ov_spec ov_gs ov_ep) (λ_. Analysis_Global ())
              Activation_Seed (exec_formals_route ov_gs) (λd. d = Bot)
              [] ov_ca (Statement 3) (STR ''p'') (alts ! i))
           (snd ov_sol)))"

abbreviation ov_alt_read :: "nat ⇒ vname ⇒ sign lifted" where
  "ov_alt_read i x ≡
     map_lift (λs. ρ⇘ov_gs⇙ s x) (ov_alt_answer i)"

definition ov_caller_store :: store where
  "ov_caller_store = (λ_. 0)(STR ''x'' := 1)"

  logical projection of this snapshot, so adding a witness does not rerun the equation
  system.›

  "(let sol = ov_sol;
        result = Solved_Table (fst sol)
          (λv ctx. ρ⇘ov_gs⇙
            (canonicalize_lift ov_ep (dg_local (snd sol (Inl (v, ctx))))));
        read = (λv ctx x. map_lift (λst. st x) (lookup_table result v ctx));

          let caller = dg_local (snd sol (Inl (Statement 3, [])));
              alts = ov_enter ov_gs ov_ep (call_info_of ov_ca (STR ''p'')) caller
          in dg_local (traverse_program
            (routed_call_alternative_program (ov_spec ov_gs ov_ep) (λ_. Analysis_Global ())
              Activation_Seed (exec_formals_route ov_gs) (λd. d = Bot)
              [] ov_ca (Statement 3) (STR ''p'') (alts ! i))
            (snd sol)))
     in TD_side_always_join_Interp_solve_c ov_eqs (cfg_exit ov_cfg, []) ≠ None
      ∧ table_contexts result p_entry = {[SPos], [STop]}
      ∧ table_contexts result (Statement 3) = {[]}
      ∧ seed [SPos] (STR ''a'') = Lifted SPos
      ∧ seed [STop] (STR ''a'') = Lifted STop
      ∧ read p_entry [SPos] (STR ''a'') = Lifted SPos
      ∧ read p_entry [STop] (STR ''a'') = Lifted STop
      ∧ read p_result [SPos] ret_var = Lifted SPos
      ∧ read p_result [STop] ret_var = Lifted STop
      ∧ map_lift (λs. ρ⇘ov_gs⇙ s (STR ''y''))
          (alt_answer 0) = Lifted SPos
      ∧ map_lift (λs. ρ⇘ov_gs⇙ s (STR ''x''))

      ∧ map_lift (λs. ρ⇘ov_gs⇙ s (STR ''y''))

      ∧ map_lift (λs. ρ⇘ov_gs⇙ s (STR ''x''))

      ∧ dg_local (snd sol (Inl (Statement 3, []))) =

      ∧ exec_formals_route ov_gs (Statement 3) []
          (transfer_lift ov_ep (generic_enter_st_for sign_ops ov_gs (call_info_of ov_ca (STR ''p'')))
            (dg_local (snd sol (Inl (Statement 3, []))))) ov_ca = [SPos]
      ∧ forget_var ov_gs (STR ''x'') STop
          (dg_local (snd sol (Inl (Statement 3, [])))) =
          default_st_set_lift (Lifted cinit_sign_st)
            (location_of ov_gs (STR ''x'')) STop
      ∧ exec_formals_route ov_gs (Statement 3) []
          (map_lift (forget_formals ov_gs (call_info_of ov_ca (STR ''p'')))
            (transfer_lift ov_ep (generic_enter_st_for sign_ops ov_gs (call_info_of ov_ca (STR ''p'')))
              (dg_local (snd sol (Inl (Statement 3, [])))))) ov_ca = [STop]
      ∧ (∀(u, ctx) ∈ fst sol. ∀(u', a, v) ∈ intra ov_cfg.

      ∧ (∀(cl, ctx) ∈ fst sol. ∀(cl', ca, ce, cont) ∈ calls ov_cfg.

  unfolding ov_sol_def ov_eqs_def ov_cfg_def ov_pi_def ov_procs_def ov_ep_def ov_program_def

subsection ‹4. The continuation holds the join of both alternatives›

subsection ‹5. Negative: an empty entry list cannot cover a live caller›

  "¬ entry_pairs_cover (λd' :: sign default_st lifted. γ (ρ⇘ov_gs⇙ d')) ov_caller_store
