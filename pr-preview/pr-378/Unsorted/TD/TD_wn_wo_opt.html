subsection ‹Definitions›

function (domintros)
    query :: "'x ⇒ 'x ⇒ 'x set ⇒ ('x, 'x list) fmap ⇒ 'x set ⇒ 'x set ⇒ ('x, 'd) map
              ⇒ 'd × ('x, 'x list) fmap × 'x set × 'x set × ('x, 'd) map" and
  iterate :: "'x ⇒ 'x set ⇒ ('x, 'x list) fmap ⇒ 'x set ⇒ 'x set ⇒ ('x, 'd) map ⇒ phase
              ⇒ 'd × ('x, 'x list) fmap × 'x set × 'x set × ('x, 'd) map" and
     eval :: "'x ⇒ ('x, 'd) strategy_tree ⇒ 'x set ⇒ ('x, 'x list) fmap ⇒ 'x set ⇒ 'x set
              ⇒ ('x, 'd) map ⇒ 'd × ('x, 'x list) fmap × 'x set × 'x set × ('x, 'd) map"
     where
  "query y x c infl stabl p σ = (
    let (xd, infl, stabl, p, σ) =
      if x ∈ c then
        (mlup σ x, infl, stabl, insert x p, σ)
      else
        iterate x (insert x c) infl stabl p σ Widening
    in (xd, fminsert infl x y, stabl, p, σ))"
| "iterate x c infl stabl p σ wn = (
    if x ∉ stabl then (
      let (xd, infl, stabl, p', σ) = eval x (T x) c infl (insert x stabl) p σ in
      let d_new = if x ∉ p then xd else if wn = Widening then mlup σ x ∇ xd else mlup σ x Δ xd in
      if mlup σ x = d_new then (
        if x ∈ p ∧ wn = Widening then (
          let (infl, stabl) = destab x infl stabl in
          iterate x c infl stabl p' σ Narrowing)
        else (mlup σ x, infl, stabl, Set.remove x p', σ))
      else (
        let (infl, stabl) = destab x infl stabl in
        iterate x c infl stabl p' (σ(x ↦ d_new)) wn))
    else
      (mlup σ x, infl, stabl, Set.remove x p, σ))"
| "eval x t c infl stabl p σ = (case t of
      Answer d ⇒ (d, infl, stabl, p, σ)
    | Query y g ⇒ (
        let (yd, infl, stabl, p, σ) = query x y c infl stabl p σ in
        eval x (g yd) c infl stabl p σ))"
  by pat_completeness auto

definition solve :: "'x ⇒ 'x set × ('x, 'd) map" where
  "solve x = (let (_, _, stabl, _, σ) = iterate x {x} fmempty {} {} Map.empty Widening in (stabl, σ))"

definition query_dom where
  "query_dom y x c infl stabl p σ = query_iterate_eval_dom (Inl (y, x, c, infl, stabl, p, σ))"
declare query_dom_def [simp]
definition iterate_dom where
  "iterate_dom x c infl stabl p σ wn = query_iterate_eval_dom (Inr (Inl (x, c, infl, stabl, p, σ, wn)))"
declare iterate_dom_def [simp]
definition eval_dom where
  "eval_dom x t c infl stabl p σ = query_iterate_eval_dom (Inr (Inr (x, t, c, infl, stabl, p, σ)))"
declare eval_dom_def [simp]

definition solve_dom where
  "solve_dom x = iterate_dom x {x} fmempty {} {} Map.empty Widening"

lemmas dom_defs = query_dom_def iterate_dom_def eval_dom_def


subsubsection ‹Refinement of auto-generated rules›

text ‹The auto-generated \texttt{pinduct} rule contains a redundant assumption. This lemma removes
this redundant assumption such that the rule is easier to instantiate and gives comprehensible
names to the cases.›

lemmas query_iterate_eval_pinduct[consumes 1, case_names Query Iterate Eval]
  = query_iterate_eval.pinduct(1)[
      folded query_dom_def iterate_dom_def eval_dom_def,
      of y x c infl stabl p σ for y x c infl stabl p σ
    ]
    query_iterate_eval.pinduct(2)[
      folded query_dom_def iterate_dom_def eval_dom_def,
      of x c infl stabl p σ wn for x c infl stabl p σ wn
    ]
    query_iterate_eval.pinduct(3)[
      folded query_dom_def iterate_dom_def eval_dom_def,
      of x t c infl stabl p σ for x t c infl stabl p σ
    ]

declare query.psimps [simp]
declare iterate.psimps [simp]
declare eval.psimps [simp]

subsubsection ‹Domain lemmas›

lemma dom_backwards_pinduct:
  shows "query_dom y x c infl stabl p σ
    ⟹ x ∉ c ⟹ iterate_dom x (insert x c) infl stabl p σ Widening"
  and "iterate_dom x c infl stabl p σ wn
    ⟹ x ∉ stabl ⟹ (eval_dom x (T x) c infl (insert x stabl) p σ ∧
        ((xd, infl1, stabl1, p1, σ1) = eval x (T x) c infl (insert x stabl) p σ
          ⟶ d_new = (if x ∉ p then xd else if wn = Widening then mlup σ1 x ∇ xd else mlup σ1 x Δ xd)
          ⟶ (mlup σ1 x = d_new ⟶ x ∈ p ⟶ wn = Widening ⟶ (infl2, stabl2) = destab x infl1 stabl1
          ⟶ iterate_dom x c infl2 stabl2 p1 σ1 Narrowing)
          ∧ (mlup σ1 x ≠ d_new ⟶ (infl2, stabl2) = destab x infl1 stabl1 ⟶
          iterate_dom x c infl2 stabl2 p1 (σ1(x ↦ d_new)) wn)))"
  and "eval_dom x (Query y g) c infl stabl p σ
    ⟹ (query_dom x y c infl stabl p σ ∧
        ((yd, infl', stabl', p', σ') = query x y c infl stabl p σ ⟶
          eval_dom x (g yd) c infl' stabl' p' σ'))"
proof (induction y x c infl stabl p σ and x c infl stabl p σ wn
    and x "Query y g" c infl stabl p σ
    arbitrary: and xd infl1 stabl1 p1 σ1 d_new infl2 stabl2 and y g yd infl' stabl' p' σ'
    rule: query_iterate_eval_pinduct)
  case (Query y x c infl stabl p σ)
  then show ?case using query_iterate_eval.domintros(2) by (auto, presburger)
next
  case (Iterate x c infl stabl p σ wn)

  have eval_dom: "eval_dom x (T x) c infl (insert x stabl) p σ"
    using Iterate.hyps(2) Iterate.prems query_iterate_eval.domintros(3) by force

  show ?case
  proof (intro impI conjI, goal_cases)
    case 1
    then show ?case using eval_dom by force
  next
    case 2
    have B: "x ∈ p ∧ wn = Widening" using 2(4,5) by simp
    show ?case
      using Iterate.hyps(3)[OF Iterate.prems 2(1) _ _ _ _ 2(2,3) B 2(6), simplified]
        query_iterate_eval.domintros(2)[folded dom_defs, of x stabl2 c infl2 p1 σ1 Narrowing]
      by fastforce
  next
    case 3
    show ?case
      using Iterate.hyps(4)[OF Iterate.prems 3(1) _ _ _ _ 3(2-4), of _ _ _ _ infl2, simplified]
        iterate_dom_def query_iterate_eval.domintros(2)[of x stabl2 c infl2 p1 "σ1(x ↦ d_new)" wn]
      by presburger
  qed
next
  case (Eval x c infl stabl p σ)
  then show ?case using query_iterate_eval.domintros(1,3) by simp
qed


subsubsection ‹Case rules›

lemma iterate_continue_fixpoint_cases[consumes 3]:
  assumes "iterate_dom x c infl stabl p σ wn"
    and "(xd, infl', stabl', p', σ') = iterate x c infl stabl p σ wn"
    and "x ∈ c"
  obtains (Stable) "x ∈ stabl"
    and "xd = mlup σ x"
    and "infl' = infl"
    and "stabl' = stabl"
    and "p' = Set.remove x p"
    and "σ' = σ"
  | (Fixpoint) p1
  where "x ∉ stabl" and "x ∉ p"
    and "eval_dom x (T x) c infl (insert x stabl) p σ"
    and "(xd, infl', stabl', p1, σ') = eval x (T x) c infl (insert x stabl) p σ"
    and "mlup σ' x = xd"
    and "p' = Set.remove x p1"
  | (W_Fixpoint) xd1 infl1 stabl1 p1 σ1 infl2 stabl2
  where "x ∉ stabl" and "x ∈ p" and "wn = Widening"
    and "eval_dom x (T x) c infl (insert x stabl) p σ"
    and "(xd1, infl1, stabl1, p1, σ1) = eval x (T x) c infl (insert x stabl) p σ"
    and "mlup σ1 x = mlup σ1 x ∇ xd1"
    and "(infl2, stabl2) = destab x infl1 stabl1"
    and "iterate_dom x c infl2 stabl2 p1 σ1 Narrowing"
    and "(xd, infl', stabl', p', σ') = iterate x c infl2 stabl2 p1 σ1 Narrowing"
  | (N_Fixpoint) xd1 p1
  where "x ∉ stabl" and "x ∈ p" and "wn = Narrowing"
    and "eval_dom x (T x) c infl (insert x stabl) p σ"
    and "(xd1, infl', stabl', p1, σ') = eval x (T x) c infl (insert x stabl) p σ"
    and "xd = mlup σ' x Δ xd1"
    and "mlup σ' x = xd"
    and "p' = Set.remove x p1"
  | (Continue) xd1 infl1 stabl1 p1 σ1 d_new infl2 stabl2
  where "x ∉ stabl"
    and "eval_dom x (T x) c infl (insert x stabl) p σ"
    and "(xd1, infl1, stabl1, p1, σ1) = eval x (T x) c infl (insert x stabl) p σ"
    and "d_new = (if x ∉ p then xd1 else if wn = Widening then mlup σ1 x ∇ xd1 else mlup σ1 x Δ xd1)"
    and "mlup σ1 x ≠ d_new"
    and "(infl2, stabl2) = destab x infl1 stabl1"
    and "iterate_dom x c infl2 stabl2 p1 (σ1(x ↦ d_new)) wn"
    and "(xd, infl', stabl', p', σ') = iterate x c infl2 stabl2 p1 (σ1(x ↦ d_new)) wn"
proof(cases "x ∈ stabl" rule: case_split[case_names Stable Unstable])
  case Stable
  then show ?thesis using that(1) assms(1,2) by auto
next
  case Unstable
  then have sldom: "eval_dom x (T x) c infl (insert x stabl) p σ"
    using assms(1) dom_backwards_pinduct(2)
    by simp
  then obtain xd1 infl1 stabl1 p1 σ1
    where slapp: "eval x (T x) c infl (insert x stabl) p σ = (xd1, infl1, stabl1, p1, σ1)"
    by (cases "eval x (T x) c infl (insert x stabl) p σ") auto
  then obtain d_new where d_new:
      "d_new = (if x ∉ p then xd1 else if wn = Widening then mlup σ1 x ∇ xd1 else mlup σ1 x Δ xd1)"
    by blast
  show ?thesis
  proof (cases "mlup σ1 x = d_new" rule: case_split[case_names Fixpoint Continue])
    case Fixpoint
    then show ?thesis
    proof (cases "x ∈ p")
      case True
      then show ?thesis
      proof (cases wn)
        case Widening
        obtain infl2 stabl2 where destab: "destab x infl1 stabl1 = (infl2, stabl2)"
          by (cases "destab x infl1 stabl1")
        have "d_new = mlup σ1 x ∇ xd1" using d_new Widening True by simp
        moreover have "iterate x c infl stabl p σ wn = iterate x c infl2 stabl2 p1 σ1 Narrowing"
          using assms iterate.psimps[folded dom_defs, OF assms(1)] Unstable slapp Fixpoint d_new Widening
            True destab
          by force
        moreover have "iterate_dom x c infl2 stabl2 p1 σ1 Narrowing"
          using Unstable slapp Fixpoint d_new Widening True assms(1) dom_backwards_pinduct(2) destab
          by presburger
        ultimately show ?thesis
          using assms(2) Fixpoint that(3)[OF Unstable True Widening sldom slapp[symmetric]] destab by simp
      next
        case Narrowing
        have "d_new = mlup σ1 x Δ xd1" using d_new Narrowing True by simp
        moreover have "iterate x c infl stabl p σ wn = (mlup σ1 x, infl1, stabl1,
            Set.remove x p1, σ1)"
          using assms iterate.psimps[folded dom_defs, OF assms(1)] Unstable slapp Fixpoint d_new Narrowing
            True
          by fastforce
        ultimately show ?thesis
          using Unstable sldom slapp d_new assms that(4)[OF Unstable True Narrowing sldom _] Fixpoint
          by auto
      qed
    next
      case False
      have "iterate x c infl stabl p σ wn = (mlup σ1 x, infl1, stabl1, Set.remove x p1, σ1)"
        using assms iterate.psimps[folded dom_defs, OF assms(1)] Unstable slapp Fixpoint d_new False
          by simp
      then show ?thesis
        using Unstable sldom slapp d_new assms that(2)[OF Unstable False sldom] False Fixpoint
        by auto
    qed
  next
    case Continue
    obtain infl2 stabl2 where destab: "destab x infl1 stabl1 = (infl2, stabl2)"
      by (cases "destab x infl1 stabl1")
    then have dom: "iterate_dom x c infl2 stabl2 p1 (σ1(x ↦ d_new)) wn"
      using Unstable Continue slapp d_new assms(1) dom_backwards_pinduct(2) by auto
    then show ?thesis
      using Unstable Continue sldom slapp d_new assms destab that(3,5)
      by (cases "x ∈ p") (auto simp add: Let_def)
  qed
qed

lemma query_iterate_lookup_cases[consumes 2]:
  assumes "query_dom y x c infl stabl p σ"
    and "(xd, infl', stabl', p', σ') = query y x c infl stabl p σ"
  obtains (Iterate) infl1
  where "x ∉ c"
    and "iterate_dom x (insert x c) infl stabl p σ Widening"
    and "(xd, infl1, stabl', p', σ') = iterate x (insert x c) infl stabl p σ Widening"
    and "infl' = fminsert infl1 x y"
  | (Lookup) "x ∈ c"
    and "xd = mlup σ x"
    and "infl' = fminsert infl x y"
    and "stabl' = stabl"
    and "p' = insert x p"
    and "σ' = σ"
  using assms that dom_backwards_pinduct(1)
  by (cases "x ∈ c"; auto split: prod.splits)

lemma eval_query_answer_cases[consumes 2]:
  assumes "eval_dom x t c infl stabl p σ"
    and "(xd, infl', stabl', p', σ') = eval x t c infl stabl p σ"
  obtains (Query) y g yd infl1 stabl1 p1 σ1
  where "t = Query y g"
    and "query_dom x y c infl stabl p σ"
    and "(yd, infl1, stabl1, p1, σ1) = query x y c infl stabl p σ"
    and "eval_dom x (g yd) c infl1 stabl1 p1 σ1"
    and "(xd, infl', stabl', p', σ') = eval x (g yd) c infl1 stabl1 p1 σ1"
  | (Answer) "t = Answer xd"
    and "infl' = infl"
    and "stabl' = stabl"
    and "p' = p"
    and "σ' = σ"
  using assms dom_backwards_pinduct(3) that
  by (cases t; auto split: prod.splits)

corollary TD_wn_TD_wn_post_equivalence: 
  assumes "solve_dom x"
      and "solve x = (stabl, σ)"
    shows "TD_post.solve_dom x ∧ TD_post.solve x = (stabl, σ)"
proof -
  obtain xd infl p where "(xd, infl, stabl, p, σ) = iterate x {x} fmempty {} {} (λx. None) Widening"
    using assms(2) unfolding solve_def by (auto split: prod.splits)
  then obtain pinfl post where "TD_post.iterate_dom x {x} {} fmempty {} {} (λx. None) Widening fmempty {}"
    "(xd, infl, stabl, p, σ, pinfl, post) = TD_post.iterate x {x} {} fmempty {} {} (λx. None) Widening fmempty {}"
    using equiv_post_solver_aux_1(2)[where c⇩w = "{x}" and c⇩n ="{}"] assms(1)
    unfolding solve_dom_def by blast
  then show ?thesis unfolding TD_post.solve_dom_def TD_post.solve_def by (auto split: prod.splits)
qed

lemma equiv_post_solver_aux_2:
  shows "TD_post.query_dom y x c⇩w c⇩n infl stabl p σ pinfl post
    ⟹ (xd, infl', stabl', p', σ', pinfl', post') =
        TD_post.query y x c⇩w c⇩n infl stabl p σ pinfl post
    ⟹ query_dom y x (c⇩w ∪ c⇩n) infl stabl p σ
        ∧ (xd, infl', stabl', p', σ') = query y x (c⇩w ∪ c⇩n) infl stabl p σ"
    and "TD_post.iterate_dom x c⇩w c⇩n infl stabl p σ wn pinfl post
    ⟹ (xd, infl', stabl', p', σ', pinfl', post') =
        TD_post.iterate x c⇩w c⇩n infl stabl p σ wn pinfl post
    ⟹ x ∈ c⇩w ∪ c⇩n
    ⟹ iterate_dom x (c⇩w ∪ c⇩n) infl stabl p σ wn
        ∧ (xd, infl', stabl', p', σ') = iterate x (c⇩w ∪ c⇩n) infl stabl p σ wn"
    and "TD_post.eval_dom x t c⇩w c⇩n infl stabl p σ pinfl post
    ⟹ (xd, infl', stabl', p', σ', pinfl', post') = TD_post.eval x t c⇩w c⇩n infl stabl p σ pinfl post
    ⟹ eval_dom x t (c⇩w ∪ c⇩n) infl stabl p σ
        ∧ (xd, infl', stabl', p', σ') = eval x t (c⇩w ∪ c⇩n) infl stabl p σ"
proof(induction y x c⇩w c⇩n infl stabl p σ pinfl post and x c⇩w c⇩n infl stabl p σ wn pinfl post
    and y t c⇩w c⇩n infl stabl p σ pinfl post arbitrary: xd infl' stabl' p' σ' pinfl' post'
    and xd infl' stabl' p' σ' pinfl' post' and xd infl' stabl' p' σ' pinfl' post'
    rule: TD_post.query_iterate_eval_pinduct)
  case (Query y x c⇩w c⇩n infl stabl p σ pinfl post)
  show ?case using Query.IH(1) Query.prems(1)
  proof(cases rule: TD_post.query_iterate_lookup_cases)
    case (Iterate infl1 pinfl1 c1⇩w c1⇩n)
    have IH: "iterate_dom x (c1⇩w ∪ c1⇩n) infl stabl p σ Widening"
      "(xd, infl1, stabl', p', σ') = iterate x (c1⇩w ∪ c1⇩n) infl stabl p σ Widening"
      using Query.IH(2) Iterate(1,3,4) by (auto split: if_splits)
    then have "query_dom y x (c⇩w ∪ c⇩n) infl stabl p σ"
      using query_iterate_eval.domintros(1)[of x "c⇩w ∪ c⇩n"] Iterate(1,4) by (auto split: if_splits)
    then show ?thesis
      using query.psimps[folded dom_defs] Iterate(1,4) IH(2) Iterate(5)
      by (auto split: prod.splits if_splits)
  next
    case Lookup
    then show ?thesis using query_iterate_eval.domintros(1)[of x "c⇩w ∪ c⇩n"] by auto
  qed
next
  case (Iterate x c⇩w c⇩n infl stabl p σ wn pinfl post)
  show ?case using Iterate.IH(1) Iterate.prems(1,2)
  proof(cases rule: TD_post.iterate_continue_fixpoint_cases)
    case Stable
    then show ?thesis using query_iterate_eval.domintros(2)[of x stabl] by fastforce
  next
    case (Fixpoint p1)

  next

    have "iterate_dom x (c⇩w ∪ c⇩n) infl stabl p σ wn"
